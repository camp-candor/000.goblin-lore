export function verifyByteZeroBuffer(buf: Buffer): boolean {
  if (buf.length < 4) return false
  return (
    buf[0] === 0x2d && // '-'
    buf[1] === 0x2d && // '-'
    buf[2] === 0x2d && // '-'
    buf[3] === 0x0a    // '\n'
  )
}
