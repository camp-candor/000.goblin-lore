export const parseVoiceRail = (cpy: VoiceModel, bal: VoiceBit, ste: State): VoiceModel => {

  const hasActualWant = content.includes('## Wants (actual)');

  const hasVoiceNotes = content.includes('## Voice notes');

  if (bal.slv != null) {
    bal.slv({

        idx: 'parse-voice-rail',
        val: isValid ? 1 : 0,
        dat: {
          hasSurfaceWant,
          hasActualWant,
          hasBlindSpot,
          hasVoiceNotes,
          hasFewShot,
          isValid
        }
      }
    });
  }
  return cpy;

import clone from 'clone-deep';
import * as Act from './voice.action';
import { VoiceModel } from './voice.model';
import * as Buzz from './voice.buzzer';
import type State from '../99.core/state';

export function reducer(
  model: VoiceModel = new VoiceModel(),
  act: Act.Actions,
  state?: State
): VoiceModel {
  const cpy: VoiceModel = clone(model);
  const ste: State = state!;

  switch (act.type) {
    case Act.INIT_VOICE:
      return Buzz.initVoice(cpy, act.bale, ste);

    case Act.UPDATE_VOICE:
      return Buzz.updateVoice(cpy, act.bale, ste);

    case Act.PARSE_VOICE_RAIL:
      return Buzz.parseVoiceRail(cpy, act.bale, ste);

    default:
      return cpy;
  }
}
