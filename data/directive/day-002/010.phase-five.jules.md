  tokensRemoved: string[];

export function executeAttenuationFenceLift(
  entityId: string,
  existingTokens: readonly string[],
  bannedTokensRemoved: readonly string[],
): AttenuationFenceLiftReceipt {
  const removalSet = new Set(bannedTokensRemoved.map((t) => t.trim().toLowerCase()));
  const remainingTokens: string[] = [];

  for (const token of existingTokens) {

      tokensRemoved.push(normalized);
    } else {
      remainingTokens.push(normalized);
    }
  }

  return Object.freeze({
    entityId,
    tokensRemoved: Object.freeze(tokensRemoved.sort()),
    remainingTokens: Object.freeze(remainingTokens.sort()),
    isBreakStateAuthorized: true,
  });

export const evaluatePostBreakThreshold = async (
  cpy: ThresholdModel,
  bal: ThresholdBit & { dat: any },
  ste: State,
): Promise<ThresholdModel> => {
  const payload = bal.dat;
  if (!payload || !payload.input) {
    if (bal.slv) bal.slv({ trhBit: { idx: 'evaluate-post-break-fail', val: -1 } });
    return cpy;
  }

  try {
    const { isBreached, receipt } = evaluateThresholdBreachInterception(
      payload.input,
      payload.activeAcousticProfile,
      payload.currentAttenuationTokens,
    );

    if (isBreached && receipt) {
      cpy.breakReceipts[receipt.entityId] = receipt;
      cpy.brokenEntities.add(receipt.entityId);

      if (ste) {
        await ste.hunt(UPDATE_CONSOLE, {
          idx: 'cns00',
          src: `>> [POST_BREAK] ${receipt.entityId} breached threshold on ${receipt.category} (${receipt.breachedValueBp}/${receipt.thresholdLimitBp} bps) [!] HALT ENGAGED [OK]`,
        });
        await ste.hunt(UPDATE_CONSOLE, {
          idx: 'cns00',
          src: `>> [FINALITY] Mutated refusal ${receipt.targetRefusalVectorId} | Added Condition: ${receipt.terminalConditionAdded} | Lifted tokens: [${receipt.bannedTokensRemoved.join(', ')}] [OK]`,
        });
      }

      if (bal.slv) {
        bal.slv({
          trhBit: {
            idx: 'evaluate-post-break-breached',
            val: receipt.breachedValueBp,
            dat: receipt,
          },
        });
      }
    } else {
      if (bal.slv) {
        bal.slv({
          trhBit: {
            idx: 'evaluate-post-break-unbreached',
            val: 0,
            dat: { isBreached: false },
          },
        });
      }
    }
  } catch (err: unknown) {
    const message = err instanceof Error ? err.message : String(err);
    if (ste) {
      await ste.hunt(UPDATE_CONSOLE, {
        idx: 'cns00',
        src: `:: [POST_BREAK FAIL] ${message}`,
      });
    }
    if (bal.slv) {
      bal.slv({
        trhBit: {
          idx: 'evaluate-post-break-err',
          val: -1,
          src: message,
        },
      });
    }
  }

  return cpy;
};

export function reducer(

  act: Act.Actions,
  state?: State,

      return model;

    default:
      return model;
  }
