#### 7. Business Logic: `packages/299.lore/29.series.unit/buz/series.buzz.ts`

```typescript
import { SeriesModel } from '../series.model.js';
import type SeriesBit from '../fce/series.bit.js';
import type State from '../../99.core/state.js';
import { resolveArchetypeInheritance, SomaticArchetypeSchema } from '../val/archetype.val.js';
import { ZodError } from 'zod';

export const initSeries = (cpy: SeriesModel, bal: SeriesBit, ste: State): SeriesModel => {
  if (bal.slv) {
    bal.slv({
      seriesBit: {
        idx: bal.idx,
        val: 1,
        dat: { initialized: true }
      }
    });
  }
  return cpy;
};

export const updateSeries = (cpy: SeriesModel, bal: SeriesBit, ste: State): SeriesModel => {
  if (bal.slv) {
    bal.slv({
      seriesBit: {
        idx: bal.idx,
        val: 1
      }
    });
  }
  return cpy;
};

export const resolveArchetype = (cpy: SeriesModel, bal: SeriesBit, ste: State): SeriesModel => {
  const archetypeId = bal.dat?.archetypeId ?? bal.dat?.delta?.archetype_id;
  const delta = bal.dat?.delta;

  try {
    if (!archetypeId) {
      throw new Error('[ARCHETYPE_ID_REQUIRED]: No archetypeId provided in payload.');
    }

    const archetype = cpy.archetypeRegistry[archetypeId];
    if (!archetype) {
      throw new Error(`[UNKNOWN_ARCHETYPE]: Archetype ID '${archetypeId}' not found in registry.`);
    }

    const resolved = resolveArchetypeInheritance(archetype, delta);

    if (bal.slv) {
      bal.slv({
        seriesBit: {
          idx: bal.idx,
          val: 1,
          dat: {
            archetypeId,
            resolvedProfile: resolved,
            isValid: true
          }
        }
      });
    }
  } catch (error) {
    const errorMsg = error instanceof ZodError
      ? `[ARCHETYPE_VALIDATION_ERROR]: ${error.issues.map(i => i.message).join(', ')}`
      : String(error);

    if (bal.slv) {
      bal.slv({
        seriesBit: {
          idx: bal.idx,
          val: 0,
          dat: {
            archetypeId,
            isValid: false,
            error: errorMsg
          }
        }
      });
    }
  }
  return cpy;
};
```

#### 9. Reducer: `packages/299.lore/29.series.unit/series.reduce.ts`

```typescript
import clone from 'clone-deep';
import * as Act from './series.action.js';
import { SeriesModel } from './series.model.js';
import * as Buzz from './series.buzzer.js';
import type State from '../99.core/state.js';

export function reducer(model: SeriesModel = new SeriesModel(), act: Act.Actions, state?: State): SeriesModel {
  const cpy = clone(model);
  switch (act.type) {
    case Act.INIT_SERIES:
      return Buzz.initSeries(cpy, act.bale, state!);
    case Act.UPDATE_SERIES:
      return Buzz.updateSeries(cpy, act.bale, state!);
    case Act.RESOLVE_ARCHETYPE:
      return Buzz.resolveArchetype(cpy, act.bale, state!);
    default:
      return cpy;
  }
}
```
