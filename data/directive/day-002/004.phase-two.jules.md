```typescript
import {
  PressureMeterSchema,
  FrictionDebitEventSchema,
  assertIntegerMetric,
  clampBps,
  type PressureMeter,
  type FrictionDebitEvent,
} from '../val/pressureMeter.val.js';
import type { ThermodynamicCycleReceiptBit } from '../fce/pressureMeter.bit.js';

export function calculateThermodynamicStep(
  meterInput: PressureMeter,
  frictionEvents: readonly FrictionDebitEvent[] = [],
  tick: number = 0,
): ThermodynamicCycleReceiptBit {
  const meter = PressureMeterSchema.parse(meterInput);
  assertIntegerMetric(tick, 'tick');

  let totalFrictionDebitsBp = 0;
  for (const event of frictionEvents) {
    const parsedEvent = FrictionDebitEventSchema.parse(event);
    assertIntegerMetric(parsedEvent.debitBp, 'debitBp');
    totalFrictionDebitsBp += parsedEvent.debitBp;
  }
  totalFrictionDebitsBp = clampBps(totalFrictionDebitsBp);

  const previousValueBp = meter.currentValueBp;
  const rawDecayed = previousValueBp - meter.decayRatePerCycleBp;
  const cooledValueBp = Math.max(meter.baselineBp, rawDecayed);
  const dissipatedBp = previousValueBp - cooledValueBp;

  const nextValueBp = clampBps(cooledValueBp + totalFrictionDebitsBp);
  const isThresholdBreached = nextValueBp >= meter.thresholdLimitBp;

  const receipt: ThermodynamicCycleReceiptBit = {
    meterId: meter.meterId,
    entityId: meter.entityId,
    category: meter.category,
    previousValueBp,
    currentValueBp: nextValueBp,
    baselineBp: meter.baselineBp,
    dissipatedBp,
    accumulatedDebitsBp: totalFrictionDebitsBp,
    isThresholdBreached,
    updatedTick: tick,
  };

  return Object.freeze(receipt);
}

export function batchStepThermodynamicMeters(
  meters: readonly PressureMeter[],
  frictionMap: Record<string, readonly FrictionDebitEvent[]> = {},
  tick: number = 0,
): ThermodynamicCycleReceiptBit[] {
  return meters.map((m) => calculateThermodynamicStep(m, frictionMap[m.meterId] || [], tick));
}
```

```typescript
import {
  InterMeterBridgeEvaluationInputSchema,
  assertIntegerMetric,
  type InterMeterBridgeEvaluationInput,
} from '../val/interMeterBridge.val.js';

export interface InterMeterBridgeReceiptBit {
  readonly entityId: string;
  readonly somaticFatigueBp: number;
  readonly originalRefusalThresholdLimitBp: number;
  readonly effectiveRefusalThresholdLimitBp: number;
  readonly isReductionActive: boolean;
  readonly reductionAppliedBp: number;
}

export function evaluateSomaticRefusalBridge(
  input: InterMeterBridgeEvaluationInput,
): InterMeterBridgeReceiptBit {
  const parsed = InterMeterBridgeEvaluationInputSchema.parse(input);
  assertIntegerMetric(parsed.somaticFatigueBp, 'somaticFatigueBp');
  assertIntegerMetric(parsed.refusalStrainTargetLimitBp, 'refusalStrainTargetLimitBp');

  const FATIGUE_TRIGGER_BP = 8000;
  const REDUCTION_RATIO_BPS = 8000;

  let effectiveLimitBp = parsed.refusalStrainTargetLimitBp;
  let isReductionActive = false;
  let reductionAppliedBp = 0;

  if (parsed.somaticFatigueBp >= FATIGUE_TRIGGER_BP) {
    isReductionActive = true;
    effectiveLimitBp = Math.floor((parsed.refusalStrainTargetLimitBp * REDUCTION_RATIO_BPS) / 10000);
    reductionAppliedBp = parsed.refusalStrainTargetLimitBp - effectiveLimitBp;
  }

  const receipt: InterMeterBridgeReceiptBit = {
    entityId: parsed.entityId,
    somaticFatigueBp: parsed.somaticFatigueBp,
    originalRefusalThresholdLimitBp: parsed.refusalStrainTargetLimitBp,
    effectiveRefusalThresholdLimitBp: effectiveLimitBp,
    isReductionActive,
    reductionAppliedBp,
  };

  return Object.freeze(receipt);
}
```

```typescript
import type { DecayModel } from './decay.model.js';
import type PressureMeterBit from './fce/pressureMeter.bit.js';
import type State from '../99.core/state.js';
import { calculateThermodynamicStep } from './buz/thermodynamicFlow.buzz.js';
import { evaluateSomaticRefusalBridge } from './buz/interMeterBridge.buzz.js';

const UPDATE_CONSOLE = '[Console action] Update Console';

export const initDecay = (cpy: DecayModel, bal: PressureMeterBit, _ste: State): DecayModel => {
  cpy.meters = {};
  cpy.cycleReceipts = {};
  if (bal?.slv) {
    bal.slv({ dcyBit: { idx: 'init-decay-ok', val: 1 } });
  }
  return cpy;
};

export const updateDecay = (cpy: DecayModel, bal: PressureMeterBit, _ste: State): DecayModel => {
  if (bal?.slv) {
    bal.slv({ dcyBit: { idx: 'update-decay-ok', val: 1 } });
  }
  return cpy;
};

export const stepThermodynamicCycle = async (
  cpy: DecayModel,
  bal: PressureMeterBit & { dat: any },
  ste: State,
): Promise<DecayModel> => {
  const payload = bal.dat;
  if (!payload || !payload.meter) {
    if (bal.slv) bal.slv({ dcyBit: { idx: 'step-thermodynamic-cycle-fail', val: -1 } });
    return cpy;
  }

  try {
    const receipt = calculateThermodynamicStep(payload.meter, payload.frictionDebits || [], payload.tick || 0);

    cpy.meters[receipt.meterId] = {
      ...payload.meter,
      currentValueBp: receipt.currentValueBp,
    };
    cpy.cycleReceipts[receipt.meterId] = receipt;

    if (ste) {
      await ste.hunt(UPDATE_CONSOLE, {
        idx: 'cns00',
        src: `>> [THERMODYNAMIC_FLOW] ${receipt.category}:${receipt.meterId} | ${receipt.previousValueBp} -> ${receipt.currentValueBp} bps | Dissipated: ${receipt.dissipatedBp} bps [OK]`,
      });
    }

    if (bal.slv) {
      bal.slv({
        dcyBit: {
          idx: 'step-thermodynamic-cycle-ok',
          val: receipt.currentValueBp,
          dat: receipt,
        },
      });
    }
  } catch (err: unknown) {
    const message = err instanceof Error ? err.message : String(err);
    if (ste) {
      await ste.hunt(UPDATE_CONSOLE, {
        idx: 'cns00',
        src: `:: [THERMODYNAMIC_FLOW FAIL] ${message}`,
      });
    }
    if (bal.slv) {
      bal.slv({
        dcyBit: {
          idx: 'step-thermodynamic-cycle-err',
          val: -1,
          src: message,
        },
      });
    }
  }

  return cpy;
};

export const evaluateInterMeterBridge = (
  cpy: DecayModel,
  bal: PressureMeterBit & { dat: any },
  _ste: State,
): DecayModel => {
  const payload = bal.dat;
  if (!payload) {
    if (bal.slv) bal.slv({ dcyBit: { idx: 'evaluate-inter-meter-bridge-fail', val: -1 } });
    return cpy;
  }

  try {
    const receipt = evaluateSomaticRefusalBridge(payload);
    if (bal.slv) {
      bal.slv({
        dcyBit: {
          idx: 'evaluate-inter-meter-bridge-ok',
          val: receipt.effectiveRefusalThresholdLimitBp,
          dat: receipt as any,
        },
      });
    }
  } catch (err: unknown) {
    const message = err instanceof Error ? err.message : String(err);
    if (bal.slv) {
      bal.slv({
        dcyBit: {
          idx: 'evaluate-inter-meter-bridge-err',
          val: -1,
          src: message,
        },
      });
    }
  }

  return cpy;
};
```

```typescript
import clone from 'clone-deep';
import * as Act from './decay.action.js';
import { DecayModel } from './decay.model.js';
import * as Buzz from './decay.buzzer.js';
import type State from '../99.core/state.js';

export function reducer(
  model: DecayModel = new DecayModel(),
  act: Act.Actions,
  state?: State,
): DecayModel {
  switch (act.type) {
    case Act.INIT_DECAY:
      return Buzz.initDecay(clone(model), act.bale, state!);

    case Act.UPDATE_DECAY:
      return Buzz.updateDecay(clone(model), act.bale, state!);

    case Act.STEP_THERMODYNAMIC_CYCLE:
      Buzz.stepThermodynamicCycle(clone(model), act.bale as any, state!);
      return model;

    case Act.EVALUATE_INTER_METER_BRIDGE:
      return Buzz.evaluateInterMeterBridge(clone(model), act.bale as any, state!);

    default:
      return model;
  }
}
```
