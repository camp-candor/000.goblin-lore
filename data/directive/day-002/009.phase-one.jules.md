## Context & Engineering Mandate
In the Generative Production Pipeline, physical embodiment and upper-limb manipulation capacity must be calculated with absolute mathematical determinism before any screenplay prose is drafted or speech synthesis is triggered[cite: 3, 15].

When a character suffers from severe physiological injuries—such as an amputated limb (`severed_left_arm`), a nerve transection (`braxial_plexus_severed`), or upper paralysis (`paralyzed_arm`)—each condition imposes an anatomical penalty[cite: 3, 5]. The **Compound Trauma Limb Clamper** enforces:
1. **Base Biological Allocation:**
   $$\text{Arms}_{\text{base}} = \begin{cases} 0 \text{ bps} & \text{if } \text{locomotion\_baseline} \in \{\texttt{"sessile"}, \texttt{"slithering"}\} \\ 20{,}000 \text{ bps} & \text{otherwise (standard bipedal/limping humanoid)} \end{cases}$$[cite: 5, 15]

   $$\text{SEVERE\_ARM\_CONDITIONS} = \{\texttt{"braxial\_plexus\_severed"}, \texttt{"severed\_right\_arm"}, \texttt{"severed\_left\_arm"}, \texttt{"paralyzed\_arm"}, \texttt{"amputated\_limb"}\}$$[cite: 5, 15]
   Each condition imposes a unit penalty of $10{,}000\text{ bps}$ ($1.0$ functional arm)[cite: 3, 5, 15].
3. **Compound Trauma Lower-Bound Clamping:**
   $$\text{availableArmsBps} = \max\left(0, \, \text{Arms}_{\text{base}} - \sum_{c \in \text{conditions}} 10{,}000 \cdot \mathbb{I}_{\text{severe}}(c)\right)$$[cite: 3, 5, 15]
4. **Discrete Scalar Derivation:**
   $$\text{armsAvailable} = \left\lfloor \frac{\text{availableArmsBps}}{10{,}000} \right\rfloor \in \{0, 1, 2\}$$[cite: 9, 15]
5. **Zero Float Divergence:** Rejects IEEE-754 floating-point scalars. All calculations operate strictly in integer basis points ($0 \dots 20{,}000\text{ bps}$)[cite: 3, 9, 16]. Non-integer basis points throw an immediate `[FLOAT_PROHIBITED]` error[cite: 9].

```typescript
  input: CompoundTraumaInput
): LimbClampingReceipt => {
  const parsed = CompoundTraumaInputSchema.parse(input);

  // 1. Determine base biological allocation in integer basis points
  let baseArmsBps = 20000; // 2.0 arms default
  if (parsed.locomotionBaseline === 'sessile' || parsed.locomotionBaseline === 'slithering') {
    baseArmsBps = 0;
  } else if (parsed.baseArmsBps !== undefined) {
    assertIntegerBps(parsed.baseArmsBps, 'baseArmsBps');
    baseArmsBps = parsed.baseArmsBps;
  }

  // 2. Audit active conditions case-insensitively
  const matchedSevereConditions: string[] = [];
  for (const rawCondition of parsed.conditions) {
    const normalized = rawCondition.trim().toLowerCase();
    if (SEVERE_ARM_CONDITIONS_SET.has(normalized)) {
      matchedSevereConditions.push(normalized);
    }
  }

  // 3. Subtract 10,000 bps (1.0 arm) per severe condition
  const penaltiesAppliedCount = matchedSevereConditions.length;
  const totalPenaltyBps = penaltiesAppliedCount * 10000;
  const rawCalculatedBps = baseArmsBps - totalPenaltyBps;

  // 4. Lower-bound clamp at 0 bps to prevent negative limb capacity
  const availableArmsBps = Math.max(0, rawCalculatedBps);
  const isClampedAtZero = rawCalculatedBps < 0;
  const armsAvailable = Math.floor(availableArmsBps / 10000);

  const receipt: LimbClampingReceipt = {
    entityId: parsed.entityId,
    locomotionBaseline: parsed.locomotionBaseline,
    baseArmsBps,
    penaltiesAppliedCount,
    totalPenaltyBps,
    rawCalculatedBps,
    availableArmsBps,
    armsAvailable,
    isClampedAtZero,
    matchedSevereConditions,
  };

  return LimbClampingReceiptSchema.parse(receipt);
};

export const initWound = (cpy: WoundModel, bal: WoundBit, _ste: State): WoundModel => {
  cpy.clampedReceipts = {};
  if (bal?.slv) bal.slv({ wndBit: { idx: 'init-wound', val: 0 } });
  return cpy;
};

export const updateWound = (cpy: WoundModel, bal: WoundBit, _ste: State): WoundModel => {
  if (bal?.slv) bal.slv({ wndBit: { idx: 'update-wound' } });
  return cpy;
};

export const clampCompoundTrauma = (
  cpy: WoundModel,
  bal: WoundBit<CompoundTraumaInput>,
  _ste: State
): WoundModel => {
  const payload = bal?.dat;
  if (!payload?.entityId) {
    if (bal?.slv) {
      bal.slv({
        wndBit: { idx: 'clamp-compound-trauma-err', src: 'MISSING_PAYLOAD', val: -1 },
      });
    }
    return cpy;
  }

  try {
    const receipt = calculateCompoundTraumaLimbClamping(payload);
    cpy.clampedReceipts[receipt.entityId] = receipt;

    if (bal?.slv) {
      bal.slv({
        wndBit: {
          idx: 'clamp-compound-trauma-ok',
          src: receipt.entityId,
          val: receipt.availableArmsBps,
          dat: receipt,
        },
      });
    }
  } catch (err: unknown) {
    const message = err instanceof Error ? err.message : String(err);
    if (bal?.slv) {
      bal.slv({
        wndBit: {
          idx: 'clamp-compound-trauma-fail',
          src: payload.entityId,
          val: -1,
          dat: { error: message },
        },
      });
    }
  }

  return cpy;
};
```

```typescript
import clone from 'clone-deep';
import * as Act from './wound.action';
import { WoundModel } from './wound.model';
import * as Buzz from './wound.buzzer';
import type State from '../99.core/state';

export function reducer(
  model: WoundModel = new WoundModel(),
  act: Act.Actions,
  state?: State
): WoundModel {
  switch (act.type) {
    case Act.INIT_WOUND:
      return Buzz.initWound(clone(model, true), act.bale, state!);

    case Act.UPDATE_WOUND:
      return Buzz.updateWound(clone(model, true), act.bale, state!);

    case Act.CLAMP_COMPOUND_TRAUMA:
      return Buzz.clampCompoundTrauma(clone(model, true), act.bale, state!);

    default:
      return model;
  }
}
```

```typescript
import { describe, it, expect } from 'vitest';
import {
  calculateCompoundTraumaLimbClamping,

import { WoundModel } from '../wound.model';

describe('40.wound.unit - Compound Trauma Limb Clamper Calculation', () => {
  it('allocates 20,000 bps (2.0 arms) for uninjured bipedal characters', () => {
    const result = calculateCompoundTraumaLimbClamping({
      entityId: 'char_bog',
      locomotionBaseline: 'bipedal_standard',
      conditions: [],
    });

    expect(result.baseArmsBps).toBe(20000);
    expect(result.penaltiesAppliedCount).toBe(0);
    expect(result.availableArmsBps).toBe(20000);
    expect(result.armsAvailable).toBe(2);
    expect(result.isClampedAtZero).toBe(false);
  });

  it('drops to 0 bps immediately for sessile or slithering baselines', () => {
    const sessileResult = calculateCompoundTraumaLimbClamping({
      entityId: 'char_barnacle',
      locomotionBaseline: 'sessile',
      conditions: [],
    });
    expect(sessileResult.availableArmsBps).toBe(0);
    expect(sessileResult.armsAvailable).toBe(0);

    const slitheringResult = calculateCompoundTraumaLimbClamping({
      entityId: 'char_worm',
      locomotionBaseline: 'slithering',
      conditions: [],
    });
    expect(slitheringResult.availableArmsBps).toBe(0);
    expect(slitheringResult.armsAvailable).toBe(0);
  });

  it('correctly deduces 10,000 bps for a single severe condition (Wart)', () => {
    const result = calculateCompoundTraumaLimbClamping({
      entityId: 'char_wart',
      locomotionBaseline: 'bipedal_hunched',
      conditions: ['severed_left_arm'],
    });

    expect(result.baseArmsBps).toBe(20000);
    expect(result.penaltiesAppliedCount).toBe(1);
    expect(result.totalPenaltyBps).toBe(10000);
    expect(result.availableArmsBps).toBe(10000);
    expect(result.armsAvailable).toBe(1);
    expect(result.isClampedAtZero).toBe(false);
    expect(result.matchedSevereConditions).toEqual(['severed_left_arm']);
  });

  it('matches severe conditions case-insensitively and ignores non-severe conditions', () => {
    const result = calculateCompoundTraumaLimbClamping({
      entityId: 'char_pilot',
      locomotionBaseline: 'limping',
      conditions: [
        'BRAXIAL_PLEXUS_SEVERED', // Severe
        'stiff_left_knee',        // Ignored
        'exhaustion_tier_1',      // Ignored
      ],
    });

    expect(result.penaltiesAppliedCount).toBe(1);
    expect(result.availableArmsBps).toBe(10000);
    expect(result.armsAvailable).toBe(1);
    expect(result.matchedSevereConditions).toEqual(['braxial_plexus_severed']);
  });

  it('clamps compound multiple severe traumas at 0 bps without negative underflow', () => {
    const compoundResult = calculateCompoundTraumaLimbClamping({
      entityId: 'char_mutilated_goblin',
      locomotionBaseline: 'bipedal_hunched',

        'paralyzed_arm',
        'amputated_limb', // 3 severe conditions = 30,000 bps penalty
      ],
    });

    expect(result => result).toBeDefined();
    expect(compoundResult.baseArmsBps).toBe(20000);
    expect(compoundResult.penaltiesAppliedCount).toBe(3);
    expect(compoundResult.totalPenaltyBps).toBe(30000);
    expect(compoundResult.rawCalculatedBps).toBe(-10000);
    expect(compoundResult.availableArmsBps).toBe(0); // Clamped!
    expect(compoundResult.armsAvailable).toBe(0);
    expect(compoundResult.isClampedAtZero).toBe(true);
  });

    expect(model.clampedReceipts['char_wart']).toBeDefined();
    expect(model.clampedReceipts['char_wart'].isClampedAtZero).toBe(false); // raw was exactly 0
  });
```
