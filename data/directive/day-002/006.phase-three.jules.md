  bal: LocomotionBit,
  ste: State

  const candidate = bal.dat?.baseline;

  try {
    if (!candidate || typeof candidate !== 'string') {
      throw new Error('[LOCOMOTION_BASELINE_REQUIRED]: Candidate baseline string must be provided in bal.dat.baseline.');
    }

    const profile = validateLocomotionBaseline(candidate);

    // Cache the validated profile in model state
    cpy.locomotionBits[profile.baseline] = profile;
    cpy.locomotionBitList.push({
      idx: bal.idx || `loc_${Date.now()}`,
      val: 1,
      dat: { baseline: profile.baseline, profile, isValid: true }
    });

    await logConsole(ste, `>> [LOCOMOTION UNIT] Baseline validated: ${profile.baseline}`);
    await logConsole(ste, `>> [VELOCITY CAP] Max velocity: ${profile.max_velocity_bps} bps`);
    await logConsole(ste, `>> [NEGATIVE MASK] Banned verbs: ${JSON.stringify(profile.default_banned_verbs)}`);
    await logConsole(ste, '>> [STATUS] Stage T_0 Physical Bounds Locked [OK]');

        locomotionBit: {
          idx: bal.idx,

            baseline: profile.baseline,

            isValid: true
          }
        }

          val: 0,
          dat: {
            baseline: candidate,
            isValid: false,
            error: errorMsg
          }
        }
      });
    }
  }

  return cpy;
};

  }
}
