export interface DialogueNodeCandidate {
  type: 'DIALOGUE' | 'BEAT' | 'ACTION';
  speakerId?: string;
  text?: string;

export function assertDialogueRhythmRatio(
  sequence: readonly DialogueNodeCandidate[],
  maxConsecutiveDialogueLines: number = 2,
): { valid: boolean; consecutiveDialogueCount: number; violationIndex?: number } {
  let consecutiveCount = 0;

  for (let i = 0; i < sequence.length; i++) {
    const node = sequence[i];
    if (node.type === 'DIALOGUE') {
      consecutiveCount++;
      if (consecutiveCount > 3) {
        throw new Error(
          `RHYTHM_BREACH: Exceeded maximum hard limit of 3 consecutive dialogue lines without action or beat at index ${i}`,
        );
      }
      if (consecutiveCount > maxConsecutiveDialogueLines) {
        return {
          valid: false,
          consecutiveDialogueCount: consecutiveCount,
          violationIndex: i,

    } else if (node.type === 'BEAT' || node.type === 'ACTION') {

import type { PayloadModel } from './payload.model.js';
import type PayloadBit from './fce/ttsPayload.bit.js';
import type State from '../99.core/state.js';
import { validateTtsGenerationPayload } from './buz/validateTtsPayload.buzz.js';

const UPDATE_CONSOLE = '[Console action] Update Console';

export const initPayload = (cpy: PayloadModel, bal: PayloadBit, _ste: State): PayloadModel => {
  cpy.validatedPayloads = {};
  cpy.validationReceipts = {};
  cpy.totalValidatedCount = 0;
  if (bal?.slv) {
    bal.slv({ pldBit: { idx: 'init-payload-ok', val: 1 } });
  }
  return cpy;
};

export const updatePayload = (cpy: PayloadModel, bal: PayloadBit, _ste: State): PayloadModel => {
  if (bal?.slv) {
    bal.slv({ pldBit: { idx: 'update-payload-ok', val: 1 } });
  }
  return cpy;
};

export const validateTtsPayload = async (
  cpy: PayloadModel,
  bal: PayloadBit & { dat: any },
  ste: State,
): Promise<PayloadModel> => {
  const payload = bal.dat?.candidatePayload || bal.dat;
  if (!payload) {
    if (bal.slv) bal.slv({ pldBit: { idx: 'validate-tts-payload-fail', val: -1 } });
    return cpy;
  }

  try {
    const { request, receipt } = validateTtsGenerationPayload(payload);

    cpy.validatedPayloads[request.lineId] = request;
    cpy.validationReceipts[request.lineId] = receipt;
    cpy.totalValidatedCount++;

    if (ste) {
      await ste.hunt(UPDATE_CONSOLE, {
        idx: 'cns00',
        src: `>> [PAYLOAD] Validated Ingress Request ${request.lineId} (${request.characterId}) | Facts: ${receipt.pinnedFactCount} [OK]`,
      });
    }

    if (bal.slv) {
      bal.slv({
        pldBit: {
          idx: 'validate-tts-payload-ok',
          val: 1,
          dat: receipt,
        },
      });
    }
  } catch (err: unknown) {
    const message = err instanceof Error ? err.message : String(err);
    if (ste) {
      await ste.hunt(UPDATE_CONSOLE, {
        idx: 'cns00',
        src: `:: [PAYLOAD FAIL] Schema breach: ${message}`,
      });
    }
    if (bal.slv) {
      bal.slv({
        pldBit: {
          idx: 'validate-tts-payload-err',
          val: -1,
          src: message,
        },
      });
    }
  }

  return cpy;
};

import clone from 'clone-deep';
import * as Act from './payload.action.js';
import { PayloadModel } from './payload.model.js';
import * as Buzz from './payload.buzzer.js';
import type State from '../99.core/state.js';

export function reducer(
  model: PayloadModel = new PayloadModel(),
  act: Act.Actions,
  state?: State,
): PayloadModel {
  switch (act.type) {
    case Act.INIT_PAYLOAD:
      return Buzz.initPayload(clone(model), act.bale, state!);

    case Act.UPDATE_PAYLOAD:
      return Buzz.updatePayload(clone(model), act.bale, state!);

    case Act.VALIDATE_TTS_PAYLOAD:
      Buzz.validateTtsPayload(clone(model), act.bale as any, state!);
      return model;

    default:
      return model;
  }
}
