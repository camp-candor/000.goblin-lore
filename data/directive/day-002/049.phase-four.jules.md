  const lemmasActual = extractSignificantLemmas(actualWant);
  const overlapBps = computeLemmaOverlapBps(lemmasSurface, lemmasActual);

  const isIdentical = surfaceWant.toLowerCase() === actualWant.toLowerCase();
  const contrastPass = !isIdentical && overlapBps <= 6000;

  let errorReason: string | undefined = undefined;
  if (!cleanOfArtifacts) {
    errorReason = 'LEAKAGE_BREACH: Unedited HTML comments or citation tags detected in voice rail';
  } else if (!hasAllSections) {
    errorReason = 'SECTION_MISSING_BREACH: One or more required Voice Rail sections are absent or empty';
  } else if (isIdentical) {
    errorReason = 'SUBTEXT_IDENTITY_BREACH: Surface want is identical to actual want';
  } else if (!contrastPass) {
    errorReason = `SUBTEXT_CONTRAST_BREACH: Lemma overlap of ${overlapBps} bps exceeds ceiling of 6000 bps (60.0%)`;
  }

  const isValid = cleanOfArtifacts && hasAllSections && contrastPass;

  const result: WantDifferentiationResult = {
    characterId,
    hasAllSections,
    surfaceWant,
    actualWant,
    blindSpot,
    voiceNotes,
    fewShotLineCount: fewShotLines.length,
    overlapPercentageBps: overlapBps,
    contrastPass,
    cleanOfArtifacts,
    isValid,
    errorReason
  };

  if (bal.slv != null) {
    bal.slv({
      vceBit: {
        idx: 'differentiate-wants',
        val: isValid ? 1 : 0,
        dat: result
      }
    });
  }

  return cpy;
};

export const validateComedicRegister = (cpy: VoiceModel, bal: VoiceBit, ste: State): VoiceModel => {
  const characterId = String(bal.idx ?? 'char_unknown');
  const candidateRegister = bal.dat?.register ?? bal.src;
  const sampleProse = String(bal.dat?.prose ?? '');

  const parseResult = ComedicRegisterEnum.safeParse(candidateRegister);
  if (!parseResult.success) {
    const failure: RegisterValidationResult = {
      characterId,
      register: 'deadpan',
      instructionDirective: '',
      detectedSlopTokens: [],
      isValid: false,
      errorCode: 'INVALID_COMEDIC_REGISTER_BREACH',
      message: `Invalid comedic register '${String(candidateRegister)}'. Must match ComedicRegisterEnum.`
    };
    if (bal.slv != null) {
      bal.slv({ vceBit: { idx: 'validate-comedic-register', val: 0, dat: failure } });
    }
    return cpy;
  }

  const validRegister: ComedicRegister = parseResult.data;
  const definition = COMEDIC_REGISTER_REGISTRY[validRegister];

  const lowerProse = sampleProse.toLowerCase();
  const detectedSlopTokens: string[] = [];

  for (const token of BANNED_LATINATE_LEMMAS) {
    const wordBoundaryRegex = new RegExp(`\\b${token}\\b`, 'i');
    if (wordBoundaryRegex.test(lowerProse)) {
      detectedSlopTokens.push(token);
    }
  }

  const hasSlop = detectedSlopTokens.length > 0;
  const isValid = !hasSlop;

  const result: RegisterValidationResult = {
    characterId,
    register: validRegister,
    instructionDirective: definition.instructionDirective,
    detectedSlopTokens,
    isValid,
    errorCode: hasSlop ? 'BANNED_LATINATE_SLOP_BREACH' : undefined,
    message: hasSlop
      ? `Prohibited Latinate slop tokens detected in voice rail: [${detectedSlopTokens.join(', ')}]`
      : `Comedic register '${validRegister}' validated with zero Latinate slop tokens.`
  };

  if (bal.slv != null) {
    bal.slv({
      vceBit: {
        idx: 'validate-comedic-register',
        val: isValid ? 1 : 0,
        dat: result
      }
    });
  }

  return cpy;
};

/**
 * Validates character risk profile schema conformance and cross-field integrity rules.
 */
export const validateRiskProfile = (cpy: VoiceModel, bal: VoiceBit, ste: State): VoiceModel => {
  const characterId = String(bal.idx ?? 'char_unknown');
  const payload = bal.dat ?? {};

  const parseResult = CharacterRiskProfileSchema.safeParse({
    characterId,
    ...payload
  });

  if (!parseResult.success) {
    const issue = parseResult.error.issues[0];
    const failure: RiskProfileValidationResult = {
      characterId,
      isValid: false,
      isExempt: Boolean(payload.isExempt),
      errorCode: issue?.message?.split(':')[0] ?? 'RISK_PROFILE_VALIDATION_ERROR',
      message: issue?.message ?? 'Invalid risk profile payload'
    };

    if (bal.slv != null) {
      bal.slv({
        vceBit: {
          idx: 'validate-risk-profile',
          val: 0,
          dat: failure
        }
      });
    }
    return cpy;
  }

  const validatedData: CharacterRiskProfile = parseResult.data;
  const result: RiskProfileValidationResult = {
    characterId,
    isValid: true,
    isExempt: validatedData.isExempt,
    message: validatedData.isExempt
      ? `Character '${characterId}' is certified exempt from risk profile mechanics.`
      : `Risk profile for '${characterId}' successfully validated and sealed at T0.`,
    parsedProfile: validatedData
  };

  if (bal.slv != null) {
    bal.slv({
      vceBit: {
        idx: 'validate-risk-profile',
        val: 1,
        dat: result
      }
    });
  }

  return cpy;
};
