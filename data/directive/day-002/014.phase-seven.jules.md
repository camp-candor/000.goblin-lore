```typescript
import {
  VocalFeasibilityCandidateSchema,
  VocalFeasibilityReceiptSchema,
  CANONICAL_APHONIC_CONDITIONS,
  BANNED_SPEAK_VERBS,
  type VocalFeasibilityCandidate,
  type VocalFeasibilityReceipt,
} from '../val/aphoniaGate.val.js';

export function assertVocalMotorFeasibility(
  candidateInput: VocalFeasibilityCandidate,
  epoch: number = 0,
): VocalFeasibilityReceipt {
  const candidate = VocalFeasibilityCandidateSchema.parse(candidateInput);

  const normalizedConditions = candidate.activeConditions.map((c) => c.trim().toLowerCase());
  const normalizedVerbs = candidate.bannedKineticVerbs.map((v) => v.trim().toLowerCase());

  const matchedAphonicConditions = normalizedConditions.filter((c) =>
    CANONICAL_APHONIC_CONDITIONS.some((aphonic) => aphonic === c || c.includes(aphonic)),
  );

  const matchedBannedVerbs = normalizedVerbs.filter((v) =>
    BANNED_SPEAK_VERBS.some((banned) => banned === v),
  );

  const isAphonic = matchedAphonicConditions.length > 0 || matchedBannedVerbs.length > 0;
  const hasVerbalDirectives =
    candidate.verbatimText.trim().length > 0 && !candidate.isNonVerbalExpression;

  if (isAphonic && hasVerbalDirectives) {
    const reasons: string[] = [];
    if (matchedAphonicConditions.length > 0) {
      reasons.push(`CONDITIONS:[${matchedAphonicConditions.join(', ')}]`);
    }
    if (matchedBannedVerbs.length > 0) {
      reasons.push(`BANNED_VERBS:[${matchedBannedVerbs.join(', ')}]`);
    }

    const breachMessage = `VOCAL_DELIVERY_BREACH: Character [${candidate.entityId}] is physically aphonic (${reasons.join(' | ')}) but verbal text was directed`;
    throw new Error(breachMessage);
  }

  const receipt: VocalFeasibilityReceipt = {
    entityId: candidate.entityId,
    isPhonationFeasible: !isAphonic || !hasVerbalDirectives,
    isAphonic,
    matchedAphonicConditions: Object.freeze(matchedAphonicConditions.sort()),
    matchedBannedVerbs: Object.freeze(matchedBannedVerbs.sort()),
    hasVerbalDirectives,
    status: 'VALID',
    verifiedAtEpoch: epoch,
  };

  return Object.freeze(VocalFeasibilityReceiptSchema.parse(receipt));
}

```

```typescript
export const initPayload = (cpy: PayloadModel, bal: PayloadBit, _ste: State): PayloadModel => {

export const validateTtsPayload = async (
  cpy: PayloadModel,
  bal: PayloadBit & { dat: any },
  ste: State,
): Promise<PayloadModel> => {
  const payload = bal.dat?.candidatePayload || bal.dat;
  if (!payload) {
    if (bal.slv) bal.slv({ pldBit: { idx: 'validate-tts-payload-fail', val: -1 } });
    return cpy;
  }

  try {
    const { request, receipt } = validateTtsGenerationPayload(payload);

    cpy.validatedPayloads[request.lineId] = request;
    cpy.validationReceipts[request.lineId] = receipt;
    cpy.totalValidatedCount++;

    if (ste) {
      await ste.hunt(UPDATE_CONSOLE, {
        idx: 'cns00',
        src: `>> [PAYLOAD] Validated Ingress Request ${request.lineId} (${request.characterId}) | Facts: ${receipt.pinnedFactCount} [OK]`,
      });
    }

    if (bal.slv) {
      bal.slv({
        pldBit: {
          idx: 'validate-tts-payload-ok',
          val: 1,
          dat: receipt,
        },
      });
    }
  } catch (err: unknown) {
    const message = err instanceof Error ? err.message : String(err);
    if (ste) {
      await ste.hunt(UPDATE_CONSOLE, {
        idx: 'cns00',
        src: `:: [PAYLOAD FAIL] Schema breach: ${message}`,
      });
    }
    if (bal.slv) {
      bal.slv({
        pldBit: {
          idx: 'validate-tts-payload-err',
          val: -1,
          src: message,
        },
      });
    }
  }

  return cpy;
};

export const verifyVocalFeasibility = async (
  cpy: PayloadModel,
  bal: AphoniaGateBit & { dat: any },

  if (!payload || !payload.entityId) {

      });
    }

    if (bal.slv) {

        aphBit: {
          idx: 'verify-vocal-feasibility-ok',
          val: receipt.isPhonationFeasible ? 1 : 0,
          dat: receipt,
        },
      });

          idx: 'verify-vocal-feasibility-err',
          val: -1,
          src: message,
        },
      });
    }
  }

  return cpy;
};
```

```typescript
import type State from '../99.core/state.js';

export function reducer(
  model: PayloadModel = new PayloadModel(),
  act: Act.Actions,
  state?: State,
): PayloadModel {
  switch (act.type) {
    case Act.INIT_PAYLOAD:
      return Buzz.initPayload(clone(model), act.bale, state!);

    case Act.UPDATE_PAYLOAD:
      return Buzz.updatePayload(clone(model), act.bale, state!);

    case Act.VALIDATE_TTS_PAYLOAD:

      return model;

    case Act.VERIFY_VOCAL_FEASIBILITY:
      Buzz.verifyVocalFeasibility(clone(model), act.bale as any, state!);
      return model;

    default:
      return model;
  }
}
```
