```typescript
  entityId: z.string().regex(/^char_[a-z0-9_]+$/),
  conditions: z.array(z.string()).default([]),
  explicitOpticalState: OpticalStateEnum.optional(),
  explicitBlindConeDegrees: z.number().int().min(0).max(360).optional(),
  explicitVisualAcuityBps: z.number().int().min(0).max(10000).optional(),
  explicitAuditoryDebuffCdB: z.number().int().min(0).max(10000).optional(),
}).strict();

export type SensoryAcuityInput = z.infer<typeof SensoryAcuityInputSchema>;

export const SensoryAcuityProfileSchema = z.object({
  entityId: z.string().regex(/^char_[a-z0-9_]+$/),
  opticalState: OpticalStateEnum,
  blindConeDegrees: z.number().int().min(0).max(360),
  visualAcuityBps: z.number().int().min(0).max(10000),
  auditoryFloorDebuffCdB: z.number().int().min(0).max(10000),
  hasAcousticTrauma: z.boolean(),
  matchedOcularConditions: z.array(z.string()),
  matchedAcousticConditions: z.array(z.string()),
}).strict().superRefine((val, ctx) => {
  if (val.opticalState === 'MONOCULAR' && val.blindConeDegrees < 60) {
    ctx.addIssue({
      code: z.ZodIssueCode.custom,
      message: '[SENSORY_SCHEMA_BREACH] Monocular vision requires blindConeDegrees >= 60',
      path: ['blindConeDegrees'],
    });
  }
  if (val.opticalState === 'BLIND' && (val.visualAcuityBps !== 0 || val.blindConeDegrees !== 360)) {
    ctx.addIssue({
      code: z.ZodIssueCode.custom,
      message: '[SENSORY_SCHEMA_BREACH] Blind vision requires visualAcuityBps = 0 and blindConeDegrees = 360',
      path: ['visualAcuityBps'],
    });
  }
});

export type SensoryAcuityProfile = z.infer<typeof SensoryAcuityProfileSchema>;
```

```typescript
import {
  SensoryAcuityInputSchema,
  SensoryAcuityProfileSchema,
  OCULAR_CONDITIONS_CATALOG,
  ACOUSTIC_TRAUMA_CONDITIONS,
  type SensoryAcuityInput,
  type SensoryAcuityProfile,
  type OpticalState,
} from '../val/sensory.val';

const BLIND_SET = new Set<string>(OCULAR_CONDITIONS_CATALOG.blind.map((c) => c.toLowerCase()));
const MONOCULAR_SET = new Set<string>(OCULAR_CONDITIONS_CATALOG.monocular.map((c) => c.toLowerCase()));
const ACOUSTIC_SET = new Set<string>(ACOUSTIC_TRAUMA_CONDITIONS.map((c) => c.toLowerCase()));

export const assertIntegerMetric = (val: number, label: string): void => {
  if (!Number.isInteger(val)) {
    throw new Error(`[FLOAT_PROHIBITED] ${label} must be integer scalar. Received: ${val}`);
  }
};

/**
 * Derives the deterministic sensory acuity profile from an actor's conditions.
 */
export const calculateSensoryAcuityProfile = (
  input: SensoryAcuityInput
): SensoryAcuityProfile => {
  const parsed = SensoryAcuityInputSchema.parse(input);

  // Validate integer metrics if explicitly supplied
  if (parsed.explicitBlindConeDegrees !== undefined) {
    assertIntegerMetric(parsed.explicitBlindConeDegrees, 'explicitBlindConeDegrees');
  }
  if (parsed.explicitVisualAcuityBps !== undefined) {
    assertIntegerMetric(parsed.explicitVisualAcuityBps, 'explicitVisualAcuityBps');
  }
  if (parsed.explicitAuditoryDebuffCdB !== undefined) {
    assertIntegerMetric(parsed.explicitAuditoryDebuffCdB, 'explicitAuditoryDebuffCdB');
  }

  const matchedOcularConditions: string[] = [];
  const matchedAcousticConditions: string[] = [];

  let isBlind = false;
  let isMonocular = false;

  for (const rawCond of parsed.conditions) {
    const normalized = rawCond.trim().toLowerCase();
    if (BLIND_SET.has(normalized)) {
      matchedOcularConditions.push(normalized);
      isBlind = true;
    } else if (MONOCULAR_SET.has(normalized)) {
      matchedOcularConditions.push(normalized);
      isMonocular = true;
    }

    if (ACOUSTIC_SET.has(normalized)) {
      matchedAcousticConditions.push(normalized);
    }
  }

  // Determine Optical State
  let opticalState: OpticalState = 'NORMAL';
  if (parsed.explicitOpticalState) {
    opticalState = parsed.explicitOpticalState;
  } else if (isBlind) {
    opticalState = 'BLIND';
  } else if (isMonocular) {
    opticalState = 'MONOCULAR';
  }

  // Derive Blind Cone Degrees & Visual Acuity Basis Points
  let blindConeDegrees = 0;
  let visualAcuityBps = 10000; // 100.00% standard

  if (opticalState === 'BLIND') {
    blindConeDegrees = 360;
    visualAcuityBps = 0;
  } else if (opticalState === 'MONOCULAR') {
    blindConeDegrees = parsed.explicitBlindConeDegrees ?? 60;
    visualAcuityBps = parsed.explicitVisualAcuityBps ?? 5000; // 50.00% acuity
  } else {
    blindConeDegrees = parsed.explicitBlindConeDegrees ?? 0;
    visualAcuityBps = parsed.explicitVisualAcuityBps ?? 10000;
  }

  // Derive Auditory Debuff in Centi-Decibels (+1,500 cdB per condition)
  let auditoryFloorDebuffCdB = 0;
  if (parsed.explicitAuditoryDebuffCdB !== undefined) {
    auditoryFloorDebuffCdB = parsed.explicitAuditoryDebuffCdB;
  } else {
    const penaltyUnits = matchedAcousticConditions.length;
    auditoryFloorDebuffCdB = Math.min(10000, penaltyUnits * 1500);
  }

  const hasAcousticTrauma = auditoryFloorDebuffCdB > 0;

  const profile: SensoryAcuityProfile = {
    entityId: parsed.entityId,
    opticalState,
    blindConeDegrees,
    visualAcuityBps,
    auditoryFloorDebuffCdB,
    hasAcousticTrauma,
    matchedOcularConditions,
    matchedAcousticConditions,
  };

  return SensoryAcuityProfileSchema.parse(profile);
};
```

```typescript
import type { TraumaModel } from '../trauma.model';
import type TraumaBit from '../fce/trauma.bit';
import type State from '../../99.core/state';
import {
  calculateSensoryAcuityProfile,
} from './sensory.buzz';
import type { SensoryAcuityInput, SensoryAcuityProfile } from '../val/sensory.val';

export const initTrauma = (cpy: TraumaModel, bal: TraumaBit, _ste: State): TraumaModel => {
  cpy.sensoryProfiles = {};
  if (bal?.slv) bal.slv({ trmBit: { idx: 'init-trauma', val: 0 } });
  return cpy;
};

export const updateTrauma = (cpy: TraumaModel, bal: TraumaBit, _ste: State): TraumaModel => {
  if (bal?.slv) bal.slv({ trmBit: { idx: 'update-trauma' } });
  return cpy;
};

export const configureSensoryAcuity = (
  cpy: TraumaModel,
  bal: TraumaBit<SensoryAcuityInput, unknown>,
  _ste: State
): TraumaModel => {
  const payload = bal?.dat;
  if (!payload?.entityId) {
    if (bal?.slv) {
      bal.slv({
        trmBit: { idx: 'configure-sensory-acuity-err', src: 'MISSING_PAYLOAD', val: -1 },
      });
    }
    return cpy;
  }

  try {
    const profile = calculateSensoryAcuityProfile(payload);
    cpy.sensoryProfiles[profile.entityId] = profile;

    if (bal?.slv) {
      bal.slv({
        trmBit: {
          idx: 'configure-sensory-acuity-ok',
          src: profile.entityId,
          val: profile.visualAcuityBps,
          dat: profile,
        },
      });
    }
  } catch (err: unknown) {
    const message = err instanceof Error ? err.message : String(err);
    if (bal?.slv) {
      bal.slv({
        trmBit: {
          idx: 'configure-sensory-acuity-fail',
          src: payload.entityId,
          val: -1,
          dat: { error: message },
        },
      });
    }
  }

  return cpy;
};
```

```typescript
  state?: State
): TraumaModel {
  switch (act.type) {

    case Act.CONFIGURE_SENSORY_ACUITY:

    default:
      return model;
  }
}
```
