/**
 * Standard stop words filtered during subtext lemma overlap analysis.
 */
export const STOP_WORDS = new Set([
  'a', 'an', 'and', 'are', 'as', 'at', 'be', 'by', 'for', 'from',
  'has', 'he', 'in', 'is', 'it', 'its', 'of', 'on', 'that', 'the',

]);

/**
 * Schema defining the five mandatory Voice Rail Markdown sections.
 */
export const VoiceRailSectionsSchema = z.object({
  surfaceWant: z.string().min(1, 'Surface want cannot be empty'),
  actualWant: z.string().min(1, 'Actual want cannot be empty'),
  blindSpot: z.string().min(1, 'Blind spot cannot be empty'),
  voiceNotes: z.string().min(1, 'Voice notes cannot be empty'),
  fewShotLines: z.array(z.string().min(1)).min(1, 'At least one few-shot line is required')
}).strict();

export type VoiceRailSections = z.infer<typeof VoiceRailSectionsSchema>;

/**
 * Verification result schema for want differentiation.
 */
export const WantDifferentiationResultSchema = z.object({
  characterId: z.string().regex(/^char_[a-z0-9_]+$/),
  hasAllSections: z.boolean(),
  surfaceWant: z.string(),
  actualWant: z.string(),
  blindSpot: z.string(),
  voiceNotes: z.string(),
  fewShotLineCount: z.number().int().nonnegative(),
  overlapPercentageBps: z.number().int().min(0).max(10000), // 0 to 10000 bps (0.00% to 100.00%)
  contrastPass: z.boolean(),
  cleanOfArtifacts: z.boolean(),
  isValid: z.boolean(),
  errorReason: z.string().optional()
}).strict();

export type WantDifferentiationResult = z.infer<typeof WantDifferentiationResultSchema>;

/**
 * Tokenizes text into lowercase 7-bit ASCII lemma words excluding stop words.
 */
export function extractSignificantLemmas(text: string): Set<string> {
  const clean = text.toLowerCase().replace(/[^a-z0-9\s]/g, ' ');
  const tokens = clean.split(/\s+/).filter(t => t.length > 2 && !STOP_WORDS.has(t));
  return new Set(tokens);
}

/**
 * Calculates Jaccard token overlap in integer basis points (0 to 10000).
 */
export function computeLemmaOverlapBps(setA: Set<string>, setB: Set<string>): number {
  if (setA.size === 0 || setB.size === 0) return 0;
  let intersectionCount = 0;
  for (const item of setA) {
    if (setB.has(item)) intersectionCount++;
  }
  const unionSize = new Set([...setA, ...setB]).size;
  if (unionSize === 0) return 0;
  return Math.round((intersectionCount / unionSize) * 10000);
}

export const differentiateWants = (cpy: VoiceModel, bal: VoiceBit, ste: State): VoiceModel => {
  const markdown = String(bal.src ?? '');
  const characterId = String(bal.idx ?? 'char_unknown');

  // Check for leaked citation markers or unedited HTML comment placeholders
  const hasCitationLeak = /\[cite:[^\]]*\]/i.test(markdown);

  const cleanOfArtifacts = !hasCitationLeak && !hasCommentLeak;

  // Extract individual markdown section texts
  const extractSection = (heading: string, nextHeadingRegex: RegExp): string => {
    const idx = markdown.indexOf(heading);
    if (idx === -1) return '';
    const slice = markdown.slice(idx + heading.length);
    const match = slice.search(nextHeadingRegex);
    const body = match === -1 ? slice : slice.slice(0, match);
    return body.trim();
  };

  const actualWant = extractSection('## Wants (actual)', /\n##\s+/);

    .split('\n')
    .map(line => line.replace(/^[-*]\s*/, '').trim())

    actualWant.length > 0 &&
    blindSpot.length > 0 &&
    voiceNotes.length > 0 &&

        dat: result
      }
    });

};

import { VoiceModel } from '../voice.model';

describe('Dual-Objective Surface Versus Actual Want Differentiator', () => {
  const validMarkdown = `
## Wants (surface)
Replace the rusted copper boiler gasket before the second work shift begins.

## Wants (actual)
Prevent Mum Grissel from discovering that he cannot read the pressure gauges.

## Blind spot
Believes absolute adherence to protocol shields him from eviction.

## Voice notes
Staccato cadence, defensive redirects toward mechanical hardware.

## Few-shot lines
- "The gasket threading is stripped, not me."
- "Read it yourself if you don't trust the copper."
`;

  it('passes when surface and actual wants have divergent subtext and valid sections', () => {
    let response: any = null;
    const model = new VoiceModel();
    differentiateWants(model, {
      idx: 'char_spleen',
      src: validMarkdown,
      slv: (res: any) => { response = res; }
    }, {} as any);

    expect(response).not.toBeNull();
    expect(response.vceBit.val).toBe(1);
    expect(response.vceBit.dat.isValid).toBe(true);
    expect(response.vceBit.dat.overlapPercentageBps).toBeLessThan(6000);
    expect(response.vceBit.dat.contrastPass).toBe(true);
  });

  it('fails when surface want and actual want are identical', () => {
    const identicalWantsMarkdown = validMarkdown.replace(
      'Prevent Mum Grissel from discovering that he cannot read the pressure gauges.',
      'Replace the rusted copper boiler gasket before the second work shift begins.'
    );

    let response: any = null;
    differentiateWants(new VoiceModel(), {
      idx: 'char_spleen',
      src: identicalWantsMarkdown,
      slv: (res: any) => { response = res; }
    }, {} as any);

    expect(response.vceBit.val).toBe(0);
    expect(response.vceBit.dat.isValid).toBe(false);
    expect(response.vceBit.dat.errorReason).toContain('SUBTEXT_IDENTITY_BREACH');
  });

  it('fails when lemma overlap exceeds 60.0% (6000 bps)', () => {
    const highOverlapMarkdown = `
## Wants (surface)
Fix the rusted pipe valve and save the fuel rations today.

## Wants (actual)
Fix the rusted pipe valve so we keep the fuel rations today.

## Blind spot
Believes safety is guaranteed.

## Voice notes
Monosyllabic.

## Few-shot lines
- "Touch the valve."
`;

    let response: any = null;
    differentiateWants(new VoiceModel(), {
      idx: 'char_bog',
      src: highOverlapMarkdown,
      slv: (res: any) => { response = res; }
    }, {} as any);

    expect(response.vceBit.val).toBe(0);
    expect(response.vceBit.dat.contrastPass).toBe(false);
    expect(response.vceBit.dat.errorReason).toContain('SUBTEXT_CONTRAST_BREACH');
  });

  it('fails when citation tags or unedited HTML comments are present', () => {
    const leakedMarkdown = validMarkdown + '\n[cite: 1]';

    let response: any = null;
    differentiateWants(new VoiceModel(), {
      idx: 'char_spleen',
      src: leakedMarkdown,
      slv: (res: any) => { response = res; }
    }, {} as any);

    expect(response.vceBit.val).toBe(0);
    expect(response.vceBit.dat.cleanOfArtifacts).toBe(false);
    expect(response.vceBit.dat.errorReason).toContain('LEAKAGE_BREACH');
  });
});
