   - `conditionsRemoved`: array of string condition tags to heal/cure.
   - `deltaFatigueBps`: signed integer basis points ($-10000 \dots +10000$).
   - `deltaPainBps`: signed integer basis points ($-10000 \dots +10000$).
   - `deltaRefusalStrainBps`: signed integer basis points ($-10000 \dots +10000$).
   - `deltaLegs`: signed integer ($-2 \dots +2$).
   - `motorLimitationsAdded`: array of uppercase tags (`['NO_SPRINT', 'IMPAIRED_FINE_MOTOR', 'UNSTEADY_GAIT']`).
   - `motorLimitationsRemoved`: array of uppercase tags.
   - `signatureTicsAdded`: array of 7-bit ASCII strings.
   - `signatureTicsRemoved`: array of 7-bit ASCII strings.
   - `tick`: integer non-negative tick counter.

2. **Unified State Roll-Up & Cross-Unit Resolution:**
   - Unifies active conditions, motor limitations, and signature tics via deterministic deduplication and lexicographical sorting.
   - Computes clamped strain meters (`fatigueBps`, `painBps`, `refusalStrainBps`) within $[0, 10000]$.
   - Sets `isRefusalStrained: true` whenever `refusalStrainBps >= 8000` (threshold breached).
   - Derives `locomotionTier`:
     * `legsAvailable <= 0` or `painBps >= 9000` $\to$ `'IMMOBILIZED'`.
     * `legsAvailable === 1` or `painBps >= 5000` or motor limitations include `'UNSTEADY_GAIT'` or `'LIMP'` $\to$ `'LIMPING'`.
     * Otherwise $\to$ `'UNRESTRICTED'`.
   - Integrates with `40.wound.unit/buz/wound.buzz.ts` by invoking `calculateCompoundTraumaLimbClamping` to calculate updated `availableArmsBps` and `armsAvailable`.
   - Integrates with `37.trauma.unit/buz/sensory.buzz.ts` by invoking `calculateSensoryAcuityProfile` to derive `opticalState`, `blindConeDegrees`, `visualAcuityBps`, and `auditoryFloorDebuffCdB`.
   - Integrates with `37.trauma.unit/buz/vocalDsp.buzz.ts` by invoking `calculateVocalDspProfile` to derive `pitchShiftCentitones`, `formantWarpingBps`, `breathinessPauseMultiplierBps`, and `bannedCadences`.

3. **Blessed Unit Model & Reducer Integration:**
   - Add action `RESOLVE_SOMATIC_DELTA = '[Trauma action] Resolve Somatic Delta'` in `trauma.action.ts`.
   - Update `TraumaModel` in `trauma.model.ts` to store `somaticStates: Record<string, SomaticTraumaReceipt>`.
   - Implement `resolveSomaticDelta` in `buz/trauma.buzz.ts` and dispatch through `trauma.reduce.ts`.

export const assertIntegerMetric = (val: number, label: string): void => {
  if (!Number.isInteger(val)) {
    throw new Error(`[FLOAT_PROHIBITED] ${label} must be integer scalar. Received: ${val}`);
  }
};

export const clampBps = (val: number, label: string): number => {
  assertIntegerMetric(val, label);
  return Math.max(0, Math.min(10000, val));
};

export const deriveLocomotionTier = (
  legsAvailable: number,
  painBps: number,
  motorLimitations: readonly string[]
): LocomotionTier => {
  if (legsAvailable <= 0 || painBps >= 9000) return 'IMMOBILIZED';
  if (
    legsAvailable === 1 ||
    painBps >= 5000 ||
    motorLimitations.includes('UNSTEADY_GAIT') ||
    motorLimitations.includes('LIMP')
  ) {
    return 'LIMPING';
  }
  return 'UNRESTRICTED';
};

export const createDefaultSomaticReceipt = (
  entityId: string,
  locomotionBaseline = 'bipedal_standard' as const
): SomaticTraumaReceipt => {
  const limbReceipt = calculateCompoundTraumaLimbClamping({
    entityId,
    locomotionBaseline,
    conditions: [],
  });

  const sensoryProfile = calculateSensoryAcuityProfile({
    entityId,
    conditions: [],
  });

  const vocalDspProfile = calculateVocalDspProfile({
    entityId,
    conditions: [],
  });

  return {
    entityId,
    armsAvailable: limbReceipt.armsAvailable,
    availableArmsBps: limbReceipt.availableArmsBps,
    legsAvailable: 2,
    locomotionTier: 'UNRESTRICTED',
    fatigueBps: 0,
    painBps: 0,
    refusalStrainBps: 0,
    isRefusalStrained: false,
    activeConditions: [],
    motorLimitations: [],
    signatureTics: [],
    limbReceipt,
    sensoryProfile,
    vocalDspProfile,
    updatedAtTick: 0,
  };
};

/**
 * Resolves a character's somatic trauma delta by computing compound limb capacity,
 * sensory acuity debuffs, vocal DSP modifiers, and integer strain meters.
 */
export const calculateSomaticDeltaResolution = (
  current: SomaticTraumaReceipt,
  delta: SomaticDeltaInput
): SomaticTraumaReceipt => {
  const parsed = SomaticDeltaInputSchema.parse(delta);

  // 1. Reconcile active conditions
  const conditionsSet = new Set<string>(
    current.activeConditions.map((c) => c.toLowerCase())
  );
  for (const c of parsed.conditionsAdded) {
    conditionsSet.add(c.trim().toLowerCase());
  }
  for (const c of parsed.conditionsRemoved) {
    conditionsSet.delete(c.trim().toLowerCase());
  }
  const activeConditions = Array.from(conditionsSet).sort();

  // 2. Reconcile motor limitations
  const motorSet = new Set<string>(
    current.motorLimitations.map((m) => m.toUpperCase())
  );
  for (const m of parsed.motorLimitationsAdded) {
    motorSet.add(m.trim().toUpperCase());
  }
  for (const m of parsed.motorLimitationsRemoved) {
    motorSet.delete(m.trim().toUpperCase());
  }
  const motorLimitations = Array.from(motorSet).sort();

  // 3. Reconcile signature tics
  const ticsSet = new Set<string>(current.signatureTics);
  for (const t of parsed.signatureTicsAdded) {
    ticsSet.add(t.trim());
  }
  for (const t of parsed.signatureTicsRemoved) {
    ticsSet.delete(t.trim());
  }
  const signatureTics = Array.from(ticsSet).sort();

  // 4. Compute clamped strain meters in integer basis points
  const fatigueBps = clampBps(current.fatigueBps + parsed.deltaFatigueBps, 'fatigueBps');
  const painBps = clampBps(current.painBps + parsed.deltaPainBps, 'painBps');
  const refusalStrainBps = clampBps(
    current.refusalStrainBps + parsed.deltaRefusalStrainBps,
    'refusalStrainBps'
  );
  const isRefusalStrained = refusalStrainBps >= 8000;

  // 5. Compute leg availability
  const rawLegs = current.legsAvailable + parsed.deltaLegs;
  const legsAvailable = Math.max(0, Math.min(2, rawLegs));

  // 6. Derive locomotion tier
  const locomotionTier = deriveLocomotionTier(legsAvailable, painBps, motorLimitations);

  // 7. Invoke cross-unit sub-systems
  const limbReceipt = calculateCompoundTraumaLimbClamping({
    entityId: parsed.entityId,
    locomotionBaseline: parsed.locomotionBaseline,
    conditions: activeConditions,
  });

  const sensoryProfile = calculateSensoryAcuityProfile({
    entityId: parsed.entityId,
    conditions: activeConditions,
  });

  const vocalDspProfile = calculateVocalDspProfile({
    entityId: parsed.entityId,
    conditions: activeConditions,
  });

  const receipt: SomaticTraumaReceipt = {
    entityId: parsed.entityId,
    armsAvailable: limbReceipt.armsAvailable,
    availableArmsBps: limbReceipt.availableArmsBps,
    legsAvailable,
    locomotionTier,
    fatigueBps,
    painBps,
    refusalStrainBps,
    isRefusalStrained,
    activeConditions,
    motorLimitations,
    signatureTics,
    limbReceipt,
    sensoryProfile,
    vocalDspProfile,
    updatedAtTick: parsed.tick > 0 ? parsed.tick : current.updatedAtTick + 1,
  };

  return SomaticTraumaReceiptSchema.parse(receipt);
};

export const configureSensoryAcuity = (
  cpy: TraumaModel,
  bal: TraumaBit<SensoryAcuityInput, unknown>,
  _ste: State
): TraumaModel => {
  const payload = bal?.dat;
  if (!payload?.entityId) {
    if (bal?.slv) {
      bal.slv({
        trmBit: { idx: 'configure-sensory-acuity-err', src: 'MISSING_PAYLOAD', val: -1 },
      });
    }
    return cpy;

  try {
    const profile = calculateSensoryAcuityProfile(payload);

    if (bal?.slv) {
      bal.slv({
        trmBit: {
          idx: 'configure-sensory-acuity-ok',
          src: profile.entityId,
          val: profile.visualAcuityBps,
          dat: profile,
        },
      });
    }
  } catch (err: unknown) {
    const message = err instanceof Error ? err.message : String(err);

        trmBit: {
          idx: 'configure-sensory-acuity-fail',
          src: payload.entityId,
          val: -1,
          dat: { error: message },
