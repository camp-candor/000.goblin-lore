    };
  }

  const statsParsed = BaseStatsSchema.safeParse(profile.base_stats);
  if (!statsParsed.success) {
    return {
      valid: false,
      errorCode: 'PERSONALITY_STAT_BOUNDS_BREACH',
      reason: statsParsed.error.issues[0]?.message ?? 'Personality stat outside [1, 5] bounds'
    };
  }

  return { valid: true };
}

/**
 * Scans a spoken dialogue turn against the character's forbidden attenuation tokens.
 */
export function scanAttenuationLeaks(
  dialogueText: string,
  attenuationTokens: readonly string[]
): AttenuationLeakDiagnosticBit[] {
  if (!dialogueText || attenuationTokens.length === 0) return [];

  const leaks: AttenuationLeakDiagnosticBit[] = [];
  const normalizedDialogue = dialogueText.toLowerCase();

  for (const token of attenuationTokens) {
    const cleanToken = token.trim().toLowerCase();
    if (cleanToken.length === 0) continue;

    // Word boundary search to prevent substring false-positives
    const regex = new RegExp(`\\b${cleanToken}\\b`, 'i');
    const match = regex.exec(normalizedDialogue);
    if (match) {
      leaks.push({
        leaked_token: cleanToken,
        token_index: match.index
      });
    }
  }

  return leaks;
}

/**
 * Evaluates a proposed dialogue turn against an actor's conversational profile at Stage T_0 = $0.00.
 */
export function evaluateDialogueTurnPreflight(
  input: DialogueTurnEvaluationInput,
  profile: CharacterConversationalProfile
): EscalationEvaluationResultBit {
  const parsedInput = DialogueTurnEvaluationInputSchema.safeParse(input);
  if (!parsedInput.success) {
    return {
      valid: false,
      actor_id: input?.actor_id ?? 'char_unknown',
      proposed_tier: input?.proposed_intensity_tier ?? 0,
      ceiling_tier: profile?.escalation_ceiling ?? 0,
      leaks: [],
      errorCode: 'SCHEMA_VALIDATION_ERROR',
      reason: parsedInput.error.issues[0]?.message ?? 'Invalid dialogue turn input'
    };
  }

  const { proposed_intensity_tier, spoken_dialogue_text, leverage_expended_count, override_authorized } = parsedInput.data;
  const ceiling = profile.escalation_ceiling;

  // 1. Check Subtext Attenuation Token Leaks
  const leaks = scanAttenuationLeaks(spoken_dialogue_text, profile.attenuation_tokens);
  if (leaks.length > 0 && !profile.is_strained_break && !override_authorized) {
    return {
      valid: false,
      actor_id: profile.actor_id,
      proposed_tier: proposed_intensity_tier,
      ceiling_tier: ceiling,
      leaks,
      errorCode: 'SUBTEXT_ATTENUATION_LEAK',
      reason: `Dialogue leaked forbidden attenuation token(s): [${leaks.map((l) => l.leaked_token).join(', ')}]`
    };
  }

  // 2. Check Intensity Escalation Ceiling
  if (proposed_intensity_tier > ceiling) {
    // Permitted only if leverage was expended or character has suffered a psychological break
    const isExemptionMet = leverage_expended_count > 0 || profile.is_strained_break || override_authorized;
    if (!isExemptionMet) {
      return {
        valid: false,
        actor_id: profile.actor_id,
        proposed_tier: proposed_intensity_tier,
        ceiling_tier: ceiling,
        leaks: [],
        errorCode: 'CONVERSATIONAL_CEILING_BREACH',
        reason: `Proposed intensity tier (${proposed_intensity_tier}) exceeds actor escalation ceiling (${ceiling}) without leverage expenditure or break state`
      };
    }
  }

  return {
    valid: true,
    actor_id: profile.actor_id,
    proposed_tier: proposed_intensity_tier,
    ceiling_tier: ceiling,
    leaks: [],
    errorCode: 'NONE'
  };
}

export const registerConversationalProfileBuzz = (
  cpy: EscalationModel,

  ste: State

    const validated = CharacterConversationalProfileSchema.parse(bal.dat);

  }
  if (bal.slv) {
    bal.slv({ escBit: { idx: bal.idx, val: 1 } });
  }
  return cpy;
};

export const validateEscalationCeilingBuzz = async (
  cpy: EscalationModel,
  bal: EscalationBit<CharacterConversationalProfile>,
  ste: State
) => {
  if (!bal.dat) {
    if (bal.slv) bal.slv({ escBit: { idx: bal.idx, val: 0 } });
    return cpy;
  }

  const boundsCheck = validateEscalationBounds(bal.dat);
  if (!boundsCheck.valid) {
    await emitTelemetry(
      'cns00',
      `:: [ESCALATION:BOUNDS_FAIL] Actor ${bal.dat.actor_id}: ${boundsCheck.errorCode} -> ${boundsCheck.reason}`
    );
  } else {
    await emitTelemetry(
      'cns00',
      `>> [ESCALATION:BOUNDS_PASS] Actor ${bal.dat.actor_id}: Escalation ceiling ${bal.dat.escalation_ceiling} in [1, 5] bounds [OK]`
    );
  }

  if (bal.slv) {
    bal.slv({ escBit: { idx: bal.idx, val: boundsCheck.valid ? 1 : 0, dat: boundsCheck } });
  }
  return cpy;
};

export const evaluateDialogueTurnBuzz = async (
  cpy: EscalationModel,
  bal: EscalationBit<DialogueTurnEvaluationInput>,
  ste: State
) => {
  if (!bal.dat) {
    if (bal.slv) bal.slv({ escBit: { idx: bal.idx, val: 0 } });
    return cpy;
  }

  const profile = cpy.profiles[bal.dat.actor_id] ?? {
    actor_id: bal.dat.actor_id,
    escalation_ceiling: 3,
    base_stats: { spite: 3, cunning: 3, charm: 3, nerve: 3, warmth: 3 },
    attenuation_tokens: [],
    is_strained_break: false
  };

  const result = evaluateDialogueTurnPreflight(bal.dat, profile);
  cpy.audit_history.push(result);

  if (!result.valid) {
    await emitTelemetry(
      'cns00',
      `:: [ESCALATION:BREACH] Actor ${result.actor_id}: ${result.errorCode} -> ${result.reason}`
    );
  } else {
    await emitTelemetry(
      'cns00',
      `>> [ESCALATION:TURN_OK] Actor ${result.actor_id}: Tier ${result.proposed_tier} <= Ceiling ${result.ceiling_tier} [OK]`
    );
  }

  if (bal.slv) {
    bal.slv({ escBit: { idx: bal.idx, val: result.valid ? 1 : 0, dat: result } });
  }
  return cpy;
};

import clone from 'clone-deep';
import * as Act from './escalation.action';
import { EscalationModel } from './escalation.model';
import * as Buzz from './escalation.buzzer';
import State from '../99.core/state';

export function reducer(
  model: EscalationModel = new EscalationModel(),
  act: Act.Actions,
  state?: State
) {
  switch (act.type) {
    case Act.INIT_ESCALATION:
      return Buzz.initEscalation(clone(model), act.bale, state!);
    case Act.UPDATE_ESCALATION:
      return Buzz.updateEscalation(clone(model), act.bale, state!);
    case Act.REGISTER_CONVERSATIONAL_PROFILE:
      return Buzz.registerConversationalProfileBuzz(clone(model), act.bale, state!);
    case Act.VALIDATE_ESCALATION_CEILING:
      return Buzz.validateEscalationCeilingBuzz(clone(model), act.bale, state!);
    case Act.EVALUATE_DIALOGUE_TURN:
      return Buzz.evaluateDialogueTurnBuzz(clone(model), act.bale, state!);
    default:
      return model;
  }
}

import { describe, it, expect } from 'vitest';
import {
  CharacterConversationalProfileSchema,
  CharacterConversationalProfile
} from '../val/escalation.val';
import {
  validateEscalationBounds,
  scanAttenuationLeaks,
  evaluateDialogueTurnPreflight

describe('10.escalation.unit :: Conversational Escalation Ceiling Bounds Validator', () => {
  const validProfile: CharacterConversationalProfile = {
    actor_id: 'char_wart',
    escalation_ceiling: 3,
    base_stats: {
      spite: 4,
      cunning: 2,
      charm: 1,
      nerve: 2,
      warmth: 1
    },
    attenuation_tokens: ['amputation', 'turbine', 'coward'],
    is_strained_break: false
  };

  it('validates a well-formed CharacterConversationalProfile within [1, 5] bounds', () => {
    const parsed = CharacterConversationalProfileSchema.parse(validProfile);
    expect(parsed.actor_id).toBe('char_wart');
    expect(parsed.escalation_ceiling).toBe(3);
    expect(parsed.base_stats.spite).toBe(4);
    expect(parsed.attenuation_tokens).toHaveLength(3);
  });

  it('rejects an escalation ceiling outside [1, 5] bounds (T_0 = $0.00)', () => {
    const invalidProfile = {
      ...validProfile,
      escalation_ceiling: 6
    };
    expect(() => CharacterConversationalProfileSchema.parse(invalidProfile)).toThrow(
      /Escalation ceiling maximum is 5/
    );

    const zeroCeiling = {
      ...validProfile,
      escalation_ceiling: 0
    };
    expect(() => CharacterConversationalProfileSchema.parse(zeroCeiling)).toThrow(
      /Escalation ceiling minimum is 1/
    );
  });

  it('rejects a floating-point escalation ceiling', () => {
    const floatCeiling = {
      ...validProfile,
      escalation_ceiling: 3.5
    };
    expect(() => CharacterConversationalProfileSchema.parse(floatCeiling)).toThrow(
      /Escalation ceiling cannot be a float/
    );
  });

  it('rejects personality base stats outside [1, 5]', () => {
    const badStats = {
      ...validProfile,
      base_stats: {
        ...validProfile.base_stats,
        spite: 0
      }
    };
    expect(() => CharacterConversationalProfileSchema.parse(badStats)).toThrow();
  });

  it('detects forbidden subtext attenuation token leaks in spoken dialogue text', () => {
    const dialogue = 'We never talk about what happened in the turbine!';
    const leaks = scanAttenuationLeaks(dialogue, validProfile.attenuation_tokens);
    expect(leaks).toHaveLength(1);
    expect(leaks[0].leaked_token).toBe('turbine');
  });

  it('does not trigger false-positive attenuation leaks on substring matches', () => {
    const dialogue = 'You are a cowardly wretch!';
    // 'coward' is forbidden, but 'cowardly' should not false-positive if word-bounded
    const leaks = scanAttenuationLeaks(dialogue, ['coward']);
    expect(leaks).toHaveLength(0);
  });

  it('fails closed when dialogue leaks an attenuation token without a break state', () => {
    const result = evaluateDialogueTurnPreflight(
      {
        actor_id: 'char_wart',
        proposed_intensity_tier: 2,
        spoken_dialogue_text: 'You left me in the turbine to die!',
        leverage_expended_count: 0,
        override_authorized: false
      },
      validProfile
    );

    expect(result.valid).toBe(false);
    expect(result.errorCode).toBe('SUBTEXT_ATTENUATION_LEAK');
    expect(result.leaks).toHaveLength(1);
  });

  it('fails closed when proposed intensity tier exceeds actor escalation ceiling without leverage', () => {
    const result = evaluateDialogueTurnPreflight(
      {
        actor_id: 'char_wart',
        proposed_intensity_tier: 5, // Ceiling is 3
        spoken_dialogue_text: 'Hand over the valve right now or else!',
        leverage_expended_count: 0,
        override_authorized: false
      },
      validProfile
    );

    expect(result.valid).toBe(false);
    expect(result.errorCode).toBe('CONVERSATIONAL_CEILING_BREACH');
    expect(result.proposed_tier).toBe(5);
    expect(result.ceiling_tier).toBe(3);
  });

  it('permits escalation beyond ceiling when leverage is expended', () => {
    const result = evaluateDialogueTurnPreflight(
      {
        actor_id: 'char_wart',
        proposed_intensity_tier: 5,
        spoken_dialogue_text: 'Hand over the valve right now or else!',
        leverage_expended_count: 1, // Expending directional leverage string
        override_authorized: false
      },
      validProfile
    );

    expect(result.valid).toBe(true);
    expect(result.errorCode).toBe('NONE');
  });

  it('permits escalation and subtext leaks when character has entered an explicit break state', () => {
    const brokenProfile: CharacterConversationalProfile = {
      ...validProfile,
      is_strained_break: true
    };

    const result = evaluateDialogueTurnPreflight(
      {
        actor_id: 'char_wart',
        proposed_intensity_tier: 4,
        spoken_dialogue_text: 'I know you called me a coward after the amputation!',
        leverage_expended_count: 0,
        override_authorized: false
      },
      brokenProfile
    );

    expect(result.valid).toBe(true);
    expect(result.errorCode).toBe('NONE');
  });
});
