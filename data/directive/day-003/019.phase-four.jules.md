```typescript
import { z } from 'zod';

export const SERIES_SLUG_MIN_LENGTH = 3;
export const SERIES_SLUG_MAX_LENGTH = 64;
export const SERIES_SLUG_REGEX = /^[a-z0-9]+(?:-[a-z0-9]+)*$/;

/**
 * Zod schema strictly validating canonical POSIX series slugs.
 */
export const SeriesSlugSchema = z
  .string()
  .min(SERIES_SLUG_MIN_LENGTH, {
    message: `Series slug must be at least ${SERIES_SLUG_MIN_LENGTH} characters long.`,
  })
  .max(SERIES_SLUG_MAX_LENGTH, {
    message: `Series slug must not exceed ${SERIES_SLUG_MAX_LENGTH} characters.`,
  })
  .regex(SERIES_SLUG_REGEX, {
    message: 'Series slug must be lowercase alphanumeric words separated by single hyphens.',
  });

export type SeriesSlug = z.infer<typeof SeriesSlugSchema>;

/**
 * Normalizes an arbitrary raw title string into a strict POSIX-compliant kebab-case slug.
 */
export const sanitizeSeriesSlug = (rawInput: string): string => {
  if (!rawInput || typeof rawInput !== 'string') {
    return '';
  }

  return rawInput
    // 1. Strip ANSI escape codes
    .replace(/\x1b\[[0-9;]*[a-zA-Z]/g, '')
    // 2. Downcase to ASCII lowercase
    .toLowerCase()
    .trim()
    // 3. Replace underscores, slashes, and whitespace with hyphens
    .replace(/[\s_/\\:]+/g, '-')
    // 4. Strip any character outside [a-z0-9-]
    .replace(/[^a-z0-9-]/g, '')
    // 5. Collapse consecutive hyphens into a single hyphen
    .replace(/-+/g, '-')
    // 6. Trim leading and trailing hyphens
    .replace(/^-+|-+$/g, '');
};

```

import {
  SeriesSlugSchema,
  sanitizeSeriesSlug,
  SERIES_SLUG_MIN_LENGTH,
  SERIES_SLUG_MAX_LENGTH,
} from '../val/seriesSlug.val.js';
import type { SeriesSlugValidationResultBit } from '../fce/seriesSlug.bit.js';

/**
 * Validates a series slug against the strict POSIX schema, returning a typed result envelope.
 */
export const validateSeriesSlug = (rawInput: string): SeriesSlugValidationResultBit => {
  const sanitized = sanitizeSeriesSlug(rawInput);

  if (sanitized.length === 0) {
    return {
      valid: false,
      rawInput,
      sanitizedAttempt: sanitized,
      errorCode: 'SLUG_EMPTY',
      message: 'Slug candidate is empty after sanitization.',
    };
  }

  if (sanitized.length < SERIES_SLUG_MIN_LENGTH) {
    return {
      valid: false,
      rawInput,
      sanitizedAttempt: sanitized,
      errorCode: 'SLUG_TOO_SHORT',
      message: `Slug '${sanitized}' is shorter than minimum length of ${SERIES_SLUG_MIN_LENGTH}.`,
    };
  }

  if (sanitized.length > SERIES_SLUG_MAX_LENGTH) {
    return {
      valid: false,
      rawInput,
      sanitizedAttempt: sanitized,
      errorCode: 'SLUG_TOO_LONG',
      message: `Slug '${sanitized}' exceeds maximum length of ${SERIES_SLUG_MAX_LENGTH}.`,
    };
  }

  const parseResult = SeriesSlugSchema.safeParse(sanitized);

  if (!parseResult.success) {
    return {
      valid: false,
      rawInput,
      sanitizedAttempt: sanitized,
      errorCode: 'SLUG_INVALID_FORMAT',
      message: parseResult.error.issues[0]?.message || 'Invalid POSIX slug format.',
    };
  }

  return {
    valid: true,
    slug: parseResult.data,
  };
};

/**
 * Asserts that a raw title can be sanitized into a valid slug, throwing a fatal Stage T0 error if invalid.
 */
export const assertSeriesSlug = (rawInput: string): string => {
  const result = validateSeriesSlug(rawInput);
  if (!result.valid) {
    throw new Error(`[SERIES_SLUG_BREACH] ${result.errorCode}: ${result.message}`);
  }
  return result.slug;
};
