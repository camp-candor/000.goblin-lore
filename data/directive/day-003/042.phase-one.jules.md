    dat: { clr0: 'black', clr1: 'yellow' },
    src: 'vertical',
    lst,
    net: gridBit?.grdBit?.dat,
  });

  const src: string = choiceBit?.chcBit?.src || '';
  cpy.lastSelected = src;

  switch (src) {
    case 'OPEN ENTITY SCAFFOLDER':
      await ste.hunt(ActMnu.SCAFFOLD_MENU, {});
      break;
    case 'COMPILE LORE INSTANCE':
      await ste.hunt(UPDATE_CONSOLE, {
        idx: 'cns00',
        src: '>> [COMPILE] Triggering 3-pass canon compiler...',
      });
      break;
    case 'AUDIT CORPUS HEALTH':
      await ste.hunt(UPDATE_CONSOLE, {
        idx: 'cns00',
        src: '>> [AUDIT] Running corpus health check...',
      });
      break;
    case 'EXIT SYSTEM':
      await ste.hunt(UPDATE_CONSOLE, {
        idx: 'cns00',
        src: '>> [HALT] Terminal control released.',
      });
      break;
  }

  if (bal.slv) bal.slv({ mnuBit: { idx: 'update-menu', src } });
  return cpy;
};

/**
 * packages/299.lore/98.menu.unit/menu.reduce.ts
 * Pure reducer switchboard delegating menu actions to clone-deep buzzers.
 */
import clone from 'clone-deep';
import * as Act from './menu.action';
import { MenuModel } from './menu.model';
import * as Buzz from './menu.buzzer';
import type State from '../99.core/state';

export function reducer(
  model: MenuModel = new MenuModel(),
  act: Act.Actions,
  state?: State
): Promise<MenuModel> | MenuModel {
  switch (act.type) {
    case Act.INIT_MENU:
      return Buzz.initMenu(clone(model), act.bale, state!);

    case Act.UPDATE_MENU:
      return Buzz.updateMenu(clone(model), act.bale, state!);

    case Act.SCAFFOLD_MENU:
      return Buzz.scaffoldMenu(clone(model), act.bale, state!);

    case Act.SCAFFOLD_CHARACTER:
      return Buzz.scaffoldCharacter(clone(model), act.bale, state!);

    case Act.SCAFFOLD_LOCATION:
      return Buzz.scaffoldLocation(clone(model), act.bale, state!);

    case Act.SCAFFOLD_POSSESSION:
      return Buzz.scaffoldPossession(clone(model), act.bale, state!);

    case Act.SCAFFOLD_GRIEVANCE:
      return Buzz.scaffoldGrievance(clone(model), act.bale, state!);

    case Act.SCAFFOLD_WORLD:
      return Buzz.scaffoldWorld(clone(model), act.bale, state!);

    default:
      return model;
  }
}
