 * 10000 bp = 1.0 arm; 20000 bp = 2.0 arms.

const SOMATIC_LIMB_PENALTIES: Record<string, number> = {
  AMPUTATION_LEFT_ARM: 10000,
  AMPUTATION_RIGHT_ARM: 10000,
  BRACHIAL_PLEXUS_LEFT: 10000,
  BRACHIAL_PLEXUS_RIGHT: 10000,
  FRACTURED_HUMERUS_LEFT: 8000,
  FRACTURED_HUMERUS_RIGHT: 8000,
  DISLOCATED_SHOULDER_LEFT: 5000,
  DISLOCATED_SHOULDER_RIGHT: 5000,
  BILATERAL_AMPUTATION_ARMS: 20000,
  TOTAL_PARALYSIS: 20000,
};

export interface LimbCapacityResult {
  readonly available_arms_bp: number;
  readonly total_penalty_bp: number;
  readonly active_penalties: Record<string, number>;
}

/**
 * Computes net available functional arms in integer basis points.
 * Clamps lower bound to 0 bp to prevent negative numbers.
 */
export function computeFunctionalLimbsBuzz(
  baseArmsBp: number = 20000,
  conditions: readonly string[] = []
): LimbCapacityResult {
  let totalPenaltyBp = 0;
  const activePenalties: Record<string, number> = {};

  for (const condition of conditions) {
    const penalty = SOMATIC_LIMB_PENALTIES[condition.toUpperCase()];
    if (penalty !== undefined) {
      totalPenaltyBp += penalty;
      activePenalties[condition] = penalty;
    }
  }

  const availableArmsBp = Math.max(0, baseArmsBp - totalPenaltyBp);

  return {
    available_arms_bp: availableArmsBp,
    total_penalty_bp: totalPenaltyBp,
    active_penalties: activePenalties,
  };
}

const VERB_TAXONOMY_ROLLUP: Record<string, readonly string[]> = {
  sprint: ["sprint", "dash", "bolt", "charge"],
  leap: ["leap", "bound", "jump", "hurdle"],
  vault: ["vault", "hop_barrier", "scale_wall"],
  scramble: ["scramble", "clamber", "rush"],
  bimanual_strike: ["bimanual_strike", "heavy_swing", "two_handed_cleave"],
};

/**
 * Consolidates, rolls up synonyms, and deduplicates banned kinetic verbs.
 */
export function enforceBannedKineticVerbsBuzz(
  baseBanned: readonly string[] = [],
  locomotionBanned: readonly string[] = [],
  somaticConditions: readonly string[] = []
): string[] {
  const resultSet = new Set<string>();

  const addVerbWithRollup = (verb: string) => {
    const normalized = verb.toLowerCase().trim();
    resultSet.add(normalized);
    const rollups = VERB_TAXONOMY_ROLLUP[normalized];
    if (rollups) {
      for (const synonym of rollups) {
        resultSet.add(synonym);
      }
    }
  };

  for (const v of baseBanned) addVerbWithRollup(v);
  for (const v of locomotionBanned) addVerbWithRollup(v);

  for (const condition of somaticConditions) {
    const uc = condition.toUpperCase();
    if (uc.includes("AMPUTATION") || uc.includes("BRACHIAL") || uc.includes("FRACTURED")) {
      addVerbWithRollup("bimanual_strike");
      addVerbWithRollup("bimanual_lift");
    }
    if (uc.includes("PARALYSIS") || uc.includes("FRACTURED_FEMUR")) {
      addVerbWithRollup("sprint");
      addVerbWithRollup("leap");
      addVerbWithRollup("vault");
    }
  }

  return Array.from(resultSet).sort();
}

 * Pure in-memory SAT solver verifying physical capability invariants at T_0 = $0.00.
 */
export function evaluateSomaticSATBuzz(
  candidate: SomaticSATCandidateBit
): SomaticSATReceiptBit {
  const limbResult = computeFunctionalLimbsBuzz(
    candidate.base_arms_bp,
    candidate.somatic_conditions
  );

      success: false,

  return {
    success: true,
    entity_id: candidate.entity_id,
    code: "OK",

    required_hands_bp: loadResult.required_hands_bp,
