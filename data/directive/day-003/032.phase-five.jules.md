```typescript
export interface CursesSuspensionResult {

  readonly exitCode: number;
  readonly headlessBypass: boolean;

export const suspendCursesAndSpawnEditor = async (
  targetFilePath: string

  const lib = (globalThis as any).LIBRARY;
  const isHeadless = !process.stdin.isTTY || process.env.CI === 'true';

  if (isHeadless) {
    await emitBusConsoleLog('cns00', `>> [HUMAN_GATE] Headless execution detected. Bypassing interactive $EDITOR for: ${targetFilePath}`);
    return {
      success: true,
      exitCode: 0,
      headlessBypass: true
    };
  }

  await emitBusConsoleLog('cns00', '>> [HUMAN_GATE] Suspending Blessed curses screen...');

  // 1. De-allocate curses raw mode & leave alternate screen buffer
  if (lib && typeof lib.hunt === 'function') {
    await lib.hunt('[Terminal action] Update Terminal', {
      dat: { suspendRawMode: true }
    }).catch(() => {});
  }

  const editorCommand = process.env.VISUAL || process.env.EDITOR || 'nano';
  let childStatus = 0;

  try {
    await emitBusConsoleLog('cns00', `>> [HUMAN_GATE] Spawning editor: ${editorCommand} ${targetFilePath}`);
    const child = spawnSync(editorCommand, [targetFilePath], {
      stdio: 'inherit',
      shell: true
    });

    childStatus = 1;
    await emitBusConsoleLog('cns00', `>> [FAIL] [HUMAN_GATE] Editor process error: ${err?.message || String(err)}`);
  } finally {
    // 2. Re-enter alternate screen buffer & re-enable raw mode
    if (lib && typeof lib.hunt === 'function') {
      await lib.hunt('[Terminal action] Update Terminal', {
        dat: { restoreRawMode: true }
      }).catch(() => {});
    }
    await emitBusConsoleLog('cns00', '>> [HUMAN_GATE] Restored Blessed curses screen buffer.');
  }

  return {
    success: childStatus === 0,
    exitCode: childStatus,
    headlessBypass: false,
    error: childStatus !== 0 ? `Editor exited with non-zero status: ${childStatus}` : undefined
  };
```

#### 5. Scratchpad Verification Gate: `packages/299.lore/41.canon.unit/buz/verifyScratchpadSave.buzz.ts`

```typescript
import * as fs from 'node:fs';
import matter from 'gray-matter';
import { verifyByteZeroIntegrity } from './byteZero.buzz.js';
import { PossessionDossierFrontmatterSchema } from '../../12.gear.unit/val/gear.val.js';
import type { EntityDomain } from '../../09.entity.unit/fce/humanGate.bit.js';

export interface ScratchpadVerificationResult {
  readonly passed: boolean;
  readonly error?: string;
  readonly parsedData?: Record<string, any>;
}

export const verifyScratchpadSave = (
  tmpFilePath: string,
  domain: EntityDomain
): ScratchpadVerificationResult => {
  if (!fs.existsSync(tmpFilePath)) {
    return { passed: false, error: `SCRATCHPAD_NOT_FOUND: ${tmpFilePath}` };
  }

  const rawBytes = fs.readFileSync(tmpFilePath);

  // 1. Byte-0 Opening Invariant
  if (!verifyByteZeroIntegrity(rawBytes)) {
    return {
      passed: false,
      error: 'BYTE_ZERO_BREACH: File does not open with delimiter --- at physical byte offset 0.'
    };
  }

  // 2. Parse Frontmatter
  const content = rawBytes.toString('utf-8');
  let parsed: matter.GrayMatterFile<string>;
  try {
    parsed = matter(content);
  } catch (err: any) {
    return {
      passed: false,
      error: `YAML_SYNTAX_ERROR: ${err?.message || String(err)}`
    };
  }

  const data = parsed.data;
  if (!data || typeof data !== 'object' || !data.id) {
    return {
      passed: false,
      error: 'SCHEMA_ERROR: Missing required entity identifier property: id'
    };
  }

  // 3. Domain-Specific Schema & SAT Invariant Gates
  if (domain === 'possessions') {
    const gearResult = PossessionDossierFrontmatterSchema.safeParse(data);
    if (!gearResult.success) {
      const issueDetails = gearResult.error.issues.map((i) => `${i.path.join('.')}: ${i.message}`).join('; ');
      return {
        passed: false,
        error: `POSSESSION_SCHEMA_FAIL: ${issueDetails}`
      };
    }
  }

  if (domain === 'characters') {
    const armsBps = data?.somatic?.available_arms_bps;
    if (armsBps !== undefined && (typeof armsBps !== 'number' || armsBps < 0 || armsBps > 20000 || !Number.isInteger(armsBps))) {
      return {
        passed: false,
        error: 'SOMATIC_SAT_FAIL: available_arms_bps must be an integer basis point between 0 and 20000.'
      };
    }
  }

  return {
    passed: true,
    parsedData: data
  };
};
```

```typescript
export const threeStrikeSaveGate = async (
  cpy: EntityModel,
  bal: HumanGateBit,
  _ste: State
): Promise<EntityModel> => {
  const entityId = bal.dat?.entityId;
  const domain = bal.dat?.domain;
  const seriesSlug = bal.src || bal.dat?.seriesSlug || 'wiregrass-basin';

  if (!entityId || !domain) {
    await emitBusConsoleLog('cns00', '>> [FAIL] [HUMAN_GATE] Missing required entityId or domain.');
    if (bal.slv) bal.slv({ entityBit: { idx: bal.idx, val: 0, dat: { error: 'INVALID_PARAMETERS' } } });
    return cpy;
  }

  await emitBusConsoleLog('cns00', `>> [HUMAN_GATE] Initializing scratchpad for ${domain}/${entityId} in '${seriesSlug}'...`);

  // Stage isolated scratchpad (.tmp)
  const scratchpad = stageEntityScratchpad({
    seriesSlug,
    domain,
    entityId,
    initialFrontmatter: bal.dat?.initialFrontmatter
  });

  const maxStrikes = 3;
  let currentStrike = 1;
  let saveSuccess = false;
  let lastError = '';

  while (currentStrike <= maxStrikes && !saveSuccess) {
    await emitBusConsoleLog('cns00', `>> [HUMAN_GATE] Strike ${currentStrike}/${maxStrikes}: Spawning editor on scratchpad...`);

    // Hand off to editor
    const suspensionResult = await suspendCursesAndSpawnEditor(scratchpad.tmpFilePath);
    if (!suspensionResult.success) {
      lastError = suspensionResult.error || 'EDITOR_SPAWN_FAILED';
      currentStrike++;
      continue;
    }

    // Verify scratchpad
    const verification = verifyScratchpadSave(scratchpad.tmpFilePath, domain);
    if (verification.passed) {
      saveSuccess = true;
      break;
    }

    lastError = verification.error || 'VERIFICATION_FAILED';
    await emitBusConsoleLog('cns00', `>> [WARN] [HUMAN_GATE] Strike ${currentStrike} Verification Failed: ${lastError}`);

    if (currentStrike < maxStrikes) {
      // Prepend commented diagnostic block inside scratchpad for the author to review
      const existingContent = fs.readFileSync(scratchpad.tmpFilePath, 'utf-8');
      const diagnosticComment = [
        '---',
        `# [DIAGNOSTIC FAILURE - STRIKE ${currentStrike}/${maxStrikes}]`,
        `# ERROR: ${lastError}`,
        `# PLEASE RESOLVE THE ABOVE DEFECT AND RE-SAVE.`,
        '---'
      ].join('\n');

      if (existingContent.startsWith('---')) {
        const withoutOpening = existingContent.slice(3);
        fs.writeFileSync(scratchpad.tmpFilePath, `${diagnosticComment}${withoutOpening}`, { encoding: 'utf-8' });
      }
    }

    currentStrike++;
  }

  // Evaluate Circuit Breaker Outcome
  if (!saveSuccess) {
    await emitBusConsoleLog('cns00', `>> [FAIL] [CIRCUIT_BREAKER_TRIPPED] Human save gate failed after ${maxStrikes} strikes.`);
    await emitBusConsoleLog('cns00', `>> [FAIL] Unlinking ephemeral scratchpad: ${scratchpad.tmpFilePath}`);

    if (fs.existsSync(scratchpad.tmpFilePath)) {
      try { fs.unlinkSync(scratchpad.tmpFilePath); } catch {}
    }

    const failurePayload: HumanGateResultPayload = {
      success: false,
      entityId,
      domain,
      canonicalPath: scratchpad.canonicalFilePath,
      attempts: maxStrikes,
      trippedCircuitBreaker: true,
      error: `STAGING_HUMAN_GATE_BREAKER_TRIPPED: ${lastError}`
    };

    if (bal.slv) bal.slv({ entityBit: { idx: bal.idx, val: 0, dat: { result: failurePayload, error: failurePayload.error } } });
    return cpy;
  }

  // Atomic POSIX Promotion over Canonical Inode
  try {
    fs.renameSync(scratchpad.tmpFilePath, scratchpad.canonicalFilePath);
    await emitBusConsoleLog('cns00', `>> [OK] [HUMAN_GATE] Promoted scratchpad to canonical path: ${scratchpad.canonicalFilePath}`);
  } catch (err: any) {
    const errorMsg = `CANONICAL_PROMOTION_ERROR: ${err?.message || String(err)}`;
    await emitBusConsoleLog('cns00', `>> [FAIL] ${errorMsg}`);
    if (bal.slv) bal.slv({ entityBit: { idx: bal.idx, val: 0, dat: { error: errorMsg } } });
    return cpy;
  }

  // Trigger Phase 5 Recompilation to seal new Merkle Root (H_root)
  let updatedMerkleRoot = '';
  try {
    const passOne = executePassOneGenesisShape(seriesSlug);
    if (passOne.passed) {
      const passTwo = executePassTwoGenesisGraph(passOne.registry);
      if (passTwo.passed) {
        const passThree = executePassThreeGenesisSeal(seriesSlug, passOne.seriesConfig, passOne.registry);
        if (passThree.passed) {
          updatedMerkleRoot = passThree.merkleRoot;
          await emitBusConsoleLog('cns00', `>> [OK] [LORE::SEAL] Resealed canonical Merkle root: ${updatedMerkleRoot.slice(0, 16)}...`);
        }
      }
    }
  } catch (err: any) {
    await emitBusConsoleLog('cns00', `>> [WARN] [LORE::SEAL] Recompilation warning: ${err?.message || String(err)}`);
  }

  const successPayload: HumanGateResultPayload = {
    success: true,
    entityId,
    domain,
    canonicalPath: scratchpad.canonicalFilePath,
    updatedMerkleRoot,
    attempts: currentStrike
  };

  if (bal.slv) {
    bal.slv({
      entityBit: {
        idx: bal.idx,
        val: 1,
        dat: { result: successPayload }
      }
    });
  }

  return cpy;
};
```
