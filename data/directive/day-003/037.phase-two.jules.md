export interface SeriesResolutionResult {

/**
 * Traverses upward to locate the monorepo root containing the series directory.
 */
export function findRepoRoot(startDir: string = process.cwd()): string {
  let current = path.resolve(startDir);
  while (true) {
    const candidateSeriesDir = path.join(current, "series");
    const candidatePackageJson = path.join(current, "package.json");
    if (fs.existsSync(candidateSeriesDir) || fs.existsSync(candidatePackageJson)) {
      if (fs.existsSync(candidateSeriesDir)) {
        return current;
      }
    }
    const parent = path.dirname(current);
    if (parent === current) {
      break;
    }
    current = parent;
  }
  return path.resolve(startDir);
}

 */
export function resolveActiveSeriesDirBuzz(

import { EntityGraphNodeBit, TargetDomain } from "./fce/target.bit";

export class TargetModel {
  private static instance: TargetModel;

  private nodes: Map<string, EntityGraphNodeBit> = new Map();
  private domainIndices: Map<TargetDomain, Set<string>> = new Map([
    ["character", new Set()],
    ["location", new Set()],
    ["possession", new Set()],
    ["grievance", new Set()],
    ["world", new Set()],
  ]);

  private constructor() {}

  public static getInstance(): TargetModel {
    if (!TargetModel.instance) {
      TargetModel.instance = new TargetModel();
    }
    return TargetModel.instance;
  }

  public clear(): void {
    this.nodes.clear();
    for (const key of this.domainIndices.keys()) {
      this.domainIndices.get(key)!.clear();
    }
  }

  public registerNode(node: EntityGraphNodeBit): void {
    this.nodes.set(node.id, node);
    if (this.domainIndices.has(node.domain)) {
      this.domainIndices.get(node.domain)!.add(node.id);
    }
  }

  public getNode(id: string): EntityGraphNodeBit | undefined {
    return this.nodes.get(id);
  }

  public hasNode(id: string): boolean {
    return this.nodes.has(id);
  }

  public getIdsByDomain(domain: TargetDomain): string[] {
    const ids = this.domainIndices.get(domain);
    return ids ? Array.from(ids) : [];
  }

  public getAllNodes(): EntityGraphNodeBit[] {
    return Array.from(this.nodes.values());
  }

  public getDomainCounts(): Record<TargetDomain, number> {
    return {
      character: this.domainIndices.get("character")?.size ?? 0,

import * as fs from "node:fs";
import * as path from "node:path";
import matter from "gray-matter";
import { TargetModel } from "../target.model";
import { EntityGraphNodeBit, TargetDomain } from "../fce/target.bit";

const DOMAIN_SUBDIRS: Array<{ dir: string; domain: TargetDomain }> = [
  { dir: "characters", domain: "character" },
  { dir: "locations", domain: "location" },
  { dir: "possessions", domain: "possession" },
  { dir: "grievances", domain: "grievance" },
  { dir: "world", domain: "world" },
];

/**
 * Extracts outbound string references from arbitrary frontmatter objects.
 */
function extractReferences(obj: unknown): string[] {
  const refs: string[] = [];
  if (!obj || typeof obj !== "object") return refs;

  for (const value of Object.values(obj)) {
    if (typeof value === "string") {
      if (/^(char|loc|item|grv)_[a-z0-9_]+$/.test(value)) {
        refs.push(value);
      }
    } else if (Array.isArray(value)) {
      for (const item of value) {
        if (typeof item === "string" && /^(char|loc|item|grv)_[a-z0-9_]+$/.test(item)) {
          refs.push(item);
        } else if (typeof item === "object") {
          refs.push(...extractReferences(item));
        }
      }
    } else if (typeof value === "object") {
      refs.push(...extractReferences(value));
    }
  }
  return refs;
}

/**
 * Scans markdown files on disk and builds the in-memory multigraph registry.
 */
export function bootstrapGraphRegistryBuzz(seriesPath: string): {
  readonly success: boolean;
  readonly totalIndexed: number;
  readonly domainCounts: Record<TargetDomain, number>;
  readonly message: string;
} {
  const model = TargetModel.getInstance();
  model.clear();

  if (!fs.existsSync(seriesPath)) {
    return {
      success: false,
      totalIndexed: 0,
      domainCounts: model.getDomainCounts(),
      message: `[FAIL] :: Path does not exist: ${seriesPath}`,
    };
  }

  let totalIndexed = 0;

  for (const { dir, domain } of DOMAIN_SUBDIRS) {
    const fullDirPath = path.join(seriesPath, dir);
    if (!fs.existsSync(fullDirPath)) continue;

    const files = fs.readdirSync(fullDirPath);
    for (const file of files) {
      if (!file.endsWith(".md")) continue;
      if (file.startsWith("_template") || file.includes("_retired")) continue;

      const fullFilePath = path.join(fullDirPath, file);
      const stat = fs.statSync(fullFilePath);
      if (!stat.isFile()) continue;

      const fileContent = fs.readFileSync(fullFilePath, "utf8");
      const parsed = matter(fileContent);

      const id = String(parsed.data.id || path.basename(file, ".md"));
      const title = String(parsed.data.name || parsed.data.title || id);
      const references = extractReferences(parsed.data);

      const node: EntityGraphNodeBit = {
        id,
        domain,
        filePath: fullFilePath,
        title,
        references,
        rawFrontmatter: parsed.data,
      };

      model.registerNode(node);
      totalIndexed++;
    }
  }

  return {
    success: true,
    totalIndexed,
    domainCounts: model.getDomainCounts(),
    message: `[OK] :: Indexed ${totalIndexed} nodes across corpus`,
  };
}

/**
 * Returns candidate entity IDs matching prefix and query substrings.
 */
export function queryForeignKeyCandidatesBuzz(
  domainPrefix?: string,
  query: string = "",
  limit: number = 20
): string[] {
  const model = TargetModel.getInstance();
  const allNodes = model.getAllNodes();

  const filtered = allNodes.filter((node) => {
    if (domainPrefix && !node.id.startsWith(domainPrefix)) {
      return false;
    }
    if (!query) return true;
    const q = query.toLowerCase();
    return node.id.toLowerCase().includes(q) || node.title.toLowerCase().includes(q);
  });

  return filtered
    .map((n) => n.id)
    .sort((a, b) => a.localeCompare(b))
    .slice(0, limit);
}

import * as fs from "node:fs";
import matter from "gray-matter";
import { TargetModel } from "../../34.target.unit/target.model";

export interface ReciprocalLinkResult {
  readonly success: boolean;
  readonly targetLocationId: string;
  readonly targetFilePath: string;
  readonly alreadyLinked: boolean;
  readonly message: string;
}

/**
 * Reconciles bidirectional location links while strictly preserving Byte-0 frontmatter alignment.
 */
export function reconcileReciprocalLinksBuzz(
  sourceLocationId: string,
  targetLocationId: string
): ReciprocalLinkResult {
  const model = TargetModel.getInstance();
  const targetNode = model.getNode(targetLocationId);

  if (!targetNode) {
    return {
      success: false,
      targetLocationId,
      targetFilePath: "",
      alreadyLinked: false,
      message: `[FAIL] :: Target location not registered in multigraph: ${targetLocationId}`,
    };
  }

  const filePath = targetNode.filePath;
  if (!fs.existsSync(filePath)) {
    return {
      success: false,
      targetLocationId,
      targetFilePath: filePath,
      alreadyLinked: false,
      message: `[FAIL] :: File does not exist: ${filePath}`,
    };
  }

  const rawContent = fs.readFileSync(filePath, "utf8");
  const parsed = matter(rawContent);

  const connected: string[] = Array.isArray(parsed.data.connected_locations)
    ? [...parsed.data.connected_locations]
    : [];

  if (connected.includes(sourceLocationId)) {
    return {
      success: true,
      targetLocationId,
      targetFilePath: filePath,
      alreadyLinked: true,
      message: `[OK] :: Reciprocal link already exists in ${targetLocationId}`,
    };
  }

  connected.push(sourceLocationId);
  connected.sort();
  parsed.data.connected_locations = connected;

  const updatedYaml = matter.stringify(parsed.content, parsed.data);

  if (!updatedYaml.startsWith("---\n")) {
    return {
      success: false,
      targetLocationId,
      targetFilePath: filePath,
      alreadyLinked: false,
      message: `[FAIL] :: Serialization failed Byte-0 alignment invariant`,
    };
  }

  const tmpPath = `${filePath}.tmp_${Date.now()}`;
  fs.writeFileSync(tmpPath, updatedYaml, "utf8");
  fs.renameSync(tmpPath, filePath);

  return {
    success: true,
    targetLocationId,
    targetFilePath: filePath,
    alreadyLinked: false,
    message: `[OK] :: Appended ${sourceLocationId} to ${targetLocationId} connected_locations`,
  };
}
