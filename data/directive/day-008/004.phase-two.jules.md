/**
 * 09.entity.unit - Entity Slug Validation Schema & Sanitizer
 * Enforces strict POSIX kebab-case formatting and domain prefix anchoring.
 */

import { z } from 'zod';

export const ENTITY_SLUG_MIN_LENGTH = 3;
export const ENTITY_SLUG_MAX_LENGTH = 64;
export const ENTITY_SLUG_REGEX = /^(?:char|loc|item|grv|law)_[a-z0-9]+(?:-[a-z0-9]+)*$/;

export const EntitySlugSchema = z
    .string()
    .min(ENTITY_SLUG_MIN_LENGTH + 5, {
        message: `Entity slug must be at least ${ENTITY_SLUG_MIN_LENGTH + 5} characters including domain prefix.`,
    })
    .max(ENTITY_SLUG_MAX_LENGTH, {
        message: `Entity slug must not exceed ${ENTITY_SLUG_MAX_LENGTH} characters.`,
    })
    .regex(ENTITY_SLUG_REGEX, {
        message: 'Entity slug must start with a valid domain prefix (char_, loc_, item_, grv_, law_) followed by lowercase alphanumeric kebab-case words.',
    });

export type EntitySlug = z.infer<typeof EntitySlugSchema>;

/**
 * Normalizes a raw entity title or ID string into a strict POSIX kebab-case slug with domain prefix.
 */
export const sanitizeEntitySlug = (
    rawInput: string,
    prefix: 'char_' | 'loc_' | 'item_' | 'grv_' | 'law_',
): string => {
    if (!rawInput || typeof rawInput !== 'string') {
        return `${prefix}draft_01`;
    }

    // 1. Strip ANSI escape codes
    let cleaned = rawInput.replace(/\x1b\[[0-9;]*[a-zA-Z]/g, '');

    // 2. Downcase to ASCII lowercase and trim whitespace
    cleaned = cleaned.toLowerCase().trim();

    // 3. Replace underscores, slashes, backslashes, and colons with hyphens
    cleaned = cleaned.replace(/[\s_/\\:]+/g, '-');

    // 4. Strip any character outside [a-z0-9-]
    cleaned = cleaned.replace(/[^a-z0-9-]/g, '');

    // 5. Collapse consecutive hyphens into a single hyphen
    cleaned = cleaned.replace(/-+/g, '-');

    // 6. Trim leading and trailing hyphens
    cleaned = cleaned.replace(/^-+|-+$/g, '');

    // If prefix is already present anywhere or at start, clean up duplicate prefixes
    const cleanPrefix = prefix.replace('_', '');
    if (cleaned.startsWith(`${cleanPrefix}_`)) {
        cleaned = cleaned.slice(cleanPrefix.length + 1);
    } else if (cleaned.startsWith(prefix)) {
        cleaned = cleaned.slice(prefix.length);
    }

    if (!cleaned || cleaned.length === 0) {
        cleaned = 'draft-01';
    }

    return `${prefix}${cleaned}`;
};

/**
 * 09.entity.unit - Entity Slug Sanitizer Buzzer
 * Functional validation wrapper for Task 1.3.
 */

import {
    EntitySlugSchema,
    sanitizeEntitySlug,
    ENTITY_SLUG_MIN_LENGTH,
    ENTITY_SLUG_MAX_LENGTH,
} from '../val/entitySlug.val.js';
import type { EntitySlugValidationResultBit } from '../fce/entitySlug.bit.js';

export const validateEntitySlug = (
    rawInput: string,
    prefix: 'char_' | 'loc_' | 'item_' | 'grv_' | 'law_',
): EntitySlugValidationResultBit => {
    const sanitized = sanitizeEntitySlug(rawInput, prefix);

    if (sanitized.length <= prefix.length) {
        return {
            valid: false,
            rawInput,
            sanitizedAttempt: sanitized,
            errorCode: 'SLUG_EMPTY',
            message: 'Entity slug candidate is empty after sanitization.',
        };
    }

    if (sanitized.length < ENTITY_SLUG_MIN_LENGTH + prefix.length) {
        return {
            valid: false,
            rawInput,
            sanitizedAttempt: sanitized,
            errorCode: 'SLUG_TOO_SHORT',
            message: `Entity slug '${sanitized}' is shorter than minimum length requirements.`,
        };
    }

    if (sanitized.length > ENTITY_SLUG_MAX_LENGTH) {
        return {
            valid: false,
            rawInput,
            sanitizedAttempt: sanitized,
            errorCode: 'SLUG_TOO_LONG',
            message: `Entity slug '${sanitized}' exceeds maximum length of ${ENTITY_SLUG_MAX_LENGTH}.`,
        };
    }

    const parseResult = EntitySlugSchema.safeParse(sanitized);
    if (!parseResult.success) {
        return {
            valid: false,
            rawInput,
            sanitizedAttempt: sanitized,
            errorCode: 'SLUG_INVALID_FORMAT',
            message: parseResult.error.issues[0]?.message || 'Invalid POSIX entity slug format.',
        };
    }

    return {
        valid: true,
        slug: parseResult.data,
    };
};

export const assertEntitySlug = (
    rawInput: string,
    prefix: 'char_' | 'loc_' | 'item_' | 'grv_' | 'law_',
): string => {
    const result = validateEntitySlug(rawInput, prefix);
    if (!result.valid) {
        throw new Error(`[ENTITY_SLUG_BREACH] ${result.errorCode}: ${result.message}`);
    }
    return result.slug;
};

/**

 * Scans all canonical entity domains in the active series workspace, returning
 * structured catalogs and available foreign-key ID arrays.

export interface SeriesForeignKeyCatalog {
    readonly seriesSlug: string
    readonly characters: readonly string[]
    readonly locations: readonly string[]
    readonly possessions: readonly string[]
    readonly grievances: readonly string[]
    readonly allKnownIds: ReadonlySet<string>
}

export function discoverDomainForeignKeys(
    seriesSlug: string,
    domain: string,
    repoRoot?: string,
): string[] {
    const root = repoRoot ? path.resolve(repoRoot) : resolveRepoRoot()
    const targetDir = path.join(root, 'series', seriesSlug, domain)
    if (!fs.existsSync(targetDir)) {
        return []
    }
    try {
        return fs
            .readdirSync(targetDir)
            .filter(
                (entry) =>
                    entry.endsWith('.md') &&
                    !entry.startsWith('.') &&
                    !entry.startsWith('_'),
            )
            .map((entry) => entry.replace(/\.md$/i, ''))
            .sort((a, b) => a.localeCompare(b))
    } catch {
        return []
    }
}

export function discoverSeriesForeignKeyCatalog(
    seriesSlug: string,
    repoRoot?: string,
): SeriesForeignKeyCatalog {
    const characters = discoverDomainForeignKeys(seriesSlug, 'characters', repoRoot)
    const locations = discoverDomainForeignKeys(seriesSlug, 'locations', repoRoot)
    const possessions = discoverDomainForeignKeys(seriesSlug, 'possessions', repoRoot)
    const grievances = discoverDomainForeignKeys(seriesSlug, 'grievances', repoRoot)

    const allKnownIds = new Set<string>([
        ...characters,
        ...locations,
        ...possessions,
        ...grievances,
    ])

    return {
        seriesSlug,
        characters,
        locations,
        possessions,
        grievances,
        allKnownIds,
    }
}

            seriesSlug = String(seriesOutcome.src).trim()
        } else {
            seriesSlug = cpy.activeSeries || 'under-the-floorboards'
        }
    }

    cpy.activeSeries = seriesSlug
    emitConsole(
        ste,
        `>> [SCAFFOLD:SERIES] Bound active series context: ${seriesSlug}`,
    )

    // -------------------------------------------------------------------------
    // TASK 1.2: Domain Entity Taxonomy Routing Switchboard
    // -------------------------------------------------------------------------
    let domainDescriptor: DomainEntityDescriptor | null = null

    if (typeof dat.domain === 'string' && dat.domain.trim().length > 0) {
        const rawDomain = dat.domain.trim().toLowerCase()
        const singularMap: Record<string, string> = {
            characters: 'character',
            locations: 'location',
            possessions: 'possession',
            grievances: 'grievance',
            world: 'world',
        }
        domainDescriptor =
            resolveDomainEntityDescriptor(singularMap[rawDomain] || rawDomain) ||
            null
    } else if (isInteractive) {
        let domainOutcome: any = null
        await promptDomainEntityTypeSelector(
            cpy,
            {
                idx: 'scaffold-domain-routing',
                dat: { defaultDomain: 'character' },
                slv: (res: any) => {
                    domainOutcome = res?.mnuBit || res
                },
            },
            ste,
        )

        if (!domainOutcome || domainOutcome.val !== 1) {
            const abortLine =
                '>> [SCAFFOLD:ABORT] Domain taxonomy selection cancelled by operator.'
            emitConsole(ste, abortLine)
            cpy.activeSelection = '[S] ATOMIC ENTITY SCAFFOLDER'
            cpy.lastLifecycleReceipt = '[SCAFFOLD:CANCELLED]'
            cpy.statusMessage = abortLine
            if (bal?.slv) {
                bal.slv({
                    mnuBit: {
                        idx: 'open-scaffold-menu',
                        val: 0,
                        dat: { cancelled: true, receipt: abortLine },
                    },
                })
            }
            return cpy
        }

        domainDescriptor =
            (domainOutcome.dat?.descriptor as DomainEntityDescriptor) ||
            resolveDomainEntityDescriptor(domainOutcome.src)
    } else {
        domainDescriptor = resolveDomainEntityDescriptor('character')
    }

    if (!domainDescriptor) {
        const errLine = `:: [SCAFFOLD:FAIL] Unrecognized entity domain: ${String(dat.domain || '')}`
        emitConsole(ste, errLine)
        cpy.lastLifecycleReceipt = '[SCAFFOLD:FAIL]'
        cpy.statusMessage = errLine
        if (bal?.slv) {
            bal.slv({
                mnuBit: {
                    idx: 'open-scaffold-menu',
                    val: -1,
                    dat: { receipt: errLine },
                },
            })
        }
        return cpy
    }

    const domain = domainDescriptor.directoryName
    const idPrefix = domainDescriptor.idPrefix
    cpy.activeEntityDomain = domainDescriptor.domain
    cpy.activeEntityPrefix = idPrefix
    cpy.activeEntityDirectory = domain

    // -------------------------------------------------------------------------
    // TASK 1.3: Canonical Entity Slug Input Sanitizer
    // -------------------------------------------------------------------------
    const rawEntityId =
        typeof dat.entityId === 'string' && dat.entityId.trim().length > 0
            ? dat.entityId.trim()
            : 'draft_01'
    const entityId = sanitizeEntitySlug(rawEntityId, idPrefix)

    // -------------------------------------------------------------------------
    // TASK 1.4: Live Foreign Key Directory Indexer Catalog
    // -------------------------------------------------------------------------
    const foreignKeyCatalog = discoverSeriesForeignKeyCatalog(
        seriesSlug,
        dat.repoRoot,
    )
    emitConsole(
        ste,
        `>> [CATALOG:INDEX] Series=${seriesSlug} Characters=${foreignKeyCatalog.characters.length} Locations=${foreignKeyCatalog.locations.length} Possessions=${foreignKeyCatalog.possessions.length} Grievances=${foreignKeyCatalog.grievances.length}`,
    )

    emitConsole(
        ste,
        `>> [SCAFFOLD:INIT] Domain=${domain} SanitizedID=${entityId}`,
    )

    let entityBit: any = null
    if (ste && typeof ste.hunt === 'function') {
        await ste.hunt(ActEnt.UPDATE_ENTITY, {
            idx: 'menu-scaffold-entity',
            src: entityId,
            dat: {
                seriesSlug,
                domain,
                idPrefix,
                entityId,
                frontmatter: dat.frontmatter || {
                    id: entityId,
                    type: domainDescriptor.domain,
                },
                proseBody:
                    dat.proseBody || '## Wants (surface)\nSurvive the shift.\n',
                repoRoot: dat.repoRoot,
                foreignKeyCatalog,
            },
            slv: (res: any) => {
                entityBit = res?.entBit || res
            },
        })
    }

    const ok = !entityBit || entityBit.val !== -1
    const receiptLine = ok
        ? `>> [SCAFFOLD:OK] Byte-0 dossier validated for ${entityId} in ${seriesSlug}/${domain}`
        : `:: [SCAFFOLD:FAIL] Pre-flight check rejected ${entityId}`

    emitConsole(ste, receiptLine)
    cpy.activeSelection = '[S] ATOMIC ENTITY SCAFFOLDER'
    cpy.lastLifecycleReceipt = ok ? '[SCAFFOLD:OK]' : '[SCAFFOLD:FAIL]'
    cpy.statusMessage = receiptLine
    if (!Array.isArray(cpy.lastTelemetryReport)) {
        cpy.lastTelemetryReport = []
    }
    cpy.lastTelemetryReport.push(receiptLine)

    if (bal?.slv) {
        bal.slv({
            mnuBit: {
                idx: 'open-scaffold-menu',
                val: ok ? 1 : 0,
                src: entityId,
                dat: {
                    receipt: receiptLine,
                    seriesSlug,
                    domain,
                    idPrefix,
                    foreignKeyCatalog,
                    entityBit,
                },
            },
        })
    }
    return cpy
}

            expect(sanitizeEntitySlug('---batman---', 'char_')).toBe('char_batman')
