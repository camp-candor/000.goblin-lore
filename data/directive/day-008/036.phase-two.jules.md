```typescript
import * as fs from 'node:fs'
import * as path from 'node:path'
import { TargetModel } from '../target.model.js'
import type TargetBit from '../fce/target.bit.js'
import type {
  FleetTargetDescriptor,
  FleetTargetKey,
} from '../fce/target.interface.js'
import type State from '../../99.core/state.js'
import { resolveRepoRoot } from '../../00.lore.unit/buz/lore.buzz.js'

const UPDATE_CONSOLE = '[Console action] Update Console'

function sanitizeAscii(raw: string): string {
  return String(raw || '')
    .replace(/[^\x20-\x7E]/g, '')
    .trim()
}

function emitConsole(ste: State | undefined, line: string): void {
  const clean = sanitizeAscii(line)
  const lib = (globalThis as any).LIBRARY || (global as any).LIBRARY
  if (lib && typeof lib.hunt === 'function') {
    lib.hunt(UPDATE_CONSOLE, { idx: 'cns00', src: clean }).catch(() => {})
    return
  }
  if (ste && typeof ste.hunt === 'function') {
    ste.hunt(UPDATE_CONSOLE, { idx: 'cns00', src: clean }).catch(() => {})
  }
}

function sanitizeSeriesSlug(raw: string | undefined): string {
  const cleaned = String(raw || '')
    .trim()
    .toLowerCase()
    .replace(/[^a-z0-9-]+/g, '-')
    .replace(/^-+|-+$/g, '')
  return cleaned.length > 0 ? cleaned : 'default-series'
}

function resolveSiblingTarget(
  targetKey: FleetTargetKey,
  repoName: string,
  siblingParentDir: string,
  explicitOverride: string | undefined,
  manifestRelativePath: string,
): FleetTargetDescriptor {
  const resolvedPath =
    typeof explicitOverride === 'string' && explicitOverride.trim().length > 0
      ? path.resolve(explicitOverride.trim())
      : path.join(siblingParentDir, repoName)

  let existsOnDisk = false
  try {
    existsOnDisk =
      fs.existsSync(resolvedPath) && fs.statSync(resolvedPath).isDirectory()
  } catch {
    existsOnDisk = false
  }

  return {
    targetKey,
    repoName,
    resolvedPath,
    existsOnDisk,
    manifestDestination: path.join(resolvedPath, manifestRelativePath),
  }
}

export const initTarget = (
  cpy: TargetModel,
  bal: TargetBit,
  _ste?: State,
): TargetModel => {
  cpy.idx = bal?.idx || '34.target'
  if (bal?.slv) {
    bal.slv({ tgtBit: { idx: 'init-target', val: 0 } })
  }
  return cpy
}

export const updateTarget = (
  cpy: TargetModel,
  bal: TargetBit,
  _ste?: State,
): TargetModel => {
  if (bal?.slv) {
    bal.slv({ tgtBit: { idx: 'update-target', val: 0 } })
  }
  return cpy
}

export const discoverFleetTargets = (
  cpy: TargetModel,
  bal: TargetBit,
  ste?: State,
): TargetModel => {
  const dat = (bal?.dat || {}) as Record<string, any>
  const seriesSlug = sanitizeSeriesSlug(dat.seriesSlug || bal?.src)
  const repoRoot =
    typeof dat.repoRoot === 'string' && dat.repoRoot.trim().length > 0
      ? path.resolve(dat.repoRoot.trim())
      : resolveRepoRoot()

  const siblingParentDir = path.resolve(repoRoot, '..')

  emitConsole(
    ste,
    `>> [TARGET:SCAN] Probing sibling fleet targets for series [${seriesSlug}]...`,
  )

  const engineTarget = resolveSiblingTarget(
    'ENGINE',
    '000.series-sower-engine',
    siblingParentDir,
    dat.targetEngineRoot,
    path.join('compiled', seriesSlug, 'bible-state.json'),
  )

  const spatialTarget = resolveSiblingTarget(
    'SPATIAL',
    '000.goblin-spatial',
    siblingParentDir,
    dat.targetSpatialRoot,
    path.join('manifests', `${seriesSlug}-topology.json`),
  )

  const conductorTarget = resolveSiblingTarget(
    'CONDUCTOR',
    '000.series-sower-conductor',
    siblingParentDir,
    dat.targetConductorRoot,
    path.join('compiled', seriesSlug, 'bible-state.json'),
  )

  const targets: Record<string, FleetTargetDescriptor> = {
    ENGINE: engineTarget,
    SPATIAL: spatialTarget,
    CONDUCTOR: conductorTarget,
  }

  const descriptorList: FleetTargetDescriptor[] = [
    engineTarget,
    spatialTarget,
    conductorTarget,
  ]

  const mountedCount = descriptorList.filter((t) => t.existsOnDisk).length

  for (const descriptor of descriptorList) {
    const statusTag = descriptor.existsOnDisk ? '[ONLINE]' : '[MISSING]'
    emitConsole(
      ste,
      `>> [TARGET:${descriptor.targetKey}] ${statusTag} ${descriptor.repoName} -> ${descriptor.manifestDestination.replace(/\\/g, '/')}`,
    )
  }

  emitConsole(
    ste,
    `>> [TARGET:SCAN] [OK] Discovered ${mountedCount}/3 active fleet target(s) for [${seriesSlug}].`,
  )

  const recordedBit: TargetBit = {
    idx: `fleet-targets-${seriesSlug}`,
    src: seriesSlug,
    val: mountedCount,
    lst: descriptorList,
    targets,
  }

  cpy.lastDiscoveredSeriesSlug = seriesSlug
  cpy.lastFleetTargets = targets
  cpy.targetBits[recordedBit.idx] = recordedBit
  cpy.targetBitList.push(recordedBit)

  if (bal?.slv) {
    bal.slv({
      tgtBit: recordedBit,
    })
  }

  return cpy
}
```

```typescript
import * as Act from './target.action.js'
import { TargetModel } from './target.model.js'
import * as Buzz from './target.buzzer.js'

const clone = <T>(val: T): T => cloneDeep(val, true)

export function reducer(
  model: TargetModel = new TargetModel(),
  act: Act.Actions,
  state?: State,
) {
  switch (act.type) {
    case Act.INIT_TARGET:

    case Act.DISCOVER_FLEET_TARGETS:
      return Buzz.discoverFleetTargets(clone(model), act.bale, state)

    default:
      return model
  }
}
```
