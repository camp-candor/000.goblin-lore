            `>> [SOMATIC:LIMBS] Available Arms: ${netBps} bp | Conditions: [${conditions.join(', ')}]`,
        )

        return {
            cancelled: false,
            availableArmsBps: netBps,
            conditions,
            traumaBannedVerbs,
        }
    }

    emitConsole(
        ste,
        '>> [SOMATIC:LIMBS] Task 2.1: Select upper-limb capacity in basis points...',
    )

    const limbModal = await promptBlessedChoice({
        title: 'SOMATIC LIMB CAPACITY (BPS)',
        choices: [...LIMB_CAPACITY_CHOICES],
        defaultChoice: LIMB_CAPACITY_CHOICES[0],
        ste,
    })

    if (limbModal.cancelled || limbModal.value === CANCEL_INTERVIEW_LABEL) {
        emitConsole(
            ste,
            '>> [SOMATIC:LIMBS] [WARN] Limb capacity selection cancelled by operator.',
        )
        return {
            cancelled: true,
            availableArmsBps: 20000,
            conditions: [],
            traumaBannedVerbs: [],
        }
    }

    let baseBps = 20000
    if (limbModal.value.startsWith('[2]')) {
        baseBps = 15000
    } else if (limbModal.value.startsWith('[3]')) {
        baseBps = 10000
    } else if (limbModal.value.startsWith('[4]')) {
        baseBps = 0
    }

    const traumaModal = await promptBlessedChoice({
        title: 'ACTIVE SOMATIC TRAUMA CONDITION',
        choices: [...TRAUMA_CONDITION_CHOICES],
        defaultChoice: TRAUMA_CONDITION_CHOICES[0],
        ste,
    })

    const conditions: string[] = []
    if (!traumaModal.cancelled && !traumaModal.value.startsWith('[1]')) {
        const tag = traumaModal.value.split(' ')[1]?.trim()
        if (tag) {
            conditions.push(tag)
        }
    }

    const penaltyBps = conditions.length * 10000
    const netBps = Math.max(0, baseBps - penaltyBps)
    const traumaBannedVerbs =
        netBps < 20000 ? [...BIMANUAL_RESTRICTED_VERBS] : []

    emitConsole(
        ste,
        `>> [SOMATIC:LIMBS] [OK] Locked available_arms_bps=${netBps} | BannedBimanual=${traumaBannedVerbs.length}`,
    )

    return {
        cancelled: false,
        availableArmsBps: netBps,
        conditions,
        traumaBannedVerbs,
    }
}

interface IndexedPossessionSpec {
    readonly itemId: string
    readonly occupiesHandsBps: number
    readonly massGrams: number
    readonly assignedOwner: string | null
}

function loadPossessionMetadataIndex(
    seriesSlug: string,
    itemIds: readonly string[],
    repoRoot?: string,
): Record<string, IndexedPossessionSpec> {
    const root = repoRoot ? path.resolve(repoRoot) : resolveRepoRoot()
    const possessionsDir = path.join(root, 'series', seriesSlug, 'possessions')
    const index: Record<string, IndexedPossessionSpec> = {}

    for (const itemId of itemIds) {
        const filePath = path.join(possessionsDir, `${itemId}.md`)
        let occupiesHandsBps = 10000
        let massGrams = 500
        let assignedOwner: string | null = null

        if (fs.existsSync(filePath)) {
            try {
                const raw = fs.readFileSync(filePath, 'utf8')
                const parsed = matter(raw)
                const fm = (parsed.data || {}) as Record<string, any>
                if (typeof fm.occupies_hands_bps === 'number') {
                    occupiesHandsBps = Math.max(
                        0,
                        Math.min(20000, Math.trunc(fm.occupies_hands_bps)),
                    )
                } else if (typeof fm.occupies_hands === 'number') {
                    occupiesHandsBps = Math.max(
                        0,
                        Math.min(20000, Math.trunc(fm.occupies_hands * 10000)),
                    )
                }
                if (typeof fm.mass_grams === 'number') {
                    massGrams = Math.max(1, Math.trunc(fm.mass_grams))
                }
                const ownerVal = fm.holder_id || fm.owner
                if (typeof ownerVal === 'string' && ownerVal.trim().length > 0) {
                    assignedOwner = ownerVal.trim()
                }
            } catch {
                // Keep safe defaults if frontmatter is sparse
            }
        }

        index[itemId] = {
            itemId,
            occupiesHandsBps,
            massGrams,
            assignedOwner,
        }
    }

    return index
}

/**
 * Task 2.4: Foreign Key Equipment Slot Binding Modal
 * Discovers verified possessions from series/<slug>/possessions/*.md, enforces
 * Hand Capacity SAT and 4-tier partition ceilings at T_0 = $0.00, and queues
 * reciprocal custody links.
 */
export const promptEquipmentSlotBindings = async (options: {
    seriesSlug: string
    characterId: string
    availableArmsBps: number
    catalog?: SeriesForeignKeyCatalog
    repoRoot?: string
    ste?: State
    presetBindings?: Array<{
        itemId: string
        partition: 'worn' | 'held' | 'carried'
        occupiesHandsBps?: number
    }>
}): Promise<CharacterEquipSlotBindingResult> => {
    const {
        seriesSlug,
        characterId,
        availableArmsBps,
        catalog,
        repoRoot,
        ste,
        presetBindings,
    } = options
    const lib = resolveLibraryBus()
    const isInteractive = Boolean(lib && typeof lib.hunt === 'function')

    const worn: string[] = []
    const held: string[] = []
    const carried: string[] = []
    const links: ReciprocalPossessionLink[] = []

    const availableItemIds = catalog?.possessions || []
    const possessionMeta = loadPossessionMetadataIndex(
        seriesSlug,
        availableItemIds,
        repoRoot,
    )

    const tryBindItem = (
        itemId: string,
        partition: 'worn' | 'held' | 'carried',
        explicitHandsBps?: number,
    ): { ok: boolean; reason?: string } => {
        if (
            worn.includes(itemId) ||
            held.includes(itemId) ||
            carried.includes(itemId)
        ) {
            return {
                ok: false,
                reason: `DUPLICATE_ITEM_LEASE: ${itemId} already bound on ${characterId}`,
            }
        }

        const spec = possessionMeta[itemId] || {
            itemId,
            occupiesHandsBps: explicitHandsBps ?? 10000,
            massGrams: 500,
            assignedOwner: null,
        }
        const itemHandsBps =
            typeof explicitHandsBps === 'number'
                ? explicitHandsBps
                : spec.occupiesHandsBps

        if (partition === 'held') {
            if (held.length >= 2) {
                return {
                    ok: false,
                    reason: 'PARTITION_CEILING_BREACH: held partition max 2 items',
                }
            }
            const projectedHeldItems = [
                ...held.map((id) => ({
                    item_id: id,
                    occupies_hands_bps:
                        possessionMeta[id]?.occupiesHandsBps ?? 10000,
                })),
                {
                    item_id: itemId,
                    occupies_hands_bps: itemHandsBps,
                },
            ]
            const satAudit = auditEntityCapacityPreFlight(
                { available_arms_bps: availableArmsBps },
                projectedHeldItems,
            )
            if (!satAudit.valid) {
                return {
                    ok: false,
                    reason: `HAND_CAPACITY_BREACH: Demanded ${satAudit.total_required_hands_bps} bp > Available ${availableArmsBps} bp`,
                }
            }
            held.push(itemId)
        } else if (partition === 'worn') {
            if (worn.length >= 4) {
                return {
                    ok: false,
                    reason: 'PARTITION_CEILING_BREACH: worn partition max 4 items',
                }
            }
            worn.push(itemId)
        } else {
            if (carried.length >= 4) {
                return {
                    ok: false,
                    reason: 'PARTITION_CEILING_BREACH: carried partition max 4 items',
                }
            }
            carried.push(itemId)
        }

        links.push({
            itemId,
            ownerId: characterId,
            partition,
        })
        return { ok: true }
    }

    // Programmatic / Headless preset bindings path
    if (Array.isArray(presetBindings) || !isInteractive) {
        for (const binding of presetBindings || []) {
            const res = tryBindItem(
                binding.itemId,
                binding.partition,
                binding.occupiesHandsBps,
            )
            if (!res.ok) {
                emitConsole(ste, `:: [SAT:BREACH] ${res.reason}`)
            }
        }
        const totalHeldHandsBps = held.reduce(
            (sum, id) => sum + (possessionMeta[id]?.occupiesHandsBps ?? 10000),
            0,
        )
        emitConsole(
            ste,
            `>> [GEAR:RESOLVE] Bound ${links.length} item(s) | Held=${totalHeldHandsBps}/${availableArmsBps} bp [OK]`,
        )
        return {
            cancelled: false,
            worn,
            held,
            carried,
            cached: {},
            totalHeldHandsBps,
            reciprocalPossessionLinks: links,
        }
    }

    // Interactive Blessed TUI loop
    const unassignedItems = availableItemIds.filter(
        (id) =>
            !possessionMeta[id]?.assignedOwner ||
            possessionMeta[id]?.assignedOwner === characterId,
    )

    if (unassignedItems.length === 0) {
        emitConsole(
            ste,
            `>> [GEAR:RESOLVE] No unassigned possessions in series/${seriesSlug}/possessions/. Skipping slot binding.`,
        )
        return {
            cancelled: false,
            worn: [],
            held: [],
            carried: [],
            cached: {},
            totalHeldHandsBps: 0,
            reciprocalPossessionLinks: [],
        }
    }

    emitConsole(
        ste,
        `>> [GEAR:BIND] Task 2.4: ${unassignedItems.length} possession(s) available | Free Arms=${availableArmsBps} bp`,
    )

    while (true) {
        const currentHeldBps = held.reduce(
            (sum, id) => sum + (possessionMeta[id]?.occupiesHandsBps ?? 10000),
            0,
        )
        const remainingArmsBps = Math.max(0, availableArmsBps - currentHeldBps)

        const remainingCandidates = unassignedItems.filter(
            (id) =>
                !worn.includes(id) &&
                !held.includes(id) &&
                !carried.includes(id),
        )

        if (remainingCandidates.length === 0) {
            break
        }

        const itemChoices = remainingCandidates.map((id) => {
            const reqBps = possessionMeta[id]?.occupiesHandsBps ?? 10000
            const satTag =
                reqBps > remainingArmsBps
                    ? `[HELD_LOCKED: ${reqBps}bp > ${remainingArmsBps}bp]`
                    : `[${reqBps}bp]`
            return `${id} ${satTag}`
        })

        const itemModal = await promptBlessedChoice({
            title: `SELECT EQUIPMENT (FREE ARMS: ${remainingArmsBps} BP)`,
            choices: [
                ...itemChoices,
                FINISH_EQUIPMENT_LABEL,
                CANCEL_INTERVIEW_LABEL,
            ],
            defaultChoice: FINISH_EQUIPMENT_LABEL,
            ste,
        })

        if (itemModal.cancelled || itemModal.value === CANCEL_INTERVIEW_LABEL) {
            emitConsole(
                ste,
                '>> [GEAR:BIND] [WARN] Equipment binding cancelled by operator.',
            )
            return {
                cancelled: true,
                worn: [],
                held: [],
                carried: [],
                cached: {},
                totalHeldHandsBps: 0,
                reciprocalPossessionLinks: [],
            }
        }

        if (itemModal.value === FINISH_EQUIPMENT_LABEL) {
            break
        }

        const chosenItemId = itemModal.value.split(' ')[0]?.trim()
        if (!chosenItemId) {
            continue
        }

        const slotModal = await promptBlessedChoice({
            title: `ASSIGN SLOT FOR ${chosenItemId}`,
            choices: [...EQUIP_PARTITION_CHOICES],
            defaultChoice: EQUIP_PARTITION_CHOICES[0],
            ste,
        })

        if (slotModal.cancelled || slotModal.value.startsWith('[<]')) {
            continue
        }

        let targetPartition: 'worn' | 'held' | 'carried' = 'held'
        if (slotModal.value.startsWith('[2]')) {
            targetPartition = 'worn'
        } else if (slotModal.value.startsWith('[3]')) {
            targetPartition = 'carried'
        }

        const bindAttempt = tryBindItem(chosenItemId, targetPartition)
        if (!bindAttempt.ok) {
            emitConsole(ste, `:: [SAT:BREACH] ${bindAttempt.reason}`)
        } else {
            emitConsole(
                ste,
                `>> [GEAR:BOUND] ${chosenItemId} -> ${targetPartition.toUpperCase()} [OK]`,
            )
        }
    }

    const totalHeldHandsBps = held.reduce(
        (sum, id) => sum + (possessionMeta[id]?.occupiesHandsBps ?? 10000),
        0,
    )

    emitConsole(
        ste,
        `>> [SAT GATE] Held: ${totalHeldHandsBps} bps <= Arms: ${availableArmsBps} bps [PASS / T0 = $0.00]`,
    )

    return {
        cancelled: false,
        worn,
        held,
        carried,
        cached: {},
        totalHeldHandsBps,
        reciprocalPossessionLinks: links,
    }
}

/**
 * Task 2.5: Epistemic Horizon Refusal Vector Collector
 */
export const promptEpistemicRefusalVector = async (options: {
    catalog?: SeriesForeignKeyCatalog
    ste?: State
    overrides?: {
        locationId?: string
        epistemicHorizon?: EpistemicHorizonCode
        escalationCeiling?: number
        refusalStrainThresholdBp?: number
    }
}): Promise<EpistemicRefusalSelectionResult> => {
    const { catalog, ste, overrides } = options
    const lib = resolveLibraryBus()
    const isInteractive = Boolean(lib && typeof lib.hunt === 'function')

    const knownLocations = catalog?.locations || []
    const fallbackLocation = knownLocations[0] || 'loc_unbound_origin'

    if (overrides || !isInteractive) {
        const locationId =
            typeof overrides?.locationId === 'string' &&
            overrides.locationId.trim().length > 0
                ? overrides.locationId.trim()
                : fallbackLocation
        const epistemicHorizon: EpistemicHorizonCode =
            overrides?.epistemicHorizon || 'FACTUAL_WITNESS'
        const escalationCeiling = Math.max(
            1,
            Math.min(5, Math.trunc(overrides?.escalationCeiling ?? 4)),
        )
        const refusalStrainThresholdBp = Math.max(
            0,
            Math.min(
                10000,
                Math.trunc(overrides?.refusalStrainThresholdBp ?? 8500),
            ),
        )

        emitConsole(
            ste,
            `>> [EPISTEMIC:VECTOR] Location=${locationId} Horizon=${epistemicHorizon} Ceiling=${escalationCeiling} Refusal=${refusalStrainThresholdBp}bp [OK]`,
        )

        return {
            cancelled: false,
            locationId,
            epistemicHorizon,
            escalationCeiling,
            refusalStrainThresholdBp,
        }
    }

    emitConsole(
        ste,
        '>> [EPISTEMIC:VECTOR] Task 2.5: Select origin location, epistemic horizon, and refusal threshold...',
    )

    // 1. Origin Location Foreign Key Selector
    let chosenLocation = fallbackLocation
    if (knownLocations.length > 0) {
        const locModal = await promptBlessedChoice({
            title: 'STARTING LOCATION FOREIGN KEY',
            choices: [...knownLocations, CANCEL_INTERVIEW_LABEL],
            defaultChoice: knownLocations[0],
            ste,
        })
        if (locModal.cancelled || locModal.value === CANCEL_INTERVIEW_LABEL) {
            return {
                cancelled: true,
                locationId: fallbackLocation,
                epistemicHorizon: 'FACTUAL_WITNESS',
                escalationCeiling: 4,
                refusalStrainThresholdBp: 8500,
            }
        }
        chosenLocation = locModal.value.trim()
    }

    // 2. Epistemic Horizon Selector
    const horizonModal = await promptBlessedChoice({
        title: 'EPISTEMIC KNOWLEDGE HORIZON',
        choices: [...EPISTEMIC_HORIZON_CHOICES],
        defaultChoice: EPISTEMIC_HORIZON_CHOICES[0],
        ste,
    })
    if (
        horizonModal.cancelled ||
        horizonModal.value === CANCEL_INTERVIEW_LABEL
    ) {
        return {
            cancelled: true,
            locationId: chosenLocation,
            epistemicHorizon: 'FACTUAL_WITNESS',
            escalationCeiling: 4,
            refusalStrainThresholdBp: 8500,
        }
    }

    const parsedHorizon = (horizonModal.value.split(' ')[1]?.trim() ||
        'FACTUAL_WITNESS') as EpistemicHorizonCode

    // 3. Escalation Ceiling Selector (1..5)
    const ceilingModal = await promptBlessedChoice({
        title: 'CONVERSATIONAL ESCALATION CEILING (1-5)',
        choices: [...ESCALATION_CEILING_CHOICES],
        defaultChoice: ESCALATION_CEILING_CHOICES[3],
        ste,
    })
    let escalationCeiling = 4
    if (!ceilingModal.cancelled) {
        const match = ceilingModal.value.match(/^\[(\d)\]/)
        if (match) {
            escalationCeiling = Math.max(
                1,
                Math.min(5, Number.parseInt(match[1], 10)),
            )
        }
    }

    // 4. Refusal Strain Threshold Selector (0..10,000 bp)
    const refusalModal = await promptBlessedChoice({
        title: 'REFUSAL VECTOR STRAIN THRESHOLD (BPS)',
        choices: [...REFUSAL_STRAIN_CHOICES],
        defaultChoice: REFUSAL_STRAIN_CHOICES[1],
        ste,
    })
    let refusalStrainThresholdBp = 8500
    if (!refusalModal.cancelled) {
        if (refusalModal.value.startsWith('[1]')) {
            refusalStrainThresholdBp = 9500
        } else if (refusalModal.value.startsWith('[2]')) {
            refusalStrainThresholdBp = 8500
        } else if (refusalModal.value.startsWith('[3]')) {
            refusalStrainThresholdBp = 6500
        } else if (refusalModal.value.startsWith('[4]')) {
            refusalStrainThresholdBp = 4000
        }
    }

    emitConsole(
        ste,
        `>> [EPISTEMIC:VECTOR] [OK] Locked Location=${chosenLocation} | Horizon=${parsedHorizon} | Ceiling=${escalationCeiling} | Refusal=${refusalStrainThresholdBp}bp`,
    )

    return {
        cancelled: false,
        locationId: chosenLocation,
        epistemicHorizon: parsedHorizon,
        escalationCeiling,
        refusalStrainThresholdBp,
    }
}

    entityId: string,
    ste?: State,
    overrides?: {
        seriesSlug?: string
        repoRoot?: string
        catalog?: SeriesForeignKeyCatalog
        availableArmsBps?: number
        conditions?: string[]
        locomotionBaseline?: LocomotionBaseline
        stats?: Partial<CombatAttributeStats>
        presetBindings?: Array<{
            itemId: string
            partition: 'worn' | 'held' | 'carried'
            occupiesHandsBps?: number
        }>
        locationId?: string
        epistemicHorizon?: EpistemicHorizonCode
        escalationCeiling?: number
        refusalStrainThresholdBp?: number
    },
): Promise<CharacterInterviewOutcome> => {
    const lib = resolveLibraryBus()
    const isInteractive = Boolean(lib && typeof lib.hunt === 'function')
    const activeSeriesSlug =
        overrides?.seriesSlug || cpy.activeSeries || 'under-the-floorboards'

    // Step 1 (Task 2.1): Somatic Limb Capacity Prompt Sequence
    const limbResult = await promptSomaticLimbCapacity(ste, {
        availableArmsBps: overrides?.availableArmsBps,
        conditions: overrides?.conditions,
    })
    if (limbResult.cancelled) {
        return { cancelled: true }
    }

    // Step 2 (Task 2.2): Locomotion Baseline Selection Modal
    let profile: LocomotionProfile
    if (overrides?.locomotionBaseline || !isInteractive) {
        profile = resolveLocomotionProfile(
            overrides?.locomotionBaseline || 'bipedal_standard',
        )
    } else {
        let locoOutcome: any = null
        await promptLocomotionChoice(
            cpy,
            {
                idx: 'character-interview-locomotion',
                slv: (res: any) => {
                    locoOutcome = res?.mnuBit || res
                },
            },
            ste,
        )
        if (!locoOutcome || locoOutcome.val !== 1) {
            return { cancelled: true }
        }
        profile =
            (locoOutcome.dat?.profile as LocomotionProfile) ||
            resolveLocomotionProfile(
                (locoOutcome.src as LocomotionBaseline) || 'bipedal_standard',
            )
    }

    // Clamp arms to 0 bps if locomotion baseline forbids upper-limb manipulation
    const finalArmsBps =
        profile.default_arms_bps === 0 ? 0 : limbResult.availableArmsBps

    // Merge and deduplicate negative motor masks
    const mergedBannedVerbs = Array.from(
        new Set<string>([
            ...profile.default_banned_verbs,
            ...limbResult.traumaBannedVerbs,
        ]),
    ).sort((a, b) => a.localeCompare(b))

    // Step 3 (Task 2.3): Opposed Two-D-Six Attribute Input Collector
    const statsResult = await promptTwoDSixAttributes(ste, overrides?.stats)
    if (statsResult.cancelled) {
        return { cancelled: true }
    }

    // Step 4 (Task 2.4): Foreign Key Equipment Slot Binding Modal
    const equipResult = await promptEquipmentSlotBindings({
        seriesSlug: activeSeriesSlug,
        characterId: entityId,
        availableArmsBps: finalArmsBps,
        catalog: overrides?.catalog,
        repoRoot: overrides?.repoRoot,
        ste,
        presetBindings: overrides?.presetBindings,
    })
    if (equipResult.cancelled) {
        return { cancelled: true }
    }

    // Step 5 (Task 2.5): Epistemic Horizon Refusal Vector Collector
    const hasEpistemicOverrides =
        overrides?.locationId !== undefined ||
        overrides?.epistemicHorizon !== undefined ||
        overrides?.escalationCeiling !== undefined ||
        overrides?.refusalStrainThresholdBp !== undefined

    const epistemicResult = await promptEpistemicRefusalVector({
        catalog: overrides?.catalog,
        ste,
        overrides: hasEpistemicOverrides
            ? {
                  locationId: overrides?.locationId,
                  epistemicHorizon: overrides?.epistemicHorizon,
                  escalationCeiling: overrides?.escalationCeiling,
                  refusalStrainThresholdBp: overrides?.refusalStrainThresholdBp,
              }
            : undefined,
    })
    if (epistemicResult.cancelled) {
        return { cancelled: true }
    }

    const frontmatter: CharacterMachineRailPayload = {
        id: entityId,
        type: 'character',
        location_id: epistemicResult.locationId,
        somatic: {
            locomotion_baseline: profile.baseline,
            max_velocity_bps: profile.max_velocity_bps,
            available_arms_bps: finalArmsBps,
            conditions: limbResult.conditions,
            banned_kinetic_verbs: mergedBannedVerbs,
        },
        stats: statsResult.stats,
        logistical: {
            worn: equipResult.worn,
            held: equipResult.held,
            carried: equipResult.carried,
            cached: equipResult.cached,
        },
        epistemic: {
            horizon: epistemicResult.epistemicHorizon,
            escalation_ceiling: epistemicResult.escalationCeiling,
            refusal_strain_threshold_bp:
                epistemicResult.refusalStrainThresholdBp,
        },
    }

    emitConsole(
        ste,
        `>> [CHARACTER:INTERVIEW] [OK] Assembled full Machine Rail for ${entityId} (Arms=${finalArmsBps}bp, Held=${equipResult.held.length}, Horizon=${epistemicResult.epistemicHorizon})`,
    )

    return {
        cancelled: false,
        frontmatter,
        reciprocalPossessionLinks: equipResult.reciprocalPossessionLinks,
    }
}
