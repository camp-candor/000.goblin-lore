  const packageRoot = resolveLorePackageDirectory(
    packageRootInput || process.cwd(),
  )
  const beePath = path.join(packageRoot, 'BEE.ts')
  const unhookedUnits: string[] = []
  const missingWiringDiagnostics: string[] = []

  const manifestReport = validateLoreResetManifest(LORE_RESET_MANIFEST)
  if (!manifestReport.valid) {
    missingWiringDiagnostics.push(...manifestReport.errors)
  }

  let mountedUnits: string[] = []
  if (fs.existsSync(beePath)) {
    const beeSrc = fs.readFileSync(beePath, 'utf8')
    const matches = [...beeSrc.matchAll(/\.\/(\d{2}\.[a-z0-9-]+\.unit)\//g)].map(
      (m) => m[1],
    )
    mountedUnits = [...new Set(matches)].sort((a, b) => a.localeCompare(b))
  } else {
    missingWiringDiagnostics.push(`Missing BEE.ts manifest at ${beePath}`)
  }

  const registryMap = new Map(
    LORE_UNIT_RESET_REGISTRY.map((entry) => [entry.unitFolder, entry]),
  )

  for (const unitFolder of mountedUnits) {
    const descriptor = registryMap.get(unitFolder)
    if (!descriptor) {
      unhookedUnits.push(unitFolder)
      missingWiringDiagnostics.push(
        `Unregistered unit in BEE.ts: ${unitFolder} missing from LORE_UNIT_RESET_REGISTRY`,
      )
      continue
    }

    if (descriptor.classification === 'STATEFUL_RESETTABLE') {
      const unitSlug = unitFolder.replace(/^\d{2}\./, '').replace(/\.unit$/, '')
      const actionPath = path.join(
        packageRoot,
        unitFolder,
        `${unitSlug}.action.ts`,
      )
      const reducePath = path.join(
        packageRoot,
        unitFolder,
        `${unitSlug}.reduce.ts`,
      )
      const expectedConstant = `RESET_${unitSlug.toUpperCase().replace(/-/g, '_')}`

      if (fs.existsSync(actionPath)) {
        const actionSrc = fs.readFileSync(actionPath, 'utf8')
        if (
          !actionSrc.includes(expectedConstant) &&
          descriptor.resetActionConstant &&
          !actionSrc.includes(descriptor.resetActionConstant)
        ) {
          unhookedUnits.push(unitFolder)
          missingWiringDiagnostics.push(
            `Stateful unit ${unitFolder} missing ${expectedConstant} in ${unitSlug}.action.ts`,
          )
        }
      }

      if (fs.existsSync(reducePath)) {
        const reduceSrc = fs.readFileSync(reducePath, 'utf8')
        if (!reduceSrc.includes(`Act.${expectedConstant}`)) {
          unhookedUnits.push(unitFolder)
          missingWiringDiagnostics.push(
            `Stateful unit ${unitFolder} missing Act.${expectedConstant} case in ${unitSlug}.reduce.ts`,
          )
        }
      }
    }
  }

  const uniqueUnhooked = [...new Set(unhookedUnits)]
  const valid =
    mountedUnits.length > 0 &&
    uniqueUnhooked.length === 0 &&
    missingWiringDiagnostics.length === 0

  const receipts: string[] = [
    '>> ============================================================',
    '>> [SENTINEL:COVERAGE] TABULA RASA RESET COVERAGE AUDIT',
    `>> Package Root     : ${packageRoot}`,
    `>> Mounted Units    : ${mountedUnits.length} BEE.ts units scanned`,
    `>> Registry Split   : ${manifestReport.statefulUnitCount} stateful | ${manifestReport.statelessUnitCount} stateless`,
    `>> Unhooked Units   : ${uniqueUnhooked.length === 0 ? 'NONE [OK]' : uniqueUnhooked.join(', ')}`,
    `>> Sentinel Verdict : ${valid ? 'PASS [OK]' : 'FAIL [BREACH]'}`,
    '>> ============================================================',
  ]

  await emitSentinelTelemetry(ste, receipts)

  return {
    valid,
    packageRoot,
    mountedUnitCount: mountedUnits.length,
    statefulUnitCount: manifestReport.statefulUnitCount,
    statelessUnitCount: manifestReport.statelessUnitCount,
    unhookedUnits: uniqueUnhooked,
    missingWiringDiagnostics,
    receipts,
  }
}

export async function verifyRoundTripVoidStateParity(
  sandboxRoot: string,
  ste?: State,
): Promise<LoreVoidStateParityReport> {
  const errors: string[] = []
  const resolvedRoot = path.resolve(sandboxRoot)

  // Step 1: Verify initial Void State
  const initialScan = scanVoidState(resolvedRoot)
  const initialVoidVerified =
    initialScan.isVoidState && initialScan.discoveredSlugs.length === 0
  if (!initialVoidVerified) {
    errors.push('Sandbox did not start in clean Void State')
  }

  // Step 2: Populate synthetic Gotham series, compiled artifacts, errata, SQLite, and models
  const seriesDir = path.join(resolvedRoot, 'series', 'gotham-trinity-crisis')
  const compiledDir = path.join(
    resolvedRoot,
    'compiled',
    'gotham-trinity-crisis',
  )
  const loreDir = path.join(resolvedRoot, 'lore')
  const sqliteDir = path.join(resolvedRoot, 'data', 'sqlite')

  fs.mkdirSync(path.join(seriesDir, 'characters'), { recursive: true })
  fs.mkdirSync(path.join(seriesDir, 'locations'), { recursive: true })
  fs.mkdirSync(path.join(seriesDir, 'possessions'), { recursive: true })
  fs.mkdirSync(path.join(seriesDir, 'grievances'), { recursive: true })
  fs.mkdirSync(compiledDir, { recursive: true })
  fs.mkdirSync(loreDir, { recursive: true })
  fs.mkdirSync(sqliteDir, { recursive: true })

  fs.writeFileSync(
    path.join(seriesDir, 'series.config.json'),
    JSON.stringify({ series_slug: 'gotham-trinity-crisis' }),
    'utf8',
  )
  fs.writeFileSync(
    path.join(seriesDir, 'characters', 'char_wonder_woman.md'),
    '---\nid: char_wonder_woman\n---\n',
    'utf8',
  )
  fs.writeFileSync(
    path.join(seriesDir, 'characters', 'char_batman.md'),
    '---\nid: char_batman\n---\n',
    'utf8',
  )
  fs.writeFileSync(
    path.join(seriesDir, 'characters', 'char_joker.md'),
    '---\nid: char_joker\n---\n',
    'utf8',
  )
  fs.writeFileSync(
    path.join(compiledDir, 'bible-state.json'),
    JSON.stringify({ seriesSlug: 'gotham-trinity-crisis' }),
    'utf8',
  )
  fs.writeFileSync(
    path.join(compiledDir, 'topology.manifest.json'),
    JSON.stringify({ locationId: 'loc_gotham_rooftop' }),
    'utf8',
  )
  fs.writeFileSync(
    path.join(loreDir, 'errata.json'),
    JSON.stringify({ records: [{ id: 'err_01' }] }),
    'utf8',
  )
  fs.writeFileSync(
    path.join(sqliteDir, 'gotham-trinity-crisis.local.db'),
    'sqlite-bytes',
    'utf8',
  )

  const mutatedScan = scanVoidState(resolvedRoot)
  const mutatedActiveVerified =
    !mutatedScan.isVoidState &&
    mutatedScan.discoveredSlugs.includes('gotham-trinity-crisis')
  if (!mutatedActiveVerified) {
    errors.push('Synthetic series mutation failed to activate corpus state')
  }

  // Dirty in-memory models
  const loreModel = new LoreModel()
  loreModel.loreBitList = [{ idx: 'dirty' }]
  const targetModel = new TargetModel()
  targetModel.activeSeriesSlug = 'gotham-trinity-crisis'
  const canonModel = new CanonModel()
  canonModel.factStandings.set('f1', { factId: 'f1' } as any)
  const collectModel = new CollectModel()
  collectModel.caboodleBits = { characters: 0 }
  const dbModel = new DatabaseModel()
  dbModel.activeDbPath = path.join(sqliteDir, 'gotham-trinity-crisis.local.db')
  const statusModel = new StatusModel()
  statusModel.isVoidState = false
  statusModel.dispatchLockEngaged = false
  statusModel.activeBorderColor = 'green'

  const localBus: any = {
    hunt: async (type: string, bale: any) => {
      switch (type) {
        case '[Target action] Reset Target':
          await resetTarget(targetModel, bale, localBus)
          break
        case '[Canon action] Reset Canon':
          await resetCanon(canonModel, bale, localBus)
          break
        case '[Collect action] Reset Collect':
          await resetCollect(collectModel, bale, localBus)
          break
        case '[Database action] Reset Database':
          await resetDatabase(dbModel, bale, localBus)
          break
        case '[Status action] Evaluate Void State':
          await voidStateScan(statusModel, bale, localBus)
          break
      }
      return {}
    },
  }

  // Step 3: Execute Triple-Lock Tabula Rasa Reset
  await executeTabulaRasaLore(
    loreModel,
    {
      idx: 'round-trip-parity-reset',
      src: resolvedRoot,
      dat: {
        repoRoot: resolvedRoot,
        confirmToken1: TABULA_RASA_TOKEN_1,
        confirmToken2: TABULA_RASA_TOKEN_2,
        confirmToken3: TABULA_RASA_TOKEN_3,
      },
    },
    localBus,
  )

  // Step 4: Assert bit-for-bit Void-State parity
  const postScan = scanVoidState(resolvedRoot)
  const postResetVoidVerified =
    postScan.isVoidState && postScan.discoveredSlugs.length === 0
  if (!postResetVoidVerified) {
    errors.push('Post-reset filesystem scan did not return to Void State')
  }

  const zeroOrphanFilesVerified =
    !fs.existsSync(seriesDir) &&
    !fs.existsSync(compiledDir) &&
    !fs.existsSync(path.join(loreDir, 'errata.json')) &&
    !fs.existsSync(path.join(sqliteDir, 'gotham-trinity-crisis.local.db'))
  if (!zeroOrphanFilesVerified) {
    errors.push('Orphan files survived Triple-Lock Tabula Rasa Reset')
  }

  const zeroModelContaminationVerified =
    loreModel.loreBitList.length === 0 &&
    targetModel.activeSeriesSlug === null &&
    canonModel.factStandings.size === 0 &&
    Object.keys(collectModel.caboodleBits).length === 0 &&
    dbModel.activeDbPath === null &&
    statusModel.isVoidState === true &&
    statusModel.dispatchLockEngaged === true &&
    statusModel.activeBorderColor === 'amber' &&
    statusModel.hudLines.some((l) => l.includes(GENESIS_MERKLE_ROOT_64))
  if (!zeroModelContaminationVerified) {
    errors.push('In-memory model contamination survived Tabula Rasa Reset')
  }

  const valid = errors.length === 0
  const receipts: string[] = [
    `>> [SENTINEL:PARITY] Round-trip Void-State parity certified [${valid ? 'PASS' : 'FAIL'}]`,
  ]
  await emitSentinelTelemetry(ste, receipts)

  return {
    valid,
    sandboxRoot: resolvedRoot,
    initialVoidVerified,
    mutatedActiveVerified,
    postResetVoidVerified,
    zeroOrphanFilesVerified,
    zeroModelContaminationVerified,
    errors,
    receipts,
  }
}
