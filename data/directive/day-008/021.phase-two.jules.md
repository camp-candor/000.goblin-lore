export const resetDatabase = async (
  cpy: DatabaseModel,
  bal?: DatabaseBit,
  ste?: State,
): Promise<DatabaseModel> => {
  if (
    cpy.activeDbHandle &&
    typeof cpy.activeDbHandle.close === 'function'
  ) {
    try {
      cpy.activeDbHandle.close()
    } catch {
      // Ignore already-closed SQLite handles
    }
  }

  const repoRoot = resolveResetLoreRoot(bal?.src || process.cwd())
  const sqliteDir = path.join(repoRoot, 'data', 'sqlite')
  let purgedDbCount = 0

  if (fs.existsSync(sqliteDir)) {
    const entries = fs.readdirSync(sqliteDir, { withFileTypes: true })
    for (const entry of entries) {
      if (!entry.isFile()) continue
      if (
        entry.name.endsWith('.local.db') ||
        entry.name.endsWith('.local.db-wal') ||
        entry.name.endsWith('.local.db-shm')
      ) {
        fs.rmSync(path.join(sqliteDir, entry.name), { force: true })
        purgedDbCount++
      }
    }
  }

  cpy.idx = '07.database'
  cpy.databaseBitList = []
  cpy.databaseBits = {}
  cpy.activeDbHandle = null
  cpy.activeDbPath = null

  await emitDatabaseTelemetry(
    ste,
    `>> [DATABASE:RESET] Relational ledger handles closed and ${purgedDbCount} local SQLite artifact(s) cleared [OK]`,
  )

  if (bal?.slv) {
    bal.slv({
      datBit: {
        idx: 'reset-database-pass',
        val: 0,
        dat: { purgedDbCount, sqliteDir },
      },
    })
  }

  return cpy
}
