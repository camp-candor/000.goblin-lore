/**
 * Task 2.1: Somatic Limb Capacity Prompt Sequence
 */
export const promptSomaticLimbCapacity = async (
    ste?: State,
    override?: {
        availableArmsBps?: number
        conditions?: string[]
    },
): Promise<SomaticLimbSelectionResult> => {
    const lib = resolveLibraryBus()
    const isInteractive = Boolean(lib && typeof lib.hunt === 'function')

    if (typeof override?.availableArmsBps === 'number' || !isInteractive) {
        const baseBps = Math.max(
            0,
            Math.min(20000, Math.trunc(override?.availableArmsBps ?? 20000)),
        )
        const conditions = Array.isArray(override?.conditions)
            ? [...override.conditions]
            : []
        const traumaPenalty = conditions.filter((c) =>
            [
                'severed_left_arm',
                'severed_right_arm',
                'braxial_plexus_severed',
                'paralyzed_arm',
                'amputated_limb',
            ].includes(c.toLowerCase()),
        ).length * 10000
        const netBps = Math.max(0, baseBps - traumaPenalty)
        const traumaBannedVerbs =
            netBps < 20000 ? [...BIMANUAL_RESTRICTED_VERBS] : []

        emitConsole(
            ste,
            `>> [SOMATIC:LIMBS] Available Arms: ${netBps} bp | Conditions: [${conditions.join(', ')}]`,
        )

        return {
            cancelled: false,
            availableArmsBps: netBps,
            conditions,
            traumaBannedVerbs,
        }
    }

    emitConsole(
        ste,
        '>> [SOMATIC:LIMBS] Task 2.1: Select upper-limb capacity in basis points...',
    )

    const limbModal = await promptBlessedChoice({
        title: 'SOMATIC LIMB CAPACITY (BPS)',
        choices: [...LIMB_CAPACITY_CHOICES],
        defaultChoice: LIMB_CAPACITY_CHOICES[0],
        ste,
    })

    if (limbModal.cancelled || limbModal.value === CANCEL_INTERVIEW_LABEL) {
        emitConsole(
            ste,
            '>> [SOMATIC:LIMBS] [WARN] Limb capacity selection cancelled by operator.',
        )
        return {
            cancelled: true,
            availableArmsBps: 20000,
            conditions: [],
            traumaBannedVerbs: [],
        }
    }

    let baseBps = 20000
    if (limbModal.value.startsWith('[2]')) {
        baseBps = 15000
    } else if (limbModal.value.startsWith('[3]')) {
        baseBps = 10000
    } else if (limbModal.value.startsWith('[4]')) {
        baseBps = 0
    }

    const traumaModal = await promptBlessedChoice({
        title: 'ACTIVE SOMATIC TRAUMA CONDITION',
        choices: [...TRAUMA_CONDITION_CHOICES],
        defaultChoice: TRAUMA_CONDITION_CHOICES[0],
        ste,
    })

    const conditions: string[] = []
    if (!traumaModal.cancelled && !traumaModal.value.startsWith('[1]')) {
        const tag = traumaModal.value.split(' ')[1]?.trim()
        if (tag) {
            conditions.push(tag)
        }
    }

    const penaltyBps = conditions.length * 10000
    const netBps = Math.max(0, baseBps - penaltyBps)
    const traumaBannedVerbs =
        netBps < 20000 ? [...BIMANUAL_RESTRICTED_VERBS] : []

    emitConsole(
        ste,
        `>> [SOMATIC:LIMBS] [OK] Locked available_arms_bps=${netBps} | BannedBimanual=${traumaBannedVerbs.length}`,
    )

    return {
        cancelled: false,
        availableArmsBps: netBps,
        conditions,
        traumaBannedVerbs,
    }
}

function clampStatInt(raw: string, fallback: number): number {
    const parsed = Number.parseInt(String(raw || '').trim(), 10)
    if (Number.isNaN(parsed)) {
        return fallback
    }
    return Math.max(-3, Math.min(3, parsed))
}

/**
 * Task 2.3: Opposed Two-D-Six Attribute Input Collector
 */
export const promptTwoDSixAttributes = async (
    ste?: State,
    override?: Partial<CombatAttributeStats>,
): Promise<TwoDSixSelectionResult> => {
    const lib = resolveLibraryBus()
    const isInteractive = Boolean(lib && typeof lib.hunt === 'function')

    if (override || !isInteractive) {
        const validated = CombatAttributeStatsSchema.parse({
            spite: Math.max(-3, Math.min(3, Math.trunc(override?.spite ?? 0))),
            cunning: Math.max(
                -3,
                Math.min(3, Math.trunc(override?.cunning ?? 0)),
            ),
            vigor: Math.max(-3, Math.min(3, Math.trunc(override?.vigor ?? 0))),
            poise: Math.max(-3, Math.min(3, Math.trunc(override?.poise ?? 0))),
        })

        emitConsole(
            ste,
            `>> [STATS:2D6] Spite=${validated.spite} Cunning=${validated.cunning} Vigor=${validated.vigor} Poise=${validated.poise} [OK]`,
        )

        return {
            cancelled: false,
            stats: validated,
        }
    }

    emitConsole(
        ste,
        '>> [STATS:2D6] Task 2.3: Select opposed 2d6 attribute profile...',
    )

    const archetypeModal = await promptBlessedChoice({
        title: 'OPPOSED 2D6 ATTRIBUTE PROFILE',
        choices: [...TWO_D_SIX_ARCHETYPE_CHOICES],
        defaultChoice: TWO_D_SIX_ARCHETYPE_CHOICES[0],
        ste,
    })

    if (
        archetypeModal.cancelled ||
        archetypeModal.value === CANCEL_INTERVIEW_LABEL
    ) {
        emitConsole(
            ste,
            '>> [STATS:2D6] [WARN] Attribute entry cancelled by operator.',
        )
        return {
            cancelled: true,
            stats: { spite: 0, cunning: 0, vigor: 0, poise: 0 },
        }
    }

    let rawStats: CombatAttributeStats = {
        spite: 0,
        cunning: 0,
        vigor: 0,
        poise: 0,
    }

    if (archetypeModal.value.startsWith('[1]')) {
        rawStats = { spite: 0, cunning: 2, vigor: 3, poise: 3 }
    } else if (archetypeModal.value.startsWith('[2]')) {
        rawStats = { spite: 2, cunning: 3, vigor: 2, poise: 2 }
    } else if (archetypeModal.value.startsWith('[3]')) {
        rawStats = { spite: 3, cunning: 3, vigor: 0, poise: -1 }
    } else if (archetypeModal.value.startsWith('[4]')) {
        rawStats = { spite: 0, cunning: 0, vigor: 0, poise: 0 }
    } else {
        const spiteRes = await promptBlessedInput({
            promptLabel: 'SPITE MODIFIER (-3 to +3)',
            defaultValue: '0',
            ste,
        })
        if (spiteRes.cancelled) {
            return { cancelled: true, stats: rawStats }
        }

        const cunningRes = await promptBlessedInput({
            promptLabel: 'CUNNING MODIFIER (-3 to +3)',
            defaultValue: '0',
            ste,
        })
        if (cunningRes.cancelled) {
            return { cancelled: true, stats: rawStats }
        }

        const vigorRes = await promptBlessedInput({
            promptLabel: 'VIGOR MODIFIER (-3 to +3)',
            defaultValue: '0',
            ste,
        })
        if (vigorRes.cancelled) {
            return { cancelled: true, stats: rawStats }
        }

        const poiseRes = await promptBlessedInput({
            promptLabel: 'POISE MODIFIER (-3 to +3)',
            defaultValue: '0',
            ste,
        })
        if (poiseRes.cancelled) {
            return { cancelled: true, stats: rawStats }
        }

        rawStats = {
            spite: clampStatInt(spiteRes.value, 0),
            cunning: clampStatInt(cunningRes.value, 0),
            vigor: clampStatInt(vigorRes.value, 0),
            poise: clampStatInt(poiseRes.value, 0),
        }
    }

    const validated = CombatAttributeStatsSchema.parse(rawStats)

    emitConsole(
        ste,
        `>> [STATS:2D6] [OK] Locked Spite=${validated.spite} Cunning=${validated.cunning} Vigor=${validated.vigor} Poise=${validated.poise}`,
    )

    return {
        cancelled: false,
        stats: validated,
    }
}

/**
 * Orchestrates Task 2.1 -> Task 2.2 -> Task 2.3 into a complete Character Machine Rail frontmatter object.
 */
export const runCharacterMachineRailInterview = async (
    cpy: MenuModel,
    entityId: string,
    ste?: State,
    overrides?: {
        availableArmsBps?: number
        conditions?: string[]
        locomotionBaseline?: LocomotionBaseline
        stats?: Partial<CombatAttributeStats>
    },
): Promise<CharacterInterviewOutcome> => {
    const lib = resolveLibraryBus()
    const isInteractive = Boolean(lib && typeof lib.hunt === 'function')

    // Step 1 (Task 2.1): Somatic Limb Capacity Prompt Sequence
    const limbResult = await promptSomaticLimbCapacity(ste, {
        availableArmsBps: overrides?.availableArmsBps,
        conditions: overrides?.conditions,
    })
    if (limbResult.cancelled) {
        return { cancelled: true }
    }

    // Step 2 (Task 2.2): Locomotion Baseline Selection Modal
    let profile: LocomotionProfile
    if (overrides?.locomotionBaseline || !isInteractive) {
        profile = resolveLocomotionProfile(
            overrides?.locomotionBaseline || 'bipedal_standard',
        )
    } else {
        let locoOutcome: any = null
        await promptLocomotionChoice(
            cpy,
            {
                idx: 'character-interview-locomotion',
                slv: (res: any) => {
                    locoOutcome = res?.mnuBit || res
                },
            },
            ste,
        )
        if (!locoOutcome || locoOutcome.val !== 1) {
            return { cancelled: true }
        }
        profile =
            (locoOutcome.dat?.profile as LocomotionProfile) ||
            resolveLocomotionProfile(
                (locoOutcome.src as LocomotionBaseline) || 'bipedal_standard',
            )
    }

    // Clamp arms to 0 bps if locomotion baseline forbids upper-limb manipulation
    const finalArmsBps =
        profile.default_arms_bps === 0 ? 0 : limbResult.availableArmsBps

    // Merge and deduplicate negative motor masks
    const mergedBannedVerbs = Array.from(
        new Set<string>([
            ...profile.default_banned_verbs,
            ...limbResult.traumaBannedVerbs,
        ]),
    ).sort((a, b) => a.localeCompare(b))

    // Step 3 (Task 2.3): Opposed Two-D-Six Attribute Input Collector
    const statsResult = await promptTwoDSixAttributes(ste, overrides?.stats)
    if (statsResult.cancelled) {
        return { cancelled: true }
    }

    const frontmatter: CharacterMachineRailPayload = {
        id: entityId,
        type: 'character',
        somatic: {
            locomotion_baseline: profile.baseline,
            max_velocity_bps: profile.max_velocity_bps,
            available_arms_bps: finalArmsBps,
            conditions: limbResult.conditions,
            banned_kinetic_verbs: mergedBannedVerbs,
        },
        stats: statsResult.stats,
    }

    emitConsole(
        ste,
        `>> [CHARACTER:INTERVIEW] [OK] Assembled Machine Rail for ${entityId} (Arms=${finalArmsBps}bp, Gait=${profile.baseline}, Banned=${mergedBannedVerbs.length})`,
    )

    return {
        cancelled: false,
        frontmatter,
    }
}
