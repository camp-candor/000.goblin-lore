function isPure7BitAscii(text: string): boolean {
    for (let i = 0; i < text.length; i++) {
        if (text.charCodeAt(i) > 127) {
            return false
        }
    }
    return true
}

function encodeBase64Utf8Safe(content: string): string {
    return btoa(unescape(encodeURIComponent(content)))
}

async function githubApiFetch(
    env: Env,
    apiPath: string,
    init?: RequestInit,
): Promise<Response> {
    const owner = (env as any).GITHUB_DEFAULT_OWNER || 'astro-kahn-it-com'
    const repo = (env as any).GITHUB_DEFAULT_REPO || '001.goblin-lore'
    const token = (env as any).GITHUB_TOKEN || ''

    return fetch(url, {

            'User-Agent': '000-goblin-lore-edge-control-plane',

        },
    })
}

export async function executeCommitRatifiedEntity(
    env: Env,
    params: CommitRatifiedEntityParams,
): Promise<EntityCommitReceipt> {
    const normalizedMd = params.markdownContent.replace(/\r\n/g, '\n')
    if (!normalizedMd.startsWith('---\n')) {
        throw new Error(
            '[BYTE_0_VIOLATION] markdownContent must begin with "---\\n" at byte offset 0',
        )
    }
    if (!isPure7BitAscii(normalizedMd)) {
        throw new Error(
            '[ASCII_VIOLATION] markdownContent contains non-ASCII bytes > 127',
        )
    }

    // Step 1: Query trunk main SHA (S_clean rollback anchor)[cite: 5]
    const refRes = await githubApiFetch(env, '/git/ref/heads/main')
    if (!refRes.ok) {
        throw new Error(
            `[S_CLEAN_FETCH_FAILED] Unable to resolve trunk main SHA (HTTP ${refRes.status})`,
        )
    }
    const refJson: any = await refRes.json()
    const sCleanSha = String(refJson?.object?.sha || '')
    if (!/^[0-9a-f]{40}$/i.test(sCleanSha)) {
        throw new Error('[S_CLEAN_INVALID] Invalid 40-hex commit SHA on main')
    }

    const shortSha = sCleanSha.slice(0, 7)
    const ephemeralBranch = `spec/ENTITY-${params.entityId}-${shortSha}`

    // Step 2: Provision isolated ephemeral branch[cite: 5]
    const branchRes = await githubApiFetch(env, '/git/refs', {
        method: 'POST',
        body: JSON.stringify({
            ref: `refs/heads/${ephemeralBranch}`,
            sha: sCleanSha,
        }),
    })
    if (!branchRes.ok && branchRes.status !== 422) {
        throw new Error(
            `[BRANCH_PROVISION_FAILED] Failed to create ${ephemeralBranch} (HTTP ${branchRes.status})`,
        )
    }

    // Step 3: Commit Byte-0 entity dossier + reciprocal mutations + bible-state.json[cite: 5]
    const filesToCommit: Array<{ filePath: string; content: string }> = [
        {
            filePath: `series/${params.seriesSlug}/${params.domain}/${params.entityId}.md`,
            content: normalizedMd,
        },
    ]

    if (Array.isArray(params.reciprocalMutations)) {
        for (const mut of params.reciprocalMutations) {
            const cleanMut = mut.content.replace(/\r\n/g, '\n')
            if (!cleanMut.startsWith('---\n') || !isPure7BitAscii(cleanMut)) {
                throw new Error(
                    `[RECIPROCAL_INVARIANT_BREACH] Invalid Byte-0 or non-ASCII content in ${mut.filePath}`,
                )
            }
            filesToCommit.push({ filePath: mut.filePath, content: cleanMut })
        }
    }

    if (params.bibleStateJson) {
        filesToCommit.push({
            filePath: `compiled/${params.seriesSlug}/bible-state.json`,
            content: params.bibleStateJson,
        })
    }

    const committedPaths: string[] = []
    for (const entry of filesToCommit) {
        let existingBlobSha: string | undefined
        const probeRes = await githubApiFetch(
            env,
            `/contents/${entry.filePath}?ref=${encodeURIComponent(ephemeralBranch)}`,
        )
        if (probeRes.ok) {
            const probeJson: any = await probeRes.json()
            existingBlobSha = probeJson?.sha
        }

        const putRes = await githubApiFetch(env, `/contents/${entry.filePath}`, {
            method: 'PUT',
            body: JSON.stringify({
                message: `feat(lore): ratify ${params.entityId} (${params.expectedHRoot.slice(0, 8)})`,
                content: encodeBase64Utf8Safe(entry.content),
                branch: ephemeralBranch,
                ...(existingBlobSha ? { sha: existingBlobSha } : {}),
            }),
        })
        if (!putRes.ok) {
            throw new Error(
                `[FILE_COMMIT_FAILED] Failed to commit ${entry.filePath} (HTTP ${putRes.status})`,
            )
        }
        committedPaths.push(entry.filePath)
    }

    // Step 4: Open trunk Pull Request[cite: 5]
    const prRes = await githubApiFetch(env, '/pulls', {
        method: 'POST',
        body: JSON.stringify({
            title: `feat(lore): ratify ${params.entityId} in ${params.seriesSlug}`,
            head: ephemeralBranch,
            base: 'main',
            body: [
                '### Ratified Byte-0 Entity Dossier Promotion',
                `- **Series:** \`${params.seriesSlug}\``,
                `- **Entity:** \`${params.entityId}\` (\`${params.domain}/\`)`,
                `- **S_clean Anchor:** \`${sCleanSha}\``,
                `- **Merkle Seal (H_root):** \`${params.expectedHRoot}\``,
                `- **Committed Files:** ${committedPaths.length}`,
            ].join('\n'),
        }),
    })

    let prNumber: number | undefined
    let prUrl: string | undefined
    if (prRes.ok) {
        const prJson: any = await prRes.json()
        prNumber = prJson?.number
        prUrl = prJson?.html_url
    }

    return {
        ok: true,
        seriesSlug: params.seriesSlug,
        entityId: params.entityId,
        domain: params.domain,
        sCleanSha,
        ephemeralBranch,
        prNumber,
        prUrl,
        hRoot: params.expectedHRoot,
        filesCommitted: committedPaths,
    }
}

export const createVerifyTrunkCanonSealTool = (env: Env): AgentTool<any> => ({
    name: 'verify_trunk_canon_seal',
    label: 'Verify Trunk Canon Seal',
    description:
        'Compares local bible-state.json Merkle root H_root against trunk main compiled canon.',
    parameters: VerifyTrunkCanonSealParamsSchema,
    execute: async (_toolCallId: string, params: VerifyTrunkCanonSealParams) => {
        const res = await githubApiFetch(
            env,
            `/contents/compiled/${params.seriesSlug}/bible-state.json?ref=main`,
        )
        let remoteHRoot = ''
        if (res.ok) {
            const json: any = await res.json()
            if (json?.content) {
                const decoded = atob(String(json.content).replace(/\n/g, ''))
                const parsed = JSON.parse(decoded)
                remoteHRoot = String(
                    parsed?.merkleRootHex || parsed?._meta?.corpusHash || '',
                ).toLowerCase()
            }
        }
        const match =
            remoteHRoot.length === 64 &&
            remoteHRoot === params.localHRoot.toLowerCase()
        const payload = redactSecrets(
            JSON.stringify({
                ok: true,
                seriesSlug: params.seriesSlug,
                localHRoot: params.localHRoot.toLowerCase(),
                remoteHRoot,
                parityMatch: match,
            }),
        )
        return {
            content: [{ type: 'text', text: payload }],
            details: JSON.parse(payload),
        }
    },
})

app.post('/api/lore/entity/commit', async (c) => {
    try {
        const body = await c.req.json()
        if (!Value.Check(CommitRatifiedEntityParamsSchema, body)) {
            return c.json(
                {
                    ok: false,
                    error: 'INVALID_ENTITY_COMMIT_PAYLOAD',

        const receipt = await executeCommitRatifiedEntity(c.env, body)
        const cleanJson = redactSecrets(JSON.stringify(receipt))
        return c.json(JSON.parse(cleanJson), 200)
    } catch (err: any) {
        const msg = redactSecrets(
            err instanceof Error ? err.message : String(err),
        )
        const status =
            msg.includes('BYTE_0_VIOLATION') || msg.includes('ASCII_VIOLATION')
                ? 422

        return c.json({ ok: false, error: msg }, status)
    }
})

            ok: true,

export function resolveEdgeWorkerBaseUrl(explicitUrl?: string): string {
    if (explicitUrl && explicitUrl.trim().length > 0) {
        return explicitUrl.trim().replace(/\/+$/, '')
    }
    const switchboardUrl = (globalThis as any).agentBaseUrl
    if (typeof switchboardUrl === 'string' && switchboardUrl.trim().length > 0) {

export const syncSealedEntityToEdge = async (
    cpy: PortModel,
    bal: PortBit,
    ste?: State,
): Promise<PortModel> => {
    const dat = (bal?.dat || {}) as Record<string, any>
    const baseUrl = resolveEdgeWorkerBaseUrl(dat.baseUrl)
    const endpoint = `${baseUrl}/api/lore/entity/commit`

    emitConsole(
        ste,
        `>> [EDGE:DISPATCH] Syncing ${dat.entityId || 'entity'} -> ${endpoint}`,
    )

    try {
        const res = await fetch(endpoint, {
            method: 'POST',
            headers: { 'Content-Type': 'application/json' },
            body: JSON.stringify({
                seriesSlug: dat.seriesSlug,
                domain: dat.domain,
                entityId: dat.entityId,
                markdownContent: dat.markdownContent,
                expectedHRoot: dat.expectedHRoot,
                reciprocalMutations: dat.reciprocalMutations,
                bibleStateJson: dat.bibleStateJson,
            }),
        })

        const json: any = await res.json()
        if (!res.ok || !json?.ok) {
            const errText = toAscii(
                json?.error || `HTTP ${res.status} from edge worker`,
            )
            emitConsole(ste, `:: [EDGE:FAIL] ${errText}`)
            if (bal?.slv) {
                bal.slv({
                    prtBit: {
                        idx: 'sync-entity-to-edge-fail',
                        val: -1,
                        src: errText,
                    },
                })
            }
            return cpy
        }

        emitConsole(
            ste,
            `>> [EDGE:S_CLEAN] Anchor=${String(json.sCleanSha || '').slice(0, 7)} | Branch=${json.ephemeralBranch} [OK]`,
        )

        if (bal?.slv) {
            bal.slv({
                prtBit: {
                    idx: 'sync-entity-to-edge',
                    val: 1,
                    src: json.ephemeralBranch,
                    dat: json,
                },
            })
        }
    } catch (err: any) {
        const msg = toAscii(err instanceof Error ? err.message : String(err))
        emitConsole(
            ste,
            `:: [EDGE:OFFLINE] Worker unreachable at ${baseUrl}: ${msg}`,
        )
        if (bal?.slv) {
            bal.slv({
                prtBit: {
                    idx: 'sync-entity-to-edge-offline',
                    val: 0,
                    src: msg,
                },
            })
        }
    }

    return cpy
}
