```typescript
import * as fs from 'node:fs'
import * as path from 'node:path'
import type State from '../../99.core/state.js'
import type { MenuModel } from '../menu.model.js'
import type MenuBit from '../fce/menu.bit.js'
import { scanVoidState } from '../../29.series.unit/buz/series.discovery.buzz.js'
import { resolveRepoRoot } from '../../29.series.unit/buz/scaffoldSeriesTree.buzz.js'
import { resolveLibraryBus } from './blessedModal.buzz.js'

const UPDATE_GRID = '[Grid action] Update Grid'
const OPEN_CHOICE = '[Open action] Open Choice'
const UPDATE_CONSOLE = '[Console action] Update Console'

function emitConsole(ste: State | undefined, line: string): void {
  const cleanLine = String(line || '')
    .replace(/[^\x20-\x7E]/g, '')
    .trim()
  const lib = resolveLibraryBus()
  if (lib && typeof lib.hunt === 'function') {
    lib.hunt(UPDATE_CONSOLE, { idx: 'cns00', src: cleanLine }).catch(() => {})
    return
  }
  if (ste && typeof ste.hunt === 'function') {
    ste.hunt(UPDATE_CONSOLE, { idx: 'cns00', src: cleanLine }).catch(() => {})
  }
}

function hasValidSeriesConfig(slug: string, repoRoot?: string): boolean {
  if (!slug || typeof slug !== 'string') return false
  const root = repoRoot ? path.resolve(repoRoot) : resolveRepoRoot()
  const configPath = path.join(root, 'series', slug.trim(), 'series.config.json')
  return fs.existsSync(configPath)
}

export const promptSeriesSelectionModal = async (
  cpy: MenuModel,
  bal: MenuBit,
  ste?: State,
): Promise<MenuModel> => {
  const lib = resolveLibraryBus()
  const repoRoot = (bal?.dat as any)?.repoRoot
  const scanResult = scanVoidState(repoRoot)

  if (scanResult.isVoidState) {
    emitConsole(
      ste,
      '>> [SERIES:DISCOVERY] [WARN] Void state detected: No series found in series/.',
    )
    emitConsole(
      ste,
      '>> [SERIES:DISCOVERY] Please initialize your first series using [G] UNIVERSAL SERIES GENESIS WIZARD.',
    )
    if (bal?.slv) {
      bal.slv({
        mnuBit: { idx: 'series-discovery-void', val: 0, dat: scanResult },
      })
    }
    return cpy
  }

  const choices = scanResult.discoveredSlugs.map(
    (slug, idx) => `[${idx + 1}] ${slug}`,
  )
  emitConsole(
    ste,
    `>> [SERIES:DISCOVERY] Discovered ${scanResult.discoveredSlugs.length} series namespace(s) on disk.`,
  )

  let selectedChoice = choices[0]
  if (lib && typeof lib.hunt === 'function') {
    const gridBit: any = await lib.hunt(UPDATE_GRID, {
      x: 0,
      y: 4,
      xSpan: 4,
      ySpan: 8,
    })
    const choiceBit: any = await lib.hunt(OPEN_CHOICE, {
      dat: { clr0: 'black', clr1: 'yellow' },
      src: 'vertical',
      lst: [...choices, '[<] RETURN TO LORE MENU'],
      net: gridBit?.grdBit?.dat,
    })
    selectedChoice = choiceBit?.chcBit?.src || choices[0]
  }

  if (selectedChoice === '[<] RETURN TO LORE MENU' || !selectedChoice) {
    emitConsole(ste, '>> [SERIES:DISCOVERY] Series selection cancelled by operator.')
    if (bal?.slv) {
      bal.slv({ mnuBit: { idx: 'series-discovery-cancel', val: 0 } })
    }
    return cpy
  }

  const cleanSlug = selectedChoice.replace(/^\[\d+\]\s*/, '').trim()
  emitConsole(
    ste,
    `>> [SERIES:ACTIVE] Bound active series namespace to: ${cleanSlug} [OK]`,
  )

  cpy.activeSelection = 'PROMPT_SERIES_SELECTION'
  cpy.activeSeries = cleanSlug
  cpy.statusMessage = `Active Series: ${cleanSlug}`

  if (bal?.slv) {
    bal.slv({
      mnuBit: {
        idx: 'series-discovery-success',
        val: 1,
        src: cleanSlug,
        dat: { scanResult, activeSlug: cleanSlug },
      },
    })
  }

  return cpy
}

/**
 * Resolves the active series slug deterministically across interactive TTY and headless runs.
 * 1. Explicit override via bal.src or bal.dat.seriesSlug
 * 2. Already bound cpy.activeSeries verified against <repoRoot>/series/<slug>/series.config.json
 * 3. Interactive TTY prompt via promptSeriesSelectionModal when unbound
 * 4. Headless deterministic fallback to first discovered series on disk (or fail-closed null in Void State)
 */
export const resolveOrPromptActiveSeries = async (
  cpy: MenuModel,
  bal?: MenuBit,
  ste?: State,
): Promise<string | null> => {
  const dat = (bal?.dat || {}) as Record<string, any>
  const repoRoot = dat.repoRoot ? path.resolve(dat.repoRoot) : resolveRepoRoot()

  const explicitSlug =
    typeof dat.seriesSlug === 'string' && dat.seriesSlug.trim().length > 0
      ? dat.seriesSlug.trim()
      : typeof bal?.src === 'string' &&
          bal.src.trim().length > 0 &&
          bal.src !== 'ROOT_ROUTE'
        ? bal.src.trim()
        : null

  if (explicitSlug) {
    cpy.activeSeries = explicitSlug
    return explicitSlug
  }

  if (cpy.activeSeries && hasValidSeriesConfig(cpy.activeSeries, repoRoot)) {
    return cpy.activeSeries
  }

  const scanResult = scanVoidState(repoRoot)
  if (scanResult.isVoidState || scanResult.discoveredSlugs.length === 0) {
    emitConsole(
      ste,
      '>> [SERIES:VOID] [WARN] No valid series found in series/. Run [G] UNIVERSAL SERIES GENESIS WIZARD first.',
    )
    return null
  }

  const lib = resolveLibraryBus()
  if (lib && typeof lib.hunt === 'function') {
    let chosenSlug: string | null = null
    await promptSeriesSelectionModal(
      cpy,
      {
        idx: 'auto-prompt-active-series',
        dat: { repoRoot },
        slv: (res: any) => {
          if (res?.mnuBit?.val === 1 && typeof res?.mnuBit?.src === 'string') {
            chosenSlug = res.mnuBit.src
          }
        },
      },
      ste,
    )
    return chosenSlug
  }

  // Headless fallback: deterministically bind the first sorted series on disk
  const fallbackSlug = scanResult.discoveredSlugs[0]
  cpy.activeSeries = fallbackSlug
  cpy.statusMessage = `Active Series: ${fallbackSlug}`
  return fallbackSlug
}
```
