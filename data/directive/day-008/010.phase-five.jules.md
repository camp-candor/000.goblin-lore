            return curr
        }
        curr = path.dirname(curr)
    }
    return path.resolve(startDir)
}

export function normalizeLocationId(raw: string): string {
    const cleaned = toAscii(raw)
        .toLowerCase()
        .replace(/[\s/\\:]+/g, '-')
        .replace(/[^a-z0-9_-]/g, '')
        .replace(/-+/g, '-')
        .replace(/^[-_]+|[-_]+$/g, '')

    if (!cleaned) {
        return 'loc_stage-01'
    }
    if (cleaned.startsWith('loc_')) {
        return cleaned
    }
    return `loc_${cleaned}`
}

/**
 * Serializes a Byte-0 compliant Location Markdown dossier with deterministic YAML keys
 * and standardized Voice Rail headings.
 */
export function renderLocationDossierMarkdown(
    params: RenderLocationDossierParams,
): string {
    const uniqueConnections = Array.from(
        new Set(
            params.connectedLocations
                .map((id) => id.trim())
                .filter((id) => id.length > 0 && id !== params.locId),
        ),
    ).sort((a, b) => a.localeCompare(b))

    const frontmatterObj: Record<string, any> = {
        id: params.locId,
        type: 'location',
        name: toAscii(params.name),
        volume_profile: params.volumeProfile,
        axial_bounds: {
            origin: {
                q: Math.trunc(params.bounds.origin.q),
                r: Math.trunc(params.bounds.origin.r),
            },
            min_q: Math.trunc(params.bounds.minQ),
            max_q: Math.trunc(params.bounds.maxQ),
            min_r: Math.trunc(params.bounds.minR),
            max_r: Math.trunc(params.bounds.maxR),
        },
        max_adult_goblins: Math.max(1, Math.trunc(params.maxAdultGoblins)),
        acoustic_damping_cdb: Math.max(0, Math.trunc(params.acousticDampingCdB)),
        ambient_noise_cdb: Math.max(0, Math.trunc(params.ambientNoiseCdB)),
        log_lux_index: Math.max(0, Math.min(448, Math.trunc(params.logLuxIndex))),
        lighting: toAscii(params.lightingTag),
        connected_locations: uniqueConnections,
    }

    const yamlDump = yaml
        .dump(frontmatterObj, {
            indent: 2,
            lineWidth: -1,
            noRefs: true,
            sortKeys: true,
        })
        .replace(/\r\n/g, '\n')
        .trimEnd()

    const bodyLines = [
        '## Tactical Geometry',
        `Bounded axial hex grid (${params.bounds.minQ}..${params.bounds.maxQ} q, ${params.bounds.minR}..${params.bounds.maxR} r) with capacity ceiling ${frontmatterObj.max_adult_goblins}.`,
        '',
        '## Sensory Atmosphere',
        `Ambient noise floor ${frontmatterObj.ambient_noise_cdb} cdB under ${frontmatterObj.lighting} illumination.`,
        '',
    ]

    return `---\n${yamlDump}\n---\n${bodyLines.join('\n')}`
}

/**
 * Verifies Byte-0 delimiter (0x2D 0x2D 0x2D 0x0A) and commits atomically via .tmp rename.
 */
export function writeLocationDossierAtomic(
    targetFilePath: string,
    markdownContent: string,
): void {
    const normalized = markdownContent.replace(/\r\n/g, '\n')
    const buf = Buffer.from(normalized, 'utf8')

    if (
        buf.length < 4 ||
        buf[0] !== 0x2d ||
        buf[1] !== 0x2d ||
        buf[2] !== 0x2d ||
        buf[3] !== 0x0a
    ) {
        throw new Error(
            '[BYTE_0_VIOLATION] Expected opening "---\\n" (0x2d2d2d0a) at byte offset 0',
        )
    }

    const parentDir = path.dirname(targetFilePath)
    fs.mkdirSync(parentDir, { recursive: true })

    const tmpPath = `${targetFilePath}.tmp_${Date.now()}_${Math.random().toString(36).slice(2, 7)}`
    fs.writeFileSync(tmpPath, buf)
    fs.renameSync(tmpPath, targetFilePath)
}

/**
 * Task 3.3: Reconciles bidirectional portal links across sibling location Markdown files
 * while preserving Byte-0 alignment and body prose via atomic POSIX renames.
 */
export function reconcileReciprocalLocationLinks(options: {
    seriesSlug: string
    sourceLocId: string
    targetLocIds: readonly string[]
    repoRoot?: string
}): { reconciledTargets: string[]; errors: string[] } {
    const root = options.repoRoot
        ? path.resolve(options.repoRoot)
        : resolveLocationRepoRoot()
    const locationsDir = path.join(root, 'series', options.seriesSlug, 'locations')

    const reconciledTargets: string[] = []
    const errors: string[] = []

    for (const rawTargetId of options.targetLocIds) {
        const targetId = rawTargetId.trim()
        if (!targetId) continue

        if (targetId === options.sourceLocId) {
            errors.push(
                `[SELF_ADJACENCY_VIOLATION] Location '${options.sourceLocId}' cannot link to itself`,
            )
            continue
        }

        const targetPath = path.join(locationsDir, `${targetId}.md`)
        if (!fs.existsSync(targetPath)) {
            errors.push(
                `[DANGLING_SPATIAL_POINTER] Target location '${targetId}' does not exist in ${locationsDir}`,
            )
            continue
        }

        const rawTarget = fs.readFileSync(targetPath, 'utf8').replace(/\r\n/g, '\n')
        if (!rawTarget.startsWith('---\n')) {
            errors.push(
                `[BYTE_0_VIOLATION] Sibling dossier '${targetId}.md' violates Byte-0 frontmatter invariant`,
            )
            continue
        }

        const parsed = matter(rawTarget)
        const data = (parsed.data || {}) as Record<string, any>
        const existing: string[] = Array.isArray(data.connected_locations)
            ? data.connected_locations.map((v: unknown) => String(v))
            : []

        if (!existing.includes(options.sourceLocId)) {
            existing.push(options.sourceLocId)
            existing.sort((a, b) => a.localeCompare(b))
            data.connected_locations = existing

            const dumpedYaml = yaml
                .dump(data, {
                    indent: 2,
                    lineWidth: -1,
                    noRefs: true,
                    sortKeys: true,
                })
                .replace(/\r\n/g, '\n')
                .trimEnd()

            const bodyClean = parsed.content.startsWith('\n')
                ? parsed.content.slice(1)
                : parsed.content

            const updatedMarkdown = `---\n${dumpedYaml}\n---\n${bodyClean}`
            writeLocationDossierAtomic(targetPath, updatedMarkdown)
            reconciledTargets.push(targetId)
        }
    }

    return { reconciledTargets, errors }
}

function listExistingLocations(seriesSlug: string, repoRoot: string): string[] {
    const locDir = path.join(repoRoot, 'series', seriesSlug, 'locations')
    if (!fs.existsSync(locDir)) {
        return []
    }
    return fs
        .readdirSync(locDir)
        .filter((f) => f.endsWith('.md') && !f.startsWith('.') && !f.startsWith('_'))
        .map((f) => f.replace(/\.md$/i, ''))
        .sort((a, b) => a.localeCompare(b))
}

async function promptGridInput(
    lib: any,
    label: string,
    defaultVal: string,
): Promise<{ cancelled: boolean; value: string }> {
    const gridBit: any = await lib.hunt(UPDATE_GRID, {
        x: 0,
        y: 4,
        xSpan: 4,
        ySpan: 4,
    })
    const inputBit: any = await lib.hunt(OPEN_INPUT, {
        txt: label,
        src: defaultVal,
        net: gridBit?.grdBit?.dat,
    })
    if (inputBit?.putBit?.val === -1) {
        return { cancelled: true, value: '' }
    }
    const raw = String(inputBit?.putBit?.src || '').trim()
    return { cancelled: false, value: raw.length > 0 ? raw : defaultVal }
}

async function promptGridChoice(
    lib: any,
    choices: string[],
    onFocus?: (item: string) => void,
): Promise<{ cancelled: boolean; value: string }> {
    const gridBit: any = await lib.hunt(UPDATE_GRID, {
        x: 0,
        y: 4,
        xSpan: 4,
        ySpan: 8,
    })
    const choiceBit: any = await lib.hunt(OPEN_CHOICE, {
        dat: { clr0: 'black', clr1: 'yellow', cb: onFocus },
        src: 'vertical',
        lst: choices,
        net: gridBit?.grdBit?.dat,
    })
    if (choiceBit?.chcBit?.val === -1) {
        return { cancelled: true, value: '' }
    }
    return {
        cancelled: false,
        value: String(choiceBit?.chcBit?.src || choices[0] || ''),
    }
}

                    portalChoice.cancelled ||
                    portalChoice.value === finishPortalsLabel

                }

                    emitConsole(
                        ste,
                        `>> [PORTAL:QUEUED] Linked adjacent node: ${pickedLoc}`,
                    )

                    !reconcilePrompt.cancelled &&
                    reconcilePrompt.value.startsWith('[1]')

    if (!LOC_ID_REGEX.test(locId)) {

        if (bal?.slv) {
            bal.slv({ locBit: { success: false, locId, error: errMsg } })
        }
        return cpy
    }

    const markdown = renderLocationDossierMarkdown({
        locId,
        name: displayName,
        volumeProfile: selectedVolume.key,
        bounds,
        maxAdultGoblins,
        acousticDampingCdB,
        ambientNoiseCdB,
        logLuxIndex,
        lightingTag,
        connectedLocations,
    })

    const targetFilePath = path.join(
        repoRoot,
        'series',
        seriesSlug,
        'locations',
        `${locId}.md`,
    )
    writeLocationDossierAtomic(targetFilePath, markdown)

    emitConsole(
        ste,
        `>> [SEALED] Location '${locId}' written to disk at Byte 0 [OK]`,
    )

    let reconciledTargets: string[] = []
    let reconcileErrors: string[] = []

    if (shouldReconcile && connectedLocations.length > 0) {
        const recResult = reconcileReciprocalLocationLinks({
            seriesSlug,
            sourceLocId: locId,
            targetLocIds: connectedLocations,
            repoRoot,

        reconciledTargets = recResult.reconciledTargets

            emitConsole(
                ste,
                `>> [RECIPROCAL] Appended ${locId} to ${target}.md [OK]`,
            )
        }
        for (const err of reconcileErrors) {
            emitConsole(ste, `:: [RECIPROCAL:WARN] ${err}`)
        }
    }

    if (bal?.slv) {
        bal.slv({
            locBit: {
                success: reconcileErrors.length === 0,
                filePath: targetFilePath,
                locId,
                reconciledTargets,
                errors: reconcileErrors.length > 0 ? reconcileErrors : undefined,
                error: reconcileErrors[0],
            },
        })
    }

    return cpy
}
