export function validateLoreResetManifest(
  candidate: unknown = LORE_RESET_MANIFEST,
): LoreResetManifestValidationReport {
  const parsed = LoreResetManifestSchema.safeParse(candidate)
  if (!parsed.success) {
    const errors = parsed.error.issues.map(
      (issue) => `${issue.path.join('.') || 'root'}: ${issue.message}`,
    )
    return {
      valid: false,
      artifactCount: 0,
      statefulUnitCount: 0,
      statelessUnitCount: 0,
      errors,
    }
  }

  const manifest = parsed.data
  const errors: string[] = []
  const seenUnits = new Set<string>()

  for (const unit of manifest.unitRegistry) {
    if (seenUnits.has(unit.unitFolder)) {
      errors.push(`Duplicate unitFolder in registry: ${unit.unitFolder}`)
    }
    seenUnits.add(unit.unitFolder)

    if (
      unit.classification === 'STATEFUL_RESETTABLE' &&
      !unit.resetActionConstant
    ) {
      errors.push(
        `Stateful unit ${unit.unitFolder} missing resetActionConstant`,
      )
    }
  }

  const statefulUnitCount = manifest.unitRegistry.filter(
    (u) => u.classification === 'STATEFUL_RESETTABLE',
  ).length
  const statelessUnitCount = manifest.unitRegistry.filter(
    (u) => u.classification === 'STATELESS_PURE',
  ).length

  return {
    valid: errors.length === 0,
    artifactCount: manifest.purgeableArtifacts.length,
    statefulUnitCount,
    statelessUnitCount,
    errors,
  }
}

async function emitCns00(

export function resolveResetLoreRoot(startDir: string = process.cwd()): string {
  let curr = path.resolve(startDir)

export function assertSafeResetTarget(
  repoRoot: string,
  candidatePath: string,
): string {
  const rawCandidate = String(candidatePath || '').trim()
  if (!rawCandidate || rawCandidate === '.' || rawCandidate === './') {
    throw new Error(
      `[RESET_BOUNDARY_BREACH] Empty or root-alias candidate rejected: "${candidatePath}"`,
    )
  }

  const normalizedSegments = rawCandidate.replace(/\\/g, '/').split('/')
  if (normalizedSegments.includes('..')) {
    throw new Error(
      `[RESET_BOUNDARY_BREACH] Parent traversal segment ".." rejected: "${candidatePath}"`,
    )
  }

  const resolvedRoot = path.resolve(repoRoot)
  const resolvedTarget = path.isAbsolute(rawCandidate)
    ? path.resolve(rawCandidate)
    : path.resolve(resolvedRoot, rawCandidate)

  if (resolvedTarget === resolvedRoot) {
    throw new Error(
      `[RESET_BOUNDARY_BREACH] Repository root self-targeting rejected: "${resolvedTarget}"`,
    )
  }

  const relativeToRoot = path.relative(resolvedRoot, resolvedTarget)
  if (
    !relativeToRoot ||
    relativeToRoot.startsWith('..') ||
    path.isAbsolute(relativeToRoot)
  ) {
    throw new Error(
      `[RESET_BOUNDARY_BREACH] Target escapes repository root boundary: "${candidatePath}"`,
    )
  }

  const relPosix = relativeToRoot.replace(/\\/g, '/')
  const baseName = path.basename(relPosix)

  if (
    baseName === '.env' ||
    baseName.startsWith('.env.') ||
    baseName === '.dev.vars' ||
    baseName.startsWith('.dev.vars.')
  ) {
    throw new Error(
      `[RESET_BOUNDARY_BREACH] Protected environment secret file rejected: "${relPosix}"`,
    )
  }

  for (const prefix of LORE_PROTECTED_PATH_PREFIXES) {
    const cleanPrefix = prefix.replace(/\\/g, '/').replace(/\/+$/, '')
    if (
      relPosix === cleanPrefix ||
      relPosix.startsWith(`${cleanPrefix}/`)
    ) {
      throw new Error(
        `[RESET_BOUNDARY_BREACH] Target intersects protected system path "${cleanPrefix}": "${relPosix}"`,
      )
    }
  }

  return resolvedTarget
}

  }

  const verifiedTargets: string[] = []
