function sanitizeAscii(raw: string): string {
  return String(raw || '')
    .replace(/[^\x20-\x7E]/g, '')
    .trim()
}

function yieldPaintCycle(): Promise<void> {
  return new Promise((resolve) => setImmediate(resolve))
}

async function emitConsoleLines(
  ste: State | undefined,
  lines: readonly string[],
): Promise<void> {
  const lib = (globalThis as any).LIBRARY || (global as any).LIBRARY
  for (const raw of lines) {
    const clean = sanitizeAscii(raw)
    if (lib && typeof lib.hunt === 'function') {
      await lib.hunt(UPDATE_CONSOLE, { idx: 'cns00', src: clean }).catch(() => {})
    } else if (ste && typeof ste.hunt === 'function') {
      await ste.hunt(UPDATE_CONSOLE, { idx: 'cns00', src: clean }).catch(() => {})
    }
  }
}

async function promptAlertChoiceModal(
  lst: readonly string[],
  tooltips: Record<string, string>,
  ste?: State,
): Promise<string> {
  const lib = (globalThis as any).LIBRARY || (global as any).LIBRARY
  if (!lib || typeof lib.hunt !== 'function') {
    return lst[0] || ''
  }

  const gridBit: any = await lib.hunt(UPDATE_GRID, {
    x: 0,
    y: 4,
    xSpan: 4,
    ySpan: 8,
  })

  const choiceBit: any = await lib.hunt(OPEN_CHOICE, {
    dat: {
      clr0: 'black',
      clr1: 'red',
      cb: (focusedLabel: string) => {
        const tip = tooltips[focusedLabel]
        if (tip) {
          emitConsoleLines(ste, [`>> [TIP] ${tip}`]).catch(() => {})
        }
      },
    },
    src: 'vertical',
    lst: [...lst],
    net: gridBit?.grdBit?.dat,
  })

  return String(choiceBit?.chcBit?.src || lst[0] || '')
}

function resolveCauldronPivot(): any {
  const g = globalThis as any
  if (g.CAULDRON && typeof g.CAULDRON.hunt === 'function') {
    return g.CAULDRON
  }
  const pivots = g.PIVOTS || {}
  return (
    pivots['@camp_candor/001.cauldron'] ||
    pivots['001.cauldron'] ||
    pivots['cauldron'] ||
    null
  )
}

function resetMenuModelToVoidState(
  cpy: MenuModel,
  receipt: LoreTabulaRasaReceipt,
): void {
  cpy.activeSelection = TABULA_RASA_MENU_LABEL
  cpy.lastActionTimestamp = Date.now()
  cpy.statusMessage = 'VOID_STATE'
  cpy.lastLifecycleReceipt = '[TABULA_RASA:OK]'
  cpy.lastSomaticReceipt = null
  cpy.activeSeries = undefined
  cpy.activeEntityDomain = undefined
  cpy.activeEntityPrefix = undefined
  cpy.activeEntityDirectory = undefined
  cpy.lastError = undefined

    cpy.lastTelemetryReport = []
  }
  cpy.lastTelemetryReport.push(...receipt.receipts)
}

/**
 * Task 5.2: Dispatches the reverse-causal 4-repo Master Fleet Reset to
 * 000.camp-candor-cauldron if mounted on globalThis, or falls back cleanly
 * to local 299.lore Void-State reset when running standalone.
 */
export const triggerCauldronFleetResetCascade = async (
  cpy: MenuModel,
  bal?: MenuBit,
  ste?: State,
): Promise<MenuModel> => {
  const dat = (bal?.dat || {}) as Record<string, any>
  const req: LoreTabulaRasaRequestBit =
    (dat.tabulaRasaRequest as LoreTabulaRasaRequestBit) || dat
  const repoRoot = resolveResetLoreRoot(
    req.repoRoot || bal?.src || process.cwd(),
  )
  const lib = (globalThis as any).LIBRARY || (global as any).LIBRARY
  const isInteractive = Boolean(lib && typeof lib.hunt === 'function')

  let token1 = req.confirmToken1
  let token2 = req.confirmToken2
  let token3 = req.confirmToken3

  if (isInteractive && (!token1 || !token2 || !token3)) {
    await emitConsoleLines(ste, [
      '>> [CAULDRON:CASCADE] Initiating Triple-Lock Master Fleet Reset Gate (4 Repositories)...',
    ])

    const c1 = await promptAlertChoiceModal(
      [STAGE_1_NO, STAGE_1_YES],
      {
        [STAGE_1_NO]: 'Abort fleet reset and preserve all 4 repositories untouched.',
        [STAGE_1_YES]:
          'Arm Stage 1/3 for reverse-causal purge (conductor -> engine -> spatial -> lore).',
      },
      ste,
    )
    if (c1 !== STAGE_1_YES) {
      await emitConsoleLines(ste, [
        '>> [RESET:ABORTED] Cauldron cascade cancelled at Stage 1/3. System untouched.',
      ])
      if (bal?.slv) {
        bal.slv({
          mnuBit: { idx: 'cauldron-fleet-reset-aborted', val: 0 },
        })
      }
      return cpy
    }

    await emitConsoleLines(ste, [
      '>> [RESET:LOCK_1_CLEARED] Stage 1/3 verified. Awaiting Stage 2/3 confirmation...',
    ])
    await yieldPaintCycle()

    const c2 = await promptAlertChoiceModal(
      [STAGE_2_NO_A, STAGE_2_NO_B, STAGE_2_YES],
      {
        [STAGE_2_NO_A]: 'Abort fleet reset immediately.',
        [STAGE_2_NO_B]: 'Preserve generated screenplays, ledgers, and series.',
        [STAGE_2_YES]:
          'Acknowledge permanent deletion across conductor, engine, spatial, and lore.',
      },
      ste,
    )
    if (c2 !== STAGE_2_YES) {
      await emitConsoleLines(ste, [
        '>> [RESET:ABORTED] Cauldron cascade cancelled at Stage 2/3. System untouched.',
      ])
      if (bal?.slv) {
        bal.slv({
          mnuBit: { idx: 'cauldron-fleet-reset-aborted', val: 0 },
        })
      }
      return cpy
    }

    await emitConsoleLines(ste, [
      '>> [RESET:LOCK_2_CLEARED] Stage 2/3 verified. ARMED FOR FINAL FLEET EXECUTION (Stage 3/3)...',
    ])
    await yieldPaintCycle()

    const c3 = await promptAlertChoiceModal(
      [STAGE_3_NO, STAGE_3_YES],
      {
        [STAGE_3_NO]: 'Stand down and disarm Master Fleet Reset.',
        [STAGE_3_YES]:
          'Execute reverse-causal 4-repo wipe and return studio to Byte-0 Void State.',
      },
      ste,
    )
    if (c3 !== STAGE_3_YES) {
      await emitConsoleLines(ste, [
        '>> [RESET:ABORTED] Cauldron cascade cancelled at Stage 3/3. System untouched.',
      ])
      if (bal?.slv) {
        bal.slv({
          mnuBit: { idx: 'cauldron-fleet-reset-aborted', val: 0 },
        })
      }
      return cpy
    }

    token1 = TABULA_RASA_TOKEN_1
    token2 = TABULA_RASA_TOKEN_2
    token3 = TABULA_RASA_TOKEN_3
  }

  if (
    token1 !== TABULA_RASA_TOKEN_1 ||
    token2 !== TABULA_RASA_TOKEN_2 ||
    token3 !== TABULA_RASA_TOKEN_3
  ) {
    await emitConsoleLines(ste, [
      ':: [CAULDRON:ABORTED] Missing or invalid Triple-Lock confirmation tokens.',
    ])
    if (bal?.slv) {
      bal.slv({
        mnuBit: { idx: 'cauldron-fleet-reset-aborted', val: 0 },
      })
    }
    return cpy
  }

  const cauldronPivot = resolveCauldronPivot()
  if (cauldronPivot) {
    await emitConsoleLines(ste, [
      '>> [CAULDRON:CASCADE] Dispatching reverse-causal 4-repo reset to 000.camp-candor-cauldron...',
      '>> [CAULDRON:ORDER] 1:conductor -> 2:engine -> 3:spatial -> 4:lore',
    ])
    await yieldPaintCycle()

    let cauldronBit: any = null
    await cauldronPivot.hunt(CAULDRON_MASTER_RESET_ACTION, {
      idx: 'lore-cauldron-cascade',
      src: repoRoot,
      dat: {
        confirmToken1: token1,
        confirmToken2: token2,
        confirmToken3: token3,
        cascadeOrigin: 'CAULDRON_MASTER_CASCADE',
        reverseCausalOrder: [
          '000.series-sower-conductor',
          '000.series-sower-engine',
          '000.goblin-spatial',
          '000.goblin-lore',
        ],
      },
      slv: (res: any) => {
        cauldronBit = res?.cldBit || res
      },
    })

    await emitConsoleLines(ste, [
      '>> [CAULDRON:CASCADE] Reverse-causal 4-repo reset completed [OK]',
    ])

    if (bal?.slv) {
      bal.slv({
        mnuBit: {
          idx: 'cauldron-fleet-reset-pass',
          val: 1,
          dat: { mode: 'CAULDRON_CASCADE', cauldronBit },
        },
      })
    }
    return cpy
  }

  // Standalone fallback when 000.camp-candor-cauldron is not mounted on shared bus
  await emitConsoleLines(ste, [
    '>> [CAULDRON:WARN] 000.camp-candor-cauldron pivot not mounted on shared bus. Executing local 299.lore Void-State reset fallback...',
  ])
  await yieldPaintCycle()

  let localReceipt: LoreTabulaRasaReceipt | undefined
  await executeTabulaRasaLore(
    new LoreModel(),
    {
      idx: 'standalone-cauldron-fallback',
      src: repoRoot,
      dat: {
        repoRoot,
        confirmToken1: token1,
        confirmToken2: token2,
        confirmToken3: token3,
        cascadeOrigin: 'LOCAL_TUI',
      },
      slv: (res: any) => {
        localReceipt = res?.lorBit?.dat as LoreTabulaRasaReceipt
      },
    },
    ste,
  )

  if (localReceipt && localReceipt.ok) {
    resetMenuModelToVoidState(cpy, localReceipt)
  }

  if (bal?.slv) {
    bal.slv({
      mnuBit: {
        idx: localReceipt?.ok
          ? 'cauldron-fleet-reset-fallback-pass'
          : 'cauldron-fleet-reset-aborted',
        val: localReceipt?.ok ? 1 : 0,
        dat: { mode: 'STANDALONE_FALLBACK', receipt: localReceipt },
      },
    })
  }

  return cpy
}

        CAULDRON_FLEET_RESET_LABEL,
        '[<] CANCEL & RETURN',

      net: modeGrid?.grdBit?.dat,

    await yieldPaintCycle()

    // Stage 2/3 Red Alert Confirmation Modal (shifted affirmative index)

      if (bal?.slv) {
        bal.slv({ mnuBit: { idx: 'tabula-rasa-menu-aborted', val: 0 } })

      return cpy

        [STAGE_3_NO]: 'Stand down and disarm reset.',

        mnuBit: {
          idx: 'tabula-rasa-menu-aborted',
          val: 0,
          dat: receipt,
        },
      })
    }

      mnuBit: {
        idx: 'tabula-rasa-menu-pass',
        val: 1,
        dat: receipt,
        lst: summaryReceipts,
      },
    })
  }

  return cpy
}
