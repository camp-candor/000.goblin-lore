    while (curr && curr !== path.dirname(curr)) {
        if (
            fs.existsSync(path.join(curr, 'series')) ||
            fs.existsSync(path.join(curr, 'packages', '299.lore'))
        ) {
            return curr
        }
        curr = path.dirname(curr)
    }
    return path.resolve(startDir)
}

export function normalizeItemId(raw: string): string {
    const cleaned = toAscii(raw)
        .toLowerCase()
        .replace(/[\s/\\:]+/g, '-')
        .replace(/[^a-z0-9_-]/g, '')
        .replace(/-+/g, '-')
        .replace(/^[-_]+|[-_]+$/g, '')

    if (!cleaned) {
        return 'item_prop-01'
    }
    return cleaned.startsWith('item_') ? cleaned : `item_${cleaned}`
}

export function classifyMassGrams(massGrams: number): PossessionWeightTier {
    const safeMass = Math.max(1, Math.trunc(massGrams))
    if (safeMass <= 50) return 'SCRAP'
    if (safeMass <= 250) return 'SHINE'
    if (safeMass <= 5000) return 'WEIGHT'
    return 'BULK'
}

export function renderPossessionDossierMarkdown(
    params: RenderPossessionDossierParams,
): string {
    const frontmatterObj: Record<string, any> = {
        id: params.itemId,
        type: 'possession',
        name: toAscii(params.name),
        mass_grams: Math.max(1, Math.trunc(params.massGrams)),
        weight_class: params.weightClass,
        occupies_hands_bps: params.occupiesHandsBps,
        equip_slot: params.equipSlot,
        custody_holder: params.custodyHolder,
        location_id: params.locationId,
        is_contested: Boolean(params.isContested),
    }

    const yamlDump = yaml
        .dump(frontmatterObj, {
            indent: 2,
            lineWidth: -1,
            noRefs: true,
            sortKeys: true,
        })
        .replace(/\r\n/g, '\n')
        .trimEnd()

    const custodyDesc = params.custodyHolder
        ? `Bound to character ${params.custodyHolder} (${params.equipSlot}).`
        : `Placed at location ${params.locationId} (${params.equipSlot}).`

    const bodyLines = [
        '## Physical Description',
        `Standardized ${params.weightClass} artifact weighing ${frontmatterObj.mass_grams}g (${params.occupiesHandsBps} bp hand demand).`,
        '',
        '## Dramatic Function',
        custodyDesc,
        '',
    ]

    return `---\n${yamlDump}\n---\n${bodyLines.join('\n')}`
}

export function writePossessionDossierAtomic(
    targetFilePath: string,
    markdownContent: string,
): void {
    const normalized = markdownContent.replace(/\r\n/g, '\n')
    const buf = Buffer.from(normalized, 'utf8')

    if (
        buf.length < 4 ||
        buf[0] !== 0x2d ||
        buf[1] !== 0x2d ||
        buf[2] !== 0x2d ||
        buf[3] !== 0x0a
    ) {
        throw new Error(
            '[BYTE_0_VIOLATION] Expected opening "---\\n" (0x2d2d2d0a) at byte offset 0',
        )
    }

    fs.mkdirSync(path.dirname(targetFilePath), { recursive: true })
    const tmpPath = `${targetFilePath}.tmp_${Date.now()}_${Math.random().toString(36).slice(2, 7)}`
    fs.writeFileSync(tmpPath, buf)
    fs.renameSync(tmpPath, targetFilePath)
}

function listDomainIds(
    seriesSlug: string,
    domain: 'characters' | 'locations',
    repoRoot: string,
): string[] {
    const dir = path.join(repoRoot, 'series', seriesSlug, domain)
    if (!fs.existsSync(dir)) return []
    return fs
        .readdirSync(dir)
        .filter((f) => f.endsWith('.md') && !f.startsWith('.') && !f.startsWith('_'))
        .map((f) => f.replace(/\.md$/i, ''))
        .sort((a, b) => a.localeCompare(b))
}

function verifyCharacterArmBudget(
    seriesSlug: string,
    characterId: string,
    itemId: string,
    requiredHandsBps: number,
    repoRoot: string,
): { ok: boolean; reason?: string } {
    if (requiredHandsBps === 0) return { ok: true }
    const charPath = path.join(
        repoRoot,
        'series',
        seriesSlug,
        'characters',
        `${characterId}.md`,
    )
    if (!fs.existsSync(charPath)) {
        return {
            ok: false,
            reason: `[DANGLING_CUSTODY_HOLDER] Character '${characterId}' does not exist on disk`,
        }
    }

    try {
        const parsed = matter(fs.readFileSync(charPath, 'utf8'))
        const fm = (parsed.data || {}) as Record<string, any>
        const availableArmsBps =
            typeof fm.somatic?.available_arms_bps === 'number'
                ? fm.somatic.available_arms_bps
                : 20000
        const existingHeld: string[] = Array.isArray(fm.logistical?.held)
            ? fm.logistical.held
            : []

        const projectedItems = [
            ...existingHeld
                .filter((id) => id !== itemId)
                .map((id) => ({ item_id: id, occupies_hands_bps: 10000 })),
            { item_id: itemId, occupies_hands_bps: requiredHandsBps },
        ]

        const audit = auditEntityCapacityPreFlight(
            { available_arms_bps: availableArmsBps },
            projectedItems,
        )
        if (!audit.valid) {
            return {
                ok: false,
                reason: `[HAND_CAPACITY_BREACH] Character '${characterId}' (${availableArmsBps} bp) cannot equip '${itemId}' (${requiredHandsBps} bp)`,
            }
        }
    } catch {
        // Non-blocking if sparse stub
    }

    return { ok: true }
}

async function promptGridInput(
    lib: any,
    label: string,
    defaultVal: string,
): Promise<{ cancelled: boolean; value: string }> {
    const gridBit: any = await lib.hunt(UPDATE_GRID, {
        x: 0,
        y: 4,
        xSpan: 4,
        ySpan: 4,
    })
    const inputBit: any = await lib.hunt(OPEN_INPUT, {
        txt: label,
        src: defaultVal,
        net: gridBit?.grdBit?.dat,
    })
    if (inputBit?.putBit?.val === -1) {
        return { cancelled: true, value: '' }
    }
    const raw = String(inputBit?.putBit?.src || '').trim()
    return { cancelled: false, value: raw.length > 0 ? raw : defaultVal }
}

async function promptGridChoice(
    lib: any,
    choices: string[],
): Promise<{ cancelled: boolean; value: string }> {
    const gridBit: any = await lib.hunt(UPDATE_GRID, {
        x: 0,
        y: 4,
        xSpan: 4,
        ySpan: 8,
    })
    const choiceBit: any = await lib.hunt(OPEN_CHOICE, {
        dat: { clr0: 'black', clr1: 'yellow' },
        src: 'vertical',
        lst: choices,
        net: gridBit?.grdBit?.dat,
    })
    if (choiceBit?.chcBit?.val === -1) {
        return { cancelled: true, value: '' }
    }
    return {
        cancelled: false,
        value: String(choiceBit?.chcBit?.src || choices[0] || ''),
    }
}

export const scaffoldPossessionWizard = async (
    cpy: ItemModel,
    bal: ItemScaffoldBit,
    ste?: State,
): Promise<ItemModel> => {
    const dat: ItemScaffoldInput = bal?.dat || {}
    const repoRoot = dat.repoRoot
        ? path.resolve(dat.repoRoot)
        : resolveItemRepoRoot()
    const seriesSlug = dat.seriesSlug || 'under-the-floorboards'
    const lib = resolveLibraryBus()
    const isInteractive = Boolean(
        lib && typeof lib.hunt === 'function' && !dat.itemId && !dat.name,
    )

    let itemId = dat.itemId ? normalizeItemId(dat.itemId) : 'item_lasso-of-truth'
    let displayName = dat.name ? toAscii(dat.name) : 'Lasso of Truth'
    let massGrams =
        typeof dat.massGrams === 'number'
            ? Math.max(1, Math.trunc(dat.massGrams))
            : 1200
    let occupiesHandsBps: 0 | 10000 | 20000 =
        dat.occupiesHandsBps === 0 ||
        dat.occupiesHandsBps === 10000 ||
        dat.occupiesHandsBps === 20000
            ? dat.occupiesHandsBps
            : 10000
    let equipSlot: PossessionEquipSlot =
        dat.equipSlot || (occupiesHandsBps > 0 ? 'held_primary' : 'worn_belt')
    let custodyHolder: string | null = dat.custodyHolder || null
    let locationId: string | null = dat.locationId || null
    let isContested = Boolean(dat.isContested)

    if (isInteractive) {
        emitConsole(
            ste,
            `>> [POSSESSION:WIZARD] Starting Possession Interview for '${seriesSlug}'...`,
        )

        const idInput = await promptGridInput(
            lib,
            'ITEM ID (item_slug)',
            'item_lasso-of-truth',
        )
        if (idInput.cancelled) {
            if (bal?.slv) bal.slv({ itmBit: { success: false, cancelled: true } })
            return cpy
        }
        itemId = normalizeItemId(idInput.value)

        const nameInput = await promptGridInput(
            lib,
            'POSSESSION DISPLAY NAME',
            'Lasso of Truth',
        )
        if (nameInput.cancelled) {
            if (bal?.slv) bal.slv({ itmBit: { success: false, cancelled: true } })
            return cpy
        }
        displayName = toAscii(nameInput.value)

        // Task 4.1: Mass in integer grams
        const massInput = await promptGridInput(
            lib,
            'MASS IN GRAMS (Integer >= 1)',
            '1200',
        )
        if (massInput.cancelled) {
            if (bal?.slv) bal.slv({ itmBit: { success: false, cancelled: true } })
            return cpy
        }
        const parsedMass = Number.parseInt(massInput.value, 10)
        massGrams = Number.isNaN(parsedMass) ? 1200 : Math.max(1, parsedMass)

        // Task 4.1: Hand demand basis points
        const handChoice = await promptGridChoice(lib, [
            ...HAND_DEMAND_PRESETS.map((p) => p.label),
            '[<] CANCEL POSSESSION WIZARD',
        ])
        if (
            handChoice.cancelled ||
            handChoice.value === '[<] CANCEL POSSESSION WIZARD'
        ) {
            if (bal?.slv) bal.slv({ itmBit: { success: false, cancelled: true } })
            return cpy
        }
        const matchedHand =
            HAND_DEMAND_PRESETS.find((p) => p.label === handChoice.value) ||
            HAND_DEMAND_PRESETS[1]
        occupiesHandsBps = matchedHand.occupiesHandsBps
        equipSlot = matchedHand.defaultSlot

        // Task 4.2: Custody Foreign Key Binder (Character Holder or Spatial Location)
        const characters = listDomainIds(seriesSlug, 'characters', repoRoot)
        const locations = listDomainIds(seriesSlug, 'locations', repoRoot)

        const custodyOptions: string[] = [
            ...characters.map((c) => `HOLDER: ${c}`),
            ...locations.map((l) => `LOCATION: ${l}`),
        ]

        if (custodyOptions.length > 0) {
            const custodyChoice = await promptGridChoice(lib, custodyOptions)
            if (custodyChoice.cancelled) {
                if (bal?.slv) {
                    bal.slv({ itmBit: { success: false, cancelled: true } })
                }
                return cpy
            }
            if (custodyChoice.value.startsWith('HOLDER: ')) {
                custodyHolder = custodyChoice.value
                    .replace('HOLDER: ', '')
                    .trim()
                locationId = null
            } else if (custodyChoice.value.startsWith('LOCATION: ')) {
                locationId = custodyChoice.value
                    .replace('LOCATION: ', '')
                    .trim()
                custodyHolder = null
                equipSlot = 'floor'
            }
        } else {
            locationId = 'loc_gotham-rooftop'
            equipSlot = 'floor'
        }

        const contestedChoice = await promptGridChoice(lib, [
            '[1] CONTESTED DISPUTE CATALYST (is_contested: true)',
            '[2] STANDARD UNCONTESTED PROP (is_contested: false)',
        ])
        isContested =
            !contestedChoice.cancelled &&
            contestedChoice.value.startsWith('[1]')
    }

    if (!ITEM_ID_REGEX.test(itemId)) {
        const errMsg = `[INVALID_ITEM_ID] '${itemId}' does not match ${ITEM_ID_REGEX.source}`
        emitConsole(ste, `:: [POSSESSION:FAIL] ${errMsg}`)
        if (bal?.slv) {
            bal.slv({ itmBit: { success: false, itemId, error: errMsg } })
        }
        return cpy
    }

    if (!custodyHolder && !locationId) {
        const errMsg =
            '[UNGROUNDED_POSSESSION_BREACH] Possession must bind to a character holder or location node'
        emitConsole(ste, `:: [POSSESSION:FAIL] ${errMsg}`)
        if (bal?.slv) {
            bal.slv({ itmBit: { success: false, itemId, error: errMsg } })
        }
        return cpy
    }

    if (custodyHolder) {
        const satCheck = verifyCharacterArmBudget(
            seriesSlug,
            custodyHolder,
            itemId,
            occupiesHandsBps,
            repoRoot,
        )
        if (!satCheck.ok) {
            emitConsole(ste, `:: [SAT:BREACH] ${satCheck.reason}`)
            if (bal?.slv) {
                bal.slv({
                    itmBit: {
                        success: false,
                        itemId,
                        error: satCheck.reason,
                    },
                })
            }
            return cpy
        }
    }

    const weightClass = classifyMassGrams(massGrams)
    const markdown = renderPossessionDossierMarkdown({
        itemId,
        name: displayName,
        massGrams,
        weightClass,
        occupiesHandsBps,
        equipSlot,
        custodyHolder,
        locationId,
        isContested,
    })

    const targetFilePath = path.join(
        repoRoot,
        'series',
        seriesSlug,
        'possessions',
        `${itemId}.md`,
    )
    writePossessionDossierAtomic(targetFilePath, markdown)

    const custodyTarget = custodyHolder || locationId || 'unbound'
    emitConsole(
        ste,
        `>> [SEALED] Possession '${itemId}' (${weightClass}, ${massGrams}g, ${occupiesHandsBps}bp -> ${custodyTarget}) written at Byte 0 [OK]`,
    )

    if (bal?.slv) {
        bal.slv({
            itmBit: {
                success: true,
                filePath: targetFilePath,
                itemId,
                weightClass,
                massGrams,
                occupiesHandsBps,
                custodyTarget,
            },
        })
    }

    return cpy
}

export function renderGrievanceDossierMarkdown(
    params: RenderGrievanceDossierParams,
): string {
    const safeCeiling = params.tonalOverride
        ? Math.min(2000, Math.max(0, Math.trunc(params.escalationCeilingBp)))
        : Math.min(10000, Math.max(0, Math.trunc(params.escalationCeilingBp)))
    const safeIntensity = Math.min(
        safeCeiling,
        Math.max(0, Math.trunc(params.intensityBp)),
    )
    const safeSpawns: GrievanceSpawnTargetRef[] = params.tonalOverride
        ? []
        : [...params.spawnsOnMaxEscalation]
    const underneathText = params.tonalOverride
        ? FARCE_UNDERNEATH_LITERAL
        : toAscii(params.theUnderneath)

    const frontmatterObj: Record<string, any> = {
        id: params.grvId,
        type: 'grievance',
        title: toAscii(params.title),
        category: params.category,
        participants_primary: [
            params.participantsPrimary[0],
            params.participantsPrimary[1],
        ],
        participants_can_involve: [...params.participantsCanInvolve],
        required_props: [...params.requiredProps],
        location_tags: [...params.locationTags],
        intensity_bp: safeIntensity,
        escalation_ceiling_bp: safeCeiling,
        cooldown_ticks: Math.max(0, Math.trunc(params.cooldownTicks)),
        tonal_override: Boolean(params.tonalOverride),
        spawns_on_max_escalation: safeSpawns,
        outcomes: buildDefaultOutcomes(),
    }

    const yamlDump = yaml
        .dump(frontmatterObj, {
            indent: 2,
            lineWidth: -1,
            noRefs: true,
            sortKeys: true,
        })
        .replace(/\r\n/g, '\n')
        .trimEnd()

    const bodyLines = [
        '## Surface Complaint',
        toAscii(params.surfaceComplaint),
        '',
        '## The Underneath',
        underneathText,
        '',
        '## Escalation Path',
        `Escalates from ${safeIntensity} bp toward ${safeCeiling} bp ceiling across ${frontmatterObj.cooldown_ticks}-tick cooldown cycles.`,
        '',
        '## Callback Phrases',
        `- "You still think rules keep this city breathing, ${params.participantsPrimary[1]}?"`,
        '',
    ]

    return `---\n${yamlDump}\n---\n${bodyLines.join('\n')}`
}

export function writeGrievanceDossierAtomic(
    targetFilePath: string,
    markdownContent: string,
): void {
    const normalized = markdownContent.replace(/\r\n/g, '\n')
    const buf = Buffer.from(normalized, 'utf8')

    if (
        buf.length < 4 ||
        buf[0] !== 0x2d ||
        buf[1] !== 0x2d ||
        buf[2] !== 0x2d ||
        buf[3] !== 0x0a
    ) {
        throw new Error(
            '[BYTE_0_VIOLATION] Expected opening "---\\n" (0x2d2d2d0a) at byte offset 0',
        )
    }

    fs.mkdirSync(path.dirname(targetFilePath), { recursive: true })
    const tmpPath = `${targetFilePath}.tmp_${Date.now()}_${Math.random().toString(36).slice(2, 7)}`
    fs.writeFileSync(tmpPath, buf)
    fs.renameSync(tmpPath, targetFilePath)
}

function listDomainFiles(
    seriesSlug: string,
    domain: 'characters' | 'possessions' | 'locations' | 'grievances',
    repoRoot: string,
): string[] {
    const dir = path.join(repoRoot, 'series', seriesSlug, domain)
    if (!fs.existsSync(dir)) return []
    return fs
        .readdirSync(dir)
        .filter((f) => f.endsWith('.md') && !f.startsWith('.') && !f.startsWith('_'))
        .map((f) => f.replace(/\.md$/i, ''))
        .sort((a, b) => a.localeCompare(b))
}

/**
 * Task 4.4: Builds the series grievance adjacency graph and verifies that adding
 * candidate spawns_on_max_escalation edges introduces zero cycles via 3-color DFS.
 */
export function verifyGrievanceDagAcyclic(options: {
    seriesSlug: string
    candidateId: string
    candidateSpawns: readonly GrievanceSpawnTargetRef[]
    repoRoot: string
}): { valid: boolean; reason?: string } {
    const { seriesSlug, candidateId, candidateSpawns, repoRoot } = options
    if (candidateSpawns.length === 0) {
        return { valid: true }
    }

    const grvDir = path.join(repoRoot, 'series', seriesSlug, 'grievances')
    const adjacency: Record<string, string[]> = {}

    if (fs.existsSync(grvDir)) {
        for (const file of fs.readdirSync(grvDir)) {
            if (!file.endsWith('.md') || file.startsWith('.') || file.startsWith('_')) {
                continue
            }
            const nodeId = file.replace(/\.md$/i, '')
            try {
                const parsed = matter(
                    fs.readFileSync(path.join(grvDir, file), 'utf8'),
                )
                const fm = (parsed.data || {}) as Record<string, any>
                const spawns = Array.isArray(fm.spawns_on_max_escalation)
                    ? fm.spawns_on_max_escalation
                    : []
                adjacency[nodeId] = spawns
                    .map((s: any) =>
                        typeof s === 'string' ? s : String(s?.target_id || ''),
                    )
                    .filter(Boolean)
            } catch {
                adjacency[nodeId] = []
            }
        }
    }

    for (const spawn of candidateSpawns) {
        if (spawn.target_id === candidateId) {
            return {
                valid: false,
                reason: `[DAG_CYCLE_BREACH] Grievance '${candidateId}' cannot spawn itself`,
            }
        }
        if (!adjacency[spawn.target_id]) {
            return {
                valid: false,
                reason: `[DANGLING_SPAWN_TARGET] Child grievance '${spawn.target_id}' does not exist in series/${seriesSlug}/grievances/`,
            }
        }
    }

    adjacency[candidateId] = candidateSpawns.map((s) => s.target_id)
    const cycleReport = detectEscalationCycles(adjacency)
    if (cycleReport.hasCycle) {
        return {
            valid: false,
            reason: `[DAG_CYCLE_BREACH] Escalation cycle detected: ${cycleReport.cyclePath.join(' -> ')}`,
        }
    }

    return { valid: true }
}

async function promptGridInput(
    lib: any,
    label: string,
    defaultVal: string,
): Promise<{ cancelled: boolean; value: string }> {
    const gridBit: any = await lib.hunt(UPDATE_GRID, {
        x: 0,
        y: 4,
        xSpan: 4,
        ySpan: 4,
    })
    const inputBit: any = await lib.hunt(OPEN_INPUT, {
        txt: label,
        src: defaultVal,
        net: gridBit?.grdBit?.dat,
    })
    if (inputBit?.putBit?.val === -1) {
        return { cancelled: true, value: '' }
    }
    const raw = String(inputBit?.putBit?.src || '').trim()
    return { cancelled: false, value: raw.length > 0 ? raw : defaultVal }
}

async function promptGridChoice(
    lib: any,
    choices: string[],
): Promise<{ cancelled: boolean; value: string }> {
    const gridBit: any = await lib.hunt(UPDATE_GRID, {
        x: 0,
        y: 4,
        xSpan: 4,
        ySpan: 8,
    })
    const choiceBit: any = await lib.hunt(OPEN_CHOICE, {
        dat: { clr0: 'black', clr1: 'yellow' },
        src: 'vertical',
        lst: choices,
        net: gridBit?.grdBit?.dat,
    })
    if (choiceBit?.chcBit?.val === -1) {
        return { cancelled: true, value: '' }
    }
    return {
        cancelled: false,
        value: String(choiceBit?.chcBit?.src || choices[0] || ''),
    }
}

                    success: false,

        title,
        category,

        participantsCanInvolve,
        requiredProps,
        locationTags,
        intensityBp,

        }
        return cpy
    }

    if (domainDescriptor.domain === 'grievance') {
        let grvOutcome: any = null
        const grvPayload = {
            idx: 'menu-scaffold-grievance',
            dat: {
                seriesSlug,
                grvId: dat.entityId ? entityId : undefined,
                title: dat.title || dat.name,
                category: dat.category,
                participantsPrimary: dat.participantsPrimary,
                participantsCanInvolve: dat.participantsCanInvolve,
                requiredProps: dat.requiredProps,
                locationTags: dat.locationTags,
                intensityBp: dat.intensityBp,
                escalationCeilingBp: dat.escalationCeilingBp,
                cooldownTicks: dat.cooldownTicks,
                tonalOverride: dat.tonalOverride,
                spawnsOnMaxEscalation: dat.spawnsOnMaxEscalation,
                surfaceComplaint: dat.surfaceComplaint,
                theUnderneath: dat.theUnderneath,
                repoRoot: dat.repoRoot,
            },
            slv: (res: any) => {
                grvOutcome = res?.grvBit || res
            },
        }

        if (ste && typeof ste.hunt === 'function') {
            await ste.hunt(ActGrv.SCAFFOLD_GRIEVANCE_WIZARD, grvPayload)
        } else {
            await scaffoldGrievanceWizard(new GrievanceModel(), grvPayload, ste)
        }

        if (grvOutcome?.cancelled) {
            const abortLine =
                '>> [SCAFFOLD:ABORT] Grievance interview cancelled by operator.'
            emitConsole(ste, abortLine)
            cpy.activeSelection = '[S] ATOMIC ENTITY SCAFFOLDER'
            cpy.lastLifecycleReceipt = '[SCAFFOLD:CANCELLED]'
            cpy.statusMessage = abortLine
            if (bal?.slv) {
                bal.slv({
                    mnuBit: {
                        idx: 'open-scaffold-menu',
                        val: 0,
                        dat: { cancelled: true, receipt: abortLine },
                    },
                })
            }
            return cpy
        }

        const grvOk = Boolean(grvOutcome?.success)
        const grvIdFinal = grvOutcome?.grvId || entityId
        const receiptLine = grvOk
            ? `>> [SCAFFOLD:OK] Byte-0 grievance dossier sealed for ${grvIdFinal} in ${seriesSlug}/grievances`
            : `:: [SCAFFOLD:FAIL] Grievance wizard error: ${grvOutcome?.error || 'unknown'}`

        emitConsole(ste, receiptLine)
        cpy.activeSelection = '[S] ATOMIC ENTITY SCAFFOLDER'
        cpy.lastLifecycleReceipt = grvOk ? '[SCAFFOLD:OK]' : '[SCAFFOLD:FAIL]'
        cpy.statusMessage = receiptLine
        if (!Array.isArray(cpy.lastTelemetryReport)) {
            cpy.lastTelemetryReport = []
        }
        cpy.lastTelemetryReport.push(receiptLine)

        if (bal?.slv) {
            bal.slv({
                mnuBit: {
                    idx: 'open-scaffold-menu',
                    val: grvOk ? 1 : -1,
                    src: grvIdFinal,
                    dat: {
                        receipt: receiptLine,
                        seriesSlug,
                        domain,
                        idPrefix,
                        grievanceResult: grvOutcome,
                    },
                },
            })
        }
        return cpy
    }
