export function inspectLoreBlastRadiusSummary(
    repoRootInput?: string,
): LoreBlastRadiusSummary {
    const repoRoot = repoRootInput
        ? path.resolve(repoRootInput)
        : resolveRepoRoot()
    const seriesDir = path.join(repoRoot, 'series')
    const compiledDir = path.join(repoRoot, 'compiled')
    const errataFile = path.join(repoRoot, 'lore', 'errata.json')

    let seriesDirectories = 0
    let markdownDossiers = 0
    let compiledManifests = 0
    let scratchpadFiles = 0

    if (fs.existsSync(seriesDir)) {
        const seriesEntries = fs.readdirSync(seriesDir, { withFileTypes: true })
        for (const entry of seriesEntries) {
            if (!entry.isDirectory()) continue
            seriesDirectories++
            const subRoot = path.join(seriesDir, entry.name)
            const stack: string[] = [subRoot]
            while (stack.length > 0) {
                const current = stack.pop()!
                const children = fs.readdirSync(current, { withFileTypes: true })
                for (const child of children) {
                    const full = path.join(current, child.name)
                    if (child.isDirectory()) {
                        stack.push(full)
                    } else if (child.isFile()) {
                        if (child.name.endsWith('.md')) markdownDossiers++
                        if (child.name.startsWith('.tmp_')) scratchpadFiles++
                    }
                }
            }
        }
    }

    if (fs.existsSync(compiledDir)) {
        const compEntries = fs.readdirSync(compiledDir, { withFileTypes: true })
        for (const entry of compEntries) {
            if (!entry.isDirectory()) continue
            const biblePath = path.join(
                compiledDir,
                entry.name,
                'bible-state.json',
            )
            if (fs.existsSync(biblePath)) compiledManifests++
        }
    }

    return {
        repoRoot,
        seriesDirectories,
        markdownDossiers,
        compiledManifests,
        scratchpadFiles,
        errataPresent: fs.existsSync(errataFile),
    }
}

async function mountRedAlertChoiceStage(
    lib: any,
    choices: readonly string[],
    expectedConfirmIndex: number,
    stageNumber: 1 | 2 | 3,
): Promise<TripleLockStageOutcome> {
    const gridBit: any = await lib.hunt(UPDATE_GRID, {
        x: 0,
        y: 4,
        xSpan: 4,
        ySpan: 8,
    })

    const choiceBit: any = await lib.hunt(OPEN_CHOICE, {
        dat: {
            clr0: 'black',
            clr1: 'red',
        },
        src: 'vertical',
        lst: [...choices],
        net: gridBit?.grdBit?.dat,
    })

    const val = choiceBit?.chcBit?.val
    const src = String(choiceBit?.chcBit?.src || '')
    const expectedLabel = choices[expectedConfirmIndex]

    if (val === -1 || choiceBit?.chcBit?.idx === 'open-choice-cancel') {
        return {
            cleared: false,
            stage: stageNumber,
            selectedLabel: src || 'ESCAPE_CANCEL',
            abortReason: `Operator pressed Escape at Stage ${stageNumber}/3`,
        }
    }

    const indexMatches =
        typeof val === 'number'
            ? val === expectedConfirmIndex
            : src === expectedLabel
    const labelMatches = src === expectedLabel

    if (!indexMatches || !labelMatches) {
        return {
            cleared: false,
            stage: stageNumber,
            selectedLabel: src || choices[0],
            abortReason: `Operator declined reset at Stage ${stageNumber}/3`,
        }
    }

    return {
        cleared: true,
        stage: stageNumber,
        selectedLabel: src,
    }
}

                    cleared: true,

                    loreResetResult,
