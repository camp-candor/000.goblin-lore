import * as fs from 'node:fs'
import * as path from 'node:path'
import type State from '../../99.core/state.js'
import type { StatusModel } from '../status.model.js'
import type StatusBit from '../fce/status.bit.js'
import { resolveRepoRoot } from '../../00.lore.unit/buz/lore.buzz.js'
import type { VoidScanResultBit } from '../fce/voidScan.bit.js'
import { GENESIS_MERKLE_ROOT_64 } from '../../38.types.unit/fce/tabulaRasa.interface.js'

/**
 * Pure filesystem evaluator scanning <repoRoot>/series for directories
 * possessing a valid, non-empty, and readable series.config.json manifest.
 */
export const scanVoidState = (repoRootInput?: string): VoidScanResultBit => {
  const repoRoot = repoRootInput || resolveRepoRoot()
  const seriesPath = path.join(repoRoot, 'series')

  if (!fs.existsSync(seriesPath)) {
    return {
      isVoidState: true,
      discoveredSlugs: [],
      repoRoot,
      seriesPath,
    }
  }

  const entries = fs.readdirSync(seriesPath, { withFileTypes: true })
  const discoveredSlugs: string[] = []

  for (const entry of entries) {
    if (entry.isDirectory()) {
      const configPath = path.join(seriesPath, entry.name, 'series.config.json')
      if (fs.existsSync(configPath)) {
        try {
          const raw = fs.readFileSync(configPath, 'utf-8')
          if (raw.trim().length > 0) {
            const parsed = JSON.parse(raw)
            if (parsed && typeof parsed === 'object' && !Array.isArray(parsed)) {
              discoveredSlugs.push(entry.name)
            }
          }
        } catch {
          // Unparseable or empty manifests are ignored; directory does not qualify as sealed
        }
      }
    }
  }

  discoveredSlugs.sort((a, b) => (a < b ? -1 : a > b ? 1 : 0))

  return {
    isVoidState: discoveredSlugs.length === 0,
    discoveredSlugs,
    repoRoot,
    seriesPath,
  }
}

/**
 * Reactive Buzzer handler for ActSts.EVALUATE_VOID_STATE.
 * Synchronizes StatusModel HUD lines, dispatch lock, border color, and genesis Merkle root.
 */
export const voidStateScan = async (
  cpy: StatusModel,
  bal: StatusBit,
  _ste?: State,
): Promise<StatusModel> => {
  const targetRoot = bal?.src || undefined
  const result = scanVoidState(targetRoot)

  cpy.isVoidState = result.isVoidState
  cpy.discoveredSlugs = [...result.discoveredSlugs]
  cpy.dispatchLockEngaged = result.isVoidState
  cpy.activeBorderColor = result.isVoidState ? 'amber' : 'green'

  if (result.isVoidState) {
    cpy.hudLines = [
      'ACTIVE SERIES : [NONE (UNBOUND)]',
      'CANON STATUS  : VOID STATE (T0=$0.00)',
      `ROOT MERKLE   : ${GENESIS_MERKLE_ROOT_64}`,
      'DISPATCH LOCK : [!] ENGAGED',
    ]
  } else {
    cpy.hudLines = [
      `ACTIVE SERIES : [${result.discoveredSlugs.join(', ')}]`,
      'CANON STATUS  : SEALED CORPUS ACTIVE',
      `SERIES COUNT  : ${result.discoveredSlugs.length}`,
      'DISPATCH LOCK : [OK] DISENGAGED',
    ]
  }

  if (bal?.slv) {
    bal.slv({
      statusBit: {
        idx: 'void-state-scan',
        val: result.isVoidState ? 1 : 0,
        dat: {
          ...result,
          merkleRootHex: result.isVoidState ? GENESIS_MERKLE_ROOT_64 : undefined,
          hudLines: [...cpy.hudLines],
        },
      },
    })
  }

  return cpy
}

  ste: State | undefined,
  lines: readonly string[],
): Promise<void> {
  const lib = (globalThis as any).LIBRARY || (global as any).LIBRARY
  for (const rawLine of lines) {
    const clean = sanitizeAscii(rawLine)
    if (lib && typeof lib.hunt === 'function') {
      await lib.hunt(UPDATE_CONSOLE, { idx: 'cns00', src: clean }).catch(() => {})
    } else if (ste && typeof ste.hunt === 'function') {
      await ste.hunt(UPDATE_CONSOLE, { idx: 'cns00', src: clean }).catch(() => {})
    }
  }
}

function collectFilesRecursive(dirPath: string): string[] {
  if (!fs.existsSync(dirPath)) {
    return []
  }
  const files: string[] = []
  const entries = fs.readdirSync(dirPath, { withFileTypes: true })
  for (const entry of entries) {
    if (entry.name === '.gitkeep') {
      continue
    }
    const fullPath = path.join(dirPath, entry.name)
    if (entry.isDirectory()) {
      files.push(...collectFilesRecursive(fullPath))
    } else if (entry.isFile()) {
      files.push(fullPath)
    }
  }
  return files
}

function isScratchpadFileName(filename: string): boolean {
  return (
    filename.startsWith('.tmp_') ||
    filename.includes('.tmp_') ||
    filename.endsWith('.tmp')
  )
}

function isErrataLedgerFileName(filename: string): boolean {
  const lower = filename.toLowerCase()
  return (
    lower === 'errata.json' ||
    (lower.startsWith('canon_errata_') && lower.endsWith('.json'))
  )
}

/**
 * Task 3.3: Purges lore/errata.json (and canon_errata_*.json files) as well as
 * orphaned .tmp_* conflict or editor scratchpad files across runtime folders.
 */
export async function purgeErrataAndScratchpads(
  repoRootInput?: string,
  ste?: State,
): Promise<{ errataFiles: number; scratchpadFiles: number }> {
  const repoRoot = resolveResetLoreRoot(repoRootInput || process.cwd())
  let errataFiles = 0
  let scratchpadFiles = 0

  // 1. Purge errata ledger files inside <repoRoot>/lore/
  const loreDir = path.join(repoRoot, 'lore')
  if (fs.existsSync(loreDir)) {
    const loreEntries = fs.readdirSync(loreDir, { withFileTypes: true })
    for (const entry of loreEntries) {
      if (!entry.isFile() || entry.name === '.gitkeep') continue
      if (isErrataLedgerFileName(entry.name)) {
        const candidate = path.join(loreDir, entry.name)
        const safePath = assertSafeResetTarget(repoRoot, candidate)
        fs.rmSync(safePath, { force: true })
        errataFiles++
      }
    }
  }

  // 2. Sweep and unlink orphaned scratchpad files across runtime directories
  const scratchpadScanDirs = [
    path.join(repoRoot, 'series'),
    path.join(repoRoot, 'compiled'),
    loreDir,
    path.join(repoRoot, 'data', 'lockfile'),
  ]

  for (const scanDir of scratchpadScanDirs) {
    const files = collectFilesRecursive(scanDir)
    for (const file of files) {
      if (isScratchpadFileName(path.basename(file))) {
        const safeScratchpad = assertSafeResetTarget(repoRoot, file)
        fs.rmSync(safeScratchpad, { force: true })
        scratchpadFiles++
      }
    }
  }

  await emitLoreResetTelemetry(ste, [
    `>> [TABULA_RASA:ERRATA] Purged ${errataFiles} errata ledger(s) and ${scratchpadFiles} scratchpad file(s) [OK]`,
  ])

  return { errataFiles, scratchpadFiles }
}

/**
 * Task 3.1: Non-destructive pre-flight scanner evaluating series/, compiled/,
 * lore/errata.json, and orphaned .tmp_* scratchpads prior to reset authorization.
 */
export async function scanLoreBlastRadius(
  repoRootInput?: string,
  ste?: State,
): Promise<LoreBlastRadiusInventory> {
  const repoRoot = resolveResetLoreRoot(repoRootInput || process.cwd())

  const seriesRoot = path.join(repoRoot, 'series')
  const discoveredSeriesSlugs: string[] = []
  let markdownDossiers = 0

  if (fs.existsSync(seriesRoot)) {
    const seriesEntries = fs.readdirSync(seriesRoot, { withFileTypes: true })
    for (const entry of seriesEntries) {
      if (
        entry.isDirectory() &&
        !entry.name.startsWith('.') &&
        entry.name !== '.gitkeep'
      ) {
        discoveredSeriesSlugs.push(entry.name)
      }
    }
    const allSeriesFiles = collectFilesRecursive(seriesRoot)
    for (const file of allSeriesFiles) {
      if (file.toLowerCase().endsWith('.md')) {
        markdownDossiers++
      }
    }
  }
  discoveredSeriesSlugs.sort((a, b) => a.localeCompare(b))

  const compiledRoot = path.join(repoRoot, 'compiled')
  let compiledBibleManifests = 0
  let topologyManifests = 0

  if (fs.existsSync(compiledRoot)) {
    const allCompiledFiles = collectFilesRecursive(compiledRoot)
    for (const file of allCompiledFiles) {
      const base = path.basename(file).toLowerCase()
      if (base === 'bible-state.json') {
        compiledBibleManifests++
      } else if (base === 'topology.manifest.json') {
        topologyManifests++
      }
    }
  }

  const loreDir = path.join(repoRoot, 'lore')
  let errataFiles = 0
  if (fs.existsSync(loreDir)) {
    const loreEntries = fs.readdirSync(loreDir, { withFileTypes: true })
    for (const entry of loreEntries) {
      if (!entry.isFile()) continue
      if (isErrataLedgerFileName(entry.name)) {
        errataFiles++
      }
    }
  }

  let scratchpadFiles = 0
  const scratchpadScanDirs = [
    seriesRoot,
    compiledRoot,
    loreDir,
    path.join(repoRoot, 'data', 'lockfile'),
  ]
  for (const scanDir of scratchpadScanDirs) {
    const files = collectFilesRecursive(scanDir)
    for (const file of files) {
      if (isScratchpadFileName(path.basename(file))) {
        scratchpadFiles++
      }
    }
  }

  const counts: LorePurgedCounts = {
    seriesDirectories: discoveredSeriesSlugs.length,
    markdownDossiers,
    compiledBibleManifests,
    topologyManifests,
    errataFiles,
    scratchpadFiles,
  }

  const targetPaths = [
    path.join(repoRoot, 'series'),
    path.join(repoRoot, 'compiled'),
    path.join(repoRoot, 'lore', 'errata.json'),
  ]

  const inventory: LoreBlastRadiusInventory = {
    repoRoot,
    counts,
    discoveredSeriesSlugs,
    targetPaths,
    scannedAtIso: new Date().toISOString(),
  }

  const telemetryLines = [
    '>> ============================================================',
    '>> [LORE:BLAST_RADIUS] PRE-FLIGHT VOID-STATE INVENTORY',
    `>> Target Root       : ${repoRoot}`,
    `>> Series Folders    : ${counts.seriesDirectories} (${discoveredSeriesSlugs.join(', ') || 'NONE'})`,
    `>> Markdown Dossiers : ${counts.markdownDossiers} .md file(s)`,
    `>> Compiled Manifests: ${counts.compiledBibleManifests} bible-state.json | ${counts.topologyManifests} topology`,
    `>> Errata & Temp     : ${counts.errataFiles} errata | ${counts.scratchpadFiles} scratchpad(s)`,
    '>> ============================================================',
  ]
  await emitLoreResetTelemetry(ste, telemetryLines)

  return inventory
}

/**
 * Tasks 3.2, 3.3 & 3.4: Complete local Void-State purge orchestrator.
 * Enforces Triple-Lock tokens, obliterates series/* and compiled/*, purges
 * errata ledgers and orphaned scratchpads, resets stateful BEE units, and
 * synchronizes 32.status.unit to Posture A Cold-Start Byte-0 Void State.
 */
export const executeTabulaRasaLore = async (
  cpy: LoreModel,
  bal?: LoreBit,
  ste?: State,
): Promise<LoreModel> => {
  const datObj = (bal?.dat || {}) as Record<string, any>
  const req: LoreTabulaRasaRequestBit =
    (datObj.tabulaRasaRequest as LoreTabulaRasaRequestBit) || datObj

  const repoRoot = resolveResetLoreRoot(
    req.repoRoot || bal?.src || process.cwd(),
  )
  const cascadeOrigin: LoreTabulaRasaCascadeOrigin =
    req.cascadeOrigin || 'LOCAL_TUI'

  const zeroCounts: LorePurgedCounts = {
    seriesDirectories: 0,
    markdownDossiers: 0,
    compiledBibleManifests: 0,
    topologyManifests: 0,
    errataFiles: 0,
    scratchpadFiles: 0,
  }

  let abortStage: 1 | 2 | 3 | undefined
  if (req.confirmToken1 !== TABULA_RASA_TOKEN_1) {
    abortStage = 1
  } else if (req.confirmToken2 !== TABULA_RASA_TOKEN_2) {
    abortStage = 2
  } else if (req.confirmToken3 !== TABULA_RASA_TOKEN_3) {
    abortStage = 3
  }

  if (abortStage !== undefined) {
    const abortLine = `:: [TABULA_RASA:ABORTED] Triple-Lock verification failed at Stage ${abortStage}/3. Disk untouched.`
    await emitLoreResetTelemetry(ste, [abortLine])

    const abortedReceipt: LoreTabulaRasaReceipt = {
      ok: false,
      aborted: true,
      abortStage,
      voidStateVerified: false,
      cascadeOrigin,
      purgedCounts: zeroCounts,
      unitsResetList: [],
      merkleRootHex: GENESIS_MERKLE_ROOT_64,
      dispatchLockEngaged: true,
      timestampIso: new Date().toISOString(),
      receipts: [abortLine],
    }

    if (bal?.slv) {
      bal.slv({
        lorBit: {
          idx: 'execute-tabula-rasa-lore-aborted',
          val: 0,
          dat: abortedReceipt,
        },
      })
    }
    return cpy
  }

  const inventory = await scanLoreBlastRadius(repoRoot, ste)
  const receipts: string[] = [
    `>> [TABULA_RASA:INIT] Executing 299.lore atomic disk purge (origin=${cascadeOrigin})...`,
  ]

  // Task 3.3: Purge errata ledgers and orphaned scratchpads across runtime dirs
  const errataPurgeStats = await purgeErrataAndScratchpads(repoRoot, ste)
  receipts.push(
    `>> [TABULA_RASA:ERRATA] Purged ${errataPurgeStats.errataFiles} errata ledger(s) and ${errataPurgeStats.scratchpadFiles} scratchpad file(s) [OK]`,
  )

  const seriesRoot = assertSafeResetTarget(
    repoRoot,
    path.join(repoRoot, 'series'),
  )
  const compiledRoot = assertSafeResetTarget(
    repoRoot,
    path.join(repoRoot, 'compiled'),
  )

  let purgedSeriesDirs = 0
  if (fs.existsSync(seriesRoot)) {
    const entries = fs.readdirSync(seriesRoot, { withFileTypes: true })
    for (const entry of entries) {
      if (entry.name === '.gitkeep') {
        continue
      }
      const childPath = path.join(seriesRoot, entry.name)
      const safeChild = assertSafeResetTarget(repoRoot, childPath)
      fs.rmSync(safeChild, { recursive: true, force: true })
      if (entry.isDirectory()) {
        purgedSeriesDirs++
      }
    }
  } else {
    fs.mkdirSync(seriesRoot, { recursive: true })
  }

  receipts.push(
    `>> [TABULA_RASA:SERIES] Purged ${purgedSeriesDirs} series directories (${inventory.counts.markdownDossiers} Markdown dossiers) [OK]`,
  )

  if (fs.existsSync(compiledRoot)) {
    const entries = fs.readdirSync(compiledRoot, { withFileTypes: true })
    for (const entry of entries) {
      if (entry.name === '.gitkeep') {
        continue
      }
      const childPath = path.join(compiledRoot, entry.name)
      const safeChild = assertSafeResetTarget(repoRoot, childPath)
      fs.rmSync(safeChild, { recursive: true, force: true })
    }
  } else {
    fs.mkdirSync(compiledRoot, { recursive: true })
  }

  receipts.push(
    `>> [TABULA_RASA:COMPILED] Purged ${inventory.counts.compiledBibleManifests} bible-state.json and ${inventory.counts.topologyManifests} topology manifests [OK]`,
  )

  const unitsResetList: string[] = [
    '00.lore.unit',
    '34.target.unit',
    '41.canon.unit',
    '97.collect.unit',
    '07.database.unit',
    '32.status.unit',
  ]

  cpy.idx = '00.lore'
  cpy.loreBitList = []
  cpy.loreBits = {}

  if (ste && typeof ste.hunt === 'function') {
    await ste.hunt(RESET_TARGET, { idx: 'tabula-rasa-target' }).catch(() => {})
    await ste.hunt(RESET_CANON, { idx: 'tabula-rasa-canon' }).catch(() => {})
    await ste.hunt(RESET_COLLECT, { idx: 'tabula-rasa-collect' }).catch(() => {})
    await ste
      .hunt(RESET_DATABASE, { idx: 'tabula-rasa-database', src: repoRoot })
      .catch(() => {})
    await ste
      .hunt(EVALUATE_VOID_STATE, {
        idx: 'tabula-rasa-void-sync',
        src: repoRoot,
      })
      .catch(() => {})
  }

  receipts.push(
    '>> [TABULA_RASA:MEMORY] Re-initialized 34.target, 41.canon, 97.collect, and 07.database [OK]',
  )

  // Task 3.4: Authoritative post-purge Void-State filesystem confirmation
  const postVoidScan = scanVoidState(repoRoot)
  const voidStateVerified =
    postVoidScan.isVoidState && postVoidScan.discoveredSlugs.length === 0

  if (voidStateVerified) {
    receipts.push(
      `>> [TABULA_RASA:HUD] Status unit synchronized -> VOID STATE (T0=$0.00) | MERKLE=${GENESIS_MERKLE_ROOT_64.slice(0, 8)}... | LOCK=ENGAGED [OK]`,
    )
  } else {
    receipts.push(
      `:: [TABULA_RASA:HUD] [FAIL] Residual series detected after purge: [${postVoidScan.discoveredSlugs.join(', ')}]`,
    )
  }

  const purgedCounts: LorePurgedCounts = {
    seriesDirectories: purgedSeriesDirs,
    markdownDossiers: inventory.counts.markdownDossiers,
    compiledBibleManifests: inventory.counts.compiledBibleManifests,
    topologyManifests: inventory.counts.topologyManifests,
    errataFiles: errataPurgeStats.errataFiles,
    scratchpadFiles: errataPurgeStats.scratchpadFiles,
  }

  const finalReceipt: LoreTabulaRasaReceipt = {
    ok: voidStateVerified,
    aborted: false,
    voidStateVerified,
    cascadeOrigin,
    purgedCounts,
    unitsResetList,
    merkleRootHex: GENESIS_MERKLE_ROOT_64,
    dispatchLockEngaged: true,
    timestampIso: new Date().toISOString(),
    receipts,
  }

  await emitLoreResetTelemetry(ste, receipts)

  if (bal?.slv) {
    bal.slv({
      lorBit: {
        idx: voidStateVerified
          ? 'execute-tabula-rasa-lore-pass'
          : 'execute-tabula-rasa-lore-fail',
        val: voidStateVerified ? 1 : -1,
        dat: finalReceipt,
      },
    })
  }

  return cpy
}
