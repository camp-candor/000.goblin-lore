/**
 * packages/299.lore/41.canon.unit/buz/verifyScratchpadSave.buzz.ts
 * Post-Save Three-Gate Scratchpad Auditor:
 * - Gate 1: Byte-0 "---\n" (0x2d2d2d0a) & Pure 7-Bit ASCII Hygiene
 * - Gate 2: YAML Parse, Domain Schema & Somatic Hand-Capacity SAT
 * - Gate 3: Foreign-Key Closure & Voice Rail Presence

import fs from 'node:fs'
import yaml from 'js-yaml'
import { validateEntity } from '../../26.schema.unit/buz/validate.buzz.js'
import { auditEntityCapacityPreFlight } from '../../09.entity.unit/buz/capacityStatValidator.buzz.js'
import type {
    ScratchpadGateFailure,
    VerifyScratchpadOptions,
    VerifyScratchpadReport,
} from '../fce/scratchpadVerify.bit.js'

export const ANNOTATION_HEADER_START =
    '/* =============================================================================='
export const ANNOTATION_HEADER_END =
    '============================================================================== */'

/**
 * Strips any prepended diagnostic error annotation comment block and normalizes CRLF to LF.
 */
export function stripDiagnosticHeaderBlock(rawContent: string): string {
    let normalized = String(rawContent || '').replace(/\r\n/g, '\n')
    const startIdx = normalized.indexOf(ANNOTATION_HEADER_START)
    const endIdx = normalized.indexOf(ANNOTATION_HEADER_END)

    if (startIdx !== -1 && endIdx !== -1 && endIdx > startIdx) {
        const before = normalized.slice(0, startIdx)
        const after = normalized.slice(endIdx + ANNOTATION_HEADER_END.length)
        normalized = `${before}${after}`.replace(/^\n+/, '')
    }

    return normalized
}

function isPure7BitAscii(text: string): { clean: boolean; badOffset?: number } {
    for (let i = 0; i < text.length; i++) {
        if (text.charCodeAt(i) > 127) {
            return { clean: false, badOffset: i }
        }
    }
    return { clean: true }
}

/**
 * Executes the Three-Gate Post-Save Verification Gauntlet over a staged entity scratchpad.
 */
export function verifyScratchpadSave(
    options: VerifyScratchpadOptions,
): VerifyScratchpadReport {
    const failures: ScratchpadGateFailure[] = []

    if (!options.scratchpadPath || !fs.existsSync(options.scratchpadPath)) {
        failures.push({
            gate: 'GATE_1_BYTE_ZERO_ASCII',
            failureClass: 'BYTE_ZERO_VIOLATION',
            message: `Scratchpad file missing on disk: ${options.scratchpadPath}`,
        })
        return {
            valid: false,
            gate1Passed: false,
            gate2Passed: false,
            gate3Passed: false,
            cleanedMarkdown: '',
            failures,
        }
    }

    const rawDiskText = fs.readFileSync(options.scratchpadPath, 'utf8')
    const cleanedMarkdown = stripDiagnosticHeaderBlock(rawDiskText)

    // -------------------------------------------------------------------------
    // GATE 1: Byte-0 Opening Delimiter & Pure 7-Bit ASCII Compliance
    // -------------------------------------------------------------------------
    const rawBuf = Buffer.from(cleanedMarkdown, 'utf8')
    const hasByteZeroOpening =
        rawBuf.length >= 4 &&
        rawBuf[0] === 0x2d &&
        rawBuf[1] === 0x2d &&
        rawBuf[2] === 0x2d &&
        rawBuf[3] === 0x0a

    if (!hasByteZeroOpening) {
        failures.push({
            gate: 'GATE_1_BYTE_ZERO_ASCII',
            failureClass: 'BYTE_ZERO_VIOLATION',
            field: 'byte_0',
            message:
                'File must begin with exact "---\\n" (0x2d2d2d0a) at physical byte offset 0 with zero leading spaces or BOM.',
        })
    }

    const closingFenceIdx = cleanedMarkdown.indexOf('\n---\n', 3)
    if (closingFenceIdx === -1) {
        failures.push({
            gate: 'GATE_1_BYTE_ZERO_ASCII',
            failureClass: 'BYTE_ZERO_VIOLATION',
            field: 'frontmatter_fence',
            message:
                'Missing closing "\\n---\\n" YAML frontmatter fence delimiter.',
        })
    }

    const asciiAudit = isPure7BitAscii(cleanedMarkdown)
    if (!asciiAudit.clean) {
        failures.push({
            gate: 'GATE_1_BYTE_ZERO_ASCII',
            failureClass: 'BYTE_ZERO_VIOLATION',
            field: `offset_${asciiAudit.badOffset}`,
            message: `Non-ASCII multi-byte character detected at character offset ${asciiAudit.badOffset}. Only 7-bit ASCII (charCode <= 127) is permitted.`,
        })
    }

    const gate1Passed = failures.length === 0
    if (!gate1Passed) {
        return {
            valid: false,
            gate1Passed: false,
            gate2Passed: false,
            gate3Passed: false,
            cleanedMarkdown,
            failures,
        }
    }

    // -------------------------------------------------------------------------
    // GATE 2: YAML Parse, Schema Bounds & Somatic Hand-Capacity SAT
    // -------------------------------------------------------------------------
    const yamlBlock = cleanedMarkdown.slice(4, closingFenceIdx)
    const proseBody = cleanedMarkdown.slice(closingFenceIdx + 5)

    let parsedFrontmatter: Record<string, any> = {}
    try {
        const loaded = yaml.load(yamlBlock)
        if (!loaded || typeof loaded !== 'object' || Array.isArray(loaded)) {
            throw new Error('YAML frontmatter did not parse into a key-value object')
        }
        parsedFrontmatter = loaded as Record<string, any>
    } catch (err: any) {
        failures.push({
            gate: 'GATE_2_SCHEMA_SOMATIC_SAT',
            failureClass: 'ZOD_SCHEMA_VIOLATION',
            field: 'yaml_frontmatter',
            message: `YAML syntax error: ${err instanceof Error ? err.message : String(err)}`,
        })
        return {
            valid: false,
            gate1Passed: true,
            gate2Passed: false,
            gate3Passed: false,
            cleanedMarkdown,
            failures,
        }
    }

    if (parsedFrontmatter.id !== options.expectedEntityId) {
        failures.push({
            gate: 'GATE_2_SCHEMA_SOMATIC_SAT',
            failureClass: 'ZOD_SCHEMA_VIOLATION',
            field: 'id',
            message: `Frontmatter id '${String(parsedFrontmatter.id)}' does not match expected '${options.expectedEntityId}'.`,
        })

    const schemaCheck = validateEntity(options.expectedDomain, parsedFrontmatter)
    if (schemaCheck && schemaCheck.valid === false && Array.isArray(schemaCheck.errors)) {
        for (const errItem of schemaCheck.errors) {
            failures.push({
                gate: 'GATE_2_SCHEMA_SOMATIC_SAT',
                failureClass: 'ZOD_SCHEMA_VIOLATION',
                field: errItem.path || 'schema',
                message: errItem.message || 'Schema validation error',
            })
        }
    }

    // Somatic Hand-Capacity SAT Audit for characters
    if (options.expectedDomain === 'character') {
        const availableArmsBps =
            typeof parsedFrontmatter.somatic?.available_arms_bps === 'number'
                ? parsedFrontmatter.somatic.available_arms_bps
                : 20000
        if (availableArmsBps < 0 || availableArmsBps > 20000) {
            failures.push({
                gate: 'GATE_2_SCHEMA_SOMATIC_SAT',
                failureClass: 'SOMATIC_SAT_VIOLATION',
                field: 'somatic.available_arms_bps',
                message: `available_arms_bps (${availableArmsBps}) must be within [0, 20000].`,
            })
        }

        const heldItems: string[] = Array.isArray(parsedFrontmatter.logistical?.held)
            ? parsedFrontmatter.logistical.held
            : []
        if (heldItems.length > 2) {
            failures.push({
                gate: 'GATE_2_SCHEMA_SOMATIC_SAT',
                failureClass: 'SOMATIC_SAT_VIOLATION',
                field: 'logistical.held',
                message: `Held equipment count (${heldItems.length}) exceeds maximum 2 hand partitions.`,
            })
        }

        const projectedItems = heldItems.map((id) => ({
            item_id: String(id),
            occupies_hands_bps: 10000,
        }))
        const satAudit = auditEntityCapacityPreFlight(
            { available_arms_bps: availableArmsBps },
            projectedItems,
        )
        if (!satAudit.valid) {
            failures.push({
                gate: 'GATE_2_SCHEMA_SOMATIC_SAT',
                failureClass: 'SOMATIC_SAT_VIOLATION',
                field: 'logistical.held',
                message: `Somatic Hand Capacity UNSAT: required ${satAudit.total_required_hands_bps} bp > available ${availableArmsBps} bp.`,
            })
        }
    }

    // Farce ceiling check for grievances
    if (options.expectedDomain === 'grievance') {
        const isFarce = Boolean(parsedFrontmatter.tonal_override)
        const ceilingBp = Number(parsedFrontmatter.escalation_ceiling_bp ?? 8500)
        if (isFarce && ceilingBp > 2000) {
            failures.push({
                gate: 'GATE_2_SCHEMA_SOMATIC_SAT',
                failureClass: 'ZOD_SCHEMA_VIOLATION',
                field: 'escalation_ceiling_bp',
                message: `Comedic farce grievance (tonal_override: true) cannot exceed 2000 bp ceiling (found ${ceilingBp} bp).`,
            })
        }
    }

    const gate2Passed = failures.length === 0
    if (!gate2Passed) {
        return {
            valid: false,
            gate1Passed: true,
            gate2Passed: false,
            gate3Passed: false,
            cleanedMarkdown,
            parsedFrontmatter,
            proseBody,
            failures,
        }
    }

    // -------------------------------------------------------------------------
    // GATE 3: Foreign-Key Closure & Voice Rail Presence
    // -------------------------------------------------------------------------
    const knownSet: Set<string> | null = options.catalogIds
        ? options.catalogIds instanceof Set
            ? new Set(options.catalogIds)
            : new Set(options.catalogIds)
        : null

    if (knownSet && knownSet.size > 0) {
        const candidateRefs: Array<{ field: string; id: string }> = []

        if (
            typeof parsedFrontmatter.location_id === 'string' &&
            parsedFrontmatter.location_id !== 'loc_unbound_origin'
        ) {
            candidateRefs.push({
                field: 'location_id',
                id: parsedFrontmatter.location_id,
            })
        }
        if (typeof parsedFrontmatter.custody_holder === 'string') {
            candidateRefs.push({
                field: 'custody_holder',
                id: parsedFrontmatter.custody_holder,
            })
        }
        if (Array.isArray(parsedFrontmatter.logistical?.held)) {
            for (const item of parsedFrontmatter.logistical.held) {
                candidateRefs.push({
                    field: 'logistical.held',
                    id: String(item),
                })
            }
        }
        if (Array.isArray(parsedFrontmatter.participants_primary)) {
            for (const actor of parsedFrontmatter.participants_primary) {
                candidateRefs.push({
                    field: 'participants_primary',
                    id: String(actor),
                })
            }
        }

        for (const ref of candidateRefs) {
            if (!knownSet.has(ref.id)) {
                failures.push({
                    gate: 'GATE_3_RELATIONAL_VOICE_RAIL',
                    failureClass: 'FOREIGN_KEY_DANGLE',
                    field: ref.field,
                    message: `Dangling foreign-key reference '${ref.id}' in field '${ref.field}' does not exist in active series catalog.`,
                })
            }
        }
    }

    const strippedProse = proseBody.replace(/[#\s-]/g, '')
    if (strippedProse.length === 0) {
        failures.push({
            gate: 'GATE_3_RELATIONAL_VOICE_RAIL',
            failureClass: 'ZOD_SCHEMA_VIOLATION',
            field: 'voice_rail_body',
            message:
                'Voice Rail Markdown body below frontmatter cannot be empty.',
        })
    }

    const gate3Passed = failures.length === 0

    return {
        valid: gate1Passed && gate2Passed && gate3Passed,
        gate1Passed,
        gate2Passed,
        gate3Passed,
        cleanedMarkdown,
        parsedFrontmatter,
        proseBody,
        failures,
    }
}

/**
 * Task 6.2: Prepends actionable 7-bit ASCII error annotations at the top of the .tmp scratchpad,
 * stripping any prior annotation block so headers never accumulate across strikes.
 */
export function annotateEntityScratchpadWithErrors(
    scratchpadPath: string,
    failures: readonly ScratchpadGateFailure[],
    attempt: number,
    maxAttempts: number = 3,
): string {
    const raw = fs.readFileSync(scratchpadPath, 'utf8')
    const strippedBody = stripDiagnosticHeaderBlock(raw)

    const lines: string[] = [
        ANNOTATION_HEADER_START,
        `>> [VALIDATION FAILURE - ATTEMPT ${attempt} OF ${maxAttempts}]`,
        '>> Resolve the following error(s) in this file and save before closing editor:',
    ]

    failures.forEach((f, idx) => {
        const fieldLabel = f.field ? ` (field: ${f.field})` : ''
        lines.push(`>>   ${idx + 1}. [${f.gate}]${fieldLabel}: ${toAscii(f.message)}`)
    })

    lines.push(
        '>> Invariant: Ensure "---" starts at Byte 0 after this header block is stripped.',
        ANNOTATION_HEADER_END,
        '',
        strippedBody,
    )

    const updated = lines.join('\n')
    fs.writeFileSync(scratchpadPath, updated, 'utf8')
    return updated

/**
 * Applies queued reciprocal custody links onto target possession dossiers atomically at Byte 0.
 */
function applyReciprocalPossessionLinksAtomic(
    seriesSlug: string,
    links: readonly ReciprocalPossessionLink[],
    repoRoot: string,
): void {
    const possessionsDir = path.join(repoRoot, 'series', seriesSlug, 'possessions')
    for (const link of links) {
        const itemPath = path.join(possessionsDir, `${link.itemId}.md`)
        if (!fs.existsSync(itemPath)) continue

        const raw = fs.readFileSync(itemPath, 'utf8').replace(/\r\n/g, '\n')
        if (!raw.startsWith('---\n')) continue

        const parsed = matter(raw)
        const data = (parsed.data || {}) as Record<string, any>
        data.custody_holder = link.ownerId
        data.location_id = null
        data.equip_slot =
            link.partition === 'held'
                ? 'held_primary'
                : link.partition === 'worn'
                  ? 'worn_belt'
                  : 'carried'

        const dumpedYaml = yaml
            .dump(data, {
                indent: 2,
                lineWidth: -1,
                noRefs: true,
                sortKeys: true,
            })
            .replace(/\r\n/g, '\n')
            .trimEnd()

        const bodyClean = parsed.content.startsWith('\n')
            ? parsed.content.slice(1)
            : parsed.content
        const updatedMarkdown = `---\n${dumpedYaml}\n---\n${bodyClean}`

        const tmpPath = `${itemPath}.tmp_${Date.now()}_${Math.random().toString(36).slice(2, 7)}`
        fs.writeFileSync(tmpPath, Buffer.from(updatedMarkdown, 'utf8'))
        fs.renameSync(tmpPath, itemPath)
    }
}

export const threeStrikeSaveGate = async (
    cpy: EntityModel,
    bal: HumanGateBit,
    ste?: State,
): Promise<EntityModel> => {
    const opts = bal.dat
    const maxAttempts =
        typeof opts.maxAttempts === 'number' && opts.maxAttempts > 0
            ? Math.trunc(opts.maxAttempts)
            : 3

    if (!opts.scratchpadPath || !fs.existsSync(opts.scratchpadPath)) {
        const errMsg = `Scratchpad file missing: ${opts.scratchpadPath}`
        emitConsole(ste, `:: [HUMAN_GATE:ERR] ${errMsg}`)
        if (bal?.slv) {
            bal.slv({
                entBit: {
                    idx: 'open-human-save-gate-error',
                    val: -1,
                    dat: {
                        resolved: false,
                        tripped: false,
                        attempts: 0,
                        canonicalFilePath: opts.canonicalFilePath,
                        scratchpadPath: opts.scratchpadPath,
                    },
                },
            })
        }
        return cpy
    }

    emitConsole(
        ste,
        `>> [HUMAN_GATE] Entering Three-Strike Post-Save Validation Loop for ${opts.entityId}...`,
    )

    let lastFailures: readonly ScratchpadGateFailure[] = []

    for (let attempt = 1; attempt <= maxAttempts; attempt++) {
        emitConsole(
            ste,

        )

        // Suspend curses and open $EDITOR on the staged scratchpad
        await suspendCursesAndSpawnEditor(

            opts.editor,
            opts.editorArgs,

        // Validate saved scratchpad across Gates 1-3
        const report = verifyScratchpadSave({
            scratchpadPath: opts.scratchpadPath,
            expectedDomain: opts.domain,
            expectedEntityId: opts.entityId,
            catalogIds: opts.catalogIds,

        if (report.valid) {
            // Task 6.4: Atomic Inode Promotion With Inline Canon Resealing
            fs.mkdirSync(path.dirname(opts.canonicalFilePath), { recursive: true })
            fs.writeFileSync(
                opts.scratchpadPath,
                Buffer.from(report.cleanedMarkdown, 'utf8'),
            )
            fs.renameSync(opts.scratchpadPath, opts.canonicalFilePath)

            const resolvedRepoRoot =
                opts.repoRoot ||
                path.resolve(opts.canonicalFilePath, '..', '..', '..', '..')

            if (
                Array.isArray(opts.reciprocalPossessionLinks) &&
                opts.reciprocalPossessionLinks.length > 0
            ) {
                applyReciprocalPossessionLinksAtomic(
                    opts.seriesSlug,
                    opts.reciprocalPossessionLinks,
                    resolvedRepoRoot,
                )
            }

            let merkleRootHex: string | undefined
            if (!opts.skipCompileReseal) {
                try {
                    const compileReport = runCorpusCompiler({
                        seriesSlug: opts.seriesSlug,
                        repoRoot: resolvedRepoRoot,
                        silent: true,
                    })
                    if (compileReport?.success && compileReport?.hash) {
                        merkleRootHex = compileReport.hash
                    }
                } catch {
                    // Non-blocking if series has partial seed state during early scaffolding
                }

            emitConsole(

            )

            const payload: HumanGateResolutionPayload = {
                resolved: true,
                tripped: false,
                attempts: attempt,
                canonicalFilePath: opts.canonicalFilePath,
                scratchpadPath: opts.scratchpadPath,
                merkleRootHex,
            }

            if (bal?.slv) {
                bal.slv({
                    entBit: {
                        idx: 'open-human-save-gate',
                        val: attempt,
                        dat: payload,
                    },
                })
            }
            return cpy
        }

        lastFailures = report.failures
        emitConsole(
            ste,
            `:: [HUMAN_GATE:WARN] Attempt ${attempt}/${maxAttempts} failed with ${lastFailures.length} validation breach(es).`,
        )

        if (attempt < maxAttempts) {
            emitConsole(
                ste,
                `>> [HUMAN_GATE] Prepending diagnostic error annotations for Attempt ${attempt + 1}...`,
            )
            annotateEntityScratchpadWithErrors(
                opts.scratchpadPath,
                lastFailures,
                attempt,
                maxAttempts,
            )

        operator: opts.operator,
        seriesSlug: opts.seriesSlug,
        entityId: opts.entityId,
        attempts: maxAttempts,
        failures: lastFailures,
        canonicalFilePath: opts.canonicalFilePath,
    })

    // Obliterate invalid scratchpad file; canonical file remains untouched
    if (fs.existsSync(opts.scratchpadPath)) {
        fs.unlinkSync(opts.scratchpadPath)
    }

    emitConsole(
        ste,
        '>> ================================================================',
    )
    emitConsole(
        ste,
        '>> [CIRCUIT BREAKER TRIPPED] HUMAN SAVE GATE CEILING EXHAUSTED',
    )
    emitConsole(ste, `>> Event         : ${dossier.escalation_event}`)
    emitConsole(ste, `>> Failure Class : [${dossier.failure_class}]`)
    emitConsole(
        ste,
        `>> Attempts      : ${dossier.attempts_recorded}/${maxAttempts} exhausted`,
    )
    emitConsole(ste, `>> Remediation   : ${dossier.remediation}`)
    emitConsole(
        ste,
        '>> ================================================================',
    )

    const trippedPayload: HumanGateResolutionPayload = {
        resolved: false,
        tripped: true,
        attempts: maxAttempts,
        canonicalFilePath: opts.canonicalFilePath,
        scratchpadPath: opts.scratchpadPath,
        dossier,
    }

    if (bal?.slv) {
        bal.slv({
            entBit: {
                idx: 'open-human-save-gate-tripped',
                val: -1,
                dat: trippedPayload,
            },
        })
    }

    return cpy
}

import fs from 'node:fs'

                seriesSlug,

                skipCompileReseal: true,

        expect(fs.existsSync(canonicalFilePath)).toBe(true)

        expect(

        ).toBe(true)
    })
