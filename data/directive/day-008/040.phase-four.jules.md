    .replace(/[^\x20-\x7E]/g, '')
    .trim()
}

function normalizeSeriesSlug(raw: unknown): string {
  return String(raw || '')
    .trim()
    .toLowerCase()
    .replace(/[^a-z0-9-]+/g, '-')
    .replace(/^-+|-+$/g, '')
}

function clampBasisPoints(raw: unknown, fallback: number): number {
  const num = Number(raw)
  if (!Number.isFinite(num)) return Math.round(fallback)
  return Math.max(0, Math.min(10000, Math.round(num)))
}

function clampPositiveInt(raw: unknown, fallback: number): number {
  const num = Number(raw)
  if (!Number.isFinite(num)) return Math.max(1, Math.round(fallback))
  return Math.max(1, Math.round(num))
}

function atomicWriteJson(targetPath: string, payload: unknown): void {
  const targetDir = path.dirname(targetPath)
  fs.mkdirSync(targetDir, { recursive: true })
  const nonce = `${Date.now()}_${crypto.randomBytes(4).toString('hex')}`
  const tempPath = path.join(targetDir, `.tmp_topology_${nonce}.json`)
  const serialized = `${JSON.stringify(payload, null, 2)}\n`
  fs.writeFileSync(tempPath, serialized, 'utf8')
  fs.renameSync(tempPath, targetPath)
}
