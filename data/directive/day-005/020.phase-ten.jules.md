/**
 * Strips pasted citation artifacts, HTML comments, and normalizes line endings to POSIX LF.
 */
export function sanitizeProseStub(rawText: string | undefined, fallback: string): string {
  if (!rawText || rawText.trim().length === 0) {
    return fallback
  }

  return rawText
    .replace(/\[cite:\s*[\d,\s]+\]/g, '')
    .replace(/<!--[\s\S]*?-->/g, '')
    .replace(/\r\n/g, '\n')
    .replace(/\r/g, '\n')
    .normalize('NFC')
    .trim()
}

/**
 * Verifies strict Byte-0 YAML frontmatter compliance:
 * - Zero UTF-8 BOM (0xEF 0xBB 0xBF)
 * - Bytes 0..3 must be exact "---\n" (0x2D 0x2D 0x2D 0x0A)
 * - Zero carriage return bytes (0x0D) anywhere in buffer
 */
export function verifyByteZeroBuffer(content: Buffer | string): ByteZeroAuditReceipt {
  const buf = Buffer.isBuffer(content) ? content : Buffer.from(content, 'utf8')
  const firstFourBytesHex = buf.subarray(0, 4).toString('hex')

  if (buf.length >= 3 && buf[0] === 0xef && buf[1] === 0xbb && buf[2] === 0xbf) {
    return {
      valid: false,
      byteLength: buf.length,
      firstFourBytesHex,
      error: '[BYTE_0_VIOLATION] UTF-8 BOM detected at physical offset 0',
    }
  }

  if (buf.length < 4 || buf[0] !== 0x2d || buf[1] !== 0x2d || buf[2] !== 0x2d || buf[3] !== 0x0a) {
    return {
      valid: false,
      byteLength: buf.length,
      firstFourBytesHex,
      error: `[BYTE_0_VIOLATION] Expected opening "---\\n" (2d2d2d0a) at offset 0, found ${firstFourBytesHex}`,
    }
  }

  for (let i = 0; i < buf.length; i++) {
    if (buf[i] === 0x0d) {
      return {
        valid: false,
        byteLength: buf.length,
        firstFourBytesHex,
        error: `[BYTE_0_VIOLATION] Prohibited carriage return (0x0D) found at byte offset ${i}`,
      }
    }
  }

  return {
    valid: true,
    byteLength: buf.length,
    firstFourBytesHex,
  }
}

    .replace(/\r/g, '\n')
    .trim()

  const voiceBody = formatVoiceRailSections(domain, voiceSections)
  const dossierText = `---\n${dumpedYaml}\n---\n\n${voiceBody}\n`

  const receipt = verifyByteZeroBuffer(dossierText)
  if (!receipt.valid) {
    throw new Error(receipt.error || '[BYTE_0_VIOLATION] Assembled dossier failed Byte-0 audit')
  }

  return { dossierText, receipt }
}

export const assembleDualRailDossierAction = (
  cpy: EntityModel,
  bal: EntityAssembleBit,
  ste?: State,
): EntityModel => {
  const dat = bal?.dat
  if (!dat) {
    if (bal?.slv) {
      bal.slv({
        entBit: {
          idx: 'assemble-dual-rail-err',
          val: -1,
          dat: {
            dossierText: '',
            receipt: {
              valid: false,
              byteLength: 0,
              firstFourBytesHex: '',
              error: '[MISSING_PAYLOAD] dat is required',
            },
          },
        },
      })
    }
    return cpy
  }

  try {
    const assembled = assembleDualRailDossier(dat.domain, dat.frontmatter, dat.voiceSections)
    if (bal?.slv) {
      bal.slv({
        entBit: {
          idx: 'assemble-dual-rail',
          val: 0,
          dat: assembled,
        },
      })
    }
  } catch (err: any) {
    if (bal?.slv) {
      bal.slv({
        entBit: {
          idx: 'assemble-dual-rail-fail',
          val: -1,
          dat: {
            dossierText: '',
            receipt: {
              valid: false,
              byteLength: 0,
              firstFourBytesHex: '',
              error: err.message,
            },
          },
        },
      })
    }
  }

  return cpy
}
