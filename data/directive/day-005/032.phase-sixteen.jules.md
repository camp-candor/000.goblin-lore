function parseByteZeroDossier(rawContent: string, fileLabel: string): ParsedDossier {
  const buf = Buffer.from(rawContent, 'utf8')
  if (buf.length >= 3 && buf[0] === 0xef && buf[1] === 0xbb && buf[2] === 0xbf) {
    throw new Error(`[BYTE_ZERO_BREACH] UTF-8 BOM detected at offset 0 in ${fileLabel}`)
  }
  if (buf.length < 4 || buf[0] !== 0x2d || buf[1] !== 0x2d || buf[2] !== 0x2d || buf[3] !== 0x0a) {
    throw new Error(`[BYTE_ZERO_BREACH] Missing opening "---\\n" at byte 0 in ${fileLabel}`)
  }

  const normalized = rawContent.replace(/\r\n/g, '\n').replace(/\r/g, '\n')
  const parts = normalized.split(/^---$/m)
  if (parts.length < 3) {
    throw new Error(`[BYTE_ZERO_BREACH] Unclosed YAML frontmatter in ${fileLabel}`)
  }

  return { frontmatter, bodyContent }

    }
  }

  let parsedChar: ParsedDossier
  try {
    parsedChar = parseByteZeroDossier(fs.readFileSync(oldPath, 'utf8'), `${characterId}.md`)
  } catch (err: any) {
    return {
      success: false,
      characterId,
      oldPath,
      newPath,
      archivedGrievances: [],
      updatedPossessions: [],
      rootMerkleHash: '',
      byteZeroVerified: false,
      errorCode: 'BYTE_ZERO_BREACH',
      errors: [err.message],
    }
  }

  // 1. Discover valid locations on disk
  const locationsDir = path.join(seriesDir, 'locations')
  const validLocationIds: string[] = fs.existsSync(locationsDir)
    ? fs
        .readdirSync(locationsDir)
        .filter((f) => f.endsWith('.md'))
        .map((f) => path.basename(f, '.md'))
    : []

  // 2. Discover active characters on disk
  const activeCharacters: Record<string, ActiveRecipientSnapshot> = {}
  const activeCharFiles = fs.readdirSync(charsDir).filter((f) => f.endsWith('.md'))
  for (const file of activeCharFiles) {
    const cId = path.basename(file, '.md')
    if (cId === characterId) continue
    const cParsed = parseByteZeroDossier(fs.readFileSync(path.join(charsDir, file), 'utf8'), file)
    const fm = cParsed.frontmatter
    const armsBps =
      typeof fm.somatic?.available_arms_bps === 'number' ? fm.somatic.available_arms_bps : 20000
    const vector: CharacterLogisticalVector = {
      entity_id: cId,
      worn: Array.isArray(fm.worn) ? fm.worn : [],
      held: Array.isArray(fm.held) ? fm.held : [],
      carried: Array.isArray(fm.carried) ? fm.carried : [],
      cached: fm.cached && typeof fm.cached === 'object' ? fm.cached : {},
    }
    activeCharacters[cId] = {
      arms_available_bps: armsBps,
      vector,
      is_retired: fm.status === 'departed' || fm.status === 'DEPRECATED_ARCHIVED',
    }
  }

  // 3. Discover possessions owned or held by the retiring entity
  const possessionsDir = path.join(seriesDir, 'possessions')
  const knownPossessions: Record<string, PossessionLogisticalVector> = {}
  const heldPossessions: PossessionLogisticalVector[] = []

  const directIds = new Set<string>([
    ...(Array.isArray(parsedChar.frontmatter.worn) ? parsedChar.frontmatter.worn : []),
    ...(Array.isArray(parsedChar.frontmatter.held) ? parsedChar.frontmatter.held : []),
    ...(Array.isArray(parsedChar.frontmatter.carried) ? parsedChar.frontmatter.carried : []),
  ])

  if (fs.existsSync(possessionsDir)) {
    const posFiles = fs.readdirSync(possessionsDir).filter((f) => f.endsWith('.md'))
    for (const file of posFiles) {
      const pId = path.basename(file, '.md')
      const pParsed = parseByteZeroDossier(
        fs.readFileSync(path.join(possessionsDir, file), 'utf8'),
        file,
      )
      const fm = pParsed.frontmatter
      const vec: PossessionLogisticalVector = {
        item_id: pId,
        occupies_hands: typeof fm.occupies_hands === 'number' ? fm.occupies_hands : 1,
        occupies_hands_bps:
          typeof fm.occupies_hands_bps === 'number' ? fm.occupies_hands_bps : 10000,
        mass_grams: typeof fm.mass_grams === 'number' ? fm.mass_grams : 100,
        weight_class: fm.weight_class || 'scrap',
        is_contested: Boolean(fm.is_contested),
        allowed_slots: Array.isArray(fm.allowed_slots) ? fm.allowed_slots : ['HELD_PRIMARY'],
        owner_id: fm.owner_id ?? null,
        location_id: fm.location_id ?? null,
      }
      knownPossessions[pId] = vec
      if (fm.owner_id === characterId || directIds.has(pId)) {
        heldPossessions.push(vec)
      }
    }
  } else {
    for (const pId of directIds) {
      const fallbackVec: PossessionLogisticalVector = {
        item_id: pId,
        occupies_hands: 1,
        occupies_hands_bps: 10000,
        mass_grams: 100,
        weight_class: 'scrap',
        is_contested: false,
        allowed_slots: ['HELD_PRIMARY'],
        owner_id: characterId,
        location_id: null,
      }
      knownPossessions[pId] = fallbackVec
      heldPossessions.push(fallbackVec)
    }
  }

  // 4. Pre-Flight Matter Conservation Gate (Aborts in RAM before disk mutation)
  const conservationReceipt = reconcileRetiredEntityPossessions({
    retiring_entity_id: characterId,
    held_possessions: heldPossessions,
    item_transfers: itemTransfers,
    active_characters: activeCharacters,
    valid_location_ids: validLocationIds,
    known_possessions: knownPossessions,
  })

  if (!conservationReceipt.valid) {
    return {
      success: false,
      characterId,
      oldPath,
      newPath,
      archivedGrievances: [],
      updatedPossessions: [],
      rootMerkleHash: '',
      byteZeroVerified: false,
      errorCode: conservationReceipt.errorCode || 'MATTER_CONSERVATION_BREACH',
      errors: conservationReceipt.errors,
    }
  }

  // 5. Update reassigned possession files & recipient character dossiers atomically
  const updatedPossessions: string[] = []
  for (const state of conservationReceipt.updated_possession_states) {
    const posPath = path.join(possessionsDir, `${state.item_id}.md`)
    if (fs.existsSync(posPath)) {
      const pParsed = parseByteZeroDossier(fs.readFileSync(posPath, 'utf8'), `${state.item_id}.md`)
      pParsed.frontmatter.owner_id = state.owner_id
      pParsed.frontmatter.location_id = state.location_id
      writeAtomicPosixFile(posPath, serializeByteZeroDossier(pParsed.frontmatter, pParsed.bodyContent))
      updatedPossessions.push(state.item_id)
    }
  }

  for (const [recId, recVec] of Object.entries(conservationReceipt.updated_character_vectors)) {
    const recPath = path.join(charsDir, `${recId}.md`)
    if (fs.existsSync(recPath)) {
      const rParsed = parseByteZeroDossier(fs.readFileSync(recPath, 'utf8'), `${recId}.md`)
      rParsed.frontmatter.worn = recVec.worn
      rParsed.frontmatter.held = recVec.held
      rParsed.frontmatter.carried = recVec.carried
      rParsed.frontmatter.cached = recVec.cached
      writeAtomicPosixFile(recPath, serializeByteZeroDossier(rParsed.frontmatter, rParsed.bodyContent))
    }
  }

  // 6. Stamp Tombstone Metadata & Relocate Character to characters/_retired/
  parsedChar.frontmatter.status = 'departed'
  parsedChar.frontmatter.banned_kinetic_verbs = ['*']
  parsedChar.frontmatter.departure_epoch = departureEpoch
  parsedChar.frontmatter.departure_type = departureType
  parsedChar.frontmatter.resolution_cause = resolutionCause
  parsedChar.frontmatter.worn = []
  parsedChar.frontmatter.held = []
  parsedChar.frontmatter.carried = []

  if (parsedChar.frontmatter.somatic && typeof parsedChar.frontmatter.somatic === 'object') {
    parsedChar.frontmatter.somatic.banned_kinetic_verbs = ['*']
  }

  const serializedRetiredChar = serializeByteZeroDossier(
    parsedChar.frontmatter,
    parsedChar.bodyContent,
  )
  writeAtomicPosixFile(newPath, serializedRetiredChar)
  fs.unlinkSync(oldPath)

  // 7. Archive linked settled grievances into grievances/_retired/
  const archivedGrievances: string[] = []
  const grievancesDir = path.join(seriesDir, 'grievances')
  const retiredGrievancesDir = path.join(grievancesDir, '_retired')
  if (fs.existsSync(grievancesDir)) {
    const grvFiles = fs.readdirSync(grievancesDir).filter((f) => f.endsWith('.md'))
    for (const file of grvFiles) {
      const gId = path.basename(file, '.md')
      const gOldPath = path.join(grievancesDir, file)
      const gParsed = parseByteZeroDossier(fs.readFileSync(gOldPath, 'utf8'), file)
      const primary = Array.isArray(gParsed.frontmatter.participants_primary)
        ? gParsed.frontmatter.participants_primary
        : []
      if (primary.includes(characterId)) {
        gParsed.frontmatter.status = 'DEPRECATED_ARCHIVED'
        gParsed.frontmatter.archived_epoch = departureEpoch
        const gNewPath = path.join(retiredGrievancesDir, file)
        writeAtomicPosixFile(gNewPath, serializeByteZeroDossier(gParsed.frontmatter, gParsed.bodyContent))
        fs.unlinkSync(gOldPath)
        archivedGrievances.push(gId)
      }
    }
  }

  // 8. Seal updated H_root Merkle digest
  const rootMerkleHash = sealCallback
    ? await sealCallback(seriesSlug, repoRoot)
    : computeCorpusMerkleSeal(seriesDir)

  const retiredBuf = fs.readFileSync(newPath)
  const byteZeroVerified =
    retiredBuf.length >= 4 &&
    retiredBuf[0] === 0x2d &&
    retiredBuf[1] === 0x2d &&
    retiredBuf[2] === 0x2d &&
    retiredBuf[3] === 0x0a

  return {
    success: true,
    characterId,
    oldPath,
    newPath,
    archivedGrievances,
    updatedPossessions,
    rootMerkleHash,
    byteZeroVerified,
    errors: [],
  }
}

export const retireCanonicalEntityAction = async (
  cpy: LogisticModel,
  bal: LogisticBit<RetireCanonicalEntityPayload>,

    })
  }
  return cpy
}

import clone from 'clone-deep'
import * as Act from './logistic.action.js'
import { LogisticModel } from './logistic.model.js'
import * as Buzz from './logistic.buzzer.js'
import type State from '../99.core/state.js'

export function reducer(
  model: LogisticModel = new LogisticModel(),
  act: Act.Actions,
  state?: State,
): LogisticModel {
  switch (act.type) {
    case Act.INIT_LOGISTIC:
      return Buzz.initLogistic(clone(model), act.bale, state)

    case Act.UPDATE_LOGISTIC:
      return Buzz.updateLogistic(clone(model), act.bale, state)

    case Act.VERIFY_HAND_SATURATION:
      return Buzz.verifyHandSaturationAction(clone(model), act.bale, state)

    case Act.ALLOCATE_EQUIP_SLOT:
      return Buzz.allocateEquipSlotAction(clone(model), act.bale, state)

    case Act.ASSERT_MASS_CONSERVATION:
      return Buzz.assertMassConservationAction(clone(model), act.bale, state)

    case Act.RECONCILE_RETIRED_POSSESSIONS:
      return Buzz.reconcileRetiredPossessionsAction(clone(model), act.bale, state)

    case Act.RETIRE_CANONICAL_ENTITY:
      return Buzz.retireCanonicalEntityAction(clone(model), act.bale, state) as unknown as LogisticModel

    default:
      return model
  }
}
