import type { CanonFactStanding } from '../fce/standing.schema.js'
import type {
  ReconsiderationPriorityRecord,
  StillGoodAffirmationResult,
  default as CanonReconsiderationBit,
} from '../fce/reconsideration.bit.js'
import type { CanonModel } from '../canon.model.js'
import type State from '../../99.core/state.js'

/**
 * Calculates topological blast-radius review priority score in integer basis points.
 * Score = DaysSinceReconsideration * log10(DependentScenes + 1) * 100
 * Bounded to discrete integers to eliminate IEEE 754 float drift.
 */
export function calculateAuditPriorityScoreBps(
  daysSinceReconsideration: number,
  downstreamDependentScenes: number,
): number {
  if (!Number.isInteger(daysSinceReconsideration)) {
    throw new Error('[FLOAT_PROHIBITED] daysSinceReconsideration must be an integer')
  }
  if (!Number.isInteger(downstreamDependentScenes)) {
    throw new Error('[FLOAT_PROHIBITED] downstreamDependentScenes must be an integer')
  }
  if (daysSinceReconsideration <= 0) return 0
  if (downstreamDependentScenes < 0) return 0

  const rawScore = daysSinceReconsideration * Math.log10(downstreamDependentScenes + 1)
  return Math.round(rawScore * 100)
}

/**
 * Evaluates all tracked facts and returns an ordered queue of audit candidates.
 */
export function rankReconsiderationQueue(
  factCatalog: Map<string, CanonFactStanding> | Record<string, CanonFactStanding>,
  currentEpoch: number = 0,
  minDaysStale: number = 0,
): ReconsiderationPriorityRecord[] {
  if (!Number.isInteger(currentEpoch)) {
    throw new Error('[FLOAT_PROHIBITED] currentEpoch must be an integer')
  }

  const factList = factCatalog instanceof Map ? Array.from(factCatalog.values()) : Object.values(factCatalog)
  const candidates: ReconsiderationPriorityRecord[] = []
  const secondsPerDay = 86400

  for (const fact of factList) {
    if (fact.standing === 'DEPRECATED_ARCHIVED') continue

    const elapsedSeconds = Math.max(0, currentEpoch - fact.last_reviewed_epoch)
    const daysStale = Math.max(fact.days_since_reconsideration, Math.floor(elapsedSeconds / secondsPerDay))

    if (daysStale >= minDaysStale || fact.standing === 'FLAGGED_FOR_REVIEW') {
      const scoreBps = calculateAuditPriorityScoreBps(daysStale, fact.in_degree_count)
      const isMonolithicSpine = fact.in_degree_count >= 12 && daysStale > 60

      candidates.push({
        fact_id: fact.fact_id,
        entity_id: fact.entity_id,
        standing: fact.standing,
        in_degree_count: fact.in_degree_count,
        days_since_reconsideration: daysStale,
        priority_score_bps: scoreBps,
        is_monolithic_spine: isMonolithicSpine,
      })
    }
  }

  return candidates.sort((a, b) => {
    if (b.priority_score_bps !== a.priority_score_bps) {
      return b.priority_score_bps - a.priority_score_bps
    }
    return a.fact_id.localeCompare(b.fact_id)
  })
}

/**
 * Executes the 5-second "Still Good" re-affirmation protocol.
 * Resets stale duration, preserves version, and triggers zero cascades.
 */
export function affirmStillGood(
  factCatalog: Map<string, CanonFactStanding>,
  factId: string,
  affirmationEpoch: number = 0,
): StillGoodAffirmationResult {
  if (!Number.isInteger(affirmationEpoch)) {
    throw new Error('[FLOAT_PROHIBITED] affirmationEpoch must be an integer')
  }

  const existing = factCatalog.get(factId)
  if (!existing) {
    throw new Error(`[FACT_NOT_FOUND] Cannot affirm non-existent fact '${factId}'`)
  }

  const previousStanding = existing.standing
  existing.last_reviewed_epoch = affirmationEpoch
  existing.days_since_reconsideration = 0

  if (existing.standing === 'FLAGGED_FOR_REVIEW') {
    existing.standing = 'ACTIVE_ENDORSED'
  }

  return {
    success: true,
    fact_id: factId,
    version_bumped: false,
    cascade_triggered: false,
    updated_last_reviewed_epoch: affirmationEpoch,
    previous_standing: previousStanding,
    current_standing: existing.standing,
  }
}

export const rankReconsiderationQueueAction = (
  cpy: CanonModel,
  bal: CanonReconsiderationBit,
  ste?: State,
): CanonModel => {
  const dat = bal?.dat
  const currentEpoch = dat?.currentEpoch ?? 0
  const minDaysStale = dat?.minDaysStale ?? 0

  const queue = rankReconsiderationQueue(cpy.factStandings, currentEpoch, minDaysStale)

  if (bal?.slv) {
    bal.slv({
      canBit: {
        idx: 'rank-reconsideration-queue',
        val: 0,
        dat: queue,
      },
    })
  }
  return cpy
}

export const affirmStillGoodAction = (
  cpy: CanonModel,
  bal: CanonReconsiderationBit,
  ste?: State,
): CanonModel => {
  const dat = bal?.dat
  if (!dat || !dat.factId) {
    if (bal?.slv) {
      bal.slv({
        canBit: {
          idx: 'affirm-still-good-err',
          val: -1,

import * as Act from './canon.action.js'
import { CanonModel } from './canon.model.js'

import type State from '../99.core/state.js'

export function reducer(
  model: CanonModel = new CanonModel(),
  act: Act.Actions,
  state?: State,
): CanonModel {
  switch (act.type) {
    case Act.INIT_CANON:
      return Buzz.initCanon(clone(model), act.bale, state)

    case Act.UPDATE_CANON:
      return Buzz.updateCanon(clone(model), act.bale, state)

    case Act.RANK_RECONSIDERATION_QUEUE:
      return Buzz.rankReconsiderationQueueAction(clone(model), act.bale, state)

    case Act.AFFIRM_STILL_GOOD:
      return Buzz.affirmStillGoodAction(clone(model), act.bale, state)

    default:
      return model
  }
}

import { describe, it, expect } from 'vitest'
import {
  calculateAuditPriorityScoreBps,
  rankReconsiderationQueue,
  affirmStillGood,
} from '../buz/standing.buzz.js'
import type { CanonFactStanding } from '../fce/standing.schema.js'

describe('41.canon.unit :: Still-Good Reconsideration Priority Calculator & Protocol', () => {
  it('calculates audit priority score using integer basis points without float drift', () => {
    // 90 days stale, 42 dependent scenes
    // log10(43) = 1.6334684555793223
    // 90 * 1.6334684555793223 * 100 = 14701.216 -> Math.round -> 14701 bps
    const scoreBps = calculateAuditPriorityScoreBps(90, 42)
    expect(scoreBps).toBe(14701)
    expect(Number.isInteger(scoreBps)).toBe(true)

    // 0 days stale yields strictly 0 score
    expect(calculateAuditPriorityScoreBps(0, 10)).toBe(0)
  })

  it('rejects floating-point numbers in priority score calculator', () => {
    expect(() => calculateAuditPriorityScoreBps(90.5, 42)).toThrow(/FLOAT_PROHIBITED/)
    expect(() => calculateAuditPriorityScoreBps(90, 42.2)).toThrow(/FLOAT_PROHIBITED/)
  })

  it('ranks facts descending by priority score and identifies monolithic spines', () => {
    const catalog = new Map<string, CanonFactStanding>([
      [
        'fact_isolated_leaf',
        {
          fact_id: 'fact_isolated_leaf',
          entity_id: 'char_bog',
          standing: 'ACTIVE_ENDORSED',
          in_degree_count: 0,
          days_since_reconsideration: 90,
          last_reviewed_epoch: 0,
        },
      ],
      [
        'fact_monolithic_spine',
        {
          fact_id: 'fact_monolithic_spine',
          entity_id: 'char_wart',
          standing: 'ACTIVE_ENDORSED',
          in_degree_count: 42,
          days_since_reconsideration: 90,
          last_reviewed_epoch: 0,
        },
      ],
      [
        'fact_active_fresh',
        {
          fact_id: 'fact_active_fresh',
          entity_id: 'loc_culvert_siphon',
          standing: 'ACTIVE_ENDORSED',
          in_degree_count: 15,
          days_since_reconsideration: 5,
          last_reviewed_epoch: 0,
        },
      ],
    ])

    const queue = rankReconsiderationQueue(catalog, 0, 0)
    expect(queue).toHaveLength(3)

    // Highest reach unreviewed fact surfaces at top
    expect(queue[0].fact_id).toBe('fact_monolithic_spine')
    expect(queue[0].is_monolithic_spine).toBe(true)
    expect(queue[0].priority_score_bps).toBe(14701)

    // Isolated leaf node has 0 score (log10(1) = 0)
    expect(queue[1].fact_id).toBe('fact_active_fresh')
    expect(queue[2].fact_id).toBe('fact_isolated_leaf')
    expect(queue[2].priority_score_bps).toBe(0)

    expect(affirmed.standing).toBe('ACTIVE_ENDORSED')

  })
