```typescript
import type {
  LimbStatusType,
  SomaticLimbProfile,
  SomaticSatVerificationReport,
  ForeignKeyResolutionReport,
  PreFlightEntityGateReport,
  default as EntityPreFlightGateBit,
} from '../fce/preFlightGate.bit.js'
import type { EntityModel } from '../entity.model.js'
import type State from '../../99.core/state.js'

const LIMB_CAPACITY_BPS: Record<LimbStatusType, number> = {
  INTACT: 10000,
  IMPAIRED: 5000,
  PARALYZED: 0,
  AMPUTATED: 0,
}

/**
 * Evaluates somatic hand-capacity satisfiability in integer basis points (0..20,000 bps).
 */
export function evaluateEntitySomaticSat(
  entityId: string,
  somatic: SomaticLimbProfile | undefined,
  requiredHandsBps: number = 0,
): SomaticSatVerificationReport {
  if (!Number.isInteger(requiredHandsBps)) {
    throw new Error('[FLOAT_PROHIBITED] requiredHandsBps must be an integer')
  }
  if (requiredHandsBps < 0 || requiredHandsBps > 20000) {
    throw new Error('[OUT_OF_BOUNDS] requiredHandsBps must be between 0 and 20000')
  }

  let availableHandsBps = 20000

  if (somatic) {
    if (somatic.available_arms_bps !== undefined) {
      if (!Number.isInteger(somatic.available_arms_bps)) {
        throw new Error('[FLOAT_PROHIBITED] available_arms_bps must be an integer')
      }
      if (somatic.available_arms_bps < 0 || somatic.available_arms_bps > 20000) {
        throw new Error('[OUT_OF_BOUNDS] available_arms_bps must be between 0 and 20000')
      }
      availableHandsBps = somatic.available_arms_bps
    } else if (somatic.left_arm !== undefined || somatic.right_arm !== undefined) {
      const leftBps = LIMB_CAPACITY_BPS[somatic.left_arm || 'INTACT']
      const rightBps = LIMB_CAPACITY_BPS[somatic.right_arm || 'INTACT']
      availableHandsBps = leftBps + rightBps
    }
  }

  const errors: string[] = []
  if (requiredHandsBps > availableHandsBps) {
    errors.push(
      `[SOMATIC_SAT_BREACH] Required ${requiredHandsBps} bps exceeds available ${availableHandsBps} bps on ${entityId}`,
    )
  }

  return {
    valid: errors.length === 0,
    entityId,
    availableHandsBps,
    requiredHandsBps,
    errors,
  }
}

/**
 * Verifies closed-world foreign-key integrity and anti-reflexivity across entity references.
 */
export function verifyEntityForeignKeys(
  entityId: string,
  frontmatter: Record<string, unknown>,
  catalogIds: Set<string> | string[],
): ForeignKeyResolutionReport {
  const knownIds = catalogIds instanceof Set ? catalogIds : new Set(catalogIds)
  const errors: string[] = []
  const danglingKeys: string[] = []
  let checkedPointersCount = 0

  const checkPointer = (field: string, targetId: unknown, forbidReflexive: boolean = false) => {
    if (typeof targetId !== 'string' || targetId.trim().length === 0) return
    checkedPointersCount++

    if (forbidReflexive && targetId === entityId) {
      errors.push(
        `[REFLEXIVE_POINTER_PROHIBITED] Entity '${entityId}' cannot reference itself in '${field}'`,
      )
      return
    }

    if (!knownIds.has(targetId)) {
      danglingKeys.push(targetId)
      errors.push(
        `[DANGLING_FOREIGN_KEY] Field '${field}' on '${entityId}' references unauthored entity '${targetId}'`,
      )
    }
  }

  if (typeof frontmatter.location_id === 'string') {
    checkPointer('location_id', frontmatter.location_id, false)
  }

  if (Array.isArray(frontmatter.possessions)) {
    for (const itemId of frontmatter.possessions) {
      checkPointer('possessions', itemId, false)
    }
  }

  if (Array.isArray(frontmatter.grievance_targets)) {
    for (const targetId of frontmatter.grievance_targets) {
      checkPointer('grievance_targets', targetId, true)
    }
  }

  if (Array.isArray(frontmatter.clue_entities)) {
    for (const clueId of frontmatter.clue_entities) {
      checkPointer('clue_entities', clueId, false)
    }
  }

  return {
    valid: errors.length === 0,
    entityId,
    checkedPointersCount,
    danglingKeys,
    errors,
  }
}

/**
 * Executes the unified Pre-Flight Somatic SAT and Foreign-Key Gate.
 */
export function runEntityPreFlightGate(
  entityId: string,
  frontmatter: Record<string, unknown>,
  catalogIds: Set<string> | string[],
  requiredHandsBps: number = 0,
): PreFlightEntityGateReport {
  const somaticProfile = frontmatter.somatic as SomaticLimbProfile | undefined
  const somatic = evaluateEntitySomaticSat(entityId, somaticProfile, requiredHandsBps)
  const foreignKeys = verifyEntityForeignKeys(entityId, frontmatter, catalogIds)
  const errors = [...somatic.errors, ...foreignKeys.errors]

  return {
    passed: somatic.valid && foreignKeys.valid,
    entityId,
    somatic,
    foreignKeys,
    errors,
  }
}

export const runEntityPreFlightGateAction = (
  cpy: EntityModel,
  bal: EntityPreFlightGateBit,
  ste?: State,
): EntityModel => {
  const dat = bal?.dat
  if (!dat) {
    if (bal?.slv) {
      bal.slv({
        entBit: {
          idx: 'run-entity-preflight-gate-err',
          val: -1,
          dat: {
            passed: false,
            entityId: '',
            somatic: {
              valid: false,
              entityId: '',
              availableHandsBps: 0,
              requiredHandsBps: 0,
              errors: ['[MISSING_PAYLOAD] dat is required'],
            },
            foreignKeys: {
              valid: false,
              entityId: '',
              checkedPointersCount: 0,
              danglingKeys: [],
              errors: [],
            },
            errors: ['[MISSING_PAYLOAD] dat is required'],
          },
        },
      })
    }
    return cpy
  }

    dat.entityId,
    dat.frontmatter,

    dat.requiredHandsBps ?? 0,
  )

  if (bal?.slv) {
    bal.slv({
      entBit: {
        idx: 'run-entity-preflight-gate',
        val: report.passed ? 0 : -1,
        dat: report,
      },
    })
  }

  return cpy
}
```

```typescript
import clone from 'clone-deep'
import * as Act from './entity.action.js'
import { EntityModel } from './entity.model.js'
import * as Buzz from './entity.buzzer.js'
import type State from '../99.core/state.js'

export function reducer(
  model: EntityModel = new EntityModel(),
  act: Act.Actions,
  state?: State,
): EntityModel {
  switch (act.type) {
    case Act.INIT_ENTITY:
      return Buzz.initEntity(clone(model), act.bale, state)

    case Act.UPDATE_ENTITY:
      return Buzz.updateEntity(clone(model), act.bale, state)

    case Act.ASSEMBLE_DUAL_RAIL_DOSSIER:
      return Buzz.assembleDualRailDossierAction(clone(model), act.bale, state)

    case Act.RUN_ENTITY_PREFLIGHT_GATE:
      return Buzz.runEntityPreFlightGateAction(clone(model), act.bale, state)

    default:
      return model
  }
}
```

```typescript
import { describe, it, expect } from 'vitest'
import {
  evaluateEntitySomaticSat,
  verifyEntityForeignKeys,
  runEntityPreFlightGate,
} from '../buz/preFlightGate.buzz.js'

describe('09.entity.unit :: Pre-Flight Somatic SAT Foreign-Key Gate', () => {
  it('computes integer basis-point hand capacities across limb statuses', () => {
    const twoArms = evaluateEntitySomaticSat(
      'char_bog',
      { left_arm: 'INTACT', right_arm: 'INTACT' },
      20000,
    )
    expect(twoArms.valid).toBe(true)
    expect(twoArms.availableHandsBps).toBe(20000)

    const oneArm = evaluateEntitySomaticSat(
      'char_wart',
      { left_arm: 'AMPUTATED', right_arm: 'INTACT' },
      10000,
    )
    expect(oneArm.valid).toBe(true)
    expect(oneArm.availableHandsBps).toBe(10000)

    const impairedArm = evaluateEntitySomaticSat(
      'char_grissel',
      { left_arm: 'IMPAIRED', right_arm: 'PARALYZED' },
      5000,
    )
    expect(impairedArm.valid).toBe(true)
    expect(impairedArm.availableHandsBps).toBe(5000)
  })

  it('halts with SOMATIC_SAT_BREACH when required hand capacity exceeds available limbs', () => {
    const breach = evaluateEntitySomaticSat(
      'char_wart',
      { left_arm: 'AMPUTATED', right_arm: 'INTACT' },
      20000,
    )
    expect(breach.valid).toBe(false)
    expect(breach.availableHandsBps).toBe(10000)
    expect(breach.errors[0]).toContain('[SOMATIC_SAT_BREACH]')
  })

  it('rejects floating-point basis-point values with FLOAT_PROHIBITED', () => {
    expect(() =>
      evaluateEntitySomaticSat('char_wart', { available_arms_bps: 10000.5 }, 10000),
    ).toThrow(/FLOAT_PROHIBITED/)

    expect(() =>
      evaluateEntitySomaticSat('char_wart', { available_arms_bps: 10000 }, 5000.25),
    ).toThrow(/FLOAT_PROHIBITED/)
  })

  it('traps dangling foreign keys and self-reflexive grievance pointers', () => {
    const catalog = ['char_wart', 'char_bog', 'loc_culvert_siphon']
    const fkReport = verifyEntityForeignKeys(
      'char_wart',
      {
        location_id: 'loc_missing_vault',
        possessions: ['item_ghost_lantern'],
        grievance_targets: ['char_wart', 'char_bog'],
      },
      catalog,
    )

    expect(fkReport.valid).toBe(false)
    expect(fkReport.danglingKeys).toContain('loc_missing_vault')
    expect(fkReport.danglingKeys).toContain('item_ghost_lantern')
    expect(fkReport.errors.some((e) => e.includes('[DANGLING_FOREIGN_KEY]'))).toBe(true)
    expect(fkReport.errors.some((e) => e.includes('[REFLEXIVE_POINTER_PROHIBITED]'))).toBe(true)
  })

  it('certifies a valid entity dossier through runEntityPreFlightGate', () => {
    const catalog = ['char_wart', 'char_bog', 'loc_culvert_siphon', 'item_tallow_tin']
    const report = runEntityPreFlightGate(
      'char_wart',
      {
        location_id: 'loc_culvert_siphon',
        possessions: ['item_tallow_tin'],
        grievance_targets: ['char_bog'],
        somatic: { left_arm: 'AMPUTATED', right_arm: 'INTACT' },
      },
      catalog,
      10000,
    )

    expect(report.passed).toBe(true)
    expect(report.errors).toHaveLength(0)
    expect(report.foreignKeys.checkedPointersCount).toBe(3)
  })
})
```
