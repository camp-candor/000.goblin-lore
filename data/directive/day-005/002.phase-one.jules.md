##### 4. Automated AST Directory Sweeper Buzzer

`packages/299.lore/00.lore.unit/buz/hygieneSentinel.buzz.ts`:

```typescript
import fs from 'node:fs'
import path from 'node:path'

export interface HygieneSentinelReport {
  totalFilesScanned: number
  hasSrcDirectory: boolean
  debuggerViolations: { file: string; line: number; text: string }[]
  nonAsciiViolations: { file: string; line: number; text: string }[]
  deadCodeViolations: { file: string; token: string }[]
  isClean: boolean
}

function sweepDirectory(dir: string, fileList: string[] = []): string[] {
  const entries = fs.readdirSync(dir, { withFileTypes: true })
  for (const entry of entries) {
    if (entry.name === 'node_modules' || entry.name === 'dist' || entry.name === '.git') {
      continue
    }
    const fullPath = path.join(dir, entry.name)
    if (entry.isDirectory()) {
      sweepDirectory(fullPath, fileList)
    } else if (entry.isFile() && (entry.name.endsWith('.ts') || entry.name.endsWith('.tsx'))) {
      fileList.push(fullPath)
    }
  }
  return fileList
}

export function runHygieneSentinelSweep(packageRoot: string): HygieneSentinelReport {
  const srcDir = path.join(packageRoot, 'src')
  const hasSrcDirectory = fs.existsSync(srcDir)
  const tsFiles = sweepDirectory(packageRoot)
  
  const debuggerViolations: { file: string; line: number; text: string }[] = []
  const nonAsciiViolations: { file: string; line: number; text: string }[] = []
  const deadCodeViolations: { file: string; token: string }[] = []

  const debuggerRegex = /\bdebugger\b/
  const nonAsciiRegex = /[\x80-\xFF\u{1F300}-\u{1FAD6}]/u

  for (const filePath of tsFiles) {
    const relPath = path.relative(packageRoot, filePath).replace(/\\/g, '/')
    const content = fs.readFileSync(filePath, 'utf8')
    const lines = content.split('\n')

    lines.forEach((line, idx) => {
      if (debuggerRegex.test(line)) {
        debuggerViolations.push({ file: relPath, line: idx + 1, text: line.trim() })
      }
      if (nonAsciiRegex.test(line)) {
        nonAsciiViolations.push({ file: relPath, line: idx + 1, text: line.trim() })
      }
    })

    if (relPath === '00.lore.unit/buz/lore.buzz.ts') {
      if (content.includes('lore.list()')) {
        deadCodeViolations.push({ file: relPath, token: 'lore.list()' })
      }
      if (content.includes('zero00-lore.onrender.com')) {
        deadCodeViolations.push({ file: relPath, token: 'zero00-lore.onrender.com' })
      }
    }
  }

  const isClean =
    !hasSrcDirectory &&
    debuggerViolations.length === 0 &&
    nonAsciiViolations.length === 0 &&
    deadCodeViolations.length === 0

  return {
    totalFilesScanned: tsFiles.length,
    hasSrcDirectory,
    debuggerViolations,
    nonAsciiViolations,
    deadCodeViolations,
    isClean,
  }
}
```
