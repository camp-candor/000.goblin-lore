import yaml from 'js-yaml'

export interface ParsedLocationDossier {
  frontmatter: Record<string, any>
  bodyContent: string
  byteZeroOk: boolean
}

export interface SymmetricalPortalEntry {
  portal_slug: string
  target_location_id: string
  damping_cdb: number
  line_of_sight: boolean
}

/**
 * Parses raw markdown bytes into frontmatter and body, asserting Byte-0 frontmatter law.
 */
export function parseLocationMarkdown(rawContent: string): ParsedLocationDossier {
  const buf = Buffer.from(rawContent, 'utf8')
  if (buf.length >= 3 && buf[0] === 0xef && buf[1] === 0xbb && buf[2] === 0xbf) {
    throw new Error('[BYTE_ZERO_BREACH] UTF-8 BOM detected at physical offset 0')
  }

  if (buf.length < 4 || buf[0] !== 0x2d || buf[1] !== 0x2d || buf[2] !== 0x2d || buf[3] !== 0x0a) {
    throw new Error('[BYTE_ZERO_BREACH] Location dossier must begin with "---\\n" at physical byte offset 0')
  }

  const normalized = rawContent.replace(/\r\n/g, '\n').replace(/\r/g, '\n')
  const parts = normalized.split(/^---$/m)
  if (parts.length < 3) {
    throw new Error('[CORRUPTED_DOSSIER] Location dossier lacks closing frontmatter delimiter')
  }

  const rawYaml = parts[1]
  const bodyContent = parts.slice(2).join('---')
  const frontmatter = (yaml.load(rawYaml) as Record<string, any>) || {}

  return {
    frontmatter,
    bodyContent,
    byteZeroOk: true,
  }
}

/**
 * Injects or synchronizes a reverse portal entry in frontmatter to guarantee bidirectional symmetry.
 */
export function injectReversePortal(
  frontmatter: Record<string, any>,
  sourceLocationId: string,
  incomingPortal: {
    portalSlug?: string
    dampingCdB?: number
    lineOfSight?: boolean
  },
): { modified: boolean; injectedPortal: SymmetricalPortalEntry } {
  frontmatter.portal_links = Array.isArray(frontmatter.portal_links) ? frontmatter.portal_links : []

  const targetId = String(frontmatter.id || 'unknown_loc')
  const reverseSlug = `portal_${targetId}_to_${sourceLocationId}`
  const expectedDamping = incomingPortal.dampingCdB !== undefined ? incomingPortal.dampingCdB : 1500
  const expectedLos = incomingPortal.lineOfSight !== undefined ? incomingPortal.lineOfSight : false

  const existing = frontmatter.portal_links.find(
    (p: any) => p && p.target_location_id === sourceLocationId,
  )

  if (existing) {
    let changed = false
    if (existing.damping_cdb !== expectedDamping) {
      existing.damping_cdb = expectedDamping
      changed = true
    }
    if (existing.line_of_sight !== expectedLos) {
      existing.line_of_sight = expectedLos
      changed = true
    }
    return { modified: changed, injectedPortal: existing }
  }

  const newPortal: SymmetricalPortalEntry = {
    portal_slug: reverseSlug,
    target_location_id: sourceLocationId,
    damping_cdb: expectedDamping,
    line_of_sight: expectedLos,
  }

  frontmatter.portal_links.push(newPortal)
  return { modified: true, injectedPortal: newPortal }
}

/**
 * Re-serializes frontmatter and prose while strictly preserving Byte-0 alignment and POSIX LF endings.
 */
export function serializeLocationMarkdown(
  frontmatter: Record<string, any>,
  bodyContent: string,
): string {
  const serializedYaml = yaml
    .dump(frontmatter, {
      indent: 2,
      lineWidth: -1,
      noRefs: true,
      sortKeys: true,
    })
    .replace(/\r\n/g, '\n')
    .replace(/\r/g, '\n')
    .trim()

  const cleanBody = bodyContent.replace(/\r\n/g, '\n').replace(/\r/g, '\n')
  return `---\n${serializedYaml}\n---${cleanBody}`
}

    if (bal.slv) {
      bal.slv({
        locationBit: {
          idx: bal.idx,
          val: 0,
          dat: {
            seriesSlug,
            mutatedFiles: [],
            injectedPortals: [],
            totalLinksChecked: 0,
            symmetrical: false,
            errors: [errorMsg],
          },
        },
      })
    }
    return cpy
  }

  const sourceFile = path.join(locationsDir, `${locationId}.md`)
  if (!fs.existsSync(sourceFile)) {
    const errorMsg = `[SOURCE_LOCATION_NOT_FOUND] ${sourceFile}`
    await emitConsole(ste, `:: [FAIL] reconcileReciprocalLinks: ${errorMsg}`)
    if (bal.slv) {
      bal.slv({
        locationBit: {
          idx: bal.idx,
          val: 0,
          dat: {
            seriesSlug,
            mutatedFiles: [],
            injectedPortals: [],
            totalLinksChecked: 0,
            symmetrical: false,
            errors: [errorMsg],
          },
        },
      })
    }
    return cpy
  }

  try {
    const sourceRaw = fs.readFileSync(sourceFile, 'utf8')
    const sourceDossier = parseLocationMarkdown(sourceRaw)
    const portalLinks: any[] = Array.isArray(sourceDossier.frontmatter.portal_links)
      ? sourceDossier.frontmatter.portal_links
      : []

    const mutatedFiles: string[] = []
    const injectedPortals: PortalLinkMutation[] = []
    const errors: string[] = []

    for (const portal of portalLinks) {
      const targetId = portal?.target_location_id
      if (!targetId) {
        errors.push(`[MALFORMED_PORTAL] Portal missing target_location_id in '${locationId}'`)
        continue
      }

      if (targetId === locationId) {
        errors.push(`[SELF_ADJACENCY_VIOLATION] Location '${locationId}' cannot link to itself`)
        continue
      }

      const targetFile = path.join(locationsDir, `${targetId}.md`)
      if (!fs.existsSync(targetFile)) {
        errors.push(
          `[DANGLING_SPATIAL_POINTER] Location '${locationId}' links to non-existent target '${targetId}'`,
        )
        continue
      }

      const targetRaw = fs.readFileSync(targetFile, 'utf8')
      const targetDossier = parseLocationMarkdown(targetRaw)

      const injection = injectReversePortal(targetDossier.frontmatter, locationId, {
        portalSlug: portal.portal_slug,
        dampingCdB: portal.damping_cdb,
        lineOfSight: portal.line_of_sight,
      })

      if (injection.modified && !payload.dryRun) {
        const updatedMarkdown = serializeLocationMarkdown(
          targetDossier.frontmatter,
          targetDossier.bodyContent,
        )

        const tmpPath = path.join(locationsDir, `.tmp_${crypto.randomUUID()}_${targetId}.md.tmp`)
        fs.writeFileSync(tmpPath, updatedMarkdown, 'utf8')
        fs.renameSync(tmpPath, targetFile)

        mutatedFiles.push(targetFile)
        injectedPortals.push({
          sourceLocationId: targetId,
          targetLocationId: locationId,
          portalSlug: injection.injectedPortal.portal_slug,
          dampingCdB: injection.injectedPortal.damping_cdb,
          lineOfSight: injection.injectedPortal.line_of_sight,
        })

        await emitConsole(
          ste,
          `>> [LINK] Reconciled reciprocal portal: ${targetId} -> ${locationId} (${injection.injectedPortal.damping_cdb} cdB)`,
        )
      }
    }

    const symmetrical = errors.length === 0
    const receipt: ReconcileSpatialLinksReceipt = {
      seriesSlug,
      mutatedFiles,
      injectedPortals,
      totalLinksChecked: portalLinks.length,
      symmetrical,
      errors: errors.length > 0 ? errors : undefined,
    }

    await emitConsole(
      ste,
      symmetrical
        ? `>> [OK] Spatial portal reciprocity verified for '${locationId}'. Mutated: ${mutatedFiles.length}`
        : `:: [FAIL] Spatial portal reconciliation encountered ${errors.length} defect(s)`,
    )

    if (bal.slv) {
      bal.slv({
        locationBit: {
          idx: bal.idx,
          val: symmetrical ? 1 : 0,
          dat: receipt,
        },
      })
    }
  } catch (err: any) {
    const errorMsg = err?.message || String(err)
    await emitConsole(ste, `:: [FAIL] reconcileReciprocalLinks: ${errorMsg}`)
    if (bal.slv) {
      bal.slv({
        locationBit: {
          idx: bal.idx,
          val: 0,
          dat: {
            seriesSlug,
            mutatedFiles: [],
            injectedPortals: [],
            totalLinksChecked: 0,
            symmetrical: false,
            errors: [errorMsg],
          },
        },
      })
    }
  }

  return cpy
}

import { LocationModel } from './location.model.js'

export function reducer(
  model: LocationModel = new LocationModel(),
  act: Act.Actions,
  state?: State,
): LocationModel {
  switch (act.type) {
    case Act.INIT_LOCATION:
      return Buzz.initLocation(clone(model), act.bale, state)

    default:
      return model
  }
}
