```typescript
import type {
  LemmaTrieNode,
  ClueDisjointnessCollision,
  TrieDisjointnessReport,
  MysteryTrieIngress,
  default as CanonTrieBit,
} from '../fce/trieDisjointness.bit.js'
import type { CanonModel } from '../canon.model.js'
import type State from '../../99.core/state.js'

export class PropositionTrie {
  private root: LemmaTrieNode = { children: new Map(), isTerminal: false }
  private lemmaCount: number = 0

  public insert(lemma: string, topicId: string, propositionId: string): void {
    const cleanLemma = lemma.toLowerCase().trim()
    if (!cleanLemma) return

    let curr = this.root
    for (const char of cleanLemma) {
      if (!curr.children.has(char)) {
        curr.children.set(char, { children: new Map(), isTerminal: false })
      }
      curr = curr.children.get(char)!
    }
    curr.isTerminal = true
    curr.topicId = topicId
    curr.propositionId = propositionId
    curr.lemma = cleanLemma
    this.lemmaCount++
  }

  public search(token: string): { topicId: string; propositionId: string; lemma: string } | null {
    const cleanToken = token.toLowerCase().trim()
    if (!cleanToken) return null

    let curr = this.root
    for (const char of cleanToken) {
      if (!curr.children.has(char)) {
        return null
      }
      curr = curr.children.get(char)!
    }

    if (curr.isTerminal && curr.topicId && curr.propositionId && curr.lemma) {
      return {
        topicId: curr.topicId,
        propositionId: curr.propositionId,
        lemma: curr.lemma,
      }
    }

    return null
  }

  public get size(): number {
    return this.lemmaCount
  }
}

export class TrieClueDisjointnessSolver {
  private trie: PropositionTrie

  constructor() {
    this.trie = new PropositionTrie()
  }

  public loadForbiddenPropositions(mysteries: MysteryTrieIngress[]): void {
    for (const mystery of mysteries) {
      const topicId = mystery.topic_id
      for (const prop of mystery.forbidden_propositions) {
        const propId = prop.id

        if (Array.isArray(prop.trigger_lemmas)) {
          for (const lemma of prop.trigger_lemmas) {
            this.trie.insert(lemma, topicId, propId)
          }
        }

        if (Array.isArray(prop.trigger_lemma_sets)) {
          for (const set of prop.trigger_lemma_sets) {
            for (const lemma of set) {
              this.trie.insert(lemma, topicId, propId)
            }
          }
        }
      }
    }
  }

  public evaluateDisjointness(
    permittedClues: string[],
    targetEntityId: string = 'system_ontology',
  ): TrieDisjointnessReport {
    const collisions: ClueDisjointnessCollision[] = []
    const evaluatedClues = permittedClues.map((c) => c.toLowerCase().trim()).filter(Boolean)

    for (const clue of evaluatedClues) {
      const match = this.trie.search(clue)
      if (match) {
        collisions.push({
          collidingToken: clue,
          topicId: match.topicId,
          propositionId: match.propositionId,
          lemma: match.lemma,
          errorVector: {
            domain: 'EPISTEMIC',
            target_entity: targetEntityId,
            rule_id: 'CLUE_DISJOINTNESS_COLLISION',
            offending_tokens: [clue, match.topicId, match.propositionId, match.lemma],
          },
        })
      }
    }

    return {
      valid: collisions.length === 0,
      totalLemmasIndexed: this.trie.size,
      totalCluesEvaluated: evaluatedClues.length,
      collisions,
    }
  }

  public static verifyDisjointnessAcrossCorpus(
    mysteries: MysteryTrieIngress[],
    targetEntityId: string = 'system_ontology',
  ): TrieDisjointnessReport {
    const solver = new TrieClueDisjointnessSolver()
    solver.loadForbiddenPropositions(mysteries)

    const allClues: string[] = []
    for (const m of mysteries) {
      if (Array.isArray(m.permitted_clue_tokens)) {
        allClues.push(...m.permitted_clue_tokens)
      }
      if (Array.isArray(m.clue_tokens)) {
        allClues.push(...m.clue_tokens)
      }
    }

    return solver.evaluateDisjointness(allClues, targetEntityId)
  }
}

export const verifyTrieClueDisjointnessAction = (
  cpy: CanonModel,
  bal: CanonTrieBit,
  ste?: State,
): CanonModel => {
  const dat = bal?.dat
  if (!dat || !dat.mysteries) {
    if (bal?.slv) {
      bal.slv({
        canBit: {
          idx: 'verify-trie-clue-disjointness-err',
          val: -1,
          dat: {
            valid: false,
            totalLemmasIndexed: 0,
            totalCluesEvaluated: 0,
            collisions: [],
          },
        },
      })
    }
    return cpy
  }

  const clues = dat.permittedClues || []
  let report: TrieDisjointnessReport

  if (clues.length > 0) {
    const solver = new TrieClueDisjointnessSolver()
    solver.loadForbiddenPropositions(dat.mysteries)
    report = solver.evaluateDisjointness(clues, dat.targetEntityId || 'system_ontology')
  } else {
    report = TrieClueDisjointnessSolver.verifyDisjointnessAcrossCorpus(
      dat.mysteries,
      dat.targetEntityId || 'system_ontology',
    )
  }

  if (bal?.slv) {
    bal.slv({
      canBit: {
        idx: 'verify-trie-clue-disjointness',
        val: report.valid ? 0 : -1,
        dat: report,
      },
    })
  }

  return cpy
}
```

#### 8. Create: `packages/299.lore/41.canon.unit/test/trieDisjointness.test.ts`

```typescript
import { describe, it, expect } from 'vitest'
import {
  PropositionTrie,
  TrieClueDisjointnessSolver,
} from '../buz/trieDisjointness.buzz.js'
import type { MysteryTrieIngress } from '../fce/trieDisjointness.bit.js'

describe('41.canon.unit :: O(L) Proposition Trie & Clue-Lemma Disjointness Scanner', () => {
  it('inserts and searches exact single-word and multi-word trigger lemmas in O(L) time', () => {
    const trie = new PropositionTrie()
    trie.insert('arsenic', 'mystery_well', 'prop_poison_01')
    trie.insert('well water', 'mystery_well', 'prop_water_02')
    trie.insert('culvert siphon', 'mystery_culvert', 'prop_culvert_01')

    expect(trie.size).toBe(3)

    // Exact matches
    const match1 = trie.search('arsenic')
    expect(match1).not.toBeNull()
    expect(match1?.topicId).toBe('mystery_well')
    expect(match1?.propositionId).toBe('prop_poison_01')
    expect(match1?.lemma).toBe('arsenic')

    const match2 = trie.search('well water')
    expect(match2).not.toBeNull()
    expect(match2?.lemma).toBe('well water')

    // Case and whitespace insensitivity
    const match3 = trie.search('  ARSENIC  ')
    expect(match3).not.toBeNull()
    expect(match3?.lemma).toBe('arsenic')

    // Non-matches and partial prefix failures
    expect(trie.search('arse')).toBeNull()
    expect(trie.search('well')).toBeNull()
    expect(trie.search('pure water')).toBeNull()
  })

  it('passes verification when permitted clues and forbidden lemmas are strictly disjoint', () => {
    const solver = new TrieClueDisjointnessSolver()
    const mysteries: MysteryTrieIngress[] = [
      {
        topic_id: 'mystery_famine_well',
        forbidden_propositions: [
          {
            id: 'prop_arsenic_taint',
            trigger_lemmas: ['poison', 'arsenic', 'taint', 'well water'],
          },
        ],
        permitted_clue_tokens: ['rusted_bucket', 'cut_rope', 'iron_grate'],
      },
    ]

    solver.loadForbiddenPropositions(mysteries)
    const report = solver.evaluateDisjointness(['rusted_bucket', 'cut_rope', 'iron_grate'], 'char_bog')

    expect(report.valid).toBe(true)
    expect(report.collisions).toHaveLength(0)
    expect(report.totalLemmasIndexed).toBe(4)
    expect(report.totalCluesEvaluated).toBe(3)
  })

  it('halts with CLUE_DISJOINTNESS_COLLISION when a clue intersects with a forbidden lemma', () => {
    const solver = new TrieClueDisjointnessSolver()
    const mysteries: MysteryTrieIngress[] = [
      {
        topic_id: 'mystery_famine_well',
        forbidden_propositions: [
          {
            id: 'prop_arsenic_taint',
            trigger_lemmas: ['arsenic', 'taint', 'sludge residue'],
          },
        ],
      },
    ]

    solver.loadForbiddenPropositions(mysteries)

    // "taint" and "sludge residue" are forbidden
    const dirtyClues = ['clean_pebble', 'taint', 'sludge residue', 'bronze_thimble']
    const report = solver.evaluateDisjointness(dirtyClues, 'char_wart')

    expect(report.valid).toBe(false)
    expect(report.collisions).toHaveLength(2)

    const col1 = report.collisions[0]
    expect(col1.collidingToken).toBe('taint')
    expect(col1.topicId).toBe('mystery_famine_well')
    expect(col1.propositionId).toBe('prop_arsenic_taint')
    expect(col1.errorVector.domain).toBe('EPISTEMIC')
    expect(col1.errorVector.rule_id).toBe('CLUE_DISJOINTNESS_COLLISION')
    expect(col1.errorVector.target_entity).toBe('char_wart')

    const col2 = report.collisions[1]
    expect(col2.collidingToken).toBe('sludge residue')
    expect(col2.lemma).toBe('sludge residue')
  })

  it('evaluates static corpus-wide disjointness helper across multiple mystery manifests', () => {
    const corpusMysteries: MysteryTrieIngress[] = [
      {
        topic_id: 'mystery_boiler_crack',
        clue_tokens: ['soot_flake', 'steam_gauge'],
        forbidden_propositions: [
          {
            id: 'prop_boiler_sabotage',
            trigger_lemma_sets: [['sabotage', 'pipe wrench']],
          },
        ],
      },
      {
        topic_id: 'mystery_missing_tallow',
        clue_tokens: ['grease_stain', 'pipe wrench'], // "pipe wrench" collides with sabotage
        forbidden_propositions: [
          {
            id: 'prop_tallow_theft',
            trigger_lemmas: ['theft', 'mum grissel'],
          },
        ],
      },
    ]

    const report = TrieClueDisjointnessSolver.verifyDisjointnessAcrossCorpus(corpusMysteries, 'system_ontology')

    expect(report.valid).toBe(false)
    expect(report.collisions).toHaveLength(1)
    expect(report.collisions[0].collidingToken).toBe('pipe wrench')
    expect(report.collisions[0].topicId).toBe('mystery_boiler_crack')
  })
})
```
