// packages/001.lore/src/health/acyclicityVerifier.ts

export const DAG_TAG = '[LORE::DAG]' as const;

export enum NodeColor {
  WHITE = 0, // Unvisited
  GRAY = 1,  // Visiting (on active recursion stack)
  BLACK = 2, // Fully explored and verified acyclic
}

export interface DirectedGraphEdge {
  fromId: string;
  toId: string;
  edgeType: string;
}

export interface AcyclicityVerificationResult {
  isAcyclic: boolean;
  cycleDetected: boolean;
  cycleLoop: string[];
  danglingPointers: string[];
  topologicalOrder: string[];
}

/**
 * Sweeps all referenced foreign keys across graph edges against known entity IDs.
 * Rejects missing dependencies with fail-closed precision at T0 = $0.00.
 */
export function verifyReferentialIntegrity(
  knownEntityIds: ReadonlySet<string>,
  edges: readonly DirectedGraphEdge[]
): string[] {
  const dangling: string[] = [];

  for (const edge of edges) {
    if (!knownEntityIds.has(edge.fromId)) {
      dangling.push(
        `${DAG_TAG} [FAIL] Dangling source pointer: '${edge.fromId}' (edgeType: ${edge.edgeType}) not in registry`
      );
    }
    if (!knownEntityIds.has(edge.toId)) {
      dangling.push(
        `${DAG_TAG} [FAIL] Dangling target pointer: '${edge.toId}' referenced by '${edge.fromId}' (edgeType: ${edge.edgeType}) not in registry`
      );
    }
  }

  return dangling;
}

/**
 * Three-Color Depth-First Search (DFS) Acyclicity & Topological Sort Verifier.
 * Detects back-edges to GRAY nodes and returns the exact cyclical chain.
 */
export function verifyGraphAcyclicity(
  nodeIds: readonly string[],
  edges: readonly DirectedGraphEdge[]
): AcyclicityVerificationResult {
  // Build adjacency list
  const adjacency = new Map<string, string[]>();
  for (const id of nodeIds) {
    adjacency.set(id, []);
  }
  for (const edge of edges) {
    if (!adjacency.has(edge.fromId)) {
      adjacency.set(edge.fromId, []);
    }
    adjacency.get(edge.fromId)!.push(edge.toId);
  }

  // Deterministic sorting of node IDs and adjacency lists by UTF-16 code units
  const sortedNodes = [...nodeIds].sort((a, b) => (a < b ? -1 : a > b ? 1 : 0));
  for (const [id, neighbors] of adjacency.entries()) {
    adjacency.set(
      id,
      [...neighbors].sort((a, b) => (a < b ? -1 : a > b ? 1 : 0))
    );
  }

  const colors = new Map<string, NodeColor>();
  for (const id of sortedNodes) {
    colors.set(id, NodeColor.WHITE);
  }

  const recursionStack: string[] = [];
  const topologicalOrder: string[] = [];
  let detectedCycle: string[] = [];

  function dfs(currentId: string): boolean {
    colors.set(currentId, NodeColor.GRAY);
    recursionStack.push(currentId);

    const neighbors = adjacency.get(currentId) || [];
    for (const neighborId of neighbors) {
      const neighborColor = colors.get(neighborId) ?? NodeColor.WHITE;

      if (neighborColor === NodeColor.GRAY) {
        // Back-edge detected! Reconstruct cycle loop from recursionStack
        const cycleStartIndex = recursionStack.indexOf(neighborId);
        detectedCycle = [...recursionStack.slice(cycleStartIndex), neighborId];
        return false;
      }

      if (neighborColor === NodeColor.WHITE) {
        if (!dfs(neighborId)) {
          return false;
        }
      }
    }

    colors.set(currentId, NodeColor.BLACK);
    recursionStack.pop();
    topologicalOrder.push(currentId);
    return true;
  }

  for (const id of sortedNodes) {
    if (colors.get(id) === NodeColor.WHITE) {
      if (!dfs(id)) {
        return {
          isAcyclic: false,
          cycleDetected: true,
          cycleLoop: detectedCycle,
          danglingPointers: [],
          topologicalOrder: [],
        };
      }
    }
  }

  // Reverse topological order to produce forward dependency traversal
  return {
    isAcyclic: true,
    cycleDetected: false,
    cycleLoop: [],
    danglingPointers: [],
    topologicalOrder: topologicalOrder.reverse(),
  };
}

import * as crypto from 'node:crypto';

export const HASH_TAG = '[LORE::H_DEP]' as const;

/**
 * RFC 8785 Canonical JSON Stringifier.
 * Recursively sorts all dictionary keys lexicographically by UTF-16 code units.
 * Leaves arrays intact to strictly preserve causal timelines and event sequences.
 */
export function canonicalJsonStringify(obj: unknown): string {
  if (obj === null || typeof obj !== 'object') {
    return JSON.stringify(obj);
  }

  if (Array.isArray(obj)) {
    const serializedElements = obj.map((item) => canonicalJsonStringify(item));
    return `[${serializedElements.join(',')}]`;
  }

  const record = obj as Record<string, unknown>;
  const sortedKeys = Object.keys(record).sort((a, b) =>
    a < b ? -1 : a > b ? 1 : 0
  );

  const keyValues: string[] = [];
  for (const key of sortedKeys) {
    const value = record[key];
    if (value !== undefined) {
      keyValues.push(
        `${JSON.stringify(key)}:${canonicalJsonStringify(value)}`
      );
    }
  }

  return `{${keyValues.join(',')}}`;
}

/**
 * Computes deterministic 64-character lowercase hexadecimal SHA-256 digest of any canonical payload.
 */
export function computeSha256(content: string | Buffer): string {
  return crypto.createHash('sha256').update(content).digest('hex');
}

/**
 * Computes discrete entity content hash from an entity's data object.
 */
export function computeEntityContentHash(entityPayload: unknown): string {
  const canonicalString = canonicalJsonStringify(entityPayload);
  return computeSha256(Buffer.from(canonicalString, 'utf8'));
}

/**
 * Computes 1-Hop Direct Dependency Hash (H_dep).
 * Formula: H_dep = SHA-256(Sort([H_target, H_fk1, H_fk2, ...]))
 * Hashes are sorted strictly by UTF-16 code units prior to concatenation.
 */
export function computeDirectOneHopHdep(
  targetContentHash: string,
  directForeignHashes: readonly string[]
): string {
  const hashSet = [targetContentHash, ...directForeignHashes];
  const sortedHashes = hashSet.sort((a, b) => (a < b ? -1 : a > b ? 1 : 0));
  const concatenated = sortedHashes.join('');
  return computeSha256(Buffer.from(concatenated, 'utf8'));
}

/**
 * Computes Multi-Hop Transitive Dependency Fingerprint (H_dep_transitive).
 * Traverses deep dependency graph to gather all reachable ancestor hashes,
 * sorts them lexicographically by UTF-16 code units, and computes the composite digest.
 */
export function computeTransitiveHdep(
  targetId: string,
  entityContentHashes: ReadonlyMap<string, string>,
  dependencyGraph: ReadonlyMap<string, readonly string[]>
): { transitiveHdep: string; visitedSet: string[] } {
  const visited = new Set<string>();

  function traverse(currentId: string) {
    if (visited.has(currentId)) return;
    visited.add(currentId);

    const neighbors = dependencyGraph.get(currentId) || [];
    for (const neighbor of neighbors) {
      traverse(neighbor);
    }
  }

  traverse(targetId);

  const gatheredHashes: string[] = [];
  for (const id of visited) {
    const hash = entityContentHashes.get(id);
    if (hash) {
      gatheredHashes.push(hash);
    }
  }

  // Sort hashes lexicographically
  gatheredHashes.sort((a, b) => (a < b ? -1 : a > b ? 1 : 0));
  const digest = computeSha256(Buffer.from(gatheredHashes.join(''), 'utf8'));

  return {
    transitiveHdep: digest,
    visitedSet: [...visited].sort((a, b) => (a < b ? -1 : a > b ? 1 : 0)),
  };
}

/**
 * Computes composite Scene Fingerprint:
 * SceneHash = SHA256(AST_hash + sum(Dependency_hashes) + Assets_hash + CompilerVersion)
 */
export function computeSceneFingerprint(params: {
  astHash: string;
  dependencyHashes: readonly string[];
  assetsHash: string;
  compilerVersion: string;
}): string {
  const sortedDepHashes = [...params.dependencyHashes].sort((a, b) =>
    a < b ? -1 : a > b ? 1 : 0
  );
  const compositeBuffer = [
    params.astHash,
    sortedDepHashes.join(''),
    params.assetsHash,
    params.compilerVersion,
  ].join('::');

  return computeSha256(Buffer.from(compositeBuffer, 'utf8'));
}

### 3. File: `packages/001.lore/scripts/compile.ts`

import * as fs from 'node:fs';
import * as path from 'node:path';
import * as crypto from 'node:crypto';
import {
  verifyReferentialIntegrity,
  verifyGraphAcyclicity,
  DirectedGraphEdge,
} from '../src/health/acyclicityVerifier';
import {
  canonicalJsonStringify,
  computeEntityContentHash,
  computeDirectOneHopHdep,
  computeTransitiveHdep,
  computeSha256,
} from './hasher';

export const COMPILER_TAG = '[LORE::COMPILER]' as const;
export const COMPILER_VERSION = '1.8.2-deterministic';

export interface RawEntityDossier {
  id: string;
  type: 'character' | 'location' | 'possession' | 'grievance' | 'fact';
  data: Record<string, unknown>;
  direct_dependencies: string[];
}

export interface CompiledEntityManifest {
  id: string;
  type: string;
  content_hash: string;
  h_dep_1hop: string;
  h_dep_transitive: string;
  data: Record<string, unknown>;
  dependencies: string[];
}

export interface CompiledBibleState {
  _meta: {
    compiler_version: string;
    state_hash: string; // Merkle H_root
    total_entities: number;
    sealed_epoch: number;
  };
  entities: Record<string, CompiledEntityManifest>;
}

/**
 * Pure Deterministic Compiler Pipeline for Cluster 01.canon.
 */
export function compileCanonState(
  rawDossiers: readonly RawEntityDossier[],
  sealedEpoch = 0
): CompiledBibleState {
  process.stdout.write(`${COMPILER_TAG} Starting 3-Pass Deterministic Compilation...\n`);

  // --- PASS 1: Entity ID Set Indexing & Structural Collection ---
  const knownIds = new Set<string>();
  const entityMap = new Map<string, RawEntityDossier>();
  for (const dossier of rawDossiers) {
    if (knownIds.has(dossier.id)) {
      throw new Error(`${COMPILER_TAG} [FAIL] Duplicate Entity ID detected: '${dossier.id}'`);
    }
    knownIds.add(dossier.id);
    entityMap.set(dossier.id, dossier);
  }

  // --- PASS 2: Relational Integrity & Three-Color DFS Acyclicity ---
  const edges: DirectedGraphEdge[] = [];
  const dependencyGraph = new Map<string, string[]>();

  for (const dossier of rawDossiers) {
    dependencyGraph.set(dossier.id, [...dossier.direct_dependencies]);
    for (const depId of dossier.direct_dependencies) {
      edges.push({
        fromId: dossier.id,
        toId: depId,
        edgeType: `${dossier.type}_reference`,
      });
    }
  }

  // 1. Foreign-key referential sweep
  const dangling = verifyReferentialIntegrity(knownIds, edges);
  if (dangling.length > 0) {
    for (const d of dangling) {
      process.stderr.write(`${d}\n`);
    }
    throw new Error(
      `${COMPILER_TAG} [FAIL] Pass 2 Relational Failure: ${dangling.length} dangling pointers detected.`
    );
  }

  // 2. Three-color DFS cycle verification
  const acyclicity = verifyGraphAcyclicity([...knownIds], edges);
  if (!acyclicity.isAcyclic) {
    const cyclePath = acyclicity.cycleLoop.join(' -> ');
    throw new Error(
      `${COMPILER_TAG} [FAIL] Pass 2 Causal Cycle Detected: ${cyclePath}`
    );
  }
  process.stdout.write(`${COMPILER_TAG} [OK] Pass 2 Relational Integrity & DAG Acyclicity Verified.\n`);

  // --- PASS 3: RFC 8785 Hashes, 1-Hop H_dep, Transitive Fingerprints & Merkle Seal ---
  const contentHashes = new Map<string, string>();
  for (const dossier of rawDossiers) {
    const hash = computeEntityContentHash(dossier.data);
    contentHashes.set(dossier.id, hash);
  }

  const compiledEntities: Record<string, CompiledEntityManifest> = {};
  const sortedEntityIds = [...knownIds].sort((a, b) => (a < b ? -1 : a > b ? 1 : 0));

  for (const id of sortedEntityIds) {
    const dossier = entityMap.get(id)!;
    const targetHash = contentHashes.get(id)!;
    const foreignHashes = dossier.direct_dependencies
      .map((depId) => contentHashes.get(depId)!)
      .filter(Boolean);

    const hDep1Hop = computeDirectOneHopHdep(targetHash, foreignHashes);
    const { transitiveHdep } = computeTransitiveHdep(id, contentHashes, dependencyGraph);

    compiledEntities[id] = {
      id: dossier.id,
      type: dossier.type,
      content_hash: targetHash,
      h_dep_1hop: hDep1Hop,
      h_dep_transitive: transitiveHdep,
      data: dossier.data,
      dependencies: [...dossier.direct_dependencies].sort((a, b) =>
        a < b ? -1 : a > b ? 1 : 0
      ),
    };
  }

  // Compute Master Merkle H_root by canonicalizing all entity manifests
  const entitiesCanonicalString = canonicalJsonStringify(compiledEntities);
  const masterRootHash = computeSha256(
    Buffer.from(`${COMPILER_VERSION}::${entitiesCanonicalString}`, 'utf8')
  );

  const compiledState: CompiledBibleState = {
    _meta: {
      compiler_version: COMPILER_VERSION,
      state_hash: masterRootHash,
      total_entities: sortedEntityIds.length,
      sealed_epoch: sealedEpoch,
    },
    entities: compiledEntities,
  };

  process.stdout.write(
    `${COMPILER_TAG} [OK] Master Root Seal (H_root): sha256:${masterRootHash}\n`
  );
  return compiledState;
}

/**
 * Atomic File Emission using temporary sibling inode and POSIX renameSync.
 */
export function writeCompiledStateAtomically(
  outputFilePath: string,
  state: CompiledBibleState
): void {
  const outputDir = path.dirname(outputFilePath);
  if (!fs.existsSync(outputDir)) {
    fs.mkdirSync(outputDir, { recursive: true });
  }

  const serialized = canonicalJsonStringify(state);
  const tempUuid = crypto.randomBytes(8).toString('hex');
  const tempFilePath = `${outputFilePath}.tmp_${tempUuid}`;

  fs.writeFileSync(tempFilePath, serialized, 'utf8');
  fs.renameSync(tempFilePath, outputFilePath);

  process.stdout.write(
    `${COMPILER_TAG} [OK] Atomically sealed canon state to ${outputFilePath}\n`
  );
}
