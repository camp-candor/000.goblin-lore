import {
  CharacterVocalProfile,
  ProposedDialogueLine,
  VocalTractCondition,
  BannedCadence,
} from "../schemas/somaticTrauma";

export interface VocalFeasibilityResult {
  satisfiable: boolean;
  errorCode?:
    | "VOCAL_DELIVERY_BREACH"
    | "PROSODIC_EXCLUSION_BREACH"
    | "POST_MORTEM_VOCAL_BREACH"
    | "UNCONSCIOUS_VOCAL_BREACH";
  violatingTokens: string[];
  reason?: string;
  errorVector?: {
    domain: "SOMATIC_VOCAL";
    target_entity: string;
    rule_id: string;
    offending_tokens: string[];
  };
}

export class VocalFeasibilityEngine {
  private static readonly VOCAL_TRAUMA_CONDITIONS: ReadonlySet<VocalTractCondition> = new Set([
    "mute",
    "crushed_larynx",
    "severed_trachea",
    "bilateral_vocal_cord_paralysis",
    "acute_asphyxiation",
  ]);

  /**
   * Evaluates the Vocal-Motor Feasibility Theorem at Stage T0 = $0.00:
   * c.has_verbal_directives => ("mute" not in c.conditions and "speak" not in c.banned_verbs and c.vitality_bps > 0)
   */
  public static evaluateFeasibility(
    profile: CharacterVocalProfile,
    proposedLine: ProposedDialogueLine
  ): VocalFeasibilityResult {
    const hasSpokenWords = proposedLine.dialogue_text.trim().length > 0;

    // Check 1: Post-Mortem Vocal Attempt
    if (profile.vitality_bps === 0) {
      if (hasSpokenWords || proposedLine.is_non_verbal_expression) {
        return {
          satisfiable: false,
          errorCode: "POST_MORTEM_VOCAL_BREACH",
          violatingTokens: ["DECEASED_ACTOR_VITALITY_ZERO"],
          reason: `POST_MORTEM_ERROR: Actor '${profile.character_id}' vitality is 0. Deceased actors cannot vocalize.`,
          errorVector: {
            domain: "SOMATIC_VOCAL",
            target_entity: profile.character_id,
            rule_id: "POST_MORTEM_VOCAL_BREACH",
            offending_tokens: ["vitality_bps: 0"],
          },
        };
      }
      return { satisfiable: true, violatingTokens: [] };
    }

    // Check 2: Unconscious Vocal Attempt
    if (!profile.is_conscious && hasSpokenWords) {
      return {
        satisfiable: false,
        errorCode: "UNCONSCIOUS_VOCAL_BREACH",
        violatingTokens: ["UNCONSCIOUS_STATE"],
        reason: `UNCONSCIOUS_ERROR: Actor '${profile.character_id}' is unconscious and cannot articulate dialogue.`,
        errorVector: {
          domain: "SOMATIC_VOCAL",
          target_entity: profile.character_id,
          rule_id: "UNCONSCIOUS_VOCAL_BREACH",
          offending_tokens: ["is_conscious: false"],
        },
      };
    }

    // Check 3: Physical Gasp & Exertion Exception
    // Non-verbal gasps, coughs, and wheezes are permitted IF dialogue_text is strictly empty
    if (!hasSpokenWords && proposedLine.is_non_verbal_expression) {
      return {
        satisfiable: true,
        violatingTokens: [],
      };
    }

    // Check 4: The Vocal-Motor Feasibility Theorem for Spoken Dialogue
    if (hasSpokenWords) {
      const activeTrauma = profile.active_conditions.filter((cond) =>
        this.VOCAL_TRAUMA_CONDITIONS.has(cond)
      );

      const bannedVerbsUpper = new Set(profile.banned_kinetic_verbs.map((v) => v.toUpperCase()));
      const isSpeechBanned =
        bannedVerbsUpper.has("SPEAK") ||
        bannedVerbsUpper.has("WHISPER") ||
        bannedVerbsUpper.has("SHOUT");

      if (activeTrauma.length > 0 || isSpeechBanned) {
        const violations = [
          ...activeTrauma,
          ...(isSpeechBanned ? ["BANNED_VERB_SPEAK"] : []),
        ];

        return {
          satisfiable: false,
          errorCode: "VOCAL_DELIVERY_BREACH",
          violatingTokens: violations,
          reason: `VOCAL_DELIVERY_BREACH: Actor '${profile.character_id}' has vocal impairment [${violations.join(", ")}] but is assigned spoken dialogue: "${proposedLine.dialogue_text}".`,
          errorVector: {
            domain: "SOMATIC_VOCAL",
            target_entity: profile.character_id,
            rule_id: "VOCAL_DELIVERY_BREACH",
            offending_tokens: violations,
          },
        };
      }
    }

    // Check 5: Prosodic Cadence Exclusion Firewall
    if (proposedLine.delivery_cadences.length > 0) {
      const bannedCadenceSet = new Set<BannedCadence>(profile.banned_cadences);
      const cadenceBreaches = proposedLine.delivery_cadences.filter((c) =>
        bannedCadenceSet.has(c)
      );

      if (cadenceBreaches.length > 0) {
        return {
          satisfiable: false,
          errorCode: "PROSODIC_EXCLUSION_BREACH",
          violatingTokens: cadenceBreaches,
          reason: `PROSODIC_EXCLUSION_BREACH: Cadences [${cadenceBreaches.join(", ")}] are blacklisted for actor '${profile.character_id}'.`,
          errorVector: {
            domain: "SOMATIC_VOCAL",
            target_entity: profile.character_id,
            rule_id: "PROSODIC_EXCLUSION_BREACH",
            offending_tokens: cadenceBreaches,
          },
        };
      }
    }

    return {
      satisfiable: true,
      violatingTokens: [],
    };
  }
}

import Database from "better-sqlite3";
import { CharacterVocalProfile, CharacterVocalProfileSchema } from "../schemas/somaticTrauma";

export class LocalVocalLedger {
  public db: Database.Database;

  constructor(dbPath: string = ":memory:") {
    this.db = new Database(dbPath);
    this.db.pragma("foreign_keys = ON");
    this.initSchema();
  }

  private initSchema(): void {
    this.db.exec(`
      CREATE TABLE IF NOT EXISTS character_vocal_profiles (
        character_id TEXT PRIMARY KEY,
        vitality_bps INTEGER NOT NULL CHECK(vitality_bps BETWEEN 0 AND 10000),
        is_conscious INTEGER NOT NULL DEFAULT 1,
        active_conditions_json TEXT NOT NULL DEFAULT '[]',
        banned_kinetic_verbs_json TEXT NOT NULL DEFAULT '[]',
        banned_cadences_json TEXT NOT NULL DEFAULT '[]',
        updated_at_epoch INTEGER NOT NULL DEFAULT 0
      );
    `);
  }

  public upsertProfile(profile: CharacterVocalProfile): void {
    const validated = CharacterVocalProfileSchema.parse(profile);
    const stmt = this.db.prepare(`
      INSERT INTO character_vocal_profiles (
        character_id, vitality_bps, is_conscious, active_conditions_json,
        banned_kinetic_verbs_json, banned_cadences_json, updated_at_epoch
      ) VALUES (?, ?, ?, ?, ?, ?, ?)
      ON CONFLICT(character_id) DO UPDATE SET
        vitality_bps = excluded.vitality_bps,
        is_conscious = excluded.is_conscious,
        active_conditions_json = excluded.active_conditions_json,
        banned_kinetic_verbs_json = excluded.banned_kinetic_verbs_json,
        banned_cadences_json = excluded.banned_cadences_json,
        updated_at_epoch = excluded.updated_at_epoch;
    `);

    stmt.run(
      validated.character_id,
      validated.vitality_bps,
      validated.is_conscious ? 1 : 0,
      JSON.stringify(validated.active_conditions),
      JSON.stringify(validated.banned_kinetic_verbs),
      JSON.stringify(validated.banned_cadences),
      validated.updated_at_epoch
    );
  }

  public getProfile(characterId: string): CharacterVocalProfile | undefined {
    const row = this.db
      .prepare("SELECT * FROM character_vocal_profiles WHERE character_id = ?")
      .get(characterId) as
      | {
          character_id: string;
          vitality_bps: number;
          is_conscious: number;
          active_conditions_json: string;
          banned_kinetic_verbs_json: string;
          banned_cadences_json: string;
          updated_at_epoch: number;
        }
      | undefined;

    if (!row) return undefined;

    return CharacterVocalProfileSchema.parse({
      character_id: row.character_id,
      vitality_bps: row.vitality_bps,
      is_conscious: row.is_conscious === 1,
      active_conditions: JSON.parse(row.active_conditions_json),
      banned_kinetic_verbs: JSON.parse(row.banned_kinetic_verbs_json),
      banned_cadences: JSON.parse(row.banned_cadences_json),
      updated_at_epoch: row.updated_at_epoch,
    });
  }

  public close(): void {
    this.db.close();
  }
}
