import { Grievance } from '../schemas/grievance.js';

export interface FarceGateConfig {
  max_farce_ratio: number; // default: 0.15 (15%)
  min_farce_ratio: number; // default: 0.05 (5%)
}

export interface FarceAuditResult {
  passed: boolean;
  farceRatio: number;
  totalGrievances: number;
  farceGrievances: number;
  violations: Array<{
    code: 'TONAL_DEGRADATION_BREACH' | 'FARCE_CHILD_SPAWN_BREACH' | 'FARCE_INTENSITY_EXCEEDED' | 'FARCE_EPISTEMIC_COLLISION';
    grievanceId?: string;
    message: string;
  }>;
}

export class FarceGate {
  private config: FarceGateConfig;

  constructor(config?: Partial<FarceGateConfig>) {
    this.config = {
      max_farce_ratio: config?.max_farce_ratio ?? 0.15,
      min_farce_ratio: config?.min_farce_ratio ?? 0.05
    };
  }

  /**
   * Audits the Farce Ratio and structural DAG constraints across all grievances.
   */
  public auditFarceInvariants(
    grievances: Grievance[],
    clueTokens: string[] = []
  ): FarceAuditResult {
    const totalGrievances = grievances.length;
    const violations: FarceAuditResult['violations'] = [];
    const clueSet = new Set(clueTokens.map(c => c.toLowerCase()));

    if (totalGrievances === 0) {
      return {
        passed: true,
        farceRatio: 0,
        totalGrievances: 0,
        farceGrievances: 0,
        violations: []
      };
    }

    const farceGrievancesList = grievances.filter(g => g.tonal_override_eligible);
    const farceRatio = farceGrievancesList.length / totalGrievances;

    // 1. Ratio Ceiling Assertion
    if (farceRatio > this.config.max_farce_ratio) {
      violations.push({
        code: 'TONAL_DEGRADATION_BREACH',
        message: `Farce ratio ${(farceRatio * 100).toFixed(1)}% exceeds configured ceiling ${(this.config.max_farce_ratio * 100).toFixed(1)}%.`
      });
    }

    // 2. Individual Farce Invariant Verification
    for (const g of farceGrievancesList) {
      // Must not spawn escalation children (must be terminal leaf)
      if (g.spawns_on_max_escalation.length > 0) {
        violations.push({
          code: 'FARCE_CHILD_SPAWN_BREACH',
          grievanceId: g.id,
          message: `Farce grievance '${g.id}' cannot spawn escalation children (${g.spawns_on_max_escalation.join(', ')}). Must be terminal leaf.`
        });
      }

      // Max intensity must not exceed 2
      if (g.intensity_envelope.max > 2) {
        violations.push({
          code: 'FARCE_INTENSITY_EXCEEDED',
          grievanceId: g.id,
          message: `Farce grievance '${g.id}' max intensity (${g.intensity_envelope.max}) exceeds ceiling of 2.`
        });
      }

      // Farce props must not collide with mystery clue tokens
      for (const prop of g.associated_props) {
        if (clueSet.has(prop.toLowerCase())) {
          violations.push({
            code: 'FARCE_EPISTEMIC_COLLISION',
            grievanceId: g.id,
            message: `Farce prop '${prop}' in '${g.id}' collides with canonical mystery clue token.`
          });
        }
      }
    }

    return {
      passed: violations.length === 0,
      farceRatio,
      totalGrievances,
      farceGrievances: farceGrievancesList.length,
      violations
    };
  }
}

export interface SpatialNode {
  id: string; // loc_*
  connected_locations: string[];
}

export interface SpatialValidationResult {
  valid: boolean;
  errors: Array<{
    sourceLocation: string;
    targetLocation: string;
    message: string;
  }>;
}

/**
 * Verifies and enforces bidirectional spatial adjacency links.
 * Prohibits "One-Way Trap" defects at compile time (T_0 = $0.00).
 */
export function validateReciprocalSpatialTopology(
  locations: Map<string, SpatialNode>
): SpatialValidationResult {
  const errors: SpatialValidationResult['errors'] = [];

  for (const [locId, node] of locations.entries()) {
    for (const neighborId of node.connected_locations) {
      const neighbor = locations.get(neighborId);
      if (!neighbor) {
        errors.push({
          sourceLocation: locId,
          targetLocation: neighborId,
          message: `Location '${locId}' links to unknown destination '${neighborId}'.`
        });
        continue;
      }

      if (!neighbor.connected_locations.includes(locId)) {
        errors.push({
          sourceLocation: locId,
          targetLocation: neighborId,
          message: `Asymmetric spatial topology: '${locId}' connects to '${neighborId}', but '${neighborId}' lacks reciprocal return link.`
        });
      }
    }
  }

  return {
    valid: errors.length === 0,
    errors
  };
}

/**
 * Automatically establishes reciprocal links across location nodes in memory.
 */
export function wireReciprocalAdjacency(
  locA: SpatialNode,
  locB: SpatialNode
): void {
  if (!locA.connected_locations.includes(locB.id)) {
    locA.connected_locations.push(locB.id);
  }
  if (!locB.connected_locations.includes(locA.id)) {
    locB.connected_locations.push(locA.id);
  }
}

export interface EntityDictionary {
  characters: Map<string, string[]>; // slug -> aliases / lemmas
  locations: Map<string, string[]>;  // slug -> aliases / lemmas
  possessions: Map<string, string[]>; // slug -> aliases / lemmas
}

export interface SoftLinkResult {
  candidateCharacters: string[];
  candidateLocations: string[];
  candidatePossessions: string[];
  annotatedTags: string[];
}

/**
 * Scans raw Markdown text drops without mutating authorial prose ("The Body is King").
 * Identifies entity mentions and matches lemma co-occurrences.
 */
export class IngestionSoftLinker {
  private dict: EntityDictionary;

  constructor(dict: EntityDictionary) {
    this.dict = dict;
  }

  public scanProse(rawMarkdown: string): SoftLinkResult {
    const normalized = rawMarkdown.toLowerCase();
    const candidateCharacters = new Set<string>();
    const candidateLocations = new Set<string>();
    const candidatePossessions = new Set<string>();

    const scanCategory = (catalog: Map<string, string[]>, targetSet: Set<string>) => {
      for (const [slug, aliases] of catalog.entries()) {
        for (const alias of aliases) {
          const pattern = new RegExp(`\\b${alias.toLowerCase()}\\b`, 'i');
          if (pattern.test(normalized)) {
            targetSet.add(slug);
            break;
          }
        }
      }
    };

    scanCategory(this.dict.characters, candidateCharacters);
    scanCategory(this.dict.locations, candidateLocations);
    scanCategory(this.dict.possessions, candidatePossessions);

    const chars = Array.from(candidateCharacters);
    const locs = Array.from(candidateLocations);
    const items = Array.from(candidatePossessions);

    return {
      candidateCharacters: chars,
      candidateLocations: locs,
      candidatePossessions: items,
      annotatedTags: [...chars, ...locs, ...items]
    };
  }
}
