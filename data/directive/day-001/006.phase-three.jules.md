import type GearBit from '../fce/gear.bit';
import type { GearModel } from '../gear.model';
import type State from '../../99.core/state';

export interface SomaticHandProfile {
  total_arms: number;
  impairment_penalty_arms: number;
}

export interface HandValidationItem {
  id: string;
  occupies_hands: number;
}

export interface HandCapacityValidationResult {
  valid: boolean;
  available_arms: number;
  occupied_hands: number;
  error?: string;
}

/**
 * Pre-Flight Hand Capacity Invariant Solver (T_0 = $0.00).
 * Enforces: sum(held_items.occupies_hands) <= max(0, total_arms - impairment_penalty)
 */
export const solveHandCapacitySAT = (
  somatic: SomaticHandProfile,
  heldItems: HandValidationItem[],
): HandCapacityValidationResult => {
  const availableArms = Math.max(
    0,
    somatic.total_arms - somatic.impairment_penalty_arms,
  );
  const occupiedHands = heldItems.reduce(
    (sum, item) => sum + item.occupies_hands,
    0,
  );

  if (occupiedHands > availableArms) {
    return {
      valid: false,
      available_arms: availableArms,
      occupied_hands: occupiedHands,
      error: `HAND_CAPACITY_EXCEEDED: Demanded ${occupiedHands} hands but only ${availableArms} functional arm(s) available.`,
    };
  }

  return {
    valid: true,
    available_arms: availableArms,
    occupied_hands: occupiedHands,
  };
};

export const initGear = (cpy: GearModel, bal: GearBit, _ste: State) => {
  if (bal?.slv) {
    bal.slv({ idx: 'init-gear', val: 0 });
  }

};

import type { Location, Portal } from '../fce/spatial.schema';

/**
 * Calculates acoustic decay across traversed topological portals.
 * Uses integer subtraction to eliminate floating-point drift.
 */
export const calculateAcousticPropagation = (
  sourceCdb: number,
  pathPortals: Portal[],
): number => {
  const totalDamping = pathPortals.reduce(
    (sum, p) => sum + p.acoustic_dampening_cdb,
    0,
  );
  return Math.max(0, sourceCdb - totalDamping);
};

/**
 * Verifies reciprocal portal symmetry:
 * If Location A has portal P_1 targeting Location B at portal P_2,
 * Location B must exist and possess portal P_2 targeting Location A at P_1.
 */
export const validatePortalReciprocity = (
  locations: Location[],
): { valid: boolean; errors: string[] } => {
  const errors: string[] = [];
  const locationMap = new Map<string, Location>();

  for (const loc of locations) {
    locationMap.set(loc.id, loc);
  }

  for (const loc of locations) {
    for (const portal of loc.portals) {
      if (portal.target_location_id === loc.id) {
        errors.push(
          `SELF_ADJACENCY_VIOLATION: Location ${loc.id} portal ${portal.portal_id} cannot target itself.`,
        );
        continue;
      }

      const targetLoc = locationMap.get(portal.target_location_id);
      if (!targetLoc) {
        errors.push(
          `DANGLING_PORTAL_TARGET: Location ${loc.id} targets missing location ${portal.target_location_id}.`,
        );
        continue;
      }

      const reciprocalPortal = targetLoc.portals.find(
        (p) => p.portal_id === portal.target_portal_id,
      );
      if (!reciprocalPortal) {
        errors.push(
          `ASYMMETRIC_PORTAL_TRAP: Location ${loc.id} connects to ${portal.target_location_id} via ${portal.portal_id}->${portal.target_portal_id}, but reciprocal portal does not exist.`,
        );
        continue;
      }

      if (reciprocalPortal.target_location_id !== loc.id) {
        errors.push(
          `PORTAL_MISDIRECTION: Reciprocal portal ${reciprocalPortal.portal_id} in ${targetLoc.id} targets ${reciprocalPortal.target_location_id} instead of ${loc.id}.`,
        );
      }
    }
  }

  return {
    valid: errors.length === 0,
    errors,
  };
};
