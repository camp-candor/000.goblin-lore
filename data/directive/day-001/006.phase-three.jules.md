```typescript
import { GrievanceMachineRail } from "../schemas/grievance";

export interface DagAcyclicityResult {
  isAcyclic: boolean;
  cycleNodes: string[];
  topologicalDepths: Map<string, number>;
  maxDepth: number;
  reason?: string;
}

enum DfsColor {
  WHITE = 0, // Unvisited
  GRAY = 1,  // Currently in recursion stack
  BLACK = 2, // Fully processed
}

/**
 * Executes Three-Color Depth-First Search (DFS) over grievance escalation links:
 * spawns_on_max_escalation
 */
export function verifyGrievanceDag(grievances: GrievanceMachineRail[]): DagAcyclicityResult {
  const adj = new Map<string, string[]>();
  const allNodes = new Set<string>();

  for (const g of grievances) {
    allNodes.add(g.id);
    adj.set(g.id, g.spawns_on_max_escalation);
  }

  const colors = new Map<string, DfsColor>();
  for (const node of allNodes) {
    colors.set(node, DfsColor.WHITE);
  }

  const cycleNodes: string[] = [];
  const topologicalDepths = new Map<string, number>();

  function dfs(node: string, currentStack: string[]): boolean {
    colors.set(node, DfsColor.GRAY);
    currentStack.push(node);

    const neighbors = adj.get(node) || [];
    for (const neighbor of neighbors) {
      if (!allNodes.has(neighbor)) {
        // Target node missing from corpus
        continue;
      }
      const neighborColor = colors.get(neighbor);
      if (neighborColor === DfsColor.GRAY) {
        // Back-edge found -> Cycle detected
        cycleNodes.push(...currentStack.slice(currentStack.indexOf(neighbor)), neighbor);
        return false;
      }
      if (neighborColor === DfsColor.WHITE) {
        if (!dfs(neighbor, currentStack)) {
          return false;
        }
      }
    }

    currentStack.pop();
    colors.set(node, DfsColor.BLACK);
    return true;
  }

  for (const node of allNodes) {
    if (colors.get(node) === DfsColor.WHITE) {
      if (!dfs(node, [])) {
        return {
          isAcyclic: false,
          cycleNodes,
          topologicalDepths: new Map(),
          maxDepth: 0,
          reason: `CIRCULAR_ESCALATION_DETECTED: Cycle detected along path: [${cycleNodes.join(" -> ")}]`,
        };
      }
    }
  }

  // Calculate topological depth delta for each node
  const memoDepth = new Map<string, number>();

  function calculateDepth(node: string): number {
    if (memoDepth.has(node)) return memoDepth.get(node)!;
    const neighbors = adj.get(node) || [];
    if (neighbors.length === 0) {
      memoDepth.set(node, 0);
      return 0;
    }
    let maxChildDepth = 0;
    for (const neighbor of neighbors) {
      if (allNodes.has(neighbor)) {
        maxChildDepth = Math.max(maxChildDepth, 1 + calculateDepth(neighbor));
      }
    }
    memoDepth.set(node, maxChildDepth);
    return maxChildDepth;
  }

  let maxDepth = 0;
  for (const node of allNodes) {
    const depth = calculateDepth(node);
    topologicalDepths.set(node, depth);
    maxDepth = Math.max(maxDepth, depth);
  }

  return {
    isAcyclic: true,
    cycleNodes: [],
    topologicalDepths,
    maxDepth,
  };
}
```

```typescript
import { GrievanceMachineRail } from "../schemas/grievance";

export interface FarceGateResult {
  valid: boolean;
  totalGrievances: number;
  farceGrievancesCount: number;
  farceRatioBps: number; // Integer basis points (1500 = 15%)
  clueCollisions: Array<{ grievanceId: string; propId: string }>;
  reason?: string;
}

/**
 * Enforces R_farce <= 0.15 (1500 bps) and prop clue-disjointness at Stage T0 = $0.00
 */
export function auditFarceRatio(
  grievances: GrievanceMachineRail[],
  permittedClues: string[] = []
): FarceGateResult {
  const totalGrievances = grievances.length;
  if (totalGrievances === 0) {
    return {
      valid: true,
      totalGrievances: 0,
      farceGrievancesCount: 0,
      farceRatioBps: 0,
      clueCollisions: [],
    };
  }

  const clueSet = new Set(permittedClues.map((c) => c.toLowerCase()));
  const farceGrievances = grievances.filter((g) => g.tonal_override_eligible);
  const farceGrievancesCount = farceGrievances.length;

  // Integer basis-point arithmetic: (count * 10000) / total
  const farceRatioBps = Math.round((farceGrievancesCount * 10000) / totalGrievances);

  const clueCollisions: Array<{ grievanceId: string; propId: string }> = [];

  for (const farce of farceGrievances) {
    for (const prop of farce.bound_props) {
      if (clueSet.has(prop.toLowerCase())) {
        clueCollisions.push({ grievanceId: farce.id, propId: prop });
      }
    }
  }

  if (farceRatioBps > 1500) {
    return {
      valid: false,
      totalGrievances,
      farceGrievancesCount,
      farceRatioBps,
      clueCollisions,
      reason: `FARCE_RATIO_BREACH: Farce ratio of ${farceRatioBps / 100}% exceeds 15.00% ceiling.`,
    };
  }

  if (clueCollisions.length > 0) {
    return {
      valid: false,
      totalGrievances,
      farceGrievancesCount,
      farceRatioBps,
      clueCollisions,
      reason: `FARCE_CLUE_COLLISION: Farce disputes bind secret mystery clues: [${clueCollisions.map((c) => c.propId).join(", ")}]`,
    };
  }

  return {
    valid: true,
    totalGrievances,
    farceGrievancesCount,
    farceRatioBps,
    clueCollisions: [],
  };
}
```

```sql
  version INTEGER NOT NULL DEFAULT 1,
  author TEXT NOT NULL,
  created_at_ms INTEGER NOT NULL,
  last_edited_at_ms INTEGER NOT NULL
);

CREATE TABLE IF NOT EXISTS scenes (
  id TEXT PRIMARY KEY,
  episode_num INTEGER NOT NULL,
  scene_num INTEGER NOT NULL,
  text TEXT NOT NULL,
  status TEXT CHECK(status IN ('clean', 'dirty', 'reviewed')) NOT NULL DEFAULT 'clean',
  is_dirty INTEGER NOT NULL DEFAULT 0
);

CREATE TABLE IF NOT EXISTS scene_fact_dependencies (
  scene_id TEXT NOT NULL,
  fact_id TEXT NOT NULL,
  pinned_version INTEGER NOT NULL,
  created_at_ms INTEGER NOT NULL,
  PRIMARY KEY (scene_id, fact_id),
  FOREIGN KEY (scene_id) REFERENCES scenes(id) ON DELETE CASCADE,
  FOREIGN KEY (fact_id) REFERENCES facts(id) ON DELETE CASCADE
);

-- Monotonic Version Bump Trigger: Cascades dirty flag on NEW.version > OLD.version
-- Setting standing to FLAGGED_FOR_REVIEW without version bump triggers ZERO dirty cascades
CREATE TRIGGER IF NOT EXISTS trg_fact_version_bump
AFTER UPDATE ON facts
FOR EACH ROW
WHEN NEW.version > OLD.version OR (NEW.standing = 'DEPRECATED_ARCHIVED' AND OLD.standing != 'DEPRECATED_ARCHIVED')
BEGIN
  UPDATE scenes
  SET status = 'dirty', is_dirty = 1
  WHERE id IN (
    SELECT scene_id FROM scene_fact_dependencies
    WHERE fact_id = NEW.id AND (pinned_version < NEW.version OR NEW.standing = 'DEPRECATED_ARCHIVED')
  );
END;

-- Passive Centrality Telemetry View (In-Degree Narrative Blast Radius)
CREATE VIEW IF NOT EXISTS v_fact_centrality_telemetry AS
SELECT 
  f.id AS fact_id,
  f.claim_text,
  f.standing,
  f.version,
  COUNT(d.scene_id) AS in_degree
FROM facts f
LEFT JOIN scene_fact_dependencies d ON f.id = d.fact_id
GROUP BY f.id;
```

##### Deliverable 5: Local SQLite Relational Core Engine

**File:** `packages/001.lore/src/db/localRelationalCore.ts`

```typescript
import Database from "better-sqlite3";
import * as fs from "node:fs";
import * as path from "node:path";

export interface FactRecord {
  id: string;
  claim_text: string;
  scope: "CHARACTER" | "LOCATION" | "LOGISTIC" | "LAW";
  authority_level?: "AXIOM" | "CANON" | "EVENT" | "RUNTIME_STATE" | "PROJECTION" | "SPECULATION";
  standing?: "ACTIVE_ENDORSED" | "FLAGGED_FOR_REVIEW" | "DEPRECATED_ARCHIVED";
  version: number;
  author: string;
  created_at_ms: number;
  last_edited_at_ms: number;
}

export interface DirtySceneTrace {
  scene_id: string;
  episode_num: number;
  scene_num: number;
  fact_id: string;
  pinned_version: number;
  current_version: number;
}

export class LocalRelationalCore {
  public db: Database.Database;

  constructor(dbPath: string = ":memory:") {
    this.db = new Database(dbPath);
    this.db.pragma("foreign_keys = ON");
    this.runMigrations();
  }

  private runMigrations(): void {
    const migrationPath = path.resolve(__dirname, "../../migrations/0001_phase0_flat_schema.sql");
    const ddl = fs.readFileSync(migrationPath, "utf8");
    this.db.exec(ddl);
  }

  public insertFact(fact: FactRecord): void {
    const stmt = this.db.prepare(`
      INSERT INTO facts (
        id, claim_text, scope, authority_level, standing, version, author, created_at_ms, last_edited_at_ms
      ) VALUES (
        @id, @claim_text, @scope, 
        COALESCE(@authority_level, 'CANON'), 
        COALESCE(@standing, 'ACTIVE_ENDORSED'), 
        @version, @author, @created_at_ms, @last_edited_at_ms
      )
    `);
    stmt.run(fact);
  }

  public bumpFactVersion(factId: string, newVersion: number): void {
    const stmt = this.db.prepare(`
      UPDATE facts
      SET version = ?, last_edited_at_ms = ?
      WHERE id = ?
    `);
    stmt.run(newVersion, Date.now(), factId);
  }

  public setFactStanding(factId: string, standing: "ACTIVE_ENDORSED" | "FLAGGED_FOR_REVIEW" | "DEPRECATED_ARCHIVED"): void {
    const stmt = this.db.prepare(`
      UPDATE facts
      SET standing = ?, last_edited_at_ms = ?
      WHERE id = ?
    `);
    stmt.run(standing, Date.now(), factId);
  }

  public registerScene(
    scene: { id: string; episode_num: number; scene_num: number; text: string },
    dependencies: Array<{ factId: string; pinnedVersion: number }>
  ): void {
    const insertScene = this.db.prepare(`
      INSERT INTO scenes (id, episode_num, scene_num, text, status, is_dirty)
      VALUES (?, ?, ?, ?, 'clean', 0)
    `);
    const insertDep = this.db.prepare(`
      INSERT INTO scene_fact_dependencies (scene_id, fact_id, pinned_version, created_at_ms)
      VALUES (?, ?, ?, ?)
    `);

    const tx = this.db.transaction(() => {
      insertScene.run(scene.id, scene.episode_num, scene.scene_num, scene.text);
      for (const dep of dependencies) {
        insertDep.run(scene.id, dep.factId, dep.pinnedVersion, Date.now());
      }
    });
    tx();
  }

  /**
   * Recursive Common Table Expression (CTE) tracing blast radius for why-dirty CLI
   */
  public queryWhyDirty(factId: string): DirtySceneTrace[] {
    const query = this.db.prepare(`
      WITH RECURSIVE dirty_cte AS (
        SELECT 
          s.id AS scene_id,
          s.episode_num,
          s.scene_num,
          d.fact_id,
          d.pinned_version,
          f.version AS current_version
        FROM scenes s
        JOIN scene_fact_dependencies d ON s.id = d.scene_id
        JOIN facts f ON d.fact_id = f.id
        WHERE f.id = ? AND (d.pinned_version < f.version OR f.standing = 'DEPRECATED_ARCHIVED')
      )
      SELECT * FROM dirty_cte;
    `);

    return query.all(factId) as DirtySceneTrace[];
  }

  public close(): void {
    this.db.close();
  }
}
```
