// Helper: Recursively discover all Markdown files (.md) in a directory
function collectMarkdownFiles(dirPath: string): string[] {
  const results: string[] = [];
  if (!fs.existsSync(dirPath)) return results;

  const entries = fs.readdirSync(dirPath, { withFileTypes: true });
  for (const entry of entries) {
    const fullPath = path.join(dirPath, entry.name);
    if (entry.isDirectory()) {
      results.push(...collectMarkdownFiles(fullPath));
    } else if (entry.isFile() && entry.name.endsWith('.md')) {

    }
  }

// Helper: Lightweight Byte-0 Frontmatter Extraction (Pure ASCII, no external grey-matter dependency)
interface ParsedDocument {
  filePath: string;
  frontmatterRaw: string;
  bodyRaw: string;
  data: Record<string, any>;
}

function parseByte0Markdown(
  filePath: string,
  failures: CompileGateFailure[]
): ParsedDocument | null {
  const buffer = fs.readFileSync(filePath);

  // 1. Zero-byte poison sentinel & empty hash check
  if (buffer.length === 0) {
    failures.push({
      gate: 'GATE_0_PREFLIGHT',
      targetFile: filePath,
      message: 'Zero-byte poison file detected.',
    });
    return null;
  }

  const hash = crypto.createHash('sha256').update(buffer).digest('hex');
  if (hash === EMPTY_STREAM_SHA256) {
    failures.push({
      gate: 'GATE_0_PREFLIGHT',
      targetFile: filePath,
      message: 'Empty-stream hash guard triggered.',
    });
    return null;
  }

  // 2. Pure 7-bit ASCII compliance scan
  for (let i = 0; i < buffer.length; i++) {
    if (buffer[i] > 127) {
      failures.push({
        gate: 'GATE_0_PREFLIGHT',
        targetFile: filePath,
        message: `Non-ASCII byte detected at offset ${i} (${buffer[i]}). Multi-byte emojis and glyphs strictly banned.`,
      });
      return null;
    }
  }

  const content = buffer.toString('ascii');

  // 3. Byte-0 delimiter check (--- strictly at index 0)
  if (!content.startsWith('---\n') && !content.startsWith('---\r\n')) {
    failures.push({
      gate: 'GATE_1_SYNTAX_BOUNDS',
      targetFile: filePath,
      message: 'Byte-0 Law breached: File must begin with standard --- delimiter at byte 0 (no BOM, whitespace, or newlines).',
    });
    return null;
  }

  const delimiterIndex = content.indexOf('\n---', 3);
  if (delimiterIndex === -1) {
    failures.push({
      gate: 'GATE_1_SYNTAX_BOUNDS',
      targetFile: filePath,
      message: 'Unclosed YAML frontmatter block. Closing --- delimiter missing.',
    });
    return null;
  }

  const frontmatterRaw = content.slice(4, delimiterIndex).trim();
  const bodyRaw = content.slice(delimiterIndex + 4).trim();

  // Parse simple key-value YAML frontmatter deterministically
  const data: Record<string, any> = {};
  const lines = frontmatterRaw.split(/\r?\n/);

  for (const line of lines) {
    const trimmed = line.trim();
    if (!trimmed || trimmed.startsWith('#')) continue;

    const colonIdx = trimmed.indexOf(':');
    if (colonIdx === -1) continue;

    const key = trimmed.slice(0, colonIdx).trim();
    let valStr = trimmed.slice(colonIdx + 1).trim();

    // Strip wrapping quotes
    if (
      (valStr.startsWith('"') && valStr.endsWith('"')) ||
      (valStr.startsWith("'") && valStr.endsWith("'"))
    ) {
      valStr = valStr.slice(1, -1);
    }

    // Parse array literals [a, b, c]
    if (valStr.startsWith('[') && valStr.endsWith(']')) {
      const items = valStr
        .slice(1, -1)
        .split(',')
        .map((s) => s.trim().replace(/^['"]|['"]$/g, ''))
        .filter(Boolean);
      data[key] = items;
    } else if (valStr === 'true') {
      data[key] = true;
    } else if (valStr === 'false') {
      data[key] = false;
    } else if (!isNaN(Number(valStr)) && valStr !== '') {
      data[key] = Number(valStr);
    } else {
      data[key] = valStr;
    }
  }

  return { filePath, frontmatterRaw, bodyRaw, data };
}

// Helper: RFC 8785 Canonical JSON Serialization (JCS)
function canonicalizeJson(obj: any): string {
  if (obj === null || typeof obj !== 'object') {
    return JSON.stringify(obj);
  }

  if (Array.isArray(obj)) {
    return '[' + obj.map((item) => canonicalizeJson(item)).join(',') + ']';
  }

  const sortedKeys = Object.keys(obj).sort((a, b) => {
    // Compare UTF-16 code units lexicographically
    const len = Math.min(a.length, b.length);
    for (let i = 0; i < len; i++) {
      const codeA = a.charCodeAt(i);
      const codeB = b.charCodeAt(i);
      if (codeA !== codeB) return codeA - codeB;
    }
    return a.length - b.length;
  });

  const members = sortedKeys.map(
    (key) => JSON.stringify(key) + ':' + canonicalizeJson(obj[key])
  );
  return '{' + members.join(',') + '}';
}

              gate: 'GATE_1_SYNTAX_BOUNDS',
              targetFile: file,
              field: k,
              message: `Basis point scalar out of bounds: ${v} (valid: 0-10,000 bp).`,

          }
        }
      }
    }

      const iMin = Number(data.intensity_min);
      const iMax = Number(data.intensity_max);
      if (iMin < 1 || iMax > 5 || iMin > iMax) {
        failures.push({
          gate: 'GATE_1_SYNTAX_BOUNDS',
          targetFile: file,
          field: 'intensity',
          message: `Intensity bounds violated: [${iMin}, ${iMax}] must satisfy 1 <= min <= max <= 5.`,

      }

          gate: 'GATE_1_SYNTAX_BOUNDS',
          targetFile: file,
          field: 'held_items',
          message: `Somatic SAT breach: Actor holds ${heldItems} items with only ${arms} functional arms.`,

    valid,
    timestamp: new Date().toISOString().replace(/:/g, '-'),

        ? undefined
        : { code: 'PASS_1_FAILED', message: cpy.lastError || 'Pass 1 bounds breach', failures },
    });

  const failures: CompileGateFailure[] = [];

  const files = collectMarkdownFiles(sourceDir);
  const entities: Map<string, { type: string; data: Record<string, any>; file: string }> = new Map();

  // Ingest catalogs
  for (const file of files) {
    const parsed = parseByte0Markdown(file, failures);
    if (!parsed) continue;

    const id = parsed.data.id || path.basename(file, '.md');
    const type = parsed.data.type || path.basename(path.dirname(file));
    entities.set(id, { type, data: parsed.data, file });
  }

  // 1. Foreign-key closure sweep
  for (const [id, entity] of entities.entries()) {
    const d = entity.data;

    // Check character participants
    if (Array.isArray(d.participants)) {
      for (const p of d.participants) {
        if (!entities.has(p)) {
          failures.push({
            gate: 'GATE_2_RELATIONAL_DAG',
            targetFile: entity.file,
            entityId: id,
            field: 'participants',
            message: `Dangling pointer: Participant [${p}] does not exist in corpus.`,
          });
        }
      }
    }

    // Check inventory held items
    if (Array.isArray(d.held_items)) {
      for (const item of d.held_items) {
        if (!entities.has(item)) {
          failures.push({
            gate: 'GATE_2_RELATIONAL_DAG',
            targetFile: entity.file,
            entityId: id,
            field: 'held_items',
            message: `Dangling pointer: Held possession [${item}] does not exist in corpus.`,
          });
        }
      }
    }

    // Check adjacent location portals
    if (Array.isArray(d.adjacent_to)) {
      for (const adj of d.adjacent_to) {
        if (!entities.has(adj)) {
          failures.push({
            gate: 'GATE_2_RELATIONAL_DAG',
            targetFile: entity.file,
            entityId: id,
            field: 'adjacent_to',
            message: `Dangling portal: Adjacent location [${adj}] does not exist.`,
          });
        } else {
          // Bidirectional portal reciprocity check
          const targetLoc = entities.get(adj);
          if (
            targetLoc &&
            Array.isArray(targetLoc.data.adjacent_to) &&
            !targetLoc.data.adjacent_to.includes(id) &&
            !d.one_way_egress
          ) {
            failures.push({
              gate: 'GATE_2_RELATIONAL_DAG',
              targetFile: entity.file,
              entityId: id,
              field: 'adjacent_to',
              message: `Reciprocity breach: [${id}] points to [${adj}], but [${adj}] lacks return link to [${id}].`,
            });
          }
        }
      }
    }

    // Clue-Lemma Disjointness Theorem: clues intersect forbidden resolution lemmas
    if (Array.isArray(d.clue_tokens) && Array.isArray(d.forbidden_lemmas)) {
      const clueSet = new Set(d.clue_tokens);
      for (const lemma of d.forbidden_lemmas) {
        if (clueSet.has(lemma)) {
          failures.push({
            gate: 'GATE_2_RELATIONAL_DAG',
            targetFile: entity.file,
            entityId: id,
            field: 'clue_tokens',
            message: `Clue-Lemma Disjointness breach: Token [${lemma}] exists in both clues and forbidden lemmas.`,
          });
        }
      }
    }
  }

  // 2. Three-Color DFS Acyclicity Check on Grievance Escalation Chains (spawns_on_max_escalation)
  const colorMap = new Map<string, number>(); // 0 = WHITE, 1 = GRAY, 2 = BLACK
  for (const id of entities.keys()) {
    colorMap.set(id, 0);
  }

  function dfsVisit(nodeId: string, trace: string[]): boolean {
    colorMap.set(nodeId, 1); // GRAY (visiting)
    trace.push(nodeId);

    const node = entities.get(nodeId);
    if (node && Array.isArray(node.data.spawns_on_max_escalation)) {
      for (const childId of node.data.spawns_on_max_escalation) {
        if (!entities.has(childId)) {
          failures.push({
            gate: 'GATE_2_RELATIONAL_DAG',
            targetFile: node.file,
            entityId: nodeId,
            field: 'spawns_on_max_escalation',
            message: `Dangling grievance child pointer: [${childId}] does not exist.`,
          });
          continue;
        }

        const childColor = colorMap.get(childId) ?? 0;
        if (childColor === 1) {
          // Circular back-edge detected
          const cyclePath = [...trace, childId].join(' -> ');
          failures.push({
            gate: 'GATE_2_RELATIONAL_DAG',
            targetFile: node.file,
            entityId: nodeId,
            field: 'spawns_on_max_escalation',
            message: `Circular narrative causal cycle detected: ${cyclePath}`,
          });
          return false;
        } else if (childColor === 0) {
          if (!dfsVisit(childId, trace)) return false;
        }
      }
    }

    trace.pop();
    colorMap.set(nodeId, 2); // BLACK (verified)
    return true;
  }

  for (const id of entities.keys()) {
    if (colorMap.get(id) === 0) {
      dfsVisit(id, []);
    }
  }

  const valid = failures.length === 0;
  const report: CompileValidationReport = {
    valid,
    timestamp: new Date().toISOString().replace(/:/g, '-'),
    seriesSlug: options.seriesSlug || cpy.activeSeriesSlug || '001.goblin-lore',
    evaluatedFiles: files.length,
    failures,
    warnings: [],
  };

  cpy.validationReport = report;
  cpy.status = valid ? 'IDLE' : 'FAILED';
  if (!valid) cpy.lastError = failures[0]?.message;

  if (bal.slv) {
    bal.slv({
      cmpBit: { idx: 'pass2-relational-sweep', val: valid ? 0 : -1 },
      resBit: { validationReport: report, cleanExit: valid },
      errBit: valid
        ? undefined
        : { code: 'PASS_2_FAILED', message: cpy.lastError || 'Pass 2 relational breach', failures },
    });
  }

  return cpy;
};

export const pass3Seal = (
  cpy: CompileModel,
  bal: CompileBit,
  ste?: State
): CompileModel => {
  cpy.status = 'PASS_3_SEALING';
  const options: CompileExecutionOptions = bal.dat || {};
  const sourceDir = options.sourceDir || cpy.sourceDirectory || 'corpus';
  const outputDir = options.outputDir || cpy.outputDirectory || 'compiled';
  const seriesSlug = options.seriesSlug || cpy.activeSeriesSlug || '001.goblin-lore';
  const targetSeriesOut = path.join(outputDir, seriesSlug);

  const files = collectMarkdownFiles(sourceDir);
  const entitiesPayload: Record<string, any> = {};
  const entityHashes: Record<string, string> = {};
  const summary: CompileCatalogSummary = {
    characterCount: 0,
    locationCount: 0,
    possessionCount: 0,
    grievanceCount: 0,
    totalEntities: 0,
  };

  for (const file of files) {
    const parsed = parseByte0Markdown(file, []);
    if (!parsed) continue;

    const id = parsed.data.id || path.basename(file, '.md');
    const type = parsed.data.type || path.basename(path.dirname(file));

    // Conform string values to Unicode NFC
    const normalizedData: Record<string, any> = {};
    for (const [k, v] of Object.entries(parsed.data)) {
      normalizedData[k] = typeof v === 'string' ? v.normalize('NFC') : v;
    }
    normalizedData._body = parsed.bodyRaw.normalize('NFC').replace(/\r\n/g, '\n');

    // Compute discrete per-entity hash
    const entityCanonical = canonicalizeJson(normalizedData);
    const entityHash = crypto.createHash('sha256').update(entityCanonical).digest('hex');
    entityHashes[id] = entityHash;

    entitiesPayload[id] = normalizedData;
    summary.totalEntities++;

    if (type.includes('char')) summary.characterCount++;
    else if (type.includes('loc')) summary.locationCount++;
    else if (type.includes('pos') || type.includes('item')) summary.possessionCount++;
    else if (type.includes('grv') || type.includes('grievance')) summary.grievanceCount++;
  }

  // Canonicalize full state without dynamic execution timestamps
  const finalState = {
    seriesSlug,
    summary,
    entityHashes,
    entities: entitiesPayload,
  };

  const canonicalJson = canonicalizeJson(finalState);
  const rootMerkleHash = crypto.createHash('sha256').update(canonicalJson).digest('hex');

  // Inject root seal and format with POSIX line endings
  const sealedPayload = {
    _seal: {
      rootMerkleHash,
      sealedByteLength: Buffer.byteLength(canonicalJson, 'utf8'),
      algorithm: 'SHA-256 (RFC 8785 JCS)',
    },
    ...finalState,
  };

  const formattedOutput = JSON.stringify(sealedPayload, null, 2).replace(/\r\n/g, '\n') + '\n';

  fs.mkdirSync(targetSeriesOut, { recursive: true });
  const canonicalPath = path.join(targetSeriesOut, 'bible-state.json');

  // Windows-safe timestamp: YYYY-MM-DDTHH-mm-ssZ
  const safeTimestamp = new Date().toISOString().replace(/:/g, '-');
  const archivePath = path.join(targetSeriesOut, `bible-state_${safeTimestamp}.json`);

  // Atomic POSIX sibling inode swap
  const tmpPath = path.join(targetSeriesOut, `.bible-state.json.tmp_${crypto.randomUUID()}`);
  fs.writeFileSync(tmpPath, formattedOutput, 'utf8');
  fs.renameSync(tmpPath, canonicalPath);

  if (!options.skipArchive) {
    fs.writeFileSync(archivePath, formattedOutput, 'utf8');
  }

  cpy.rootMerkleHash = rootMerkleHash;
  cpy.lastSealedAt = safeTimestamp;
  cpy.catalogSummary = summary;
  cpy.status = 'SEALED';

  // Recursive deep-freeze in-memory representation
  Object.freeze(sealedPayload);

  if (bal.slv) {
    bal.slv({
      cmpBit: { idx: 'pass3-deterministic-seal', val: 0 },
      resBit: {
        seriesSlug,
        rootMerkleHash,
        entityHashes,
        canonicalPath,
        archivePath,
        catalogSummary: summary,
        cleanExit: true,
      },
    });
  }

  return cpy;
};

export const runCompile = async (
  cpy: CompileModel,
  bal: CompileBit,
  ste?: State
): Promise<CompileModel> => {
  const startTime = Date.now();
  await emitConsole(ste, '>> [COMPILE:INIT] Commencing complete legislative compilation gauntlet...');

  // 1. Pass 1: Bounds and Shapes
  pass1Evaluate(cpy, bal, ste);
  if (cpy.status === 'FAILED') {
    await emitConsole(ste, `:: [COMPILE:FAIL] Pass 1 bounds breach: ${cpy.lastError}`);
    return cpy;
  }
  await emitConsole(ste, '>> [PASS_1:BOUNDS] Evaluated Byte-0 laws and scalar bounds [OK]');

  // 2. Pass 2: Relational Topology & DAG Traversal
  pass2Traverse(cpy, bal, ste);
  if (cpy.status === 'FAILED') {
    await emitConsole(ste, `:: [COMPILE:FAIL] Pass 2 relational topology breach: ${cpy.lastError}`);
    return cpy;
  }
  await emitConsole(ste, '>> [PASS_2:TOPOLOGY] Traversed relational foreign keys and grievance DAG [OK]');

  // 3. Pass 3: Deterministic Seal & Persistence
  pass3Seal(cpy, bal, ste);
  const elapsed = Date.now() - startTime;
  await emitConsole(
    ste,
    `>> [PASS_3:SEAL] Sealed universal Merkle root ${cpy.rootMerkleHash} in ${elapsed}ms [SEALED]`
  );

  return cpy;
};

export const gate0Verify = async (
  cpy: CompileModel,
  bal: CompileBit,
  ste?: State
): Promise<CompileModel> => {
  await emitConsole(ste, '>> [GATE_0:AUDIT] Running Gate 0 verification gauntlet...');

  // Level 0: Structural Perimeter Audit
  try {
    const dirtyHarness = execSync('git status -s apps/995.library/995.library/').toString().trim();
    if (dirtyHarness) {
      throw new Error(`Immutable runner boundary breached in apps/995.library/995.library/: ${dirtyHarness}`);
    }

    const strayFiles = execSync('find packages/300.engine/21.compile.unit/ -name "*.js" -o -name "*.d.ts"')
      .toString()
      .trim();
    if (strayFiles) {
      throw new Error(`Stray build artifacts staged directly in source tree: ${strayFiles}`);
    }
  } catch (err: any) {
    cpy.status = 'FAILED';
    cpy.lastError = err.message;
    await emitConsole(ste, `:: [GATE_0:FAIL] Level 0 perimeter violation: ${err.message}`);
    if (bal.slv) {
      bal.slv({
        cmpBit: { idx: 'gate0-verify', val: -1 },
        errBit: { code: 'GATE_0_LEVEL_0_FAILED', message: err.message },
      });
    }
    return cpy;
  }

  // Level 1: Static TypeScript Project Reference Check
  try {
    execSync('npm run check:types', { stdio: 'pipe' });
  } catch (err: any) {
    cpy.status = 'FAILED';
    cpy.lastError = 'TypeScript project references compile error (tsc -b failed)';
    await emitConsole(ste, `:: [GATE_0:FAIL] Level 1 type check failed.`);
    if (bal.slv) {
      bal.slv({
        cmpBit: { idx: 'gate0-verify', val: -1 },
        errBit: { code: 'GATE_0_LEVEL_1_FAILED', message: cpy.lastError },
      });
    }
    return cpy;
  }

  // Level 2: Three-Pass Compilation Execution
  await runCompile(cpy, bal, ste);
  if (cpy.status === 'FAILED') {
    return cpy;
  }

  // Level 3: Zero-Drift Tripwire Check
  if (bal.dat?.verifyDriftTripwire) {
    try {
      execSync('git diff --exit-code compiled/', { stdio: 'pipe' });
      await emitConsole(ste, '>> [GATE_0:TRIPWIRE] Zero-drift tripwire clean: zero uncommitted drift [PASS]');
    } catch {
      cpy.status = 'FAILED';
      cpy.lastError = 'Zero-drift tripwire tripped: Uncompiled changes or desynchronized JSON in compiled/';
      await emitConsole(ste, `:: [GATE_0:FAIL] ${cpy.lastError}`);
      if (bal.slv) {
        bal.slv({
          cmpBit: { idx: 'gate0-verify', val: -1 },
          errBit: { code: 'GATE_0_DRIFT_DETECTED', message: cpy.lastError },
        });
      }
      return cpy;
    }
  }

  await emitConsole(ste, '>> [GATE_0:SUCCESS] Gate 0 verification gauntlet completed clean [EXIT 0]');
  if (bal.slv) {
    bal.slv({
      cmpBit: { idx: 'gate0-verify', val: 0 },
      resBit: { cleanExit: true, rootMerkleHash: cpy.rootMerkleHash },
    });
  }

  return cpy;
};
