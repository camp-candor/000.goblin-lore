import * as fs from 'node:fs';
import * as path from 'node:path';
import * as yaml from 'js-yaml';
import { CharacterSchema, PossessionSchema, LocationSchema, RefusalVector } from '../schemas/entities.js';

export interface DualRailVoiceStubs {
  wantsSurface: string;
  wantsActual: string;
  blindSpot: string;
  voiceNotes: string;
  refusalVectors?: RefusalVector[];
}

/**
 * Enforces the Byte-0 Frontmatter Law:
 * 1. Output starts strictly at byte 0 with "---" (0x2D 0x2D 0x2D 0x0A).
 * 2. Prohibits UTF-8 BOM, leading whitespace, and carriage returns (\r).
 * 3. Formats the human voice rail under standardized markdown section stubs.
 */
export function buildByteZeroDossier(frontmatter: object, voice: DualRailVoiceStubs): string {
  const yamlContent = yaml.dump(frontmatter, {
    indent: 2,
    lineWidth: -1,
    noRefs: true,
    sortKeys: true
  }).trim();

  let refusalSection = '';
  if (voice.refusalVectors && voice.refusalVectors.length > 0) {
    refusalSection = `\n## Refusal Vectors\n` + voice.refusalVectors.map(rv => (
      `* **Trigger:** ${rv.trigger_condition}\n` +
      `  * **Mandatory Refusal:** ${rv.mandatory_refusal}\n` +
      `  * **Unavoidable Cost:** ${rv.unavoidable_cost}`
    )).join('\n');
  }

  const voiceRail = [
    `## Wants (surface)`,
    voice.wantsSurface.trim(),
    ``,
    `## Wants (actual)`,
    voice.wantsActual.trim(),
    ``,
    `## Blind spot`,
    voice.blindSpot.trim(),
    ``,
    `## Voice notes`,
    voice.voiceNotes.trim(),
    refusalSection
  ].join('\n').trim();

  // POSIX LF normalized, byte-exact opening delimiter
  const dossier = `---\n${yamlContent}\n---\n\n${voiceRail}\n`;
  return dossier.replace(/\r\n/g, '\n');
}

/**
 * Validates that an in-memory buffer or file adheres to the Byte-0 Frontmatter Law.
 */
export function verifyByteZeroIntegrity(content: Buffer | string): { valid: boolean; error?: string } {
  const buf = Buffer.isBuffer(content) ? content : Buffer.from(content, 'utf8');

  // Check for UTF-8 BOM (0xEF, 0xBB, 0xBF)
  if (buf.length >= 3 && buf[0] === 0xEF && buf[1] === 0xBB && buf[2] === 0xBF) {
    return { valid: false, error: 'BYTE_0_VIOLATION: UTF-8 BOM detected at byte offset 0.' };
  }

  // Check for --- delimiter at index 0 (0x2D, 0x2D, 0x2D)
  if (buf.length < 3 || buf[0] !== 0x2D || buf[1] !== 0x2D || buf[2] !== 0x2D) {
    return { valid: false, error: 'BYTE_0_VIOLATION: File does not start with "---" at byte index 0.' };
  }

  // Ensure byte 3 is LF (0x0A) or CR+LF (0x0D 0x0A)
  if (buf[3] !== 0x0A && !(buf[3] === 0x0D && buf[4] === 0x0A)) {
    return { valid: false, error: 'BYTE_0_VIOLATION: Delimiter not followed immediately by newline.' };
  }

  return { valid: true };
}

/**
 * Persists an entity dossier to disk atomically using sibling tmp files.
 */
export function commitDossierAtomically(targetPath: string, fileContent: string): void {
  const integrity = verifyByteZeroIntegrity(fileContent);
  if (!integrity.valid) {
    throw new Error(integrity.error);
  }

  const dir = path.dirname(targetPath);
  fs.mkdirSync(dir, { recursive: true });
  
  const tmpPath = path.join(dir, `.tmp_${path.basename(targetPath)}_${Date.now()}`);
  fs.writeFileSync(tmpPath, fileContent, { encoding: 'utf8' });
  fs.renameSync(tmpPath, targetPath);
}

import { Character, Possession } from '../schemas/entities.js';

export interface SatCheckResult {
  passed: boolean;
  code?: 'SOMATIC_SAT_BREACH' | 'LOGISTICAL_OVERALLOCATION' | 'MOTOR_LIMITATION_MISMATCH';
  message?: string;
  armsDemanded?: number;
  armsAvailable?: number;
}

/**
 * Pre-Flight Invariant SAT Solver ($T_0 = $0.00 compute cost)
 * Evaluates hand saturation equation: sum(item.occupies_hands) <= character.available_arms
 * Validates condition-to-banned-verb consistency.
 */
export function evaluateSomaticSAT(
  character: Character,
  inventoryCatalog: Map<string, Possession>
): SatCheckResult {
  let netArmsAvailable = character.somatic.available_arms;

  // Sessile or slithering baseline organisms have 0 functional arms
  if (character.somatic.locomotion_baseline === 'sessile' || character.somatic.locomotion_baseline === 'slithering') {
    netArmsAvailable = 0;
  }

  // Penalize functional arms for severe upper-limb trauma
  for (const condition of character.somatic.conditions) {
    if (condition.includes('severed_arm') || condition.includes('paralyzed_arm') || condition.includes('amputated')) {
      netArmsAvailable = Math.max(0, netArmsAvailable - 1);
    }
  }

  // Calculate hand demand across equipped held slots
  let handsDemanded = 0;
  for (const itemId of character.logistical.held) {
    const item = inventoryCatalog.get(itemId);
    if (!item) {
      return {
        passed: false,
        code: 'LOGISTICAL_OVERALLOCATION',
        message: `Equipped item '${itemId}' not found in registered inventory catalog.`
      };
    }
    handsDemanded += item.logistical.occupies_hands;
  }

  if (handsDemanded > netArmsAvailable) {
    return {
      passed: false,
      code: 'SOMATIC_SAT_BREACH',
      message: `Hand capacity equation violated: Demanded ${handsDemanded} arms, but only ${netArmsAvailable} functional arms available.`,
      armsDemanded: handsDemanded,
      armsAvailable: netArmsAvailable
    };
  }

  // Verify kinetic prohibition consistency (e.g., NO_SPRINT requires 'sprint' in banned_kinetic_verbs)
  if (character.somatic.motor_limitations.includes('NO_SPRINT')) {
    const hasSprintBanned = character.somatic.banned_kinetic_verbs.some(v => v.toLowerCase().includes('sprint'));
    if (!hasSprintBanned) {
      return {
        passed: false,
        code: 'MOTOR_LIMITATION_MISMATCH',
        message: "Motor limitation 'NO_SPRINT' declared without corresponding 'sprint' verb in banned_kinetic_verbs."
      };
    }
  }

  return { passed: true };
}
