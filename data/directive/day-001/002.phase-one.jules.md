import * as fs from "node:fs";
import * as path from "node:path";
import * as yaml from "js-yaml";
import { CharacterMachineRailSchema, CharacterMachineRail } from "../schemas/entities";

export interface ByteZeroValidationResult {
  valid: boolean;
  byteOffset: number;
  hasBom: boolean;
  hasCarriageReturns: boolean;
  reason?: string;
}

/**
 * Asserts that the buffer starts strictly at physical byte offset 0 with --- (0x2D 0x2D 0x2D 0x0A)
 */
export function verifyByteZeroIntegrity(buffer: Buffer): ByteZeroValidationResult {
  if (buffer.length < 4) {
    return {
      valid: false,
      byteOffset: 0,
      hasBom: false,
      hasCarriageReturns: false,
      reason: "FILE_TOO_SHORT",
    };
  }

  // Detect UTF-8 BOM (0xEF 0xBB 0xBF)
  const hasBom = buffer[0] === 0xef && buffer[1] === 0xbb && buffer[2] === 0xbf;
  if (hasBom) {
    return {
      valid: false,
      byteOffset: 3,
      hasBom: true,
      hasCarriageReturns: false,
      reason: "BOM_CORRUPTION_DETECTED",
    };
  }

  // Check for carriage returns (\r / 0x0D) in the opening slice
  const hasCarriageReturns = buffer.subarray(0, 100).includes(0x0d);

  // Exact Byte-0 frontmatter delimiter: --- followed by \n
  const isByteZero =
    buffer[0] === 0x2d && buffer[1] === 0x2d && buffer[2] === 0x2d && buffer[3] === 0x0a;

  if (!isByteZero) {
    return {
      valid: false,
      byteOffset: 0,
      hasBom: false,
      hasCarriageReturns,
      reason: "BYTE_ZERO_FRONTMATTER_MISALIGNED",
    };
  }

  if (hasCarriageReturns) {
    return {
      valid: false,
      byteOffset: 0,
      hasBom: false,
      hasCarriageReturns: true,
      reason: "CARRIAGE_RETURN_NON_POSIX_DETECTED",
    };
  }

  return {
    valid: true,
    byteOffset: 0,
    hasBom: false,
    hasCarriageReturns: false,
  };
}

/**
 * Serializes frontmatter and voice prose with POSIX LF newlines starting at Byte 0
 */
export function buildByteZeroDossier(frontmatter: object, voiceBody: string): string {
  const yamlContent = yaml.dump(frontmatter, {
    indent: 2,
    lineWidth: -1,
    sortKeys: true,
  });

  // Strict POSIX LF normalization
  const normalizedYaml = yamlContent.replace(/\r\n/g, "\n").trim();
  const normalizedBody = voiceBody.replace(/\r\n/g, "\n").trim();

  return `---\n${normalizedYaml}\n---\n\n${normalizedBody}\n`;
}

/**
 * Scaffolds an atomic character dossier to disk using atomic POSIX rename
 */
export function scaffoldCharacterDossier(
  targetDir: string,
  frontmatter: CharacterMachineRail,
  voiceBody: string
): string {
  // Validate schema before disk serialization
  const validatedFrontmatter = CharacterMachineRailSchema.parse(frontmatter);
  const content = buildByteZeroDossier(validatedFrontmatter, voiceBody);
  const targetFilePath = path.join(targetDir, `${validatedFrontmatter.id}.md`);
  const tempFilePath = path.join(targetDir, `.tmp_${validatedFrontmatter.id}_${Date.now()}`);

  fs.writeFileSync(tempFilePath, content, { encoding: "utf8" });

  const rawBuffer = fs.readFileSync(tempFilePath);
  const integrity = verifyByteZeroIntegrity(rawBuffer);

  if (!integrity.valid) {
    fs.unlinkSync(tempFilePath);
    throw new Error(`CRITICAL: Byte-0 integrity failure: ${integrity.reason}`);
  }

  fs.renameSync(tempFilePath, targetFilePath);
  return targetFilePath;
}

import { CharacterMachineRail, PossessionMachineRail } from "../schemas/entities";

export interface SomaticSatEvaluation {
  satisfiable: boolean;
  netArmsAvailable: number;
  handsDemanded: number;
  activeTraumaPenalties: number;
  violatedBannedVerbs: string[];
  errorCode?: "SOMATIC_SAT_BREACH" | "KINETIC_VERB_BREACH";
  reason?: string;
}

/**
 * Enforces the Hand Capacity Invariant and Kinetic Exclusion Masks at Stage T0 = $0.00:
 * sum(occupies_hands) <= max(0, available_arms - traumaPenalties)
 */
export function evaluateSomaticSAT(
  character: CharacterMachineRail,
  heldPossessions: PossessionMachineRail[],
  attemptedActions: string[] = []
): SomaticSatEvaluation {
  const traumaPenalties = character.somatic.active_conditions.reduce(
    (acc, cond) => acc + cond.trauma_penalty_arms,
    0
  );

  const netArmsAvailable = Math.max(0, character.somatic.available_arms - traumaPenalties);

  const handsDemanded = heldPossessions.reduce(
    (acc, item) => acc + item.logistical.occupies_hands,
    0
  );

  // Check 1: Hand capacity invariant
  if (handsDemanded > netArmsAvailable) {
    return {
      satisfiable: false,
      netArmsAvailable,
      handsDemanded,
      activeTraumaPenalties: traumaPenalties,
      violatedBannedVerbs: [],
      errorCode: "SOMATIC_SAT_BREACH",
      reason: `OVERCAPACITY: Demanded ${handsDemanded} hands but only ${netArmsAvailable} functional limbs available.`,
    };
  }

  // Check 2: Banned kinetic verbs intersection
  const bannedSet = new Set(character.somatic.banned_kinetic_verbs.map((v) => v.toUpperCase()));
  const violatedVerbs = attemptedActions.filter((action) => bannedSet.has(action.toUpperCase()));

  if (violatedVerbs.length > 0) {
    return {
      satisfiable: false,
      netArmsAvailable,
      handsDemanded,
      activeTraumaPenalties: traumaPenalties,
      violatedBannedVerbs: violatedVerbs,
      errorCode: "KINETIC_VERB_BREACH",
      reason: `MOTOR_VIOLATION: Attempted banned verbs: [${violatedVerbs.join(", ")}]`,
    };
  }

  return {
    satisfiable: true,
    netArmsAvailable,
    handsDemanded,
    activeTraumaPenalties: traumaPenalties,
    violatedBannedVerbs: [],
  };
}

  id: string;
  episode_num: number;
  scene_num: number;
  text: string;
  status: "clean" | "dirty" | "reviewed";
  is_dirty: number;
}

export class LocalSqliteLedger {
  public db: Database.Database;

  constructor(dbPath: string = ":memory:") {
    this.db = new Database(dbPath);
    this.db.pragma("foreign_keys = ON");
    this.initSchema();
  }

  private initSchema(): void {
    this.db.exec(`
      CREATE TABLE IF NOT EXISTS facts (
        id TEXT PRIMARY KEY,
        claim_text TEXT NOT NULL,
        scope TEXT CHECK(scope IN ('CHARACTER', 'LOCATION', 'LOGISTIC', 'LAW')) NOT NULL,
        version INTEGER NOT NULL DEFAULT 1,
        author TEXT NOT NULL,
        created_at INTEGER NOT NULL,
        last_edited_at INTEGER NOT NULL
      );

      CREATE TABLE IF NOT EXISTS scenes (
        id TEXT PRIMARY KEY,
        episode_num INTEGER NOT NULL,
        scene_num INTEGER NOT NULL,
        text TEXT NOT NULL,
        status TEXT CHECK(status IN ('clean', 'dirty', 'reviewed')) NOT NULL DEFAULT 'clean',
        is_dirty INTEGER NOT NULL DEFAULT 0
      );

      CREATE TABLE IF NOT EXISTS scene_fact_dependencies (
        scene_id TEXT NOT NULL,
        fact_id TEXT NOT NULL,
        pinned_version INTEGER NOT NULL,
        created_at INTEGER NOT NULL,
        PRIMARY KEY (scene_id, fact_id),
        FOREIGN KEY (scene_id) REFERENCES scenes(id) ON DELETE CASCADE,
        FOREIGN KEY (fact_id) REFERENCES facts(id) ON DELETE CASCADE
      );

      CREATE TRIGGER IF NOT EXISTS trg_fact_version_bump
      AFTER UPDATE ON facts
      FOR EACH ROW
      WHEN NEW.version > OLD.version
      BEGIN
        UPDATE scenes
        SET status = 'dirty', is_dirty = 1
        WHERE id IN (
          SELECT scene_id FROM scene_fact_dependencies
          WHERE fact_id = NEW.id AND pinned_version < NEW.version
        );
      END;

      CREATE VIEW IF NOT EXISTS v_fact_centrality_telemetry AS
      SELECT 
        f.id AS fact_id,
        f.claim_text,
        f.version,
        COUNT(d.scene_id) AS in_degree
      FROM facts f
      LEFT JOIN scene_fact_dependencies d ON f.id = d.fact_id
      GROUP BY f.id;
    `);
  }

  public insertFact(fact: FactRecord): void {
    const stmt = this.db.prepare(`
      INSERT INTO facts (id, claim_text, scope, version, author, created_at, last_edited_at)
      VALUES (@id, @claim_text, @scope, @version, @author, @created_at, @last_edited_at)
    `);
    stmt.run(fact);
  }

  public bumpFactVersion(factId: string, newVersion: number): void {
    const stmt = this.db.prepare(`
      UPDATE facts
      SET version = ?, last_edited_at = ?
      WHERE id = ?
    `);
    stmt.run(newVersion, Date.now(), factId);
  }

  public registerScene(scene: SceneRecord, dependencies: Array<{ factId: string; pinnedVersion: number }>): void {
    const insertScene = this.db.prepare(`
      INSERT INTO scenes (id, episode_num, scene_num, text, status, is_dirty)
      VALUES (@id, @episode_num, @scene_num, @text, @status, @is_dirty)
    `);
    const insertDep = this.db.prepare(`
      INSERT INTO scene_fact_dependencies (scene_id, fact_id, pinned_version, created_at)
      VALUES (?, ?, ?, ?)
    `);

    const tx = this.db.transaction(() => {
      insertScene.run(scene);
      for (const dep of dependencies) {
        insertDep.run(scene.id, dep.factId, dep.pinnedVersion, Date.now());
      }
    });
    tx();
  }

  public getScene(sceneId: string): SceneRecord | undefined {
    return this.db.prepare("SELECT * FROM scenes WHERE id = ?").get(sceneId) as SceneRecord | undefined;
  }

  public close(): void {
    this.db.close();
  }
}
