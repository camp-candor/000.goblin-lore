import Database from "better-sqlite3";
import { MaterialStateVector, MaterialStateVectorSchema } from "../schemas/materialState";
import { EnvironmentalZone, EnvironmentalZoneSchema } from "../schemas/environmentalOverlay";

export interface ShardDebrisRecord {
  item_id: string;
  parent_item_id: string;
  room_id: string;
  carrier_type: "ROOM_SURFACE";
  occupies_hands: number;
  structural_integrity_bps: number;
  is_operable: boolean;
  created_at_tick: number;
}

export class LocalEnvironmentalLedger {
  public db: Database.Database;

  constructor(dbPath: string = ":memory:") {
    this.db = new Database(dbPath);
    this.db.pragma("foreign_keys = ON");
    this.initSchema();
  }

  private initSchema(): void {
    this.db.exec(`
      CREATE TABLE IF NOT EXISTS material_states (
        item_id TEXT PRIMARY KEY,
        material_class TEXT NOT NULL CHECK(material_class IN ('METALLIC', 'ORGANIC_WOOD', 'TEXTILE', 'GLASS_CERAMIC', 'STONE', 'CHEMICAL_LIQUID')),
        wetness_bps INTEGER NOT NULL CHECK(wetness_bps BETWEEN 0 AND 10000),
        corrosion_bps INTEGER NOT NULL CHECK(corrosion_bps BETWEEN 0 AND 10000),
        structural_integrity_bps INTEGER NOT NULL CHECK(structural_integrity_bps BETWEEN 0 AND 10000),
        damage_threshold_bps INTEGER NOT NULL CHECK(damage_threshold_bps BETWEEN 0 AND 10000),
        combustibility_tier TEXT NOT NULL CHECK(combustibility_tier IN ('IMMUNE', 'IGNITABLE', 'HIGHLY_VOLATILE')),
        is_operable INTEGER NOT NULL CHECK(is_operable IN (0, 1)),
        updated_at_epoch INTEGER NOT NULL DEFAULT 0
      );

      CREATE TABLE IF NOT EXISTS environment_zones (
        zone_id TEXT PRIMARY KEY,
        ambient_lux INTEGER NOT NULL CHECK(ambient_lux >= 0),
        flooded_depth_cm INTEGER NOT NULL CHECK(flooded_depth_cm >= 0),
        mud_friction_bps INTEGER NOT NULL CHECK(mud_friction_bps BETWEEN 0 AND 10000),
        airborne_particulate_bps INTEGER NOT NULL CHECK(airborne_particulate_bps BETWEEN 0 AND 10000),
        acoustic_damping_cdb INTEGER NOT NULL CHECK(acoustic_damping_cdb >= 0),
        updated_at_epoch INTEGER NOT NULL DEFAULT 0
      );

      CREATE TABLE IF NOT EXISTS item_placements (
        item_id TEXT PRIMARY KEY,
        carrier_type TEXT NOT NULL CHECK(carrier_type IN ('ROOM_SURFACE', 'CONTAINER_FIXTURE', 'ACTOR_HELD', 'ACTOR_WORN', 'ACTOR_CARRIED')),
        carrier_id TEXT NOT NULL,
        room_id TEXT NOT NULL,
        slot_name TEXT,
        occupies_hands INTEGER NOT NULL DEFAULT 1 CHECK(occupies_hands BETWEEN 0 AND 2),
        is_secured INTEGER NOT NULL DEFAULT 1 CHECK(is_secured IN (0, 1)),
        updated_at_tick INTEGER NOT NULL DEFAULT 0,
        updated_at_ms INTEGER NOT NULL DEFAULT 0
      );
    `);
  }

  public upsertMaterialState(state: MaterialStateVector): void {
    const validated = MaterialStateVectorSchema.parse(state);
    const stmt = this.db.prepare(`
      INSERT INTO material_states (
        item_id, material_class, wetness_bps, corrosion_bps, structural_integrity_bps,
        damage_threshold_bps, combustibility_tier, is_operable, updated_at_epoch
      ) VALUES (?, ?, ?, ?, ?, ?, ?, ?, ?)
      ON CONFLICT(item_id) DO UPDATE SET
        material_class = excluded.material_class,
        wetness_bps = excluded.wetness_bps,
        corrosion_bps = excluded.corrosion_bps,
        structural_integrity_bps = excluded.structural_integrity_bps,
        damage_threshold_bps = excluded.damage_threshold_bps,
        combustibility_tier = excluded.combustibility_tier,
        is_operable = excluded.is_operable,
        updated_at_epoch = excluded.updated_at_epoch;
    `);

    stmt.run(
      validated.item_id,
      validated.material_class,
      validated.wetness_bps,
      validated.corrosion_bps,
      validated.structural_integrity_bps,
      validated.damage_threshold_bps,
      validated.combustibility_tier,
      validated.is_operable ? 1 : 0,
      validated.updated_at_epoch
    );
  }

  public getMaterialState(itemId: string): MaterialStateVector | undefined {
    const row = this.db
      .prepare("SELECT * FROM material_states WHERE item_id = ?")
      .get(itemId) as
      | {
          item_id: string;
          material_class: MaterialStateVector["material_class"];
          wetness_bps: number;
          corrosion_bps: number;
          structural_integrity_bps: number;
          damage_threshold_bps: number;
          combustibility_tier: MaterialStateVector["combustibility_tier"];
          is_operable: number;
          updated_at_epoch: number;
        }
      | undefined;

    if (!row) return undefined;

    return MaterialStateVectorSchema.parse({
      ...row,
      is_operable: row.is_operable === 1,
    });
  }

  public upsertZone(zone: EnvironmentalZone): void {
    const validated = EnvironmentalZoneSchema.parse(zone);
    this.db.prepare(`
      INSERT INTO environment_zones (
        zone_id, ambient_lux, flooded_depth_cm, mud_friction_bps,
        airborne_particulate_bps, acoustic_damping_cdb, updated_at_epoch
      ) VALUES (?, ?, ?, ?, ?, ?, ?)
      ON CONFLICT(zone_id) DO UPDATE SET
        ambient_lux = excluded.ambient_lux,
        flooded_depth_cm = excluded.flooded_depth_cm,
        mud_friction_bps = excluded.mud_friction_bps,
        airborne_particulate_bps = excluded.airborne_particulate_bps,
        acoustic_damping_cdb = excluded.acoustic_damping_cdb,
        updated_at_epoch = excluded.updated_at_epoch;
    `).run(
      validated.zone_id,
      validated.ambient_lux,
      validated.flooded_depth_cm,
      validated.mud_friction_bps,
      validated.airborne_particulate_bps,
      validated.acoustic_damping_cdb,
      validated.updated_at_epoch
    );
  }

  public getZone(zoneId: string): EnvironmentalZone | undefined {
    const row = this.db
      .prepare("SELECT * FROM environment_zones WHERE zone_id = ?")
      .get(zoneId) as EnvironmentalZone | undefined;

    return row ? EnvironmentalZoneSchema.parse(row) : undefined;
  }

  /**
   * Atomically destroys an item and deposits sharp debris shards on the room floor
   */
  public breakItemAndSpawnDebris(
    itemId: string,
    roomId: string,
    currentTick: number
  ): ShardDebrisRecord {
    const shardId = `${itemId}_shards`;

    const tx = this.db.transaction(() => {
      // 1. Destroy original item
      this.db.prepare(`
        UPDATE material_states
        SET structural_integrity_bps = 0, is_operable = 0, updated_at_epoch = ?
        WHERE item_id = ?
      `).run(Date.now(), itemId);

      // 2. Clear item placement from actor/container and mark it on room floor
      this.db.prepare(`
        UPDATE item_placements
        SET carrier_type = 'ROOM_SURFACE', carrier_id = ?, slot_name = 'RUINED', updated_at_tick = ?, updated_at_ms = ?
        WHERE item_id = ?
      `).run(roomId, currentTick, Date.now(), itemId);

      // 3. Register shards in material_states
      this.db.prepare(`
        INSERT INTO material_states (
          item_id, material_class, wetness_bps, corrosion_bps, structural_integrity_bps,
          damage_threshold_bps, combustibility_tier, is_operable, updated_at_epoch
        ) VALUES (?, 'GLASS_CERAMIC', 0, 0, 1000, 0, 'IMMUNE', 0, ?)
        ON CONFLICT(item_id) DO UPDATE SET structural_integrity_bps = 1000;
      `).run(shardId, Date.now());

      // 4. Place shards on room surface
      this.db.prepare(`
        INSERT INTO item_placements (
          item_id, carrier_type, carrier_id, room_id, slot_name, occupies_hands, is_secured, updated_at_tick, updated_at_ms
        ) VALUES (?, 'ROOM_SURFACE', ?, ?, 'FLOOR_DEBRIS', 0, 0, ?, ?)
        ON CONFLICT(item_id) DO UPDATE SET room_id = excluded.room_id;
      `).run(shardId, roomId, roomId, currentTick, Date.now());
    });

    tx();

    return {
      item_id: shardId,
      parent_item_id: itemId,
      room_id: roomId,
      carrier_type: "ROOM_SURFACE",
      occupies_hands: 0,
      structural_integrity_bps: 1000,
      is_operable: false,
      created_at_tick: currentTick,
    };
  }

  public close(): void {
    this.db.close();
  }
}

import { LocalEnvironmentalLedger } from "../db/localEnvironmentalLedger";
import { AttemptedOperation } from "../schemas/materialState";

export interface FeasibilityAuditResult {
  satisfiable: boolean;
  errorCode?:
    | "MECHANICAL_FAILURE_IMMUTABLE"
    | "COMBUSTION_IMPOSSIBILITY_HALT"
    | "ZERO_INTEGRITY_HALT"
    | "BRITTLE_SHATTER_HALT";
  reason?: string;
  spawnedDebrisId?: string;
  errorVector?: {
    domain: "PHYSICAL_MATERIAL";
    target_entity: string;
    rule_id: string;
    offending_tokens: string[];
  };
}

export interface ImpactDropContext {
  actor_id: string;
  item_id: string;
  room_id: string;
  surface_material: "STONE" | "METALLIC" | "ORGANIC_WOOD" | "MUD";
  impact_energy_bps: number; // 2000 bps is shatter threshold for glass/ceramic
  current_tick: number;
}

export class MaterialFeasibilityGate {
  constructor(private ledger: LocalEnvironmentalLedger) {}

  /**
   * Verifies action against active material state stored in local SQLite
   */
  public auditOperation(operation: AttemptedOperation): FeasibilityAuditResult {
    const state = this.ledger.getMaterialState(operation.item_id);
    if (!state) {
      throw new Error(`ITEM_NOT_FOUND: Material state for '${operation.item_id}' not found in SQLite.`);
    }

    // 1. Zero Integrity / Inoperable Check
    if (state.structural_integrity_bps === 0 || !state.is_operable) {
      return {
        satisfiable: false,
        errorCode: "ZERO_INTEGRITY_HALT",
        reason: `DESTROYED_ITEM: Item '${state.item_id}' has 0 structural integrity and cannot be operated.`,
        errorVector: {
          domain: "PHYSICAL_MATERIAL",
          target_entity: state.item_id,
          rule_id: "ZERO_INTEGRITY_HALT",
          offending_tokens: ["structural_integrity_bps: 0", "is_operable: false"],
        },
      };
    }

    // 2. Mechanical Oxidation Seizure Check (Corrosion > 9000 bps)
    if (
      operation.operation_type === "OPERATE_MECHANISM" ||
      operation.operation_type === "PRY_LEVERAGE"
    ) {
      if (state.material_class === "METALLIC" && state.corrosion_bps > 9000) {
        return {
          satisfiable: false,
          errorCode: "MECHANICAL_FAILURE_IMMUTABLE",
          reason: `OXIDATION_SEIZURE: Metallic item '${state.item_id}' corrosion is ${state.corrosion_bps} bps (>9000 bps). Mechanical operation seized.`,
          errorVector: {
            domain: "PHYSICAL_MATERIAL",
            target_entity: state.item_id,
            rule_id: "MECHANICAL_FAILURE_IMMUTABLE",
            offending_tokens: [`corrosion_bps: ${state.corrosion_bps}`],
          },
        };
      }
    }

    // 3. Submerged Combustion Check (Wetness > 3000 bps or IMMUNE tier)
    if (operation.operation_type === "IGNITE") {
      if (state.wetness_bps > 3000 || state.combustibility_tier === "IMMUNE") {
        return {
          satisfiable: false,
          errorCode: "COMBUSTION_IMPOSSIBILITY_HALT",
          reason: `COMBUSTION_BLOCKED: Item '${state.item_id}' has wetness of ${state.wetness_bps} bps (>3000 bps) or tier 'IMMUNE'. Ignition impossible.`,
          errorVector: {
            domain: "PHYSICAL_MATERIAL",
            target_entity: state.item_id,
            rule_id: "COMBUSTION_IMPOSSIBILITY_HALT",
            offending_tokens: [`wetness_bps: ${state.wetness_bps}`, `tier: ${state.combustibility_tier}`],
          },
        };
      }
    }

    return { satisfiable: true };
  }

  /**
   * Evaluates fragile drops, executes shatter dynamics, and spawns room debris shards
   */
  public evaluateImpactDrop(ctx: ImpactDropContext): FeasibilityAuditResult {
    const state = this.ledger.getMaterialState(ctx.item_id);
    if (!state) {
      throw new Error(`ITEM_NOT_FOUND: Material state for '${ctx.item_id}' not found in SQLite.`);
    }

    const isBrittle = state.material_class === "GLASS_CERAMIC";
    const isHardSurface = ctx.surface_material === "STONE" || ctx.surface_material === "METALLIC";
    const exceedsShatterThreshold = ctx.impact_energy_bps >= 2000;

    if (isBrittle && isHardSurface && exceedsShatterThreshold) {
      // Execute atomic shatter & debris spawning in local SQLite
      const debris = this.ledger.breakItemAndSpawnDebris(ctx.item_id, ctx.room_id, ctx.current_tick);

      return {
        satisfiable: false,
        errorCode: "BRITTLE_SHATTER_HALT",
        spawnedDebrisId: debris.item_id,
        reason: `BRITTLE_SHATTER: Item '${ctx.item_id}' impacted '${ctx.surface_material}' with ${ctx.impact_energy_bps} bps energy. Item shattered into '${debris.item_id}'.`,
        errorVector: {
          domain: "PHYSICAL_MATERIAL",
          target_entity: ctx.item_id,
          rule_id: "BRITTLE_SHATTER_HALT",
          offending_tokens: [`impact_energy_bps: ${ctx.impact_energy_bps}`, `surface: ${ctx.surface_material}`],
        },
      };
    }

    return { satisfiable: true };
  }
}

      is_operable: true,
      updated_at_epoch: 0,
    });

    const result = gate.auditOperation({
      actor_id: "char_bog",
      item_id: "item_rusted_winch",
      operation_type: "OPERATE_MECHANISM",
      applied_energy_bps: 1500,
    });

    expect(result.satisfiable).toBe(false);
    expect(result.errorCode).toBe("MECHANICAL_FAILURE_IMMUTABLE");
    expect(result.errorVector?.rule_id).toBe("MECHANICAL_FAILURE_IMMUTABLE");
    expect(result.errorVector?.offending_tokens).toContain("corrosion_bps: 9500");
  });

  it("halts ignition of submerged or wet objects with COMBUSTION_IMPOSSIBILITY_HALT", () => {
    ledger.upsertMaterialState({
      item_id: "item_waterlogged_torch",
      material_class: "ORGANIC_WOOD",
      wetness_bps: 5000, // Wet (> 3000)
      corrosion_bps: 0,
      structural_integrity_bps: 10000,
      damage_threshold_bps: 200,
      combustibility_tier: "IMMUNE",
      is_operable: true,
      updated_at_epoch: 0,
    });

    const result = gate.auditOperation({
      actor_id: "char_silas",
      item_id: "item_waterlogged_torch",
      operation_type: "IGNITE",
      applied_energy_bps: 100,
    });

    expect(result.satisfiable).toBe(false);
    expect(result.errorCode).toBe("COMBUSTION_IMPOSSIBILITY_HALT");
    expect(result.errorVector?.rule_id).toBe("COMBUSTION_IMPOSSIBILITY_HALT");
  });

  it("executes brittle shatter dynamics and spawns debris shards in local SQLite", () => {
    ledger.upsertMaterialState({
      item_id: "item_glass_vial",
      material_class: "GLASS_CERAMIC",
      wetness_bps: 0,
      corrosion_bps: 0,
      structural_integrity_bps: 3000,
      damage_threshold_bps: 200,
      combustibility_tier: "IMMUNE",
      is_operable: true,
      updated_at_epoch: 0,
    });

    // Drop onto stone floor with 2500 bps impact energy
    const result = gate.evaluateImpactDrop({
      actor_id: "char_spleen",
      item_id: "item_glass_vial",
      room_id: "loc_pump_station",
      surface_material: "STONE",
      impact_energy_bps: 2500,
      current_tick: 10,
    });

    expect(result.satisfiable).toBe(false);
    expect(result.errorCode).toBe("BRITTLE_SHATTER_HALT");
    expect(result.spawnedDebrisId).toBe("item_glass_vial_shards");

    // Verify target item destroyed in SQLite
    const originalItem = ledger.getMaterialState("item_glass_vial");
    expect(originalItem?.structural_integrity_bps).toBe(0);
    expect(originalItem?.is_operable).toBe(false);

    // Verify debris shards registered and resting on room surface
    const debrisState = ledger.getMaterialState("item_glass_vial_shards");
    expect(debrisState).toBeDefined();

    const placement = ledger.db
      .prepare("SELECT * FROM item_placements WHERE item_id = 'item_glass_vial_shards'")
      .get() as { carrier_type: string; room_id: string; slot_name: string };

    expect(placement.carrier_type).toBe("ROOM_SURFACE");
    expect(placement.room_id).toBe("loc_pump_station");
    expect(placement.slot_name).toBe("FLOOR_DEBRIS");

    // Subsequent operation on the shattered vial must fail with ZERO_INTEGRITY_HALT
    const postShatterOp = gate.auditOperation({
      actor_id: "char_spleen",
      item_id: "item_glass_vial",
      operation_type: "OPERATE_MECHANISM",
      applied_energy_bps: 500,
    });

    expect(postShatterOp.satisfiable).toBe(false);
    expect(postShatterOp.errorCode).toBe("ZERO_INTEGRITY_HALT");
  });

  it("permits fragile item drops on soft surfaces without shatter", () => {
    ledger.upsertMaterialState({
      item_id: "item_ceramic_mug",
      material_class: "GLASS_CERAMIC",
      wetness_bps: 0,
      corrosion_bps: 0,
      structural_integrity_bps: 4000,
      damage_threshold_bps: 200,
      combustibility_tier: "IMMUNE",

    const result = gate.evaluateImpactDrop({

      item_id: "item_ceramic_mug",
      room_id: "loc_canal_bank",
      surface_material: "MUD", // Soft surface
      impact_energy_bps: 2500,
      current_tick: 11,
    });

    expect(result.satisfiable).toBe(true);

    const intactItem = ledger.getMaterialState("item_ceramic_mug");
    expect(intactItem?.structural_integrity_bps).toBe(4000);
    expect(intactItem?.is_operable).toBe(true);
  });
