 */
export function deepFreeze<T>(obj: T): DeepReadonly<T> {
  if (obj === null || typeof obj !== 'object') {
    return obj as DeepReadonly<T>;
  }

  // Freeze properties first
  const propNames = Object.getOwnPropertyNames(obj);
  for (const name of propNames) {
    const value = (obj as Record<string, unknown>)[name];
    if (value && typeof value === 'object' && !Object.isFrozen(value)) {
      deepFreeze(value);
    }
  }

  return Object.freeze(obj) as DeepReadonly<T>;
}

export interface BibleLoaderOptions {
  expectedRootHash?: string;
  skipHashVerification?: boolean;
}

/**
 * Loads and defensively seals the compiled canonical universe manifest from disk.
 * Enforces H_root cryptographic integrity and applies recursive immutability.
 */
export function loadSealedBibleState(
  manifestPath: string,
  options: BibleLoaderOptions = {}
): DeepReadonly<CompiledBibleState> {
  if (!fs.existsSync(manifestPath)) {
    throw new Error(`${LOADER_TAG} [FAIL] Canonical bible manifest not found: ${manifestPath}`);
  }

  const rawBytes = fs.readFileSync(manifestPath, 'utf8');
  let parsed: CompiledBibleState;

  try {
    parsed = JSON.parse(rawBytes) as CompiledBibleState;
  } catch (err) {
    throw new Error(
      `${LOADER_TAG} [FAIL] Malformed JSON in canonical manifest: ${(err as Error).message}`
    );
  }

  if (!parsed._meta || !parsed._meta.state_hash || !parsed.entities) {
    throw new Error(
      `${LOADER_TAG} [FAIL] Invalid manifest schema: missing '_meta.state_hash' or 'entities'`
    );
  }

  // Verify H_root if expected hash is provided
  if (!options.skipHashVerification && options.expectedRootHash) {
    const normalizedExpected = options.expectedRootHash.replace(/^sha256:/, '');
    const normalizedActual = parsed._meta.state_hash.replace(/^sha256:/, '');

    if (normalizedActual !== normalizedExpected) {
      throw new Error(
        `${LOADER_TAG} [FAIL] Cryptographic Hash Mismatch :: Expected H_root [${normalizedExpected}], received [${normalizedActual}]. Downstream execution blocked.`
      );
    }
  }

  // Apply Deep-Freeze recursive immutability
  const sealedState = deepFreeze(parsed);
  return sealedState;
}

/**
 * Helper to query an entity by ID from the deep-frozen manifest.
 */
export function getSealedEntity(
  state: DeepReadonly<CompiledBibleState>,
  entityId: string
) {
  const entity = state.entities[entityId];
  if (!entity) {
    throw new Error(`${LOADER_TAG} [FAIL] Entity '${entityId}' not found in sealed state.`);
  }
  return entity;
}
