```sql
-- packages/001.lore/migrations/0001_phase0_flat_schema.sql
-- ============================================================================
-- STAGE A: CANONICAL NARRATIVE FACTS LEDGER & SCENE DEPENDENCY PROPAGATION
-- ============================================================================

-- 1. Master Fact Registry: Ground-Truth Somatic, Logistical, and Physical State
CREATE TABLE IF NOT EXISTS canonical_facts (
    id TEXT PRIMARY KEY,
    claim_text TEXT NOT NULL,
    scope TEXT NOT NULL CHECK(scope IN ('CHARACTER', 'LOCATION', 'LOGISTIC', 'LAW')),
    authority_level TEXT NOT NULL DEFAULT 'CANON' CHECK(authority_level IN ('AXIOM', 'CANON', 'EVENT', 'RUNTIME_STATE', 'PROJECTION', 'SPECULATION')),
    standing TEXT NOT NULL DEFAULT 'ACTIVE_ENDORSED' CHECK(standing IN ('ACTIVE_ENDORSED', 'FLAGGED_FOR_REVIEW', 'DEPRECATED_ARCHIVED')),
    author TEXT NOT NULL,
    version INTEGER NOT NULL DEFAULT 1 CHECK(version >= 1),
    content_hash TEXT NOT NULL,
    severity_bps INTEGER NOT NULL DEFAULT 0 CHECK(severity_bps >= 0 AND severity_bps <= 10000),
    confidence_bps INTEGER NOT NULL DEFAULT 10000 CHECK(confidence_bps >= 0 AND confidence_bps <= 10000),
    created_at_epoch INTEGER NOT NULL DEFAULT 0,
    last_edited_at_epoch INTEGER NOT NULL DEFAULT 0,
    last_reconsidered_at_epoch INTEGER NOT NULL DEFAULT 0
);

CREATE INDEX IF NOT EXISTS idx_canonical_facts_scope_standing 
ON canonical_facts (scope, standing);

-- 2. Episodic Scene Registry: Prose Units & Lifecycle Cache Flags
CREATE TABLE IF NOT EXISTS scenes (
    id TEXT PRIMARY KEY,
    episode_num INTEGER NOT NULL CHECK(episode_num >= 1),
    scene_num INTEGER NOT NULL CHECK(scene_num >= 1),
    prose_text TEXT NOT NULL,
    status TEXT NOT NULL DEFAULT 'DRAFT' CHECK(status IN ('DRAFT', 'CLEAN', 'DIRTY', 'PENDING_REVALIDATION', 'REVIEWED', 'REVIEWED_NO_CHANGE')),
    is_dirty INTEGER NOT NULL DEFAULT 0 CHECK(is_dirty IN (0, 1)),
    dirty_reason TEXT,
    asserts_zero_dependencies INTEGER NOT NULL DEFAULT 0 CHECK(asserts_zero_dependencies IN (0, 1)),
    owner_id TEXT NOT NULL,
    updated_at_epoch INTEGER NOT NULL DEFAULT 0
);

CREATE INDEX IF NOT EXISTS idx_scenes_dirty_status 
ON scenes (is_dirty, status);

CREATE INDEX IF NOT EXISTS idx_scenes_episode_sequence 
ON scenes (episode_num, scene_num);

-- 3. Scene-Fact Dependency Junction: Explicit Version-Pinned Links
CREATE TABLE IF NOT EXISTS scene_fact_dependencies (
    scene_id TEXT NOT NULL REFERENCES scenes(id) ON DELETE CASCADE,
    fact_id TEXT NOT NULL REFERENCES canonical_facts(id) ON DELETE CASCADE,
    entity_type TEXT NOT NULL DEFAULT 'character' CHECK(entity_type IN ('character', 'location', 'possession', 'grievance', 'axiom')),
    pinned_version INTEGER NOT NULL CHECK(pinned_version >= 1),
    pinned_hash TEXT,
    is_dirty INTEGER NOT NULL DEFAULT 0 CHECK(is_dirty IN (0, 1)),
    invalidation_reason TEXT,
    PRIMARY KEY (scene_id, fact_id)
);

CREATE INDEX IF NOT EXISTS idx_scene_fact_deps_fact_pinned 
ON scene_fact_dependencies (fact_id, pinned_version);

CREATE INDEX IF NOT EXISTS idx_scene_fact_deps_scene_dirty 
ON scene_fact_dependencies (scene_id, is_dirty);

-- 4. Monotonic Invalidation Trigger: Automatically Degrades Dependent Scenes
-- Fires on version increment OR retirement to DEPRECATED_ARCHIVED.
-- Transitions to FLAGGED_FOR_REVIEW explicitly trigger ZERO cascade.
CREATE TRIGGER IF NOT EXISTS propagate_fact_mutation
AFTER UPDATE ON canonical_facts
WHEN (NEW.version > OLD.version) OR (OLD.standing != 'DEPRECATED_ARCHIVED' AND NEW.standing = 'DEPRECATED_ARCHIVED')
BEGIN
    -- Step A: Flag dependency edges where pinned_version is obsolete or fact retired
    UPDATE scene_fact_dependencies
    SET is_dirty = 1,
        invalidation_reason = CASE
            WHEN NEW.standing = 'DEPRECATED_ARCHIVED' THEN 'Fact [' || NEW.id || '] retired to DEPRECATED_ARCHIVED'
            ELSE 'Fact [' || NEW.id || '] mutated v' || OLD.version || '->v' || NEW.version || ': ' || OLD.claim_text || ' -> ' || NEW.claim_text
        END
    WHERE fact_id = NEW.id
      AND (pinned_version < NEW.version OR NEW.standing = 'DEPRECATED_ARCHIVED');

    -- Step B: Degrade parent scene status to DIRTY and record invalidation audit string
    UPDATE scenes
    SET is_dirty = 1,
        status = 'DIRTY',
        dirty_reason = CASE
            WHEN NEW.standing = 'DEPRECATED_ARCHIVED' THEN 'Retired Fact Reference: [' || NEW.id || ']'
            ELSE 'Upstream fact mutated: [' || NEW.id || '] advanced from v' || OLD.version || ' to v' || NEW.version
        END
    WHERE id IN (
        SELECT scene_id 
        FROM scene_fact_dependencies
        WHERE fact_id = NEW.id AND is_dirty = 1
    );
END;

-- 5. Passive Centrality Telemetry View: Zero-Cost Blast-Radius In-Degree Calculation
CREATE VIEW IF NOT EXISTS v_fact_centrality_telemetry AS
SELECT 
    f.id AS fact_id,
    f.scope,
    f.version AS current_version,
    f.standing,
    f.last_reconsidered_at_epoch,
    COUNT(sfd.scene_id) AS active_dependent_scenes_count,
    COALESCE(MAX(s.episode_num), 0) AS latest_dependent_episode,
    COALESCE(MIN(s.episode_num), 0) AS earliest_dependent_episode
FROM canonical_facts f
LEFT JOIN scene_fact_dependencies sfd ON f.id = sfd.fact_id
LEFT JOIN scenes s ON sfd.scene_id = s.id
GROUP BY f.id, f.scope, f.version, f.standing, f.last_reconsidered_at_epoch;

```

Implement **TASK-GL-MV-3.3**: Diagnostic CLI utility executing the forward-walking recursive CTE over `CANON_DB` / SQLite, reporting direct scene matches, downstream transitive dependents, and estimated $T_0 = \$0.00$ rebuild costs.

```typescript
// packages/001.lore/scripts/why-dirty.ts
import { Database } from 'node:sqlite';

export const WHY_DIRTY_TAG = '[LORE::WHY_DIRTY]' as const;

export interface DownstreamBlastNode {
  scene_id: string;
  episode_num: number;
  scene_num: number;
  initial_status: string;
  cascade_depth: number;
  fact_id: string;
}

export interface BlastRadiusReport {
  target_fact_id: string;
  claim_text: string;
  current_version: number;
  total_invalidated_scenes: number;
  max_cascade_depth: number;
  rebuild_cost_usd: string;
  affected_episodes: number[];
  nodes: DownstreamBlastNode[];
}

/**
 * Authoritative Recursive Common Table Expression (CTE) Query.
 * Recursively walks forward from a mutated canonical fact through scene_fact_dependencies
 * and sequential continuity dependencies, clamped to depth < 6 to eliminate circular loops.
 */
export const DOWNSTREAM_BLAST_RADIUS_CTE = `
WITH RECURSIVE downstream_cascade AS (
  -- Anchor Member: Scenes directly binding the target fact
  SELECT 
    d.fact_id,
    d.scene_id,
    s.episode_num,
    s.scene_num,
    s.status AS initial_status,
    1 AS depth
  FROM scene_fact_dependencies d
  JOIN scenes s ON d.scene_id = s.id
  WHERE d.fact_id = :target_fact_id

  UNION ALL

  -- Recursive Member: Forward sequential continuity scenes sharing dependencies
  SELECT 
    dc.fact_id,
    s_next.id AS scene_id,
    s_next.episode_num,
    s_next.scene_num,
    s_next.status AS initial_status,
    dc.depth + 1 AS depth
  FROM downstream_cascade dc
  JOIN scenes s_prev ON dc.scene_id = s_prev.id
  JOIN scenes s_next ON (
    (s_next.episode_num = s_prev.episode_num AND s_next.scene_num > s_prev.scene_num)
    OR (s_next.episode_num > s_prev.episode_num)
  )
  JOIN scene_fact_dependencies d_next ON s_next.id = d_next.scene_id
  WHERE dc.depth < 6 -- Strict recursion safety brake
)
SELECT 
  scene_id,
  episode_num,
  scene_num,
  initial_status,
  MIN(depth) AS cascade_depth,
  fact_id
FROM downstream_cascade
GROUP BY scene_id, episode_num, scene_num, initial_status, fact_id
ORDER BY episode_num ASC, scene_num ASC;
`;

export function executeBlastRadiusAnalysis(
  db: Database,
  targetFactId: string
): BlastRadiusReport {
  const factStmt = db.prepare('SELECT id, claim_text, version FROM canonical_facts WHERE id = ?;');
  const factRow = factStmt.get(targetFactId) as { id: string; claim_text: string; version: number } | undefined;

  if (!factRow) {
    throw new Error(`${WHY_DIRTY_TAG} [FAIL] Fact '${targetFactId}' not found in canonical ledger.`);
  }

  const cteStmt = db.prepare(DOWNSTREAM_BLAST_RADIUS_CTE);
  const rows = cteStmt.all({ target_fact_id: targetFactId }) as DownstreamBlastNode[];

  const episodeSet = new Set<number>();
  let maxDepth = 0;
  for (const r of rows) {
    episodeSet.add(r.episode_num);
    if (r.cascade_depth > maxDepth) {
      maxDepth = r.cascade_depth;
    }
  }

  return {
    target_fact_id: factRow.id,
    claim_text: factRow.claim_text,
    current_version: factRow.version,
    total_invalidated_scenes: rows.length,
    max_cascade_depth: maxDepth,
    rebuild_cost_usd: '$0.00', // T0 Economic Firewall guarantee
    affected_episodes: [...episodeSet].sort((a, b) => a - b),
    nodes: rows,
  };
}

export function formatBlastRadiusReport(report: BlastRadiusReport): string {
  const lines: string[] = [
    '================================================================================',
    `>> ${WHY_DIRTY_TAG} BLAST RADIUS REPORT FOR [${report.target_fact_id}]`,
    '================================================================================',
    `  Claim Text        : "${report.claim_text}"`,
    `  Active Version    : v${report.current_version}`,
    `  Total Scenes Dirtied: ${report.total_invalidated_scenes}`,
    `  Max Causal Depth  : ${report.max_cascade_depth}`,
    `  Rebuild Cost (T0) : ${report.rebuild_cost_usd} (Compile-time interception)`,
    `  Affected Episodes : [${report.affected_episodes.join(', ')}]`,
    '--------------------------------------------------------------------------------',
    '  DOWNSTREAM SCENE CASCADE TREE:',
  ];

  if (report.nodes.length === 0) {
    lines.push('    (None - 0 downstream dependencies registered)');
  } else {
    for (const node of report.nodes) {
      const indent = '  '.repeat(node.cascade_depth);
      lines.push(
        `  ${indent}|-- [Scene ${node.scene_id}] (Ep.${node.episode_num} Sc.${node.scene_num}) | Status: ${node.initial_status} -> DIRTY`
      );
    }
  }
  lines.push('================================================================================');
  return lines.join('\n');
}

if (require.main === module) {
  const targetId = process.argv[2];
  if (!targetId) {
    process.stderr.write(`${WHY_DIRTY_TAG} Usage: npx tsx packages/001.lore/scripts/why-dirty.ts <fact_id>\n`);
    process.exit(1);
  }
  const dbPath = process.env.CANON_DB_PATH || ':memory:';
  const db = new Database(dbPath);
```
