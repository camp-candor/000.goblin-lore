##### Deliverable 1: RFC 8785 JSON Canonicalization Scheme (JCS) & Merkle Verifier

**File:** `packages/001.lore/src/core/canon/merkleVerifier.ts`

```typescript
import * as crypto from "node:crypto";

/**
 * Implements RFC 8785 JSON Canonicalization Scheme (JCS):
 * 1. Recursively sorts dictionary keys lexicographically by UTF-16 code units.
 * 2. Normalizes all string values to Unicode Normalization Form C (NFC).
 * 3. Prohibits NaN, Infinity, and undefined.
 * 4. Strips non-literal whitespace.
 */
export function canonicalizeJson(obj: unknown): string {
  if (obj === null) {
    return "null";
  }

  if (typeof obj === "boolean" || typeof obj === "number") {
    if (typeof obj === "number" && !Number.isFinite(obj)) {
      throw new TypeError("CANONICALIZE_ERROR: Non-finite numbers (NaN, Infinity) are prohibited by RFC 8785.");
    }
    return JSON.stringify(obj);
  }

  if (typeof obj === "string") {
    return JSON.stringify(obj.normalize("NFC"));
  }

  if (Array.isArray(obj)) {
    const elements = obj.map((item) => {
      const val = item === undefined ? null : item;
      return canonicalizeJson(val);
    });
    return `[${elements.join(",")}]`;
  }

  if (typeof obj === "object") {
    const record = obj as Record<string, unknown>;
    const sortedKeys = Object.keys(record).sort((a, b) => {
      const len = Math.min(a.length, b.length);
      for (let i = 0; i < len; i++) {
        const codeA = a.charCodeAt(i);
        const codeB = b.charCodeAt(i);
        if (codeA !== codeB) return codeA - codeB;
      }
      return a.length - b.length;
    });

    const entries: string[] = [];
    for (const key of sortedKeys) {
      const val = record[key];
      if (val === undefined || typeof val === "function" || typeof val === "symbol") {
        continue;
      }
      const canonicalKey = JSON.stringify(key.normalize("NFC"));
      const canonicalVal = canonicalizeJson(val);
      entries.push(`${canonicalKey}:${canonicalVal}`);
    }

    return `{${entries.join(",")}}`;
  }

  return "";
}

/**
 * Computes 64-character lowercase SHA-256 Merkle root digest over RFC 8785 canonical representation
 */
export function computeMerkleRoot(payload: unknown): string {
  const canonicalBytes = Buffer.from(canonicalizeJson(payload), "utf8");
  return crypto.createHash("sha256").update(canonicalBytes).digest("hex");
}

/**
 * Strips volatile _meta.state_hash before verifying canonical Merkle parity
 */
export function verifyStateMerkleParity(
  rawState: Record<string, unknown>,
  expectedHash: string
): { valid: boolean; calculatedHash: string } {
  const clone = { ...rawState };
  if (clone._meta && typeof clone._meta === "object") {
    const metaClone = { ...(clone._meta as Record<string, unknown>) };
    delete metaClone.state_hash;
    clone._meta = metaClone;
  }

  const calculatedHash = computeMerkleRoot(clone);
  return {
    valid: calculatedHash.toLowerCase() === expectedHash.toLowerCase(),
    calculatedHash,
  };
}
```

```typescript
    for (const f of dossiers.characters) {
      const raw = this.extractByteZeroFrontmatter(f);
      const parsed = CharacterMachineRailSchema.parse(raw);

      // Pre-flight Somatic SAT check
      const heldItems = parsed.logistical.slots.held
        .map((itemId) => possessions[itemId])
        .filter((item): item is PossessionMachineRail => item !== undefined);

      const sat = evaluateSomaticSAT(parsed, heldItems);
      if (!sat.satisfiable) {
        throw new Error(`PASS_1_SOMATIC_SAT_BREACH: Character '${parsed.id}' failed SAT check: ${sat.reason}`);
      }

      characters[parsed.id] = parsed;
    }

    for (const f of dossiers.grievances) {
      const raw = this.extractByteZeroFrontmatter(f);
      const parsed = GrievanceMachineRailSchema.parse(raw);
      grievances[parsed.id] = parsed;
    }

    return { characters, possessions, grievances };
  }

  public executePass2(state: {
    characters: Record<string, CharacterMachineRail>;
    possessions: Record<string, PossessionMachineRail>;
    grievances: Record<string, GrievanceMachineRail>;
  }): void {
    // 1. Foreign-Key Sweep
    for (const char of Object.values(state.characters)) {
      for (const slotKey of ["worn", "held", "carried", "cached"] as const) {
        for (const itemId of char.logistical.slots[slotKey]) {
          if (!state.possessions[itemId]) {
            throw new Error(`COMPILER_FK_ERROR: Character '${char.id}' references non-existent possession '${itemId}'.`);
          }
        }
      }
    }

    for (const grv of Object.values(state.grievances)) {
      for (const participantId of grv.participants_primary) {
        if (!state.characters[participantId]) {
          throw new Error(`COMPILER_FK_ERROR: Grievance '${grv.id}' references unknown character '${participantId}'.`);
        }
      }
      for (const propId of grv.bound_props) {
        if (!state.possessions[propId]) {
          throw new Error(`COMPILER_FK_ERROR: Grievance '${grv.id}' references unknown bound prop '${propId}'.`);
        }
      }
    }

    // 2. Three-Color DFS Grievance DAG Acyclicity
    const dagResult = verifyGrievanceDag(Object.values(state.grievances));
    if (!dagResult.isAcyclic) {
      throw new Error(`COMPILER_DAG_CYCLE: ${dagResult.reason}`);
    }

    // 3. Farce Ratio Cap Linter
    const farceResult = auditFarceRatio(Object.values(state.grievances));
    if (!farceResult.valid) {
      throw new Error(`COMPILER_FARCE_BREACH: ${farceResult.reason}`);
    }
  }

  public executePass3(
    data: {
      characters: Record<string, CharacterMachineRail>;
      possessions: Record<string, PossessionMachineRail>;
      grievances: Record<string, GrievanceMachineRail>;
    },
    outputDir: string,
    fixedEpoch: number = 0
  ): { headPath: string; wormPath: string; stateHash: string } {
    const rawPayload: Omit<BibleState, "_meta"> & { _meta: Partial<BibleMeta> } = {
      _meta: {
        seriesSlug: this.seriesSlug,
        version: 1,
        compiled_at_epoch: fixedEpoch,
        entity_count:
          Object.keys(data.characters).length +
          Object.keys(data.possessions).length +
          Object.keys(data.grievances).length,
      },
      characters: data.characters,
      possessions: data.possessions,
      grievances: data.grievances,
      locations: {},
    };

    // Calculate Merkle root hash
    const stateHash = computeMerkleRoot(rawPayload);
    rawPayload._meta.state_hash = stateHash;

    const finalizedState = rawPayload as BibleState;
    const canonicalString = canonicalizeJson(finalizedState);

    // Format Windows NTFS-safe WORM timestamp (YYYY-MM-DDTHH-mm-ssZ)
    const wormTimestamp = new Date(fixedEpoch)
      .toISOString()
      .replace(/:/g, "-")
      .replace(/\..+/, "Z");

    const seriesCompiledDir = path.join(outputDir, this.seriesSlug);
    fs.mkdirSync(seriesCompiledDir, { recursive: true });

    const headPath = path.join(seriesCompiledDir, "bible-state.json");
    const wormPath = path.join(seriesCompiledDir, `bible-state_${wormTimestamp}.json`);

    const tempPath = path.join(seriesCompiledDir, `.tmp_bible_${crypto.randomUUID()}`);
    fs.writeFileSync(tempPath, canonicalString + "\n", "utf8");

    fs.copyFileSync(tempPath, wormPath);
    fs.renameSync(tempPath, headPath);

    return { headPath, wormPath, stateHash };
  }
}
```

##### Deliverable 3: Deep-Freeze Runtime Ingestion Client

**File:** `packages/001.lore/src/core/canon/bibleLoader.ts`

```typescript
import * as fs from "node:fs";
import Database from "better-sqlite3";
import { BibleState } from "../../scripts/compile";
import { verifyStateMerkleParity } from "./merkleVerifier";

/**
 * Recursively freezes an object tree in place, enforcing strict runtime immutability
 */
export function deepFreeze<T>(obj: T): T {
  if (obj === null || typeof obj !== "object") {
    return obj;
  }

  Object.freeze(obj);

  for (const key of Object.keys(obj)) {
    const val = (obj as Record<string, unknown>)[key];
    if (val !== null && typeof val === "object" && !Object.isFrozen(val)) {
      deepFreeze(val);
    }
  }

  return obj;
}

export interface LoadBibleOptions {
  localSqliteDbPath?: string;
  expectedStateHash?: string;
}

export interface LoadBibleResult {
  state: BibleState;
  stateHash: string;
  isParityVerified: boolean;
}

/**
 * Loads compiled bible-state.json from disk, verifies cryptographic Merkle root parity,
 * cross-references local SQLite metadata table, and applies recursive deepFreeze
 */
export function loadBibleState(filePath: string, options: LoadBibleOptions = {}): LoadBibleResult {
  const content = fs.readFileSync(filePath, "utf8");
  const rawJson = JSON.parse(content) as BibleState;

  if (!rawJson._meta || !rawJson._meta.state_hash) {
    throw new Error("CANON_LOAD_ERROR: Payload missing mandatory _meta.state_hash.");
  }

  const expectedHash = options.expectedStateHash || rawJson._meta.state_hash;
  const parity = verifyStateMerkleParity(rawJson as unknown as Record<string, unknown>, expectedHash);

  if (!parity.valid) {
    throw new Error(
      `CANON_DESYNC_HALT: Merkle root mismatch. Expected ${expectedHash}, calculated ${parity.calculatedHash}.`
    );
  }

  // Verify against local SQLite instance before cloud D1
  if (options.localSqliteDbPath) {
    const db = new Database(options.localSqliteDbPath);
    try {
      db.exec(`
        CREATE TABLE IF NOT EXISTS canon_metadata (
          series_slug TEXT PRIMARY KEY,
          state_hash TEXT NOT NULL,
          is_dirty INTEGER NOT NULL DEFAULT 0,
          updated_at_epoch INTEGER NOT NULL
        );
      `);

      const record = db
        .prepare("SELECT state_hash, is_dirty FROM canon_metadata WHERE series_slug = ?")
        .get(rawJson._meta.seriesSlug) as { state_hash: string; is_dirty: number } | undefined;

      if (record) {
        if (record.is_dirty === 1) {
          throw new Error("CANON_DESYNC_HALT: Local SQLite flags active canon as DIRTY (is_dirty = 1).");
        }
        if (record.state_hash !== parity.calculatedHash) {
          throw new Error(
            `CANON_DESYNC_HALT: Local SQLite state_hash mismatch. Db has ${record.state_hash}, calculated ${parity.calculatedHash}.`
          );
        }
      }
    } finally {
      db.close();
    }
  }

  const frozenState = deepFreeze(rawJson);

  return {
    state: frozenState,
    stateHash: parity.calculatedHash,
    isParityVerified: true,
  };
}
```
