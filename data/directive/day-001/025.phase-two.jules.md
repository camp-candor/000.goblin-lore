    return obj;
  }

export function verifyByteZeroIntegrity(content: Buffer | Uint8Array | string): ByteZeroCheckResult {
  const buf = Buffer.isBuffer(content)
    ? content
    : typeof content === 'string'
    ? Buffer.from(content, 'utf8')
    : Buffer.from(content);

  if (buf.length < 4) {
    return {
      valid: false,
      byteOffset: 0,
      hasBom: false,
      hasCarriageReturns: false,
      errorCode: 'PAYLOAD_TOO_SHORT',

  if (hasBom) {
    return {
      valid: false,
      byteOffset: 3,
      hasBom: true,
      hasCarriageReturns: false,
      errorCode: 'UTF8_BOM_DETECTED',
      message: 'Byte-0 violation: UTF-8 BOM detected at byte offset 0.'
    };
  }

  // Detect leading whitespace or newlines
  if (buf[0] === 0x20 || buf[0] === 0x09 || buf[0] === 0x0a || buf[0] === 0x0d) {
    return {
      valid: false,
      byteOffset: 0,
      hasBom: false,
      hasCarriageReturns: buf[0] === 0x0d,
      errorCode: 'LEADING_WHITESPACE_DETECTED',
      message: 'Byte-0 violation: Leading whitespace or newline before frontmatter delimiter.'
    };
  }

  // Check exact 0x2D 0x2D 0x2D (---)
  const startsWithDelimiter = buf[0] === 0x2d && buf[1] === 0x2d && buf[2] === 0x2d;
  if (!startsWithDelimiter) {
    return {
      valid: false,
      byteOffset: 0,
      hasBom: false,
      hasCarriageReturns: false,
      errorCode: 'MISSING_BYTE_ZERO_DELIMITER',
      message: 'Byte-0 violation: File does not start with "---" at byte index 0.'
    };

  if (buf[3] !== 0x0a) {
    const isCrLf = buf[3] === 0x0d && buf[4] === 0x0a;
    return {
      valid: false,
      byteOffset: 3,
      hasBom: false,
      hasCarriageReturns: buf[3] === 0x0d,
      errorCode: isCrLf ? 'CARRIAGE_RETURN_NON_POSIX' : 'DELIMITER_NOT_FOLLOWED_BY_LF',
      message: isCrLf

        : 'Byte-0 violation: Opening delimiter not immediately followed by POSIX LF (0x0A).'
    };
  }

  // Scan frontmatter header block for non-POSIX carriage returns (\r / 0x0D).
  // Bound the header block by locating the closing '---' delimiter on its own line.
  let headerEnd = buf.length;
  let pos = 4;
  while (pos < buf.length) {

    if (idx === -1) {
      break;
    }
    const prevByte = buf[idx - 1];
    const isLineStart = prevByte === 0x0a || prevByte === 0x0d;
    if (isLineStart) {
      const afterDash = idx + 3;
      if (afterDash === buf.length) {
        headerEnd = afterDash;
        break;
      } else if (buf[afterDash] === 0x0a) {
        headerEnd = afterDash + 1;
        break;

          headerEnd = afterDash + 2;
        } else {
          headerEnd = afterDash + 1;
        }
        break;
      }

    };

    byteOffset: 0,

    if (text.endsWith('\n---')) {
      const rawYaml = text.slice(4, text.length - 4);
      return { rawYaml, proseBody: '' };
    }
    throw new Error('[BYTE_0_EXTRACT_ERROR] Missing closing frontmatter delimiter (\\n---\\n)');
  }

  const rawYaml = text.slice(4, closingIndex);
  const proseBody = text.slice(closingIndex + 5);

  return { rawYaml, proseBody };
}

/**
 * Extracts and validates a Markdown dossier with strict schema checking and deep freezing.
 */
export function extractByteZeroFactDossier<T>(
  filePath: string,
  content: Buffer | Uint8Array | string,
  validator?: (data: unknown) => T
): DossierExtractionResult<T> {
  const byteCheck = verifyByteZeroIntegrity(content);
  if (!byteCheck.valid) {
    return {
      valid: false,
      filePath,
      errorCode: byteCheck.errorCode || 'MISSING_BYTE_ZERO_DELIMITER',
      message: byteCheck.message || 'Byte-0 verification failed'
    };
  }

  let extracted: { rawYaml: string; proseBody: string };
  try {
    extracted = extractRawFrontmatterAndBody(content);
  } catch (err: unknown) {
    const errorMsg = err instanceof Error ? err.message : String(err);
    return {
      valid: false,
      filePath,
      errorCode: 'MISSING_CLOSING_DELIMITER',
      message: errorMsg
    };
  }

  let parsedYaml: unknown;
  try {
    parsedYaml = yaml.load(extracted.rawYaml);
  } catch (err: unknown) {
    const errorMsg = err instanceof Error ? err.message : String(err);
    return {
      valid: false,
      filePath,
      errorCode: 'MALFORMED_YAML_PAYLOAD',
      message: `Failed to parse YAML frontmatter: ${errorMsg}`
    };
  }

  let validatedData: T;
  if (validator) {
    try {
      validatedData = validator(parsedYaml);
    } catch (err: unknown) {
      const errorMsg = err instanceof Error ? err.message : String(err);
      return {
        valid: false,
        filePath,
        errorCode: 'SCHEMA_VALIDATION_FAILURE',
        message: `Frontmatter schema validation failed: ${errorMsg}`
      };
    }
  } else {
    validatedData = parsedYaml as T;
  }

  const frozenFrontmatter = deepFreeze(validatedData);

  return {
    valid: true,
    filePath,
    frontmatter: frozenFrontmatter,
    proseBody: extracted.proseBody,
    rawFrontmatterYaml: extracted.rawYaml,
    byteSize: Buffer.isBuffer(content) ? content.length : Buffer.from(content).length
  };
}

/**
 * Hard fail-closed assertion for Byte-0 dossier extraction.
 * Throws an Error immediately at Stage T0 ($0.00) if any extraction invariant fails.
 */
export function assertByteZeroFactDossier<T>(
  filePath: string,
  content: Buffer | Uint8Array | string,
  validator?: (data: unknown) => T
): ExtractedFactDossierBit<T> {
  const result = extractByteZeroFactDossier(filePath, content, validator);
  if (!result.valid) {
    throw new Error(`[BYTE_ZERO_BREACH] ${result.errorCode}: ${result.message} in file '${filePath}'`);
  }
  return result;
}

import * as path from 'node:path';
import { ExtractedFactDossierBit, DossierExtractionFailureBit } from '../../41.canon.unit/fce/byteZero.bit.js';
import { extractByteZeroFactDossier, deepFreeze } from '../../41.canon.unit/buz/byteZero.buzz.js';
import { FactCollectionBit, CollectBatchResultBit } from '../fce/collect.bit.js';

/**
 * Scans a directory recursively for .md files and extracts them under the Byte-0 Frontmatter Law.
 */
export function collectDossiersFromDirectory<T>(
  dirPath: string,
  validator?: (data: unknown) => T
): CollectBatchResultBit {
  if (!fs.existsSync(dirPath)) {
    throw new Error(`[COLLECT_DIR_NOT_FOUND] Directory does not exist: '${dirPath}'`);
  }

  const dossiers = new Map<string, ExtractedFactDossierBit>();
  const failures: DossierExtractionFailureBit[] = [];

  function scanRecursive(currentDir: string): void {
    const entries = fs.readdirSync(currentDir, { withFileTypes: true });
    for (const entry of entries) {
      const fullPath = path.join(currentDir, entry.name);
      if (entry.isDirectory()) {
        scanRecursive(fullPath);
      } else if (entry.isFile() && entry.name.endsWith('.md')) {
        const fileBuffer = fs.readFileSync(fullPath);
        const result = extractByteZeroFactDossier(fullPath, fileBuffer, validator);
        if (result.valid) {
          dossiers.set(fullPath, result);
        } else {
          failures.push(result);
        }
      }
    }
  }

  scanRecursive(dirPath);

  const collection: FactCollectionBit = {
    totalFilesScanned: dossiers.size + failures.length,
    successfulExtractions: dossiers.size,
    failedExtractions: failures.length,
    dossiers: deepFreeze(dossiers),
    failures: deepFreeze(failures)
  };

  if (failures.length > 0) {
    return {
      valid: false,
      errorCode: 'COLLECTION_VALIDATION_BREACH',
      failureCount: failures.length,
      errors: failures.map((f) => `[${f.errorCode}] ${f.filePath}: ${f.message}`)
    };
  }

  return {
    valid: true,
    collection
  };
}
