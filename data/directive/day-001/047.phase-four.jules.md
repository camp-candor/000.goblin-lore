  let curr = path.resolve(startDir);
  while (curr) {
    const pkgPath = path.join(curr, 'package.json');
    if (fs.existsSync(pkgPath)) {
      try {
        const pkg = fs.readJsonSync(pkgPath);
        if (pkg.workspaces || fs.existsSync(path.join(curr, '.git'))) {

        }
      } catch {
        // Continue upward on JSON read error
      }

    const parent = path.dirname(curr);
    if (parent === curr) break;
    curr = parent;
  }
  return process.cwd();

        cmpBit: { idx: 'gate0-verify-fail', val: -1 },
        errBit: { code: 'GATE0_LEVEL1_FAILED', message: lvl1.error || 'Level 1 failed' },
      });
    }
    return cpy;
  }
  await emitTelemetry(ste, '>> [GATE_0:LVL_1] [OK] Type soundness verified.');

  // LEVEL 2: Three-Pass In-Memory Compiler Gauntlet
  await emitTelemetry(ste, '>> [GATE_0:LVL_2] Executing 3-pass compiler gauntlet...');
  const { result: lvl2, updatedModel } = await auditLevel2(cpy, bal, ste);
  cpy = updatedModel;

  if (!lvl2.passed) {
    const errMsg = `:: [FAIL] Level 2 breached: ${lvl2.error}`;
    await emitTelemetry(ste, errMsg);
    cpy.status = 'FAILED';
    cpy.lastError = lvl2.error;

    if (bal.slv) {
      bal.slv({
        cmpBit: { idx: 'gate0-verify-fail', val: -1 },
        errBit: { code: 'GATE0_LEVEL2_FAILED', message: lvl2.error || 'Level 2 failed' },
      });
    }
    return cpy;
  }
  await emitTelemetry(ste, '>> [GATE_0:LVL_2] [OK] Compiler gauntlet cleared.');

  // LEVEL 3: Zero-Drift Tripwire
  if (verifyDrift) {
    await emitTelemetry(ste, '>> [GATE_0:LVL_3] Checking zero-drift tripwire (git diff --exit-code)...');
    const lvl3 = await auditLevel3(repoRoot);
    if (!lvl3.passed) {
      const errMsg = `:: [FAIL] Level 3 breached: ${lvl3.error}`;
      await emitTelemetry(ste, errMsg);
      cpy.status = 'FAILED';
      cpy.lastError = lvl3.error;

      if (bal.slv) {
        bal.slv({
          cmpBit: { idx: 'gate0-verify-fail', val: -1 },
          errBit: { code: 'GATE0_LEVEL3_FAILED', message: lvl3.error || 'Level 3 failed' },
        });
      }
      return cpy;
    }
    await emitTelemetry(ste, '>> [GATE_0:LVL_3] [OK] Zero-drift tripwire passed.');
  }

  const durationMs = Date.now() - startTime;
  await emitTelemetry(ste, `>> [GATE_0] [ALL_GATES_PASSED] Certified in ${durationMs}ms.`);

  cpy.status = 'SEALED';
  cpy.lastError = undefined;

  if (bal.slv) {
    bal.slv({
      cmpBit: { idx: 'gate0-verify-success', val: 0 },
      resBit: {
        seriesSlug: cpy.activeSeriesSlug || '001.goblin-lore',
        rootMerkleHash: cpy.rootMerkleHash || '',
        canonicalPath: path.join(repoRoot, 'compiled', cpy.activeSeriesSlug || '001.goblin-lore', 'bible-state.json'),
        executionDurationMs: durationMs,
        cleanExit: true,
      },
    });
  }

  return cpy;
};
