```typescript
import { Database } from 'node:sqlite';

export const STANDING_TAG = '[LORE::STANDING]' as const;

export type EpistemologicalStanding =
  | 'ACTIVE_ENDORSED'
  | 'FLAGGED_FOR_REVIEW'
  | 'DEPRECATED_ARCHIVED';

export type FactScope = 'CHARACTER' | 'LOCATION' | 'LOGISTIC' | 'LAW';

export interface FactAuditCandidate {
  fact_id: string;
  claim_text: string;
  scope: FactScope;
  standing: EpistemologicalStanding;
  current_version: number;
  last_reconsidered_at_epoch: number;
  days_stale: number;
  dependent_scenes_count: number;
  priority_score_bps: number; // Stored in basis points (100 = 1.00 score)
}

export const SCOPE_WEIGHT_BPS: Readonly<Record<FactScope, number>> = Object.freeze({
  LAW: 20000,       // 2.0x weight for fundamental world laws
  CHARACTER: 15000, // 1.5x weight for character somatic invariants
  LOCATION: 10000,  // 1.0x baseline weight
  LOGISTIC: 8000,   // 0.8x weight for transient inventory items
});

/**
 * Calculates Topological Blast-Radius Priority Score:
 * PriorityScore = DaysStale * log10(DependentScenes + 1) * ScopeMultiplier
 * Computed in integer basis points (bps) to eliminate IEEE 754 floating-point drift.
 */
export function calculateAuditPriorityScoreBps(
  daysStale: number,
  dependentScenesCount: number,
  scope: FactScope
): number {
  if (daysStale <= 0) return 0;
  const logScenes = Math.log10(dependentScenesCount + 1);
  const scopeMultiplier = (SCOPE_WEIGHT_BPS[scope] ?? 10000) / 10000;
  const score = daysStale * logScenes * scopeMultiplier;
  return Math.round(score * 100);
}

export class StandingEngine {
  private db: Database;

  constructor(db: Database) {
    this.db = db;
  }

  /**
   * Evaluates all canonical facts and returns an ordered queue of audit candidates.
   * Only facts unreviewed for >= minDaysStale days are prioritized.
   */
  public getAuditQueue(
    currentEpoch: number,
    minDaysStale = 30
  ): FactAuditCandidate[] {
    const query = `
      SELECT 
        f.id AS fact_id,
        f.claim_text,
        f.scope,
        f.standing,
        f.version AS current_version,
        f.last_reconsidered_at_epoch,
        COUNT(sfd.scene_id) AS dependent_scenes_count
      FROM canonical_facts f
      LEFT JOIN scene_fact_dependencies sfd ON f.id = sfd.fact_id
      WHERE f.standing != 'DEPRECATED_ARCHIVED'
      GROUP BY f.id, f.claim_text, f.scope, f.standing, f.version, f.last_reconsidered_at_epoch;
    `;

    const rows = this.db.prepare(query).all() as Array<{
      fact_id: string;
      claim_text: string;
      scope: FactScope;
      standing: EpistemologicalStanding;
      current_version: number;
      last_reconsidered_at_epoch: number;
      dependent_scenes_count: number;
    }>;

    const candidates: FactAuditCandidate[] = [];

    for (const row of rows) {
      const elapsedEpochs = Math.max(0, currentEpoch - row.last_reconsidered_at_epoch);
      // Assuming 1 day = 86400 logical epoch ticks (or integer seconds)
      const daysStale = Math.floor(elapsedEpochs / 86400);

      if (daysStale >= minDaysStale || row.standing === 'FLAGGED_FOR_REVIEW') {
        const scoreBps = calculateAuditPriorityScoreBps(
          daysStale,
          row.dependent_scenes_count,
          row.scope
        );

        candidates.push({
          fact_id: row.fact_id,
          claim_text: row.claim_text,
          scope: row.scope,
          standing: row.standing,
          current_version: row.current_version,
          last_reconsidered_at_epoch: row.last_reconsidered_at_epoch,
          days_stale: daysStale,
          dependent_scenes_count: row.dependent_scenes_count,
          priority_score_bps: scoreBps,
        });
      }
    }

    // Sort descending by priority_score_bps, tie-breaking by UTF-16 fact_id
    return candidates.sort((a, b) => {
      if (b.priority_score_bps !== a.priority_score_bps) {
        return b.priority_score_bps - a.priority_score_bps;
      }
      return a.fact_id < b.fact_id ? -1 : a.fact_id > b.fact_id ? 1 : 0;
    });
  }

      .get(factId) as { id: string; version: number; standing: string } | undefined;

    }

    const updateStmt = this.db.prepare(`

      versionBumpped: false,
      cascadeTriggered: false,

  /**
   * Action 2: Flag for Review (The Liminal State)
   * Places fact under thematic scrutiny. Suppresses dirty cascades completely.
   */
  public flagForReview(
    factId: string,
    currentEpoch: number,
    reviewerId: string,
    scrutinyNote: string
  ): { success: boolean; cascadeTriggered: boolean } {
    const existing = this.db
      .prepare('SELECT id, version FROM canonical_facts WHERE id = ?;')
      .get(factId);

    if (!existing) {
      throw new Error(`${STANDING_TAG} [FAIL] Fact '${factId}' does not exist.`);
    }

    const updateStmt = this.db.prepare(`
      UPDATE canonical_facts
      SET standing = 'FLAGGED_FOR_REVIEW',
          last_reconsidered_at_epoch = ?
      WHERE id = ?;
    `);

    updateStmt.run(currentEpoch, factId);

    process.stdout.write(
      `${STANDING_TAG} [OK] Fact [${factId}] entered FLAGGED_FOR_REVIEW by [${reviewerId}] :: "${scrutinyNote}". Cascade suppressed ($0.00).\n`
    );

    return {
      success: true,
      cascadeTriggered: false,
    };
  }

  /**
   * Action 3: Retcon & Deprecate Fact
   * Retires fact to DEPRECATED_ARCHIVED and activates the invalidation cascade.
   */
  public retconAndDeprecate(
    factId: string,
    currentEpoch: number,
    showrunnerId: string,
    retconRationale: string
  ): { success: boolean; cascadeTriggered: boolean; dirtiedScenesCount: number } {
    const existing = this.db
      .prepare('SELECT id, version FROM canonical_facts WHERE id = ?;')
      .get(factId);

    if (!existing) {
      throw new Error(`${STANDING_TAG} [FAIL] Fact '${factId}' does not exist.`);
    }

    // Trigger in migrations/0001_phase0_flat_schema.sql catches this transition
    const updateStmt = this.db.prepare(`
      UPDATE canonical_facts
      SET standing = 'DEPRECATED_ARCHIVED',
          last_reconsidered_at_epoch = ?,
          last_edited_at_epoch = ?
      WHERE id = ?;
    `);

    updateStmt.run(currentEpoch, currentEpoch, factId);

    const dirtiedScenes = this.db
      .prepare('SELECT COUNT(*) AS count FROM scenes WHERE dirty_reason LIKE ?;')
      .get(`%${factId}%`) as { count: number };

    process.stdout.write(
      `${STANDING_TAG} [OK] Fact [${factId}] retired to DEPRECATED_ARCHIVED by [${showrunnerId}]. Invalidation cascade dirtied ${dirtiedScenes.count} scenes :: "${retconRationale}".\n`
    );

    return {
      success: true,
      cascadeTriggered: true,
      dirtiedScenesCount: dirtiedScenes.count,
    };
  }
```

```typescript
// packages/001.lore/src/core/ledger/errataManager.ts
import * as fs from 'node:fs';
import * as path from 'node:path';
import * as crypto from 'node:crypto';
import { z } from 'zod';

export const ERRATA_TAG = '[LORE::ERRATA]' as const;

export const ErrataRecordSchema = z
  .object({
    errata_id: z
      .string()
      .regex(
        /^err_[a-z0-9_.-]+$/,
        `${ERRATA_TAG} [FAIL] errata_id must match pattern ^err_[a-z0-9_.-]+$`
      ),
    superseded_canon_sha: z
      .string()
      .regex(
        /^[0-9a-f]{40}$/,
        `${ERRATA_TAG} [FAIL] superseded_canon_sha must be a 40-character hex commit SHA`
      ),
    active_canon_sha: z
      .string()
      .regex(
        /^[0-9a-f]{40}$/,
        `${ERRATA_TAG} [FAIL] active_canon_sha must be a 40-character hex commit SHA`
      ),
    affected_entity_ids: z
      .array(z.string().min(1))
      .min(1, `${ERRATA_TAG} [FAIL] affected_entity_ids cannot be empty`),
    affected_artifact_shas: z
      .array(z.string().regex(/^[0-9a-f]{64}$/))
      .default([]),
    reason: z
      .string()
      .min(10, `${ERRATA_TAG} [FAIL] Reason must be at least 10 characters long`),
    authorized_by: z.string().min(1),
    created_at_epoch: z.number().int().min(0),
  })
  .strict();

export type ErrataRecord = z.infer<typeof ErrataRecordSchema>;

export class ErrataManager {
  private ledgerFilePath: string;

  constructor(ledgerFilePath: string) {
    this.ledgerFilePath = ledgerFilePath;
  }

  /**
   * Reads and parses all errata records. Returns empty array if file does not exist yet.
   */
  public loadRecords(): ErrataRecord[] {
    if (!fs.existsSync(this.ledgerFilePath)) {
      return [];
    }

    const raw = fs.readFileSync(this.ledgerFilePath, 'utf8').trim();
    if (!raw) return [];

    let parsed: unknown;
    try {
      parsed = JSON.parse(raw);
    } catch (err) {
      throw new Error(
        `${ERRATA_TAG} [FAIL] Malformed errata ledger JSON: ${(err as Error).message}`
      );
    }

    const arraySchema = z.array(ErrataRecordSchema);
    return arraySchema.parse(parsed);
  }

  /**
   * Appends an ErrataRecord to the ledger using an atomic sibling swap.
   * Enforces uniqueness of errata_id and strict forward SHA progression.
   */
  public appendRecord(record: ErrataRecord): void {
    const validated = ErrataRecordSchema.parse(record);
    const existingRecords = this.loadRecords();

    // Check uniqueness
    if (existingRecords.some((r) => r.errata_id === validated.errata_id)) {
      throw new Error(
        `${ERRATA_TAG} [FAIL] Duplicate errata_id detected: '${validated.errata_id}' already exists in ledger.`
      );
    }

    if (validated.superseded_canon_sha === validated.active_canon_sha) {
      throw new Error(
        `${ERRATA_TAG} [FAIL] Cyclic SHA Reference: superseded_canon_sha cannot equal active_canon_sha (${validated.active_canon_sha}).`
      );
    }

    const updatedLedger = [...existingRecords, validated];

    const dir = path.dirname(this.ledgerFilePath);
    if (!fs.existsSync(dir)) {
      fs.mkdirSync(dir, { recursive: true });
    }

    const tempUuid = crypto.randomBytes(8).toString('hex');
    const tempFilePath = `${this.ledgerFilePath}.tmp_${tempUuid}`;

    fs.writeFileSync(
      tempFilePath,
      JSON.stringify(updatedLedger, null, 2) + '\n',
      'utf8'
    );
    fs.renameSync(tempFilePath, this.ledgerFilePath);

    process.stdout.write(
      `${ERRATA_TAG} [OK] Appended ErrataRecord [${validated.errata_id}] targeting [${validated.affected_entity_ids.join(', ')}].\n`
    );
  }
}
```

```typescript
// packages/001.lore/scripts/retire.ts
import * as fs from 'node:fs';
import * as path from 'node:path';
import { Database } from 'node:sqlite';
import { assertByteZeroFrontmatter } from '../schemas/schemas';

export const RETIRE_TAG = '[LORE::RETIRE]' as const;

export interface PreFlightRetireAuditResult {
  canRetireSafely: boolean;
  blockingReferences: string[];
  downstreamScenesToDirty: number;
}

/**
 * Pre-flight audit checking if an entity is actively bound in character equipment slots
 * or active grievances before allowing retirement to proceed.
 */
export function preFlightRetireAudit(
  db: Database,
  entityId: string
): PreFlightRetireAuditResult {
  const blockingReferences: string[] = [];

  // Check if character actively equips this possession
  const equipCheckStmt = db.prepare(`
    SELECT id, claim_text FROM canonical_facts 
    WHERE claim_text LIKE ? AND standing = 'ACTIVE_ENDORSED';
  `);
  const equipHits = equipCheckStmt.all(`%${entityId}%`) as Array<{ id: string; claim_text: string }>;
  for (const hit of equipHits) {
    if (hit.id !== entityId) {
      blockingReferences.push(
        `Referenced by active fact: [${hit.id}] "${hit.claim_text}"`
      );
    }
  }

  // Count downstream scenes that will be dirtied
  const sceneDepsStmt = db.prepare(`
    SELECT COUNT(DISTINCT scene_id) AS count 
    FROM scene_fact_dependencies 
    WHERE fact_id = ? AND is_dirty = 0;
  `);
  const sceneCount = (sceneDepsStmt.get(entityId) as { count: number })?.count ?? 0;

  return {
    canRetireSafely: blockingReferences.length === 0,
    blockingReferences,
    downstreamScenesToDirty: sceneCount,
  };
}

/**
 * Executes guardrailed retirement: modifies Markdown frontmatter in-place to DEPRECATED_ARCHIVED
 * without deleting the file from disk, preserving historical git fold replay.
 */
export function executeEntityRetirement(
  dossierFilePath: string,
  entityId: string,
  retireRationale: string,
  authorId: string
): void {
  if (!fs.existsSync(dossierFilePath)) {
    throw new Error(`${RETIRE_TAG} [FAIL] Dossier file not found: ${dossierFilePath}`);
  }

  const rawContent = fs.readFileSync(dossierFilePath, 'utf8');
  assertByteZeroFrontmatter(rawContent);

  // Parse and update YAML frontmatter status
  const frontmatterMatch = /^---\r?\n([\s\S]*?)\r?\n---/.exec(rawContent);
  if (!frontmatterMatch) {
    throw new Error(`${RETIRE_TAG} [FAIL] Missing YAML frontmatter in ${dossierFilePath}`);
  }

  let yamlBody = frontmatterMatch[1];

  // Update standing or status to DEPRECATED_ARCHIVED
  if (/standing:\s*['"]?[a-zA-Z_]+['"]?/.test(yamlBody)) {
    yamlBody = yamlBody.replace(
      /standing:\s*['"]?[a-zA-Z_]+['"]?/,
      "standing: 'DEPRECATED_ARCHIVED'"
    );
  } else if (/status:\s*['"]?[a-zA-Z_]+['"]?/.test(yamlBody)) {
    yamlBody = yamlBody.replace(
      /status:\s*['"]?[a-zA-Z_]+['"]?/,
      "status: 'DEPRECATED_ARCHIVED'"
    );
  } else {
    yamlBody += `\nstanding: 'DEPRECATED_ARCHIVED'`;
  }

  // Append retirement metadata
  yamlBody += `\nretired_by: '${authorId}'\nretirement_rationale: '${retireRationale}'`;

  const remainder = rawContent.slice(frontmatterMatch[0].length);
  const updatedContent = `---\n${yamlBody}\n---${remainder}`;

  // Atomic sibling write
  const tempPath = `${dossierFilePath}.tmp_retire`;
  fs.writeFileSync(tempPath, updatedContent, 'utf8');
  fs.renameSync(tempPath, dossierFilePath);

  process.stdout.write(
    `${RETIRE_TAG} [OK] Inode preserved. Dossier '${entityId}' marked DEPRECATED_ARCHIVED in ${dossierFilePath}.\n`
  );
}

if (require.main === module) {
  const args = process.argv.slice(2);
  if (args.length < 3) {
    process.stderr.write(
      `${RETIRE_TAG} Usage: npx tsx packages/001.lore/scripts/retire.ts <entity_id> <path/to/dossier.md> "<rationale>"\n`
    );
    process.exit(1);
  }

  const [entityId, dossierPath, rationale] = args;
  const dbPath = process.env.CANON_DB_PATH || ':memory:';
  const db = new Database(dbPath);

  try {
    const audit = preFlightRetireAudit(db, entityId);
    if (!audit.canRetireSafely) {
      process.stderr.write(`${RETIRE_TAG} [FAIL] Pre-Flight Block: Active references detected:\n`);
      for (const ref of audit.blockingReferences) {
        process.stderr.write(`  >> ${ref}\n`);
      }
      process.exit(1);
    }

    executeEntityRetirement(dossierPath, entityId, rationale, 'showrunner_cli');
    process.stdout.write(
      `${RETIRE_TAG} [OK] Retirement complete. Downstream scenes queued for reconciliation: ${audit.downstreamScenesToDirty}\n`
    );
  } catch (err) {
    process.stderr.write(`${(err as Error).message}\n`);
    process.exit(1);
  } finally {
    db.close();
  }
}
```
