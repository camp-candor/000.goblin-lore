import { ErrorSignatureVector } from "../types/dialogueAst";

interface TrieNode {
  children: Map<string, TrieNode>;
  isTerminal: boolean;
  topicId?: string;
  propositionId?: string;
}

export interface ClueDisjointnessCollision {
  collidingToken: string;
  topicId: string;
  propositionId: string;
  errorVector: ErrorSignatureVector;
}

export class PropositionTrie {
  private root: TrieNode = { children: new Map(), isTerminal: false };

  /**
   * Inserts a forbidden trigger lemma into the prefix tree
   */
  public insert(lemma: string, topicId: string, propositionId: string): void {
    let curr = this.root;
    const cleanLemma = lemma.toLowerCase().trim();
    for (const char of cleanLemma) {
      if (!curr.children.has(char)) {
        curr.children.set(char, { children: new Map(), isTerminal: false });
      }
      curr = curr.children.get(char)!;
    }
    curr.isTerminal = true;
    curr.topicId = topicId;
    curr.propositionId = propositionId;
  }

  /**
   * Searches for a matching forbidden prefix in O(length(token))
   */
  public search(token: string): { topicId: string; propositionId: string } | null {
    let curr = this.root;
    const cleanToken = token.toLowerCase().trim();
    for (const char of cleanToken) {
      if (!curr.children.has(char)) {
        return null;
      }
      curr = curr.children.get(char)!;
    }
    return curr.isTerminal && curr.topicId && curr.propositionId
      ? { topicId: curr.topicId, propositionId: curr.propositionId }
      : null;
  }
}

export class TrieClueDisjointnessSolver {
  private trie: PropositionTrie;

  constructor() {
    this.trie = new PropositionTrie();
  }

  /**
   * Populates the search trie with forbidden trigger lemmas
   */
  public loadForbiddenPropositions(
    mysteries: Array<{
      topic_id: string;
      forbidden_propositions: Array<{ id: string; trigger_lemmas: string[] }>;
    }>
  ): void {
    for (const mystery of mysteries) {
      for (const prop of mystery.forbidden_propositions) {
        for (const lemma of prop.trigger_lemmas) {
          this.trie.insert(lemma, mystery.topic_id, prop.id);
        }
      }
    }
  }

  /**
   * Asserts P_clues intersect Union(T_lemmas) === EmptySet in O(L) time
   */
  public evaluateDisjointness(
    permittedClues: string[],
    targetEntityId: string = "system_ontology"
  ): { valid: boolean; collisions: ClueDisjointnessCollision[] } {
    const collisions: ClueDisjointnessCollision[] = [];

    for (const clue of permittedClues) {
      const match = this.trie.search(clue);
      if (match) {
        collisions.push({
          collidingToken: clue,
          topicId: match.topicId,
          propositionId: match.propositionId,
          errorVector: {
            domain: "EPISTEMIC",
            target_entity: targetEntityId,
            rule_id: "CLUE_DISJOINTNESS_COLLISION",
            offending_tokens: [clue, match.topicId, match.propositionId],
          },
        });
      }
    }

    return {
      valid: collisions.length === 0,
      collisions,
    };
  }
}

import * as fs from "node:fs";
import * as path from "node:path";
import { DialogueLineAST, ErrorSignatureVector } from "../types/dialogueAst";

export interface LexicalGateOptions {
  ontologyPath?: string;
}

export interface LexicalAuditResult {
  valid: boolean;
  selfReportedActions: string[];
  extractedProseActions: string[];
  combinedActions: string[];
  bannedViolations: string[];
  errorVector?: ErrorSignatureVector;
}

export class LexicalGate {
  private verbFamilies: Record<string, { capability_required: string; lemmas: string[] }>;
  private abstractWhitelist: Set<string>;
  private lemmaToFamily: Map<string, string>;

  constructor(options: LexicalGateOptions = {}) {
    const defaultPath = path.resolve(__dirname, "../../schemas/ontology/verb_families.json");
    const ontologyRaw = fs.readFileSync(options.ontologyPath || defaultPath, "utf8");
    const parsed = JSON.parse(ontologyRaw);

    this.verbFamilies = parsed.families || {};
    this.abstractWhitelist = new Set((parsed.abstract_dobj_whitelist || []).map((w: string) => w.toLowerCase()));
    this.lemmaToFamily = new Map();

    for (const [familyName, family] of Object.entries(this.verbFamilies)) {
      for (const lemma of family.lemmas) {
        this.lemmaToFamily.set(lemma.toLowerCase(), familyName);
      }
    }
  }

  /**
   * Extracts action verbs from natural language prose while disambiguating
   * polarity negations and abstract direct objects (e.g., "clutching fate")
   */
  public extractProseActions(prose: string): string[] {
    const actions: string[] = [];
    const normalized = prose.toLowerCase().replace(/[^\w\s]/g, " ");
    const words = normalized.split(/\s+/).filter(Boolean);

    for (let i = 0; i < words.length; i++) {
      const word = words[i];

      if (this.lemmaToFamily.has(word)) {
        // Polarity negation check: check if preceded by "not", "couldn't", "never", "cannot"
        const prevWord = i > 0 ? words[i - 1] : "";
        const prevPrevWord = i > 1 ? words[i - 2] : "";
        const isNegated =
          prevWord === "not" ||
          prevWord === "couldn" ||
          prevWord === "cannot" ||
          prevWord === "never" ||
          (prevPrevWord === "could" && prevWord === "not");

        if (isNegated) {
          continue; // Negated action is legally bypassed
        }

        // Direct object abstract disambiguation check: inspect subsequent word
        const nextWord = i + 1 < words.length ? words[i + 1] : "";
        const nextNextWord = i + 2 < words.length ? words[i + 2] : "";

        if (this.abstractWhitelist.has(nextWord) || this.abstractWhitelist.has(nextNextWord)) {
          continue; // Metaphorical/abstract action bypassed
        }

        actions.push(word.toUpperCase());
      }
    }

    return Array.from(new Set(actions));
  }

  /**
   * Enforces Union-Intersection Discrepancy Formulation:
   * CombinedActions = Actions_self_reported UNION Actions_prose_extracted
   * Assert: CombinedActions INTERSECT BannedVerbs(Subject) === EmptySet
   */
  public auditTurn(
    line: DialogueLineAST,
    bannedKineticVerbs: string[],
    localDbQuery?: (entityId: string) => string[]
  ): LexicalAuditResult {
    const effectiveBanned = new Set(
      (localDbQuery ? localDbQuery(line.speaker_id) : bannedKineticVerbs).map((v) => v.toUpperCase())
    );

    const selfReportedActions = line.extracted_motor_actions.map((a) => a.toUpperCase());
    const extractedProseActions = this.extractProseActions(
      `${line.observable_parenthetical} ${line.dialogue_text}`
    );

    // Union calculation
    const combinedActions = Array.from(new Set([...selfReportedActions, ...extractedProseActions]));

    // Intersection with banned verbs
    const bannedViolations = combinedActions.filter((action) => effectiveBanned.has(action));

    if (bannedViolations.length > 0) {
      return {
        valid: false,
        selfReportedActions,
        extractedProseActions,
        combinedActions,
        bannedViolations,
        errorVector: {
          domain: "EPISTEMIC",
          target_entity: line.speaker_id,
          rule_id: "BANNED_KINETIC_MOTOR_BREACH",
          offending_tokens: bannedViolations,
          beat_span: line.id,
        },
      };
    }

    return {
      valid: true,
      selfReportedActions,
      extractedProseActions,
      combinedActions,
      bannedViolations: [],
    };
  }
}
