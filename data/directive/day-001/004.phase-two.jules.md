```typescript
export interface TrieNode {
  children: Map<string, TrieNode>;
  isEndOfWord: boolean;
  topicId?: string;
  lemma?: string;
}

export class TrieClueDisjointnessSolver {
  private root: TrieNode = { children: new Map(), isEndOfWord: false };

  /**
   * Inserts a forbidden trigger lemma into the search Trie.
   */
  public insertForbiddenLemma(lemma: string, topicId: string): void {
    const normalized = lemma.trim().toLowerCase();
    if (!normalized) return;

    let current = this.root;
    for (const char of normalized) {
      if (!current.children.has(char)) {
        current.children.set(char, { children: new Map(), isEndOfWord: false });
      }
      current = current.children.get(char)!;
    }
    current.isEndOfWord = true;
    current.topicId = topicId;
    current.lemma = normalized;
  }

  /**
   * Queries whether a clue token matches any forbidden trigger lemma in O(length(token)) time.
   */
  public searchCollision(clueToken: string): { collision: boolean; topicId?: string; lemma?: string } {
    const normalized = clueToken.trim().toLowerCase();
    let current = this.root;

    for (const char of normalized) {
      if (!current.children.has(char)) {
        return { collision: false };
      }
      current = current.children.get(char)!;
    }

    if (current.isEndOfWord) {
      return {
        collision: true,
        topicId: current.topicId,
        lemma: current.lemma
      };
    }

    return { collision: false };
  }

  /**
   * Enforces the Epistemic Clue Disjointness Invariant:
   * P_clues ∩ (⋃ T_lemmas) == ∅
   */
  public static verifyDisjointness(
    mysteries: Array<{
      topic_id: string;
      permitted_clue_tokens: string[];
      trigger_lemma_sets: string[][];
    }>
  ): { valid: boolean; collisions: Array<{ clue: string; topicId: string; lemma: string }> } {
    const solver = new TrieClueDisjointnessSolver();
    const collisions: Array<{ clue: string; topicId: string; lemma: string }> = [];

    // 1. Populate Trie with all forbidden lemmas across all active mysteries
    for (const mystery of mysteries) {
      for (const lemmaSet of mystery.trigger_lemma_sets) {
        for (const lemma of lemmaSet) {
          solver.insertForbiddenLemma(lemma, mystery.topic_id);
        }
      }
    }

    // 2. Query all permitted clue tokens against the Trie in O(L) time
    for (const mystery of mysteries) {
      for (const clue of mystery.permitted_clue_tokens) {
        const result = solver.searchCollision(clue);
        if (result.collision) {
          collisions.push({
            clue,
            topicId: result.topicId!,
            lemma: result.lemma!
          });
        }
      }
    }

    return {
      valid: collisions.length === 0,
      collisions
    };
  }
}
```

```typescript
import * as fs from 'node:fs';
import * as path from 'node:path';
import { DialogueLineAST, ErrorSignatureVector } from '../types/dialogueAst.js';

export interface VerbFamilyData {
  families: Record<string, { capability_required: string; required_limbs: number; lemmas: string[] }>;
  abstract_dobj_whitelist: string[];
}

export class LexicalGate {
  private ontology: VerbFamilyData;
  private lemmaToFamily: Map<string, string> = new Map();
  private abstractWhitelist: Set<string>;

  constructor(ontologyPath?: string) {
    const resolvedPath = ontologyPath || path.resolve(__dirname, '../../schemas/ontology/verb_families.json');
    const raw = fs.readFileSync(resolvedPath, 'utf8');
    this.ontology = JSON.parse(raw);
    this.abstractWhitelist = new Set(this.ontology.abstract_dobj_whitelist.map(w => w.toLowerCase()));

    for (const [familyKey, family] of Object.entries(this.ontology.families)) {
      for (const lemma of family.lemmas) {
        this.lemmaToFamily.set(lemma.toLowerCase(), familyKey);
      }
    }
  }

  /**
   * Tokenizes and extracts verbs from a natural language prose parenthetical.
   */
  public extractProseVerbs(prose: string): Array<{ verb: string; dobj?: string; isNegated: boolean }> {
    const results: Array<{ verb: string; dobj?: string; isNegated: boolean }> = [];
    const normalized = prose.toLowerCase().replace(/[^a-z0-9_\s]/g, ' ');
    const tokens = normalized.split(/\s+/).filter(t => t.length > 0);

    const negationTokens = new Set(['not', 'never', 'unable', 'cannot', 'couldnt', 'neither', 'hardly']);

    for (let i = 0; i < tokens.length; i++) {
      const token = tokens[i];
      if (this.lemmaToFamily.has(token)) {
        // Lookback window for negation (up to 3 preceding tokens)
        let isNegated = false;
        for (let j = Math.max(0, i - 3); j < i; j++) {
          if (negationTokens.has(tokens[j])) {
            isNegated = true;
            break;
          }
        }

        // Lookahead window for direct object (up to 2 succeeding tokens)
        let dobj: string | undefined = undefined;
        if (i + 1 < tokens.length) {
          const next = tokens[i + 1];
          if (!this.lemmaToFamily.has(next)) {
            dobj = next;
          }
        }

        results.push({ verb: token, dobj, isNegated });
      }
    }

    return results;
  }

  /**
   * Dual-Field Cross-Check: Evaluates CombinedActions ∩ BannedVerbs(Subject)
   */
  public auditDialogueLine(
    line: DialogueLineAST,
    bannedKineticVerbs: string[]
  ): { passed: boolean; errorVector?: ErrorSignatureVector } {
    const bannedSet = new Set(bannedKineticVerbs.map(v => v.toLowerCase()));

    // 1. Evaluate self-reported machine AST actions
    for (const action of line.extracted_motor_actions) {
      const actionNormalized = action.toLowerCase();
      if (bannedSet.has(actionNormalized)) {
        return {
          passed: false,
          errorVector: {
            domain: 'SOMATIC',
            target_entity: line.speaker_id,
            rule_id: 'FATAL_SOMATIC_BREACH',
            offending_tokens: [actionNormalized],
            span: { line_id: line.id, text_snippet: line.dialogue_text }
          }
        };
      }
    }

    // 2. Extract implied prose actions across both camera and observable parentheticals
    const combinedProse = `${line.camera_parenthetical} ${line.observable_parenthetical}`;
    const proseActions = this.extractProseVerbs(combinedProse);

    for (const item of proseActions) {
      if (item.isNegated) {
        // Negation filter: "could not sprint" accurately depicts inability
        continue;
      }

      if (item.dobj && this.abstractWhitelist.has(item.dobj)) {
        // Abstract metaphor filter: "clutched fate" bypassed as non-physical
        continue;
      }

      if (bannedSet.has(item.verb)) {
        return {
          passed: false,
          errorVector: {
            domain: 'SOMATIC',
            target_entity: line.speaker_id,
            rule_id: 'FATAL_SOMATIC_PROSE_LEAK',
            offending_tokens: [item.verb, item.dobj || ''],
            span: { line_id: line.id, text_snippet: combinedProse }
          }
        };
      }
    }

    return { passed: true };
  }
}
```
