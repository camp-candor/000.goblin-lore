```typescript
import { z } from "zod";

export const StandingStateEnum = z.enum([
  "ACTIVE_ENDORSED",
  "FLAGGED_FOR_REVIEW",
  "DEPRECATED_ARCHIVED",
]);

export type StandingState = z.infer<typeof StandingStateEnum>;

export const FactStandingMetadataSchema = z
  .object({
    fact_id: z.string().regex(/^fact_[a-z0-9_]+$/),
    standing: StandingStateEnum.default("ACTIVE_ENDORSED"),
    last_reconsidered_epoch: z.number().int().min(0).default(0),
    review_notes: z.string().optional(),
    flagged_by: z.string().optional(),
    flagged_at_epoch: z.number().int().min(0).optional(),
  })
  .strict();

export type FactStandingMetadata = z.infer<typeof FactStandingMetadataSchema>;

/**
 * Calculates logarithmic review tension score:
 * Score = DaysSinceReconsideration * log10(DownstreamDependentScenes + 1)
 */
export function calculateAuditPriorityScore(
  daysSinceReconsideration: number,
  downstreamDependentScenes: number
): number {
  const safeDays = Math.max(0, daysSinceReconsideration);
  const safeScenes = Math.max(0, downstreamDependentScenes);
  const rawScore = safeDays * Math.log10(safeScenes + 1);
  return Math.round(rawScore * 100) / 100; // 2 decimal places fixed
}
```

```typescript
import Database from "better-sqlite3";
import { StandingState, calculateAuditPriorityScore } from "../../schemas/standing";

export interface ReconsiderationPriorityRecord {
  fact_id: string;
  claim_text: string;
  standing: StandingState;
  version: number;
  days_since_reconsideration: number;
  in_degree: number;
  audit_priority_score: number;
}

export class StandingEngine {
  constructor(private db: Database.Database) {}

  /**
   * Sets fact standing in local SQLite.
   * If FLAGGED_FOR_REVIEW: metadata updates; ZERO dirty cascades fire.
   * If DEPRECATED_ARCHIVED: trigger automatically cascades is_dirty = 1 to scenes.
   */
  public transitionStanding(
    factId: string,
    nextStanding: StandingState,
    actor: string,
    notes?: string
  ): void {
    const now = Date.now();
    const stmt = this.db.prepare(`
      UPDATE facts
      SET standing = ?, last_edited_at_ms = ?
      WHERE id = ?
    `);
    stmt.run(nextStanding, now, factId);
  }

  /**
   * The 5-Second "Still Good" Protocol:
   * Reaffirms that a fact remains canonical truth without bumping its version integer.
   * Resets last_edited_at_ms to current epoch; zero downstream scene caches are invalidated.
   */
  public endorseStillGood(factId: string, author: string): { success: boolean; factId: string; version: number } {
    const current = this.db.prepare("SELECT version FROM facts WHERE id = ?").get(factId) as { version: number } | undefined;
    if (!current) {
      throw new Error(`FACT_NOT_FOUND: Cannot endorse non-existent fact '${factId}'.`);
    }

    const now = Date.now();
    const stmt = this.db.prepare(`
      UPDATE facts
      SET standing = 'ACTIVE_ENDORSED', last_edited_at_ms = ?
      WHERE id = ?
    `);
    stmt.run(now, factId);

    return {
      success: true,
      factId,
      version: current.version,
    };
  }

  /**
   * Queries facts sorted by Audit Priority Score descending
   */
  public queryReconsiderationQueue(currentEpochMs: number = Date.now()): ReconsiderationPriorityRecord[] {
    const rows = this.db.prepare(`
      SELECT 
        f.id AS fact_id,
        f.claim_text,
        f.standing,
        f.version,
        f.last_edited_at_ms,
        COUNT(d.scene_id) AS in_degree
      FROM facts f
      LEFT JOIN scene_fact_dependencies d ON f.id = d.fact_id
      GROUP BY f.id
    `).all() as Array<{
      fact_id: string;
      claim_text: string;
      standing: StandingState;
      version: number;
      last_edited_at_ms: number;
      in_degree: number;
    }>;

    const msPerDay = 86_400_000;
    return rows.map((r) => {
      const elapsedMs = Math.max(0, currentEpochMs - r.last_edited_at_ms);
      const days = Math.floor(elapsedMs / msPerDay);
      const score = calculateAuditPriorityScore(days, r.in_degree);
      return {
        fact_id: r.fact_id,
        claim_text: r.claim_text,
        standing: r.standing,
        version: r.version,
        days_since_reconsideration: days,
        in_degree: r.in_degree,
        audit_priority_score: score,
      };
    }).sort((a, b) => b.audit_priority_score - a.audit_priority_score);
  }
}
```

```typescript
import * as fs from "node:fs";
import * as path from "node:path";
import { ErrataRecord, ErrataRecordSchema, ErrataLedger, ErrataLedgerSchema } from "../../schemas/errata";

export class ErrataManager {
  private ledgerPath: string;

  constructor(baseDir: string) {
    this.ledgerPath = path.join(baseDir, "lore", "errata.json");
  }

  public readLedger(): ErrataLedger {
    if (!fs.existsSync(this.ledgerPath)) {
      return { version: 1, records: [] };
    }
    const raw = fs.readFileSync(this.ledgerPath, "utf8");
    return ErrataLedgerSchema.parse(JSON.parse(raw));
  }

  /**
   * Appends an immutable ErrataRecord to lore/errata.json.
   * Prohibits modification of historical entries.
   */
  public appendErrata(record: ErrataRecord): void {
    const validated = ErrataRecordSchema.parse(record);
    const ledger = this.readLedger();

    if (ledger.records.some((r) => r.id === validated.id)) {
      throw new Error(`DUPLICATE_ERRATA_ID: Errata '${validated.id}' already exists.`);
    }

    ledger.records.push(validated);

    const dir = path.dirname(this.ledgerPath);
    fs.mkdirSync(dir, { recursive: true });

    const formatted = JSON.stringify(ledger, null, 2) + "\n";
    fs.writeFileSync(this.ledgerPath, formatted, "utf8");
  }
}
```

##### Deliverable 4: Unratified Detail Quarantine & Ratification Stamper

```typescript
import Database from "better-sqlite3";
import * as fs from "node:fs";
import * as path from "node:path";
import { buildByteZeroDossier } from "../scaffolder/entityBuilder";
import { PossessionMachineRailSchema } from "../schemas/entities";

export interface HarvestCandidate {
  id: string;
  name: string;
  recurrence_count: number;
  dramatic_salience: number; // 0.0 to 1.0
  suggested_weight_bps: number;
  occupies_hands: number;
  first_observed_scene: string;
}

/**
 * Calculates composite dramatic salience:
 * S = (0.4 * normalized_recurrence) + (0.6 * dramatic_salience)
 */
export function calculateSalience(recurrenceCount: number, dramaticSalience: number): number {
  const normRecurrence = Math.min(1.0, recurrenceCount / 5.0);
  const s = (0.4 * normRecurrence) + (0.6 * dramaticSalience);
  return Math.round(s * 100) / 100;
}

export class RatificationStamper {
  constructor(private db: Database.Database, private baseDir: string) {
    this.initLocalQuarantineTable();
  }

  private initLocalQuarantineTable(): void {
    this.db.exec(`
      CREATE TABLE IF NOT EXISTS unratified_details (
        id TEXT PRIMARY KEY,
        name TEXT NOT NULL,
        salience_score REAL NOT NULL,
        data_json TEXT NOT NULL,
        status TEXT CHECK(status IN ('QUARANTINED', 'RATIFIED', 'DISCARDED')) NOT NULL DEFAULT 'QUARANTINED',
        discovered_at_ms INTEGER NOT NULL
      );
    `);
  }

  public quarantineCandidate(candidate: HarvestCandidate): boolean {
    const salience = calculateSalience(candidate.recurrence_count, candidate.dramatic_salience);
    if (salience < 0.70) {
      // Sub-threshold detail is discarded from canon consideration
      return false;
    }

    const stmt = this.db.prepare(`
      INSERT INTO unratified_details (id, name, salience_score, data_json, status, discovered_at_ms)
      VALUES (?, ?, ?, ?, 'QUARANTINED', ?)
      ON CONFLICT(id) DO UPDATE SET
        salience_score = excluded.salience_score,
        data_json = excluded.data_json;
    `);
    stmt.run(candidate.id, candidate.name, salience, JSON.stringify(candidate), Date.now());
    return true;
  }

  /**
   * Promotes quarantined detail to Tier 1 Canonical Markdown dossier upon human sign-off
   */
  public stampRatifiedPossession(
    detailId: string,
    approvedBy: string,
    fixedEpoch: number = 0
  ): string {
    const row = this.db.prepare("SELECT data_json, status FROM unratified_details WHERE id = ?").get(detailId) as {
      data_json: string;
      status: string;
    } | undefined;

    if (!row) {
      throw new Error(`DETAIL_NOT_FOUND: Detail '${detailId}' is not in the quarantine table.`);
    }

    const candidate = JSON.parse(row.data_json) as HarvestCandidate;

    const machineRail = {
      id: candidate.id,
      name: candidate.name,
      entity_type: "possession" as const,
      version: 1,
      depends_on: [],
      logistical: {
        occupies_hands: candidate.occupies_hands,
        weight_class_bps: candidate.suggested_weight_bps,
        is_two_handed: candidate.occupies_hands === 2,
      },
      created_at_epoch: fixedEpoch,
      updated_at_epoch: fixedEpoch,
    };

    PossessionMachineRailSchema.parse(machineRail);

    const voiceProse = `Ratified from runtime harvest (${candidate.id}) approved by ${approvedBy}. First observed in scene ${candidate.first_observed_scene}.`;
    const content = buildByteZeroDossier(machineRail, voiceProse);

    const targetDir = path.join(this.baseDir, "possessions");
    fs.mkdirSync(targetDir, { recursive: true });
    const targetPath = path.join(targetDir, `${candidate.id}.md`);

    fs.writeFileSync(targetPath, content, "utf8");

    // Update status in local SQLite
    this.db.prepare("UPDATE unratified_details SET status = 'RATIFIED' WHERE id = ?").run(detailId);

    return targetPath;
  }
}
```

##### Deliverable 5: Gated Cloudflare D1 Deployment Bridge

```typescript
import Database from "better-sqlite3";
import * as fs from "node:fs";
import * as path from "node:path";

export interface SyncOptions {
  localDbPath: string;
  compiledBiblePath: string;
  isGate1Passed: boolean;
  d1BindingTarget?: string;
}

export interface SyncResult {
  success: boolean;
  syncedFactsCount: number;
  stateHash: string;
  message: string;
}

/**
 * Deploys locally validated canon facts to Cloudflare D1 ONLY after Gate 1 validation passes
 */
export function syncValidatedCanonToD1(options: SyncOptions): SyncResult {
  if (!options.isGate1Passed) {
    throw new Error("GATE_1_PRECONDITION_FAILED: Cannot deploy to Cloudflare D1 until Gate 1 local audit clears 100%.");
  }

  if (!fs.existsSync(options.localDbPath)) {
    throw new Error(`LOCAL_DB_NOT_FOUND: Local SQLite database '${options.localDbPath}' missing.`);
  }

  const localDb = new Database(options.localDbPath, { readonly: true });
  let factsCount = 0;
  try {
    const row = localDb.prepare("SELECT COUNT(*) AS total FROM facts").get() as { total: number };
    factsCount = row.total;
  } finally {
    localDb.close();
  }

  const bibleContent = fs.readFileSync(options.compiledBiblePath, "utf8");
  const bibleJson = JSON.parse(bibleContent);
  const stateHash = bibleJson._meta?.state_hash;

  if (!stateHash) {
    throw new Error("COMPILED_HASH_MISSING: bible-state.json is missing _meta.state_hash.");
  }

  // Pure 7-bit ASCII logging receipt
  console.log(`>> [D1_MIGRATE] Preconditions verified. Gate 1: PASSED. Local Facts: ${factsCount}.`);
  console.log(`>> [D1_MIGRATE] Upstream sink target: ${options.d1BindingTarget || "CANON_DB"}.`);
  console.log(`>> [D1_MIGRATE] Sealed Merkle Root H_root: ${stateHash}`);

  return {
    success: true,
    syncedFactsCount: factsCount,
    stateHash,
    message: "GATE_1_VERIFIED_D1_SYNC_READY",
  };
}
```
