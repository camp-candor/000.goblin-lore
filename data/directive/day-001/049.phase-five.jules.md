export async function printDetailedCompileReceipt(
  ste: State | undefined,
  seriesSlug: string,
  resBit: any,
  elapsedMs: number

  const divider = '>> ==============================================================================';
  const subDivider = '>> ------------------------------------------------------------------------------';

  await emitConsoleLine(ste, divider);
  await emitConsoleLine(ste, `>> [COMPILE:COMPLETE] Series: [${seriesSlug}] in ${elapsedMs}ms`);

  if (resBit?.validationReport) {

    await emitConsoleLine(
      ste,
      `>> [VALIDATION] Evaluated: ${r.evaluatedFiles} dossiers | Failures: ${r.failures.length} | Warnings: ${r.warnings.length}`
    );
  }

  if (resBit?.rootMerkleHash) {
    await emitConsoleLine(ste, `>> [SEAL:MERKLE_ROOT] H_root: ${resBit.rootMerkleHash}`);
  }

  if (resBit?.canonicalPath) {
    await emitConsoleLine(ste, `>> [SNAPSHOT:CANONICAL] ${resBit.canonicalPath}`);

  if (resBit?.archivePath) {
    await emitConsoleLine(ste, `>> [SNAPSHOT:ARCHIVE]   ${resBit.archivePath}`);
  }

  await emitConsoleLine(ste, `>> [STATUS:FINAL] [SEALED] Bit-identical canon verified and persisted.`);
  await emitConsoleLine(ste, divider);

  await emitConsoleLine(ste, divider);
}

import MenuBit from '../fce/menu.bit';
import State from '../../99.core/state';
import * as ActMnu from '../menu.action';

/**

export const updateMenu = async (cpy: MenuModel, bal: MenuBit, ste: State) => {
  const activeSeries = cpy.activeSeries || '001.goblin-lore';

  const menuOptions = [
    'CHARACTERS',

/**
 * Opens the dedicated Compile Menu Flight Deck.
 */
export const openCompileMenu = async (cpy: MenuModel, bal: MenuBit, ste: State) => {
  const activeSeries = bal.dat?.activeSeries || cpy.activeSeries || '001.goblin-lore';

  await emitConsoleLine(ste, '>> [MENU:COMPILE] Opening Compile Flight Deck for series: ' + activeSeries);

  const optionLabels = COMPILE_MENU_OPTIONS.map((opt) => opt.label);

  if (bal.slv) {
    bal.slv({
      mnuBit: {
        idx: 'open-compile-menu',
        lst: optionLabels,
        dat: {
          activeSeries,
          options: COMPILE_MENU_OPTIONS,
        },
      },
    });
  }

  return cpy;
};

/**
 * Triggers an operation from the compile menu and streams full telemetry.
 */
export const triggerCompileAction = async (cpy: MenuModel, bal: MenuBit, ste: State) => {
  const activeSeries = bal.dat?.activeSeries || cpy.activeSeries || '001.goblin-lore';
  const selectedIndex = bal.val ?? 0;
  const option = COMPILE_MENU_OPTIONS[selectedIndex];

  if (!option) {
    await emitConsoleLine(ste, ':: [MENU:ERR] Invalid compile menu option selected.');
    if (bal.slv) bal.slv({ mnuBit: { idx: 'compile-action-err', val: -1 } });
    return cpy;
  }

  // Return to Lore Menu option
  if (option.key === '<') {
    await emitConsoleLine(ste, '>> [MENU:COMPILE] Returning to Lore Menu...');
    return updateMenu(cpy, bal, ste);
  }

  const startTime = Date.now();
  await emitConsoleLine(ste, `>> [COMPILE:DISPATCH] Triggered: ${option.label}`);
  await emitConsoleLine(ste, `>> [TARGET:SERIES]   slug=${activeSeries} source=corpus/ output=compiled/`);

  let compileResult: any = null;
  let compileError: any = null;

  try {
    // Dispatch to 21.compile.unit via ste.hunt
    await ste.hunt(option.actionType, {
      idx: 'cmp-tui-dispatch',
      src: process.cwd(),
      val: 1, // Verbose telemetry flag
      dat: {
        seriesSlug: activeSeries,
        verifyDriftTripwire: option.key === '5', // Enforce tripwire on Gate 0 option
      },
      slv: (res: any) => {
        if (res.resBit) compileResult = res.resBit;
        if (res.errBit) compileError = res.errBit;
      },
    });

    const elapsedMs = Date.now() - startTime;

    if (compileError) {
      await printCompileFailureReport(ste, activeSeries, compileError);
      cpy.lastError = compileError.message;
    } else {
      await printDetailedCompileReceipt(ste, activeSeries, compileResult, elapsedMs);
      cpy.lastError = undefined;
    }

    if (bal.slv) {
      bal.slv({
        mnuBit: {
          idx: 'trigger-compile-action',
          val: compileError ? -1 : 0,
          dat: {
            action: option.actionType,
            elapsedMs,
            success: !compileError,
            resBit: compileResult,
            errBit: compileError,
          },
        },
      });
    }
  } catch (err: any) {
    const elapsedMs = Date.now() - startTime;
    const errorPayload = { code: 'DISPATCH_EXCEPTION', message: err.message };
    await printCompileFailureReport(ste, activeSeries, errorPayload);

    if (bal.slv) {
      bal.slv({
        mnuBit: {
          idx: 'trigger-compile-action-err',
          val: -1,
          src: err.message,
        },
      });
    }
  }

  return cpy;
};
