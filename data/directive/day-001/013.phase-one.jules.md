import { z } from "zod";

// ---------------------------------------------------------------------------
// 1. Axial Hex Coordinates (Pointy-Top / Doubled Integer Lattice)
// ---------------------------------------------------------------------------
export const AxialCoordinateSchema = z
  .object({
    q: z.number().int(),
    r: z.number().int(),
    s: z.number().int(),
  })
  .strict()
  .refine((coord) => coord.q + coord.r + coord.s === 0, {
    message: "AXIAL_COORDINATE_CUBE_INVARIANT_BREACH: q + r + s must equal 0.",
  });

export type AxialCoordinate = z.infer<typeof AxialCoordinateSchema>;

// ---------------------------------------------------------------------------
// 2. Actor Physical Placement Schema
// ---------------------------------------------------------------------------
export const ActorScaleEnum = z.enum(["STAGED", "IN_TRANSIT"]);

export const ActorPhysicalPlacementSchema = z
  .object({
    actor_id: z.string().regex(/^char_[a-z0-9_]+$/),
    room_id: z.string().regex(/^loc_[a-z0-9_]+$/),
    scale: ActorScaleEnum.default("STAGED"),
    axial_coord: AxialCoordinateSchema.optional(),
    transit_edge_id: z.string().optional(),
    updated_at_tick: z.number().int().min(0).default(0),
  })
  .strict()
  .refine(
    (data) => {
      if (data.scale === "IN_TRANSIT") {
        return !!data.transit_edge_id;
      }
      return true;
    },
    {
      message: "TRANSIT_EDGE_REQUIRED: Actors IN_TRANSIT must specify transit_edge_id.",
      path: ["transit_edge_id"],
    }
  );

export type ActorPhysicalPlacement = z.infer<typeof ActorPhysicalPlacementSchema>;

// ---------------------------------------------------------------------------
// 3. Item Physical Placement Schema (Single Carrier / Conservation of Matter)
// ---------------------------------------------------------------------------
export const ItemCarrierTypeEnum = z.enum([
  "ROOM_SURFACE",
  "CONTAINER_FIXTURE",
  "ACTOR_HELD",
  "ACTOR_WORN",
  "ACTOR_CARRIED",
]);

export const ItemPhysicalPlacementSchema = z
  .object({
    item_id: z.string().regex(/^item_[a-z0-9_]+$/),
    carrier_type: ItemCarrierTypeEnum,
    carrier_id: z.string().min(1), // Can be loc_id, item_id (container), or char_id
    room_id: z.string().regex(/^loc_[a-z0-9_]+$/), // Physical room where item resides
    slot_name: z.string().optional(), // 'held_main', 'held_off', 'torso', etc.
    is_secured: z.boolean().default(true),
    updated_at_tick: z.number().int().min(0).default(0),
  })
  .strict();

export type ItemPhysicalPlacement = z.infer<typeof ItemPhysicalPlacementSchema>;

// ---------------------------------------------------------------------------
// 4. World Physical Manifest & Global Disjointness Invariant
// ---------------------------------------------------------------------------
export const WorldPhysicalManifestSchema = z
  .object({
    tick: z.number().int().min(0).default(0),
    actors: z.record(z.string(), ActorPhysicalPlacementSchema).default({}),
    items: z.record(z.string(), ItemPhysicalPlacementSchema).default({}),
  })
  .strict()
  .refine(
    (manifest) => {
      // Conservation of Mass Check: Item location must match carrying actor's location
      for (const item of Object.values(manifest.items)) {
        if (
          item.carrier_type === "ACTOR_HELD" ||
          item.carrier_type === "ACTOR_WORN" ||
          item.carrier_type === "ACTOR_CARRIED"
        ) {
          const actor = manifest.actors[item.carrier_id];
          if (!actor) {
            return false; // Carried by non-existent actor
          }
          if (actor.room_id !== item.room_id) {
            return false; // Spatial desync: Item claimed by actor in different room
          }
        }
      }
      return true;
    },
    {
      message: "MASS_CONSERVATION_DESYNC: Carried item room_id does not match carrier actor room_id.",
    }
  );

export type WorldPhysicalManifest = z.infer<typeof WorldPhysicalManifestSchema>;

import Database from "better-sqlite3";
import {
  WorldPhysicalManifest,
  WorldPhysicalManifestSchema,
  ItemPhysicalPlacement,
} from "../schemas/physicalPlacement";

export interface CoPresenceEvaluation {
  satisfiable: boolean;
  actorRoomId: string;
  targetRoomId: string;
  errorCode?: "SPATIAL_CO_PRESENCE_VIOLATION" | "MASS_CONSERVATION_BREACH" | "ENTITY_NOT_FOUND";
  reason?: string;
}

export class SpatialPresenceSolver {
  /**
   * Evaluates topological co-presence between an actor and a target entity in O(1) time
   */
  public static evaluateCoPresence(
    manifest: WorldPhysicalManifest,
    actorId: string,
    targetEntityId: string
  ): CoPresenceEvaluation {
    const actor = manifest.actors[actorId];
    if (!actor) {
      return {
        satisfiable: false,
        actorRoomId: "UNKNOWN",
        targetRoomId: "UNKNOWN",
        errorCode: "ENTITY_NOT_FOUND",
        reason: `ACTOR_NOT_FOUND: Actor '${actorId}' does not exist in spatial manifest.`,
      };
    }

    let targetRoomId: string | undefined;

    if (targetEntityId.startsWith("char_")) {
      const targetActor = manifest.actors[targetEntityId];
      targetRoomId = targetActor ? targetActor.room_id : undefined;
    } else if (targetEntityId.startsWith("item_")) {
      const targetItem = manifest.items[targetEntityId];
      targetRoomId = targetItem ? targetItem.room_id : undefined;
    } else if (targetEntityId.startsWith("loc_")) {
      targetRoomId = targetEntityId;
    }

    if (!targetRoomId) {
      return {
        satisfiable: false,
        actorRoomId: actor.room_id,
        targetRoomId: "UNKNOWN",
        errorCode: "ENTITY_NOT_FOUND",
        reason: `TARGET_NOT_FOUND: Target entity '${targetEntityId}' does not exist in spatial manifest.`,
      };
    }

    if (actor.room_id !== targetRoomId) {
      return {
        satisfiable: false,
        actorRoomId: actor.room_id,
        targetRoomId,
        errorCode: "SPATIAL_CO_PRESENCE_VIOLATION",
        reason: `TOPOLOGICAL_DISCONNECT: Actor '${actorId}' in room '${actor.room_id}' cannot interact with target '${targetEntityId}' in room '${targetRoomId}'.`,
      };
    }

    return {
      satisfiable: true,
      actorRoomId: actor.room_id,
      targetRoomId,
    };
  }

  /**
   * Asserts Absolute Item Disjointness across a collection of item placement proposals
   */
  public static assertItemDisjointness(items: ItemPhysicalPlacement[]): {
    valid: boolean;
    duplicateItemIds: string[];
  } {
    const seen = new Set<string>();
    const duplicates: string[] = [];

    for (const item of items) {
      if (seen.has(item.item_id)) {
        duplicates.push(item.item_id);
      }
      seen.add(item.item_id);
    }

    return {
      valid: duplicates.length === 0,
      duplicateItemIds: duplicates,
    };
  }

  /**
   * Verifies spatial state against a local SQLite database
   */
  public static verifyLocalSqliteParity(
    db: Database.Database,
    manifest: WorldPhysicalManifest
  ): boolean {
    WorldPhysicalManifestSchema.parse(manifest);

    db.exec(`
      CREATE TABLE IF NOT EXISTS spatial_actors (
        actor_id TEXT PRIMARY KEY,
        room_id TEXT NOT NULL,
        scale TEXT NOT NULL,
        updated_at_tick INTEGER NOT NULL
      );

      CREATE TABLE IF NOT EXISTS spatial_items (
        item_id TEXT PRIMARY KEY,
        carrier_type TEXT NOT NULL,
        carrier_id TEXT NOT NULL,
        room_id TEXT NOT NULL,
        updated_at_tick INTEGER NOT NULL
      );
    `);

    const insertActor = db.prepare(`
      INSERT OR REPLACE INTO spatial_actors (actor_id, room_id, scale, updated_at_tick)
      VALUES (?, ?, ?, ?)
    `);

    const insertItem = db.prepare(`
      INSERT OR REPLACE INTO spatial_items (item_id, carrier_type, carrier_id, room_id, updated_at_tick)
      VALUES (?, ?, ?, ?, ?)
    `);

    const tx = db.transaction(() => {
      for (const actor of Object.values(manifest.actors)) {
        insertActor.run(actor.actor_id, actor.room_id, actor.scale, actor.updated_at_tick);
      }
      for (const item of Object.values(manifest.items)) {
        insertItem.run(item.item_id, item.carrier_type, item.carrier_id, item.room_id, item.updated_at_tick);
      }
    });

    tx();
    return true;
  }
}
