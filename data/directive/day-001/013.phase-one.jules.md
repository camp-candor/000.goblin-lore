 * Byte-0 Frontmatter Invariant Verifier
 * Asserts that raw dossier buffers begin strictly at byte index 0 with "---\n" or "---\r\n"
 * and contain zero UTF-8 BOM (0xEF, 0xBB, 0xBF) or leading whitespace.
 */
export function assertByteZeroFrontmatter(rawBuffer: Buffer | Uint8Array | string): void {
  const bytes =
    typeof rawBuffer === 'string' ? Buffer.from(rawBuffer, 'utf8') : Buffer.from(rawBuffer);

  if (bytes.length >= 3 && bytes[0] === 0xef && bytes[1] === 0xbb && bytes[2] === 0xbf) {
    throw new Error(
      `${LORE_SCHEMA_TAG} [FAIL] Byte-0 Violation :: UTF-8 BOM detected at byte 0. Dossiers must start with 0x2D 0x2D 0x2D.`
    );
  }

  if (bytes.length < 4 || bytes[0] !== 0x2d || bytes[1] !== 0x2d || bytes[2] !== 0x2d) {
    throw new Error(
      `${LORE_SCHEMA_TAG} [FAIL] Byte-0 Violation :: YAML frontmatter must begin strictly at byte index 0 with '---'.`
    );

  const fourthByte = bytes[3];
  const isLf = fourthByte === 0x0a;

      `${LORE_SCHEMA_TAG} [FAIL] Byte-0 Violation :: Opening '---' must be immediately followed by LF (0x0A) or CRLF (0x0D 0x0A).`

  if (oldFact.id !== newFact.id) {
    throw new Error(
      `${LORE_SCHEMA_TAG} [FAIL] Identity Mismatch :: Cannot transition between distinct fact IDs (${oldFact.id} != ${newFact.id}).`
    );
  }

  // Enforce Monotonic Authority Principle
  if (
    AUTHORITY_RANK[actorAuthority] < AUTHORITY_RANK[oldFact.authority_level] ||
    actorAuthority === 'PROJECTION' ||
    actorAuthority === 'SPECULATION'
  ) {
    throw new Error(
      `${LORE_SCHEMA_TAG} [FAIL] Authority Boundary Breach :: Actor with authority '${actorAuthority}' cannot mutate fact governed by '${oldFact.authority_level}'.`
    );
  }

  const contentMutated =
    oldFact.claim_text !== newFact.claim_text ||
    oldFact.scope !== newFact.scope ||
    oldFact.content_hash !== newFact.content_hash ||
    oldFact.severity_bps !== newFact.severity_bps;

  const standingDeprecated =
    oldFact.standing !== 'DEPRECATED_ARCHIVED' &&
    newFact.standing === 'DEPRECATED_ARCHIVED';

  if (contentMutated) {
    if (newFact.version !== oldFact.version + 1) {
      throw new Error(
        `${LORE_SCHEMA_TAG} [FAIL] Monotonic Version Violation :: Fact '${oldFact.id}' mutated content, so version must increment from ${oldFact.version} to ${oldFact.version + 1} (received ${newFact.version}).`
      );
    }
    return { versionIncremented: true, requiresCascade: true };
  }

  // "Still Good" / FLAGGED_FOR_REVIEW upward scrutiny path: zero version bump, zero cascade
  if (newFact.version < oldFact.version) {
    throw new Error(
      `${LORE_SCHEMA_TAG} [FAIL] Monotonic Regression :: Fact '${oldFact.id}' version regressed from ${oldFact.version} to ${newFact.version}.`
    );
  }

  return {
    versionIncremented: newFact.version > oldFact.version,
    requiresCascade: newFact.version > oldFact.version || standingDeprecated,
  };
}

      scene.status === 'CLEAN' ||
      scene.status === 'REVIEWED' ||
      scene.status === 'REVIEWED_NO_CHANGE';

      scene.depends_on.length === 0 &&

        message: `${DEP_SCHEMA_TAG} [FAIL] Status Consistency Violation :: Scene '${scene.id}' has status='DIRTY' but is_dirty=false`,

        code: z.ZodIssueCode.custom,
        message: `${DEP_SCHEMA_TAG} [FAIL] Status Consistency Violation :: Scene '${scene.id}' cannot have status='${scene.status}' while is_dirty=true`,

      });

/**
 * Deterministic Staleness Predicate Evaluator
 * Evaluates: Exists d in Scene.depends_on | d.pinned_version < Fact.version OR Fact.standing == 'DEPRECATED_ARCHIVED'
 */
export interface StalenessEvaluationResult {
  scene_id: string;
  is_dirty: boolean;
  status: SceneLifecycleStatus;
  dirty_reason: string | null;
  updated_edges: DependencyEdge[];
}

export function evaluateSceneStaleness(
  scene: SceneDependencyContract,
  activeFactsMap: ReadonlyMap<string, FactNode>
): StalenessEvaluationResult {
  const reasons: string[] = [];
  const updatedEdges: DependencyEdge[] = [];

  for (const edge of scene.depends_on) {
    const activeFact = activeFactsMap.get(edge.fact_id);
    if (!activeFact) {
      const reason = `Dangling Fact Reference: [${edge.fact_id}] is missing from active facts ledger`;
      reasons.push(reason);
      updatedEdges.push({
        ...edge,
        is_dirty: true,
        invalidation_reason: reason,
      });
      continue;
    }

    if (activeFact.standing === 'DEPRECATED_ARCHIVED') {
      const reason = `Fact [${activeFact.id}] retired to DEPRECATED_ARCHIVED`;
      reasons.push(reason);
      updatedEdges.push({
        ...edge,
        is_dirty: true,
        invalidation_reason: reason,
      });
      continue;
    }

    // Note: FLAGGED_FOR_REVIEW intentionally triggers ZERO dirty cascade (Zero-Cascade Liminal Guarantee)
    if (edge.pinned_version < activeFact.version) {
      const reason = `Fact [${activeFact.id}] mutated v${edge.pinned_version}->v${activeFact.version}: ${activeFact.claim_text}`;
      reasons.push(reason);
      updatedEdges.push({
        ...edge,
        is_dirty: true,
        invalidation_reason: reason,
      });
      continue;
    }

    updatedEdges.push({
      ...edge,
      is_dirty: false,
      invalidation_reason: undefined,
    });
  }

  const isDirty = reasons.length > 0;
  return {
    scene_id: scene.id,
    is_dirty: isDirty,
    status: isDirty ? 'DIRTY' : scene.status,
    dirty_reason: isDirty ? reasons.join(' | ') : null,
    updated_edges: updatedEdges,
  };
}

import * as fs from 'node:fs';
import * as path from 'node:path';

export const DETERMINISM_TAG = '[LORE::DETERMINISM]' as const;

export interface DeterminismViolation {
  filePath: string;
  line: number;
  ruleId:
    | 'BAN_WALL_CLOCK'
    | 'BAN_MATH_RANDOM'
    | 'BAN_PROCESS_ENV'
    | 'BAN_NETWORK_IO'
    | 'BAN_MTIME_STAT'
    | 'BAN_NON_ASCII';
  snippet: string;
  message: string;
}

interface ForbiddenPatternRule {
  ruleId: DeterminismViolation['ruleId'];
  regex: RegExp;
  message: string;
}

const FORBIDDEN_RULES: readonly ForbiddenPatternRule[] = [
  {
    ruleId: 'BAN_WALL_CLOCK',
    regex: /\b(?:Date\.now\s*\(|new\s+Date\s*\(|performance\.now\s*\()/g,
    message:
      'Ambient wall-clock calls (Date.now, new Date, performance.now) are banned. Use monotonic integer versions or logical epochs.',
  },

    regex: /\bMath\.random\s*\(/g,
    message:
      'Unseeded Math.random() is banned in canonical schemas and compilers.',

    message:

  },
  {

  },
  {

    regex: /\.(?:mtime|mtimeMs|ctime|ctimeMs|birthtime|birthtimeMs)\b/g,
    message:
      'Filesystem modification timestamps (mtime/ctime) are banned for staleness evaluation. Use monotonic integer versions.',
  },
  {
    ruleId: 'BAN_NON_ASCII',
    regex: /[^\x00-\x7F]/g,
    message:
      'Non-ASCII characters or multi-byte UTF-8 emojis detected. Enforce pure 7-bit ASCII telemetry.',
  },
];

/**
 * Strips single-line and multi-line comments for code token scanning while preserving line numbers,
 * EXCEPT for the 7-bit ASCII check which scans every raw byte including comments.
 */
function stripCommentsPreservingLines(source: string): string {
  return source
    .replace(/\/\*[\s\S]*?\*\//g, (match) => match.replace(/[^\n]/g, ' '))
    .replace(/\/\/.*$/gm, '');
}

/**
 * Scans a single TypeScript/JavaScript source string for determinism and 7-bit ASCII violations.
 */
export function scanSourceForDeterminismViolations(
  sourceText: string,
  filePath = 'inline.ts'
): DeterminismViolation[] {
  const violations: DeterminismViolation[] = [];
  const rawLines = sourceText.split(/\r?\n/);
  const codeOnlyText = stripCommentsPreservingLines(sourceText);
  const codeLines = codeOnlyText.split(/\r?\n/);

  for (let i = 0; i < rawLines.length; i++) {
    const lineNumber = i + 1;
    const rawLine = rawLines[i];
    const codeLine = codeLines[i] ?? '';

    for (const rule of FORBIDDEN_RULES) {
      const targetLine = rule.ruleId === 'BAN_NON_ASCII' ? rawLine : codeLine;
      rule.regex.lastIndex = 0;
      const match = rule.regex.exec(targetLine);
      if (match) {
        violations.push({
          filePath,
          line: lineNumber,
          ruleId: rule.ruleId,
          snippet: rawLine.trim(),
          message: `${DETERMINISM_TAG} [FAIL] [${rule.ruleId}] ${filePath}:${lineNumber} :: ${rule.message}`,
        });
      }
    }
  }

  return violations;
}

/**
 * Recursively collects .ts and .js files in a directory, excluding node_modules, dist, and this linter itself.
 */
export function collectTargetFiles(dirPath: string, results: string[] = []): string[] {
  if (!fs.existsSync(dirPath)) {
    return results;
  }

  const entries = fs.readdirSync(dirPath, { withFileTypes: true });
  for (const entry of entries) {
    if (entry.name === 'node_modules' || entry.name === 'dist' || entry.name === '.git') {
      continue;
    }
    const fullPath = path.join(dirPath, entry.name);
    if (entry.isDirectory()) {
      collectTargetFiles(fullPath, results);
    } else if (
      entry.isFile() &&
      (entry.name.endsWith('.ts') || entry.name.endsWith('.js')) &&
      entry.name !== 'lint-determinism.ts' &&
      !entry.name.endsWith('.test.ts')
    ) {
      results.push(fullPath);
    }
  }
  return results;
}

/**
 * Audits an entire package directory and throws on any determinism breach.
 */
export function auditDirectoryDeterminism(rootDir: string): DeterminismViolation[] {
  const files = collectTargetFiles(rootDir);
  const allViolations: DeterminismViolation[] = [];

  for (const file of files) {
    const content = fs.readFileSync(file, 'utf8');
    const fileViolations = scanSourceForDeterminismViolations(content, file);
    allViolations.push(...fileViolations);
  }

  return allViolations;
}

if (require.main === module) {
  const targetRoot = path.resolve(__dirname, '..');
  const violations = auditDirectoryDeterminism(targetRoot);
  if (violations.length > 0) {
    for (const v of violations) {
      process.stderr.write(`${v.message}\n  >> ${v.snippet}\n`);
    }
    process.exit(1);
  }
  process.stdout.write(
    `${DETERMINISM_TAG} [OK] Zero volatile contamination or non-ASCII violations detected.\n`
  );
}
