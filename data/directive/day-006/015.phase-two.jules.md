#### 1. Domain Entity Selector Bit Contract: `packages/299.lore/98.menu.unit/fce/domainSelector.bit.ts`

```typescript
export type DomainEntityType =
  | 'character'
  | 'location'
  | 'possession'
  | 'grievance'
  | 'world';

export interface DomainEntityDescriptor {
  readonly domain: DomainEntityType;
  readonly label: string;
  readonly idPrefix: 'char_' | 'loc_' | 'item_' | 'grv_' | 'law_';
  readonly directoryName: 'characters' | 'locations' | 'possessions' | 'grievances' | 'world';
}

export const DOMAIN_ENTITY_REGISTRY: Readonly<Record<DomainEntityType, DomainEntityDescriptor>> =
  Object.freeze({
    character: Object.freeze({
      domain: 'character',
      label: '[1] CHARACTER DOSSIER (char_)',
      idPrefix: 'char_',
      directoryName: 'characters',
    }),
    location: Object.freeze({
      domain: 'location',
      label: '[2] LOCATION TOPOLOGY (loc_)',
      idPrefix: 'loc_',
      directoryName: 'locations',
    }),
    possession: Object.freeze({
      domain: 'possession',
      label: '[3] POSSESSION ARTIFACT (item_)',
      idPrefix: 'item_',
      directoryName: 'possessions',
    }),
    grievance: Object.freeze({
      domain: 'grievance',
      label: '[4] GRIEVANCE VECTOR (grv_)',
      idPrefix: 'grv_',
      directoryName: 'grievances',
    }),
    world: Object.freeze({
      domain: 'world',
      label: '[5] WORLD AXIOM LAW (law_)',
      idPrefix: 'law_',
      directoryName: 'world',
    }),
  });

export const DOMAIN_ENTITY_CHOICES: readonly string[] = Object.freeze([
  DOMAIN_ENTITY_REGISTRY.character.label,
  DOMAIN_ENTITY_REGISTRY.location.label,
  DOMAIN_ENTITY_REGISTRY.possession.label,
  DOMAIN_ENTITY_REGISTRY.grievance.label,
  DOMAIN_ENTITY_REGISTRY.world.label,
]);

export function resolveDomainEntityDescriptor(selection: string): DomainEntityDescriptor | null {
  const normalized = String(selection || '').trim();
  for (const descriptor of Object.values(DOMAIN_ENTITY_REGISTRY)) {
    if (
      normalized === descriptor.label ||
      normalized.toLowerCase() === descriptor.domain
    ) {
      return descriptor;
    }
  }
  return null;
}

export default interface DomainSelectorBit {
  readonly idx: string;
  readonly src?: string;
  readonly val?: number;
  readonly dat?: {
    readonly defaultDomain?: DomainEntityType;
    readonly descriptor?: DomainEntityDescriptor;
  };
  readonly slv?: Function;
}
```

#### 2. Domain Entity Type Selector Buzzer: `packages/299.lore/98.menu.unit/buz/domainEntitySelector.buzz.ts`

```typescript
import type State from '../../99.core/state.js';
import type DomainSelectorBit from '../fce/domainSelector.bit.js';
import {
  DOMAIN_ENTITY_CHOICES,
  DOMAIN_ENTITY_REGISTRY,
  resolveDomainEntityDescriptor,
  type DomainEntityType,
} from '../fce/domainSelector.bit.js';

const UPDATE_GRID = '[Grid action] Update Grid';
const OPEN_CHOICE = '[Open action] Open Choice';
const UPDATE_CONSOLE = '[Console action] Update Console';
const CANCEL_LABEL = '[<] CANCEL DOMAIN SELECTION';

function resolveLibraryBus(): any {
  return (globalThis as any).LIBRARY || (global as any).LIBRARY || null;
}

function emitConsole(ste: State | undefined, line: string): void {
  const cleanLine = String(line || '').replace(/[^\x20-\x7E]/g, '').trim();
  const lib = resolveLibraryBus();
  if (lib && typeof lib.hunt === 'function') {
    lib.hunt(UPDATE_CONSOLE, { idx: 'cns00', src: cleanLine });
    return;
  }
  if (ste && typeof ste.hunt === 'function') {
    ste.hunt(UPDATE_CONSOLE, { idx: 'cns00', src: cleanLine });
  }
}

export const promptDomainEntityTypeSelector = async (
  cpy: any,
  bal: DomainSelectorBit,
  ste?: State,
): Promise<any> => {
  const lib = resolveLibraryBus();
  const defaultDomain: DomainEntityType = bal?.dat?.defaultDomain || 'character';
  const defaultDescriptor = DOMAIN_ENTITY_REGISTRY[defaultDomain];

  emitConsole(ste, '>> [DOMAIN:SELECTOR] Opening canonical entity domain selector...');

  let selectedRaw = bal?.src || defaultDescriptor.label;

  if (!bal?.src && lib && typeof lib.hunt === 'function') {
    const gridBit: any = await lib.hunt(UPDATE_GRID, {
      x: 0,
      y: 4,
      xSpan: 4,
      ySpan: 8,
    });
    const choiceBit: any = await lib.hunt(OPEN_CHOICE, {
      dat: { clr0: 'black', clr1: 'cyan' },
      src: 'vertical',
      lst: [...DOMAIN_ENTITY_CHOICES, CANCEL_LABEL],
      net: gridBit?.grdBit?.dat,
    });

    if (choiceBit?.chcBit?.val === -1 || choiceBit?.chcBit?.src === CANCEL_LABEL) {
      emitConsole(ste, '>> [DOMAIN:SELECTOR] [WARN] Entity domain selection cancelled by operator.');
      if (bal?.slv) {
        bal.slv({
          mnuBit: {
            idx: 'domain-selector-cancelled',
            val: 0,
          },
        });
      }
      return cpy;
    }

    selectedRaw = choiceBit?.chcBit?.src || defaultDescriptor.label;
  }

  const resolved = resolveDomainEntityDescriptor(selectedRaw);

  if (!resolved) {
    emitConsole(ste, `:: [DOMAIN:SELECTOR] [FAIL] Invalid entity domain selection: ${selectedRaw}`);
    if (bal?.slv) {
      bal.slv({
        mnuBit: {
          idx: 'domain-selector-invalid',
          val: -1,
          src: selectedRaw,
        },
      });
    }
    return cpy;
  }

  cpy.activeEntityDomain = resolved.domain;
  cpy.activeEntityPrefix = resolved.idPrefix;
  cpy.activeEntityDirectory = resolved.directoryName;
  cpy.statusMessage = `Domain Locked: ${resolved.domain} (${resolved.idPrefix})`;

  emitConsole(
    ste,
    `>> [DOMAIN:SELECTOR] [OK] Bound domain '${resolved.domain}' (prefix: ${resolved.idPrefix}, dir: ${resolved.directoryName}/)`,
  );

  if (bal?.slv) {
    bal.slv({
      mnuBit: {
        idx: 'domain-selector-success',
        val: 1,
        src: resolved.domain,
        dat: { descriptor: resolved },
      },
    });
  }

  return cpy;
};
```
