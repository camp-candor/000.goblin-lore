export const injectCustomSeedIdentifiers = async (
  cpy: any,
  bal: CustomSeedBit,
  _ste?: State,
): Promise<any> => {
  const dat = bal.dat || {};
  const seriesSlug = bal.src || dat.seriesSlug;
  const customIds = dat.customIdentifiers || {};

  if (!seriesSlug) {
    if (bal.slv) {
      bal.slv({ seedBit: { idx: bal.idx, val: 0, dat: { error: 'MISSING_SERIES_SLUG' } } });

    return cpy;

  const itemId = customIds.possessionId || 'item_lasso_of_truth';
  const grvId = customIds.grievanceId || 'grv_gotham_standoff';

  fs.writeFileSync(locPath, locContent, { encoding: 'utf8' });

  const itemPath = path.join(possDir, `${itemId}.md`);
  const itemContent = [
    '---',
    `id: "${itemId}"`,
    'name: "Lasso of Truth"',
    'mass_grams: 1200',
    'weight_class: "weight"',
    'slot_category: "held"',

  fs.writeFileSync(itemPath, itemContent, { encoding: 'utf8' });

    `id: "${grvId}"`,
