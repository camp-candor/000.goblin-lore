#### 1. Custom Seed Bit Contract: `packages/299.lore/27.seed.unit/fce/customSeed.bit.ts`

```typescript
export interface CustomSeedIdentifiers {
  readonly characterId?: string;
  readonly locationId?: string;
  readonly possessionId?: string;
  readonly grievanceId?: string;
}

export interface CustomSeedConfigBit {
  readonly seriesSlug: string;
  readonly customIdentifiers?: CustomSeedIdentifiers;
  readonly repoRoot?: string;
}

export interface CustomSeedResultBit {
  readonly seriesSlug: string;
  readonly injectedIds: CustomSeedIdentifiers;
  readonly totalInjected: number;
  readonly byteZeroVerified: boolean;
}

export default interface CustomSeedBit {
  readonly idx: string;
  readonly src?: string;
  readonly val?: number;
  readonly dat?: CustomSeedConfigBit;
  readonly slv?: Function;
}
```

#### 2. Custom Seed Injector Buzzer: `packages/299.lore/27.seed.unit/buz/customSeed.buzz.ts`

```typescript
import * as fs from 'node:fs';
import * as path from 'node:path';
import * as crypto from 'node:crypto';
import type State from '../../99.core/state.js';
import type CustomSeedBit from '../fce/customSeed.bit.js';
import { resolveRepoRoot } from '../../29.series.unit/buz/scaffoldSeriesTree.buzz.js';
import { verifyByteZeroBuffer } from './seedWorldDefinitions.buzz.js';

export const injectCustomSeedIdentifiers = async (
  cpy: any,
  bal: CustomSeedBit,
  _ste?: State,
): Promise<any> => {
  const dat = bal.dat || {};
  const seriesSlug = bal.src || dat.seriesSlug;
  const customIds = dat.customIdentifiers || {};

  if (!seriesSlug) {
    if (bal.slv) {
      bal.slv({ seedBit: { idx: bal.idx, val: 0, dat: { error: 'MISSING_SERIES_SLUG' } } });
    }
    return cpy;
  }

  const repoRoot = dat.repoRoot || resolveRepoRoot();
  const seriesDir = path.join(repoRoot, 'series', seriesSlug);

  const charId = customIds.characterId || 'char_wonder_woman';
  const locId = customIds.locationId || 'loc_gotham_rooftop';
  const itemId = customIds.possessionId || 'item_lasso_of_truth';
  const grvId = customIds.grievanceId || 'grv_gotham_standoff';

  const charsDir = path.join(seriesDir, 'characters');
  const locsDir = path.join(seriesDir, 'locations');
  const possDir = path.join(seriesDir, 'possessions');
  const grvDir = path.join(seriesDir, 'grievances');

  for (const d of [charsDir, locsDir, possDir, grvDir]) {
    if (!fs.existsSync(d)) {
      fs.mkdirSync(d, { recursive: true });
    }
  }

  const charPath = path.join(charsDir, `${charId}.md`);
  const charContent = [
    '---',
    `id: "${charId}"`,
    'title: "Wonder Woman"',
    'available_arms_bps: 20000',
    'locomotion_baseline: "bipedal_standard"',
    `held_primary: "${itemId}"`,
    'tags:',
    '  - "protagonist"',
    '  - "hero"',
    '---',
    '',
    '## Wants (surface)',
    'To preserve civilian order without compromising absolute ethical boundaries.',
    '',
    '## Wants (actual)',
    'To reconcile the chasm between divine mandate and flawed human chaos.',
    '',
  ].join('\n');
  fs.writeFileSync(charPath, charContent, { encoding: 'utf8' });

  const locPath = path.join(locsDir, `${locId}.md`);
  const locContent = [
    '---',
    `id: "${locId}"`,
    'title: "Gotham Rooftop"',
    'axial_coords:',
    '  q: 0',
    '  r: 0',
    '  s: 0',
    'max_adult_occupants: 4',
    'acoustic_damping_cdb: 1200',
    'tags:',
    '  - "location"',
    '  - "exterior"',
    '---',
    '',
    '## Overview',
    'Cold limestone gargoyles overlook the endless neon canyons of Gotham City.',
    '',
  ].join('\n');
  fs.writeFileSync(locPath, locContent, { encoding: 'utf8' });

  const itemPath = path.join(possDir, `${itemId}.md`);
  const itemContent = [
    '---',
    `id: "${itemId}"`,
    'name: "Lasso of Truth"',
    'mass_grams: 1200',
    'weight_class: "weight"',
    'slot_category: "held"',
    'hands_required_bps: 10000',
    `custody_holder: "${charId}"`,
    'tags:',
    '  - "artifact"',
    '  - "weapon"',
    '---',
    '',
    '## Description',
    'A golden cord forged from divine metallurgy that compels absolute veracity.',
    '',
  ].join('\n');
  fs.writeFileSync(itemPath, itemContent, { encoding: 'utf8' });

  const grvPath = path.join(grvDir, `${grvId}.md`);
  const grvContent = [
    '---',
    `id: "${grvId}"`,
    'title: "Gotham Standoff"',
    `participants_primary: ["${charId}"]`,
    `location_anchor: "${locId}"`,
    `physical_anchor_prop: "${itemId}"`,
    'intensity_bounds_bps: [3000, 9000]',
    'cooldown_ticks: 5',
    'tags:',
    '  - "grievance"',
    '  - "conflict"',
    '---',
    '',
    '## Surface Argument',
    'Jurisdiction and custody over the detonator amidst torrential rain.',
    '',
  ].join('\n');
  fs.writeFileSync(grvPath, grvContent, { encoding: 'utf8' });

  const allBytes = Buffer.concat([
    fs.readFileSync(charPath),
    fs.readFileSync(locPath),
    fs.readFileSync(itemPath),
    fs.readFileSync(grvPath),
  ]);
  const byteZeroVerified = verifyByteZeroBuffer(allBytes);

  const receipt = {
    seriesSlug,
    injectedIds: {
      characterId: charId,
      locationId: locId,
      possessionId: itemId,
      grievanceId: grvId,
    },
    totalInjected: 4,
    byteZeroVerified,
  };

  if (bal.slv) {
    bal.slv({
      seedBit: {
        idx: bal.idx,
        val: 1,
        dat: receipt,
      },
    });
  }

  return cpy;
};
```

#### 3. Unit Verification Suite: `packages/299.lore/27.seed.unit/test/customSeed.test.ts`

```typescript
import { describe, it, expect, afterEach } from 'vitest';
import fs from 'node:fs';
import path from 'node:path';
import os from 'node:os';
import { injectCustomSeedIdentifiers } from '../buz/customSeed.buzz.js';
import { SeedModel } from '../seed.model.js';

describe('299.lore - [Feature 1.5] Custom Starter Seed Identifier Injection', () => {
  let tempRepoRoot: string;

  afterEach(() => {
    if (tempRepoRoot && fs.existsSync(tempRepoRoot)) {
      fs.rmSync(tempRepoRoot, { recursive: true, force: true });
    }
  });

  it('injects Gotham-specific custom starter seed IDs while maintaining Byte-0 frontmatter', async () => {
    tempRepoRoot = fs.mkdtempSync(path.join(os.tmpdir(), 'custom-seed-'));
    const seriesSlug = 'gotham-trinity-crisis';

    const model = new SeedModel();
    let resolved: any = null;

    const bal: any = {
      idx: 'test-custom-seed',
      src: seriesSlug,
      dat: {
        seriesSlug,
        repoRoot: tempRepoRoot,
        customIdentifiers: {
          characterId: 'char_wonder_woman',
          locationId: 'loc_gotham_rooftop',
          possessionId: 'item_lasso_of_truth',
          grievanceId: 'grv_gotham_standoff',
        },
      },
      slv: (res: any) => {
        resolved = res;
      },
    };

    await injectCustomSeedIdentifiers(model, bal);

    expect(resolved?.seedBit?.val).toBe(1);
    expect(resolved?.seedBit?.dat?.totalInjected).toBe(4);
    expect(resolved?.seedBit?.dat?.byteZeroVerified).toBe(true);

    const charFile = path.join(tempRepoRoot, 'series', seriesSlug, 'characters', 'char_wonder_woman.md');
    const locFile = path.join(tempRepoRoot, 'series', seriesSlug, 'locations', 'loc_gotham_rooftop.md');
    const itemFile = path.join(tempRepoRoot, 'series', seriesSlug, 'possessions', 'item_lasso_of_truth.md');
    const grvFile = path.join(tempRepoRoot, 'series', seriesSlug, 'grievances', 'grv_gotham_standoff.md');

    expect(fs.existsSync(charFile)).toBe(true);
    expect(fs.existsSync(locFile)).toBe(true);
    expect(fs.existsSync(itemFile)).toBe(true);
    expect(fs.existsSync(grvFile)).toBe(true);

    // Verify Byte-0 invariant
    const charBuf = fs.readFileSync(charFile);
    expect(charBuf[0]).toBe(0x2d); // '-'
    expect(charBuf[1]).toBe(0x2d); // '-'
    expect(charBuf[2]).toBe(0x2d); // '-'
    expect(charBuf[3]).toBe(0x0a); // '\n'
  });
});

```
