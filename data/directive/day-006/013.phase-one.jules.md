#### 2. Series Discovery Buzzer: `packages/299.lore/29.series.unit/buz/series.discovery.buzz.ts`

```typescript
import * as fs from 'node:fs';
import * as path from 'node:path';
import type State from '../../99.core/state.js';
import type SeriesDiscoveryBit from '../fce/seriesDiscovery.bit.js';
import { resolveRepoRoot } from './scaffoldSeriesTree.buzz.js';
import type { VoidScanResultBit } from '../fce/seriesDiscovery.bit.js';

export const scanVoidState = (repoRootInput?: string): VoidScanResultBit => {
  const repoRoot = repoRootInput || resolveRepoRoot();
  const seriesPath = path.join(repoRoot, 'series');

  if (!fs.existsSync(seriesPath)) {
    return {
      isVoidState: true,
      discoveredSlugs: [],
      repoRoot,
      seriesPath,
    };
  }

  const entries = fs.readdirSync(seriesPath, { withFileTypes: true });
  const discoveredSlugs: string[] = [];

  for (const entry of entries) {
    if (entry.isDirectory()) {
      const configPath = path.join(seriesPath, entry.name, 'series.config.json');
      if (fs.existsSync(configPath)) {
        try {
          const raw = fs.readFileSync(configPath, 'utf-8');
          if (raw.trim().length > 0) {
            const parsed = JSON.parse(raw);
            if (parsed && typeof parsed === 'object' && !Array.isArray(parsed)) {
              discoveredSlugs.push(entry.name);
            }
          }
        } catch {
          // Ignore unparseable configs
        }
      }
    }
  }

  discoveredSlugs.sort((a, b) =>
    a === 'under-the-floorboards' ? -1 : b === 'under-the-floorboards' ? 1 : a.localeCompare(b),
  );

  return {
    isVoidState: discoveredSlugs.length === 0,
    discoveredSlugs: Object.freeze(discoveredSlugs),
    repoRoot,
    seriesPath,
  };
};

export const listDiscoveredSeriesAction = async (
  cpy: any,
  bal: SeriesDiscoveryBit,
  _ste?: State,
): Promise<any> => {
  const repoRoot = bal.src || bal.dat?.repoRoot;
  const scanResult = scanVoidState(repoRoot);

  if (bal.slv) {
    bal.slv({
      seriesBit: {
        idx: bal.idx,
        val: scanResult.isVoidState ? 0 : 1,
        dat: scanResult,
      },
    });
  }

  return cpy;
};

```

#### 3. Blessed Series Selection Menu Buzzer: `packages/299.lore/98.menu.unit/buz/seriesSelectionMenu.buzz.ts`

```typescript
import type State from '../../99.core/state.js';
import type { MenuModel } from '../menu.model.js';
import type MenuBit from '../fce/menu.bit.js';
import { scanVoidState } from '../../29.series.unit/buz/series.discovery.buzz.js';
import { resolveLibraryBus } from './blessedModal.buzz.js';

const UPDATE_GRID = '[Grid action] Update Grid';
const OPEN_CHOICE = '[Open action] Open Choice';
const UPDATE_CONSOLE = '[Console action] Update Console';

function emitConsole(ste: State | undefined, line: string): void {
  const cleanLine = String(line || '').replace(/[^\x20-\x7E]/g, '').trim();
  const lib = resolveLibraryBus();
  if (lib && typeof lib.hunt === 'function') {
    lib.hunt(UPDATE_CONSOLE, { idx: 'cns00', src: cleanLine });
    return;
  }
  if (ste && typeof ste.hunt === 'function') {
    ste.hunt(UPDATE_CONSOLE, { idx: 'cns00', src: cleanLine });
  }
}

export const promptSeriesSelectionModal = async (
  cpy: MenuModel,
  bal: MenuBit,
  ste?: State,
): Promise<MenuModel> => {
  const lib = resolveLibraryBus();
  const scanResult = scanVoidState();

  if (scanResult.isVoidState) {
    emitConsole(ste, '>> [SERIES:DISCOVERY] [WARN] Void state detected: No series found in series/.');
    emitConsole(ste, '>> [SERIES:DISCOVERY] Please initialize your first series using the Genesis Wizard.');
    if (bal?.slv) {
      bal.slv({ mnuBit: { idx: 'series-discovery-void', val: 0, dat: scanResult } });
    }
    return cpy;
  }

  const choices = scanResult.discoveredSlugs.map((slug, idx) => `[${idx + 1}] ${slug}`);
  emitConsole(ste, `>> [SERIES:DISCOVERY] Discovered ${scanResult.discoveredSlugs.length} series namespace(s) on disk.`);

  let selectedChoice = choices[0];
  if (lib && typeof lib.hunt === 'function') {
    const gridBit: any = await lib.hunt(UPDATE_GRID, {
      x: 0,
      y: 4,
      xSpan: 4,
      ySpan: 10,
    });
    const choiceBit: any = await lib.hunt(OPEN_CHOICE, {
      dat: { clr0: 'black', clr1: 'yellow' },
      src: 'vertical',
      lst: [...choices, '[<] RETURN TO LORE MENU'],
      net: gridBit?.grdBit?.dat,
    });
    selectedChoice = choiceBit?.chcBit?.src || choices[0];
  }

  if (selectedChoice === '[<] RETURN TO LORE MENU' || !selectedChoice) {
    emitConsole(ste, '>> [SERIES:DISCOVERY] Series selection cancelled by operator.');
    if (bal?.slv) {
      bal.slv({ mnuBit: { idx: 'series-discovery-cancel', val: 0 } });
    }
    return cpy;
  }

  // Extract clean slug from "[1] under-the-floorboards"
  const cleanSlug = selectedChoice.replace(/^\[\d+\]\s*/, '').trim();
  emitConsole(ste, `>> [SERIES:ACTIVE] Bound active series namespace to: ${cleanSlug} [OK]`);

  cpy.activeSelection = 'PROMPT_SERIES_SELECTION';
  cpy.activeSeries = cleanSlug;
  cpy.statusMessage = `Active Series: ${cleanSlug}`;

  if (bal?.slv) {
    bal.slv({
      mnuBit: {
        idx: 'series-discovery-success',
        val: 1,
        src: cleanSlug,
        dat: { scanResult, activeSlug: cleanSlug },
      },
    });
  }

  return cpy;
};

```

#### 4. Unit Verification Suite: `packages/299.lore/29.series.unit/test/seriesDiscovery.test.ts`

```typescript
import { describe, it, expect, afterEach } from 'vitest';
import fs from 'node:fs';
import path from 'node:path';
import os from 'node:os';
import { scanVoidState } from '../buz/series.discovery.buzz.js';

describe('299.lore - [Feature 1.1] Series Disk Discovery & Target Selection', () => {
  let tempRepoRoot: string;

  afterEach(() => {
    if (tempRepoRoot && fs.existsSync(tempRepoRoot)) {
      fs.rmSync(tempRepoRoot, { recursive: true, force: true });
    }
  });

  it('detects void state when series/ directory is absent or empty', () => {
    tempRepoRoot = fs.mkdtempSync(path.join(os.tmpdir(), 'void-test-'));
    const scan = scanVoidState(tempRepoRoot);

    expect(scan.isVoidState).toBe(true);
    expect(scan.discoveredSlugs).toEqual([]);
  });

  it('discovers valid series namespaces containing series.config.json', () => {
    tempRepoRoot = fs.mkdtempSync(path.join(os.tmpdir(), 'discovery-test-'));
    const seriesDir = path.join(tempRepoRoot, 'series', 'gotham-trinity-crisis');
    fs.mkdirSync(seriesDir, { recursive: true });
    fs.writeFileSync(
      path.join(seriesDir, 'series.config.json'),
      JSON.stringify({ series_slug: 'gotham-trinity-crisis', schema_version: '1.0.0' }),
      'utf-8',
    );

    const scan = scanVoidState(tempRepoRoot);

    expect(scan.isVoidState).toBe(false);
    expect(scan.discoveredSlugs).toContain('gotham-trinity-crisis');
  });
});
```
