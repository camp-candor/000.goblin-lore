import type {
  BlessedChoiceModalOptions,
  BlessedInputModalOptions,
  ModalPromptResult,

const UPDATE_GRID = '[Grid action] Update Grid';
const OPEN_CHOICE = '[Open action] Open Choice';
const OPEN_INPUT = '[Open action] Open Input';
const UPDATE_CONSOLE = '[Console action] Update Console';

export function resolveLibraryBus(): any {
  return (globalThis as any).LIBRARY || (global as any).LIBRARY || null;
}

export function sanitizeSeriesSlug(raw: string, fallback = 'ashfall-basin'): string {
  const cleaned = String(raw || '')
    .toLowerCase()
    .trim()
    .replace(/[\s_]+/g, '-')
    .replace(/[^a-z0-9-]/g, '')
    .replace(/-+/g, '-')
    .replace(/^-+|-+$/g, '');
  return cleaned.length > 0 ? cleaned : fallback;
}

/**
 * [Task 1.1.1] Reusable 85.choice.unit modal invocation anchored to the left 4-column grid.
 */
export async function promptBlessedChoice(
  options: BlessedChoiceModalOptions,
): Promise<ModalPromptResult<string>> {
  const lib = resolveLibraryBus();
  const fallbackChoice =
    options.defaultValue || (options.choices.length > 0 ? options.choices[0] : '');

  if (!lib || typeof lib.hunt !== 'function') {
    return { cancelled: false, value: fallbackChoice };
  }

  await lib.hunt(UPDATE_CONSOLE, {
    idx: 'cns00',
    src: `>> [WIZARD:PROMPT] ${options.promptHeader}`,
  });

  const gridBit: any = await lib.hunt(UPDATE_GRID, {
    x: 0,
    y: 4,
    xSpan: 4,
    ySpan: 8,
  });

  const choiceList = [...options.choices, '[CANCEL]'];
  const choiceBit: any = await lib.hunt(OPEN_CHOICE, {
    dat: {
      clr0: 'black',
      clr1: 'yellow',
    },
    src: 'vertical',
    lst: choiceList,
    net: gridBit?.grdBit?.dat,
  });

  const selected = choiceBit?.chcBit?.src ?? '';
  const exitVal = choiceBit?.chcBit?.val ?? 0;

  if (exitVal === -1 || selected === '[CANCEL]' || selected === '') {
    return { cancelled: true, value: '' };
  }

  return { cancelled: false, value: selected };
}

/**
 * [Task 1.1.3] Reusable 84.input.unit text modal invocation with slug sanitization.
 */
export async function promptBlessedInput(
  options: BlessedInputModalOptions,
): Promise<ModalPromptResult<string>> {
  const lib = resolveLibraryBus();
  const fallbackText = options.defaultValue || '';

  if (!lib || typeof lib.hunt !== 'function') {
    const val = options.sanitizeSlug
      ? sanitizeSeriesSlug(fallbackText, fallbackText || 'ashfall-basin')
      : fallbackText;
    return { cancelled: false, value: val };
  }

  await lib.hunt(UPDATE_CONSOLE, {
    idx: 'cns00',
    src: `>> [WIZARD:INPUT] ${options.promptLabel}`,
  });

  const gridBit: any = await lib.hunt(UPDATE_GRID, {
    x: 0,
    y: 4,
    xSpan: 4,
    ySpan: 4,
  });

  const inputBit: any = await lib.hunt(OPEN_INPUT, {
    txt: options.promptLabel,
    src: fallbackText,
    dat: {
      clr0: 'green',
      clr1: 'cyan',
    },
    net: gridBit?.grdBit?.dat,
  });

  const exitVal = inputBit?.putBit?.val ?? 0;
  if (exitVal === -1) {
    return { cancelled: true, value: '' };
  }

  const rawEntered = (inputBit?.putBit?.src ?? '').trim() || fallbackText;
  const finalValue = options.sanitizeSlug
    ? sanitizeSeriesSlug(rawEntered, sanitizeSeriesSlug(fallbackText, 'ashfall-basin'))
    : rawEntered;

  return { cancelled: false, value: finalValue };
}

 * Chains 84.input.unit and 85.choice.unit modals before invoking 29.series.unit.

export const runSeriesGenesisMenu = async (
  cpy: MenuModel,
  bal: MenuBit,
  ste?: State,
): Promise<MenuModel> => {
  const dat = (bal?.dat || {}) as Record<string, any>;
  const abortWizard = (reason: string): MenuModel => {
    const cancelLine = `>> [GENESIS:CANCELLED] ${reason}`;
    emitConsole(ste, cancelLine);
    cpy.activeSelection = '[G] UNIVERSAL SERIES GENESIS WIZARD';
    cpy.lastLifecycleReceipt = '[GENESIS:CANCELLED]';
    cpy.statusMessage = cancelLine;
    if (bal?.slv) {
      bal.slv({
        mnuBit: {
          idx: 'open-genesis-menu',
          val: 0,
          dat: { cancelled: true, receipt: cancelLine },
        },
      });
    }
    return cpy;
  };

  emitConsole(ste, '>> [GENESIS:WIZARD] Initiating Sequential 4-Tuple Parameterization...');

  // Step 1: Series Slug via 84.input.unit (unless explicitly pre-supplied in payload)
  let seriesSlug = dat.seriesSlug ? sanitizeSeriesSlug(dat.seriesSlug) : '';
  if (!seriesSlug) {
    const slugRes = await promptBlessedInput({
      promptLabel: 'Step 1/6: Enter POSIX Series Slug (e.g. gotham-trinity-crisis)',
      defaultValue: 'ashfall-basin',
      sanitizeSlug: true,
    });
    if (slugRes.cancelled) {
      return abortWizard('Aborted at Step 1 (Series Slug).');
    }
    seriesSlug = slugRes.value;
  }

  const displayName =
    dat.displayName ||
    seriesSlug
      .split('-')
      .map((w) => (w.length > 0 ? w[0].toUpperCase() + w.slice(1) : w))
      .join(' ');

  // Step 2: Genre Modality via 85.choice.unit
  let genreTag: GenreClassification = dat.genreTag || 'LOW_FANTASY_FARCE';
  if (!dat.genreTag && !dat.config) {
    const genreOptions = Object.keys(GENRE_CHOICE_MAP);
    const genreRes = await promptBlessedChoice({
      promptHeader: 'Step 2/6: Select Genre Classification',
      choices: genreOptions,
      defaultValue: genreOptions[0],
    });
    if (genreRes.cancelled) {
      return abortWizard('Aborted at Step 2 (Genre Classification).');
    }
    genreTag = GENRE_CHOICE_MAP[genreRes.value] || 'LOW_FANTASY_FARCE';
  }

  // Step 3: Tuple Element <S> State Space via 85.choice.unit
  let stateSpaceSelection = STATE_SPACE_PRESETS['[1] HIGH SCARCITY (8500 bp / Paranoia 6500 bp)'];
  if (!dat.stateSpacePreset && !dat.config) {
    const stateOptions = Object.keys(STATE_SPACE_PRESETS);
    const stateRes = await promptBlessedChoice({
      promptHeader: 'Step 3/6: Configure <S> State Space Accumulators',
      choices: stateOptions,
      defaultValue: stateOptions[0],
    });
    if (stateRes.cancelled) {
      return abortWizard('Aborted at Step 3 (State Space <S>).');
    }
    stateSpaceSelection = STATE_SPACE_PRESETS[stateRes.value] || stateSpaceSelection;
  }

  // Step 4: Tuple Element <A> Action Vocabulary via 85.choice.unit
  let actionVocabSelection =
    ACTION_VOCAB_PRESETS['[1] CONSTRAINED SCAVENGER (LIMP, SIPHON, CONCEAL)'];
  if (!dat.actionVocabPreset && !dat.config) {
    const actionOptions = Object.keys(ACTION_VOCAB_PRESETS);
    const actionRes = await promptBlessedChoice({
      promptHeader: 'Step 4/6: Configure <A> Bounded Action Vocabulary',
      choices: actionOptions,
      defaultValue: actionOptions[0],
    });
    if (actionRes.cancelled) {
      return abortWizard('Aborted at Step 4 (Action Vocabulary <A>).');
    }
    actionVocabSelection = ACTION_VOCAB_PRESETS[actionRes.value] || actionVocabSelection;
  }

  // Step 5: Tuple Element <I> Invariant Laws via 85.choice.unit
  let invariantSelection =
    INVARIANT_PRESETS['[1] STRICT DRAMA (Farce <= 1500 bp / Floor 1500 cdB)'];
  if (!dat.invariantPreset && !dat.config) {
    const invOptions = Object.keys(INVARIANT_PRESETS);
    const invRes = await promptBlessedChoice({
      promptHeader: 'Step 5/6: Configure <I> Invariant Ceilings',
      choices: invOptions,
      defaultValue: invOptions[0],
    });
    if (invRes.cancelled) {
      return abortWizard('Aborted at Step 5 (Invariant Laws <I>).');
    }
    invariantSelection = INVARIANT_PRESETS[invRes.value] || invariantSelection;
  }

  // Step 6: Central Irony Dramatic Spine via 84.input.unit
  let centralIrony = dat.centralIrony || '';
  if (!centralIrony && !dat.config) {
    const ironyRes = await promptBlessedInput({
      promptLabel: 'Step 6/6: Enter Central Irony Dramatic Spine',
      defaultValue: 'Scavengers hoard broken brass valves while starving above clean water.',
      sanitizeSlug: false,
    });
    if (ironyRes.cancelled) {
      return abortWizard('Aborted at Step 6 (Central Irony).');
    }
    centralIrony = sanitizeAscii(ironyRes.value);
  }

  const assembledConfig: AssembledFourTupleConfig = dat.config || {
    schema_version: '1.0.0',
    series_slug: seriesSlug,
    display_name: sanitizeAscii(displayName),
    genre_tag: genreTag,
    central_irony:
      centralIrony ||
      'Scavengers hoard broken brass valves while starving above clean water.',
    state_space: {
      scarcity_index_bps: stateSpaceSelection.scarcity_index_bps,
      paranoia_baseline_bps: stateSpaceSelection.paranoia_baseline_bps,
      somatic_fatigue_enabled: true,
      epistemic_ignorance_enabled: true,
      gram_mass_tiers: {
        scrap_max_grams: 50,
        weight_max_grams: 5000,
        bulk_min_grams: 5001,
      },
    },
    action_vocabulary: {
      permitted_kinetic_verbs: [...actionVocabSelection.permitted_kinetic_verbs],
      forbidden_verbs_global: [...actionVocabSelection.forbidden_verbs_global],
      locomotion_taxonomies: [...actionVocabSelection.locomotion_taxonomies],
    },
    invariant_rules: {
      max_farce_ratio_bps: invariantSelection.max_farce_ratio_bps,
      min_grievance_depth_delta: 1,
      global_acoustic_floor_cdb: invariantSelection.global_acoustic_floor_cdb,
      voiceover_ducking_db: -14,
      min_normalized_category_entropy_bps: 7000,
      thematic_categories: ['C1', 'C2', 'C3', 'C4', 'C5', 'C6', 'C7'],
    },
    medium_projections: {
      target_fps: 24,
      audio_sample_rate_hz: 48000,
      tts_service_port: 8000,
      video_service_port: 8188,
      aspect_ratio: '16:9',
    },
  };

  emitConsole(
    ste,
    `>> [GENESIS:SEAL] Dispatching 4-Tuple for '${seriesSlug}' (Genre: ${assembledConfig.genre_tag})...`,
  );

  let scaffoldBit: any = null;
  const repoRoot = dat.repoRoot || resolveRepoRoot();
  const seriesDir = path.join(repoRoot, 'series', seriesSlug);

  if (ste && typeof ste.hunt === 'function') {
    const scaffoldRes = await ste.hunt(ActSer.SCAFFOLD_SERIES_TREE, {
      idx: 'menu-genesis-scaffold',
      src: seriesSlug,
      dat: {
        seriesSlug,
        config: assembledConfig,
        repoRoot: dat.repoRoot,
      },
    });
    scaffoldBit = scaffoldRes?.seriesBit || scaffoldRes;

    if (scaffoldBit?.val === 1 || fs.existsSync(seriesDir)) {
      try {
        await ste.hunt(ActSed.POPULATE_GENESIS_SEEDS, {
          idx: 'menu-genesis-seed',
          src: seriesSlug,
          dat: {
            seriesSlug,
            repoRoot: dat.repoRoot,
          },
        });
      } catch {
        // Fallback handled by seedDirectoryExamples
      }

      seedDirectoryExamples(seriesDir, seriesSlug);
    }
  } else if (fs.existsSync(seriesDir)) {
    seedDirectoryExamples(seriesDir, seriesSlug);
  }

  const ok = !scaffoldBit || scaffoldBit.val === 1;
  const receiptLine = ok
    ? `>> [GENESIS:OK] Series created: ${seriesSlug}`
    : `:: [GENESIS:FAIL] ${scaffoldBit?.dat?.error || 'Scaffolding rejected'}`;

  emitConsole(ste, receiptLine);
  cpy.activeSelection = '[G] UNIVERSAL SERIES GENESIS WIZARD';
  cpy.lastLifecycleReceipt = ok ? '[GENESIS:OK]' : '[GENESIS:FAIL]';
  cpy.statusMessage = receiptLine;
  if (!Array.isArray(cpy.lastTelemetryReport)) {
    cpy.lastTelemetryReport = [];
  }
  cpy.lastTelemetryReport.push(receiptLine);

  if (bal?.slv) {
    bal.slv({
      mnuBit: {
        idx: 'open-genesis-menu',
        val: ok ? 1 : 0,
        src: seriesSlug,
        dat: { receipt: receiptLine, config: assembledConfig, scaffoldBit },
      },
    });
  }

  return cpy;
};
