#### 1. Branded Basis Point Primitives: `packages/299.lore/38.types.unit/fce/capacityMetric.interface.ts`

```typescript
import { z } from 'zod';

export const BASIS_POINTS_MAX = 10000;
export const BASIS_POINTS_MIN = 0;

/**
 * Strict Basis Points Zod Schema.
 * Rejects floating-point numbers and out-of-range scalars with explicit error codes.
 */
export const BasisPointSchema = z
  .number({
    required_error: '[BASIS_POINTS_REQUIRED] Basis points value is mandatory',
    invalid_type_error: '[FLOAT_PROHIBITED] Basis points must be a pure integer number',
  })
  .int({ message: '[FLOAT_PROHIBITED] Basis points must be an integer (no decimals permitted)' })
  .min(BASIS_POINTS_MIN, { message: '[OUT_OF_BOUNDS] Basis points cannot be negative (< 0)' })
  .max(BASIS_POINTS_MAX, { message: '[OUT_OF_BOUNDS] Basis points cannot exceed 10,000 (1.0)' });

export type BasisPoints = z.infer<typeof BasisPointSchema>;

export interface BasisPointMetricBit {
  readonly raw_value: number;
  readonly basis_points: BasisPoints;
  readonly normalized_ratio: number; // raw_value / 10000.0 for read-only telemetry
}

export function sanitizeBasisPoints(value: number): BasisPoints {
  if (!Number.isInteger(value)) {
    throw new TypeError(`[FLOAT_PROHIBITED] Non-integer basis points provided: ${value}`);
  }
  if (value < BASIS_POINTS_MIN || value > BASIS_POINTS_MAX) {
    throw new RangeError(`[OUT_OF_BOUNDS] Value ${value} lies outside [0, 10000] basis-point range.`);
  }
  return value as BasisPoints;
}

```

import { z } from 'zod';
import { BasisPointSchema } from '../../38.types.unit/fce/capacityMetric.interface.js';

export const LocomotionBaselineEnum = z.enum([
  'bipedal_standard',
  'bipedal_hunched',
  'limping',
  'quadrupedal',
  'slithering',
  'aerial',
]);

export const SomaticLimbCapacitySchema = z
  .object({
    arms_available_bps: BasisPointSchema, // e.g. 20000 bp for 2 arms, 10000 bp for 1 arm
    locomotion_baseline: LocomotionBaselineEnum,
    banned_kinetic_verbs: z.array(z.string().min(1)).default([]),
  })
  .strict();

export type SomaticLimbCapacity = z.infer<typeof SomaticLimbCapacitySchema>;

export const CombatAttributeStatsSchema = z
  .object({
    spite: z.number().int().min(1).max(5).default(3),
    cunning: z.number().int().min(1).max(5).default(3),
    vigor: z.number().int().min(1).max(5).default(3),
    poise: z.number().int().min(1).max(5).default(3),

  .strict();

export type CombatAttributeStats = z.infer<typeof CombatAttributeStatsSchema>;

export const EntityCapacityConfigurationSchema = z
  .object({
    entity_id: z.string().regex(/^(char|loc|item|grv)_[a-z0-9_]+$/),
    somatic_capacity: SomaticLimbCapacitySchema,
    combat_stats: CombatAttributeStatsSchema,
  })
  .strict();

export type EntityCapacityConfiguration = z.infer<typeof EntityCapacityConfigurationSchema>;

#### 3. Pre-Flight Capacity Gating Buzzer: `packages/299.lore/09.entity.unit/buz/capacityStatValidator.buzz.ts`

```typescript
import type State from '../../99.core/state.js';
import type { SomaticLimbCapacity, CombatAttributeStats } from '../val/capacityStat.val.js';

export interface HandLoadRequirement {
  readonly item_id: string;
  readonly occupies_hands_bps: number;
}

export interface CapacityAuditReceipt {
  readonly valid: boolean;
  readonly available_arms_bps: number;
  readonly required_hands_bps: number;
  readonly deficit_bps: number;
  readonly errorCode?: string;
  readonly message?: string;
}

export const auditEntityCapacityPreFlight = (
  somatic: SomaticLimbCapacity,
  equippedItems: readonly HandLoadRequirement[] = [],
  _stats: CombatAttributeStats,
): CapacityAuditReceipt => {
  let requiredHandsBps = 0;
  for (const item of equippedItems) {
    if (!Number.isInteger(item.occupies_hands_bps)) {
      return {
        valid: false,
        available_arms_bps: somatic.arms_available_bps,
        required_hands_bps: 0,
        deficit_bps: 0,
        errorCode: 'FLOAT_PROHIBITED',
        message: `[FLOAT_PROHIBITED] Item '${item.item_id}' hand requirement must be an integer basis point.`,
      };
    }
    requiredHandsBps += item.occupies_hands_bps;
  }

  const deficitBps = requiredHandsBps - somatic.arms_available_bps;
  if (deficitBps > 0) {
    return {
      valid: false,
      available_arms_bps: somatic.arms_available_bps,
      required_hands_bps: requiredHandsBps,
      deficit_bps,
      errorCode: 'ANATOMICAL_CAPACITY_EXCEEDED',
      message: `[ANATOMICAL_CAPACITY_EXCEEDED] Required hand load (${requiredHandsBps} bp) exceeds available somatic capacity (${somatic.arms_available_bps} bp) by ${deficitBps} bp.`,
    };
  }

  return {
    valid: true,
    available_arms_bps: somatic.arms_available_bps,
    required_hands_bps: requiredHandsBps,
    deficit_bps: 0,
  };
};

export const validateCapacityAction = async (
  cpy: any,
  bal: any,
  _ste?: State,
): Promise<any> => {
  const dat = bal?.dat || {};
  const audit = auditEntityCapacityPreFlight(dat.somatic, dat.equippedItems || [], dat.stats);

  if (bal?.slv) {
    bal.slv({
      entBit: {
        idx: bal.idx,
        val: audit.valid ? 1 : 0,
        dat: audit,
      },
    });
  }

  return cpy;
};

```

import { describe, it, expect } from 'vitest';
import { BasisPointSchema, sanitizeBasisPoints } from '../../38.types.unit/fce/capacityMetric.interface.js';
import { auditEntityCapacityPreFlight } from '../buz/capacityStatValidator.buzz.js';
import { CombatAttributeStatsSchema } from '../val/capacityStat.val.js';

describe('299.lore - [Feature 2.2] Integer Basis Point Capacity & Stat Configuration', () => {
  it('enforces BasisPointSchema integer bounds [0, 10000] and rejects floats', () => {
    expect(BasisPointSchema.safeParse(5000).success).toBe(true);
    expect(BasisPointSchema.safeParse(0).success).toBe(true);
    expect(BasisPointSchema.safeParse(10000).success).toBe(true);

    const floatRes = BasisPointSchema.safeParse(5000.5);
    expect(floatRes.success).toBe(false);
    expect(floatRes.error?.issues[0].message).toContain('FLOAT_PROHIBITED');

    const overRes = BasisPointSchema.safeParse(10500);
    expect(overRes.success).toBe(false);
    expect(overRes.error?.issues[0].message).toContain('OUT_OF_BOUNDS');

    expect(() => sanitizeBasisPoints(5000.25)).toThrowError(/FLOAT_PROHIBITED/);
  });

  it('validates 2d6 combat statistics within [1, 5] integer bounds', () => {
    const statsValid = CombatAttributeStatsSchema.safeParse({
      spite: 4,
      cunning: 5,
      vigor: 3,
      poise: 2,
    });
    expect(statsValid.success).toBe(true);

    const statsInvalid = CombatAttributeStatsSchema.safeParse({
      spite: 6, // Out of bounds
      cunning: 2,
      vigor: 3,
      poise: 2,
    });
    expect(statsInvalid.success).toBe(false);
  });

  it('audits somatic limb capacity and flags anatomical capacity breaches', () => {
    const somatic = {
      arms_available_bps: 10000, // 1 arm available (e.g. Wonder Woman holding Lasso)
      locomotion_baseline: 'bipedal_standard' as const,
      banned_kinetic_verbs: [],
    };

    const stats = { spite: 3, cunning: 5, vigor: 4, poise: 5 };

    const validAudit = auditEntityCapacityPreFlight(
      somatic,
      [{ item_id: 'item_lasso_of_truth', occupies_hands_bps: 10000 }],
      stats,
    );
    expect(validAudit.valid).toBe(true);

    const invalidAudit = auditEntityCapacityPreFlight(
      somatic,
      [
        { item_id: 'item_lasso_of_truth', occupies_hands_bps: 10000 },
        { item_id: 'item_heavy_shield', occupies_hands_bps: 10000 },
      ],
      stats,
    );
    expect(invalidAudit.valid).toBe(false);
    expect(invalidAudit.errorCode).toBe('ANATOMICAL_CAPACITY_EXCEEDED');
  });
});
